1/*!
2Handlers for HTTP requests to `/api/v0/moocfi/outbound-handoffs`.
3
4The other direction from [`super::credit_registration_handoffs`]: a student leaving courses for mooc.fi
5carries a token minted by
6[`crate::controllers::main_frontend::credit_registrations::create_my_moocfi_handoff`], and mooc.fi's
7backend redeems it here to learn whom to sign in and where to send them.
8*/

23/// Per student, so that one student's retries cannot use up the quota of everyone behind mooc.fi.
24static PER_STUDENT_LIMITS: Lazy<EndpointLimiters> = Lazy::new(|| {
25    EndpointLimiters::from_config(&RateLimitConfig {
26        per_minute: Some(20),
27        per_hour: Some(200),
28        ..Default::default()
29    })
30});

32#[derive(Debug, Deserialize)]
33pub struct RedeemOutboundHandoffRequest {
34    token: Option<SecretString>,
35    /// The IP of the browser that brought the token to mooc.fi, which must be the one it was
36    /// minted from.
37    client_ip: Option<String>,
38}

40#[derive(Debug, Serialize)]
41pub struct RedeemedOutboundHandoff {
42    /// The student's TMC user id.
43    upstream_id: i32,
44    moocfi_course_id: Uuid,
45    destination: MoocfiHandoffDestination,
46}

48/**
49POST `/api/v0/moocfi/outbound-handoffs/redeem` - Consumes a handoff a student brought to mooc.fi.
50
51Authorized by `Authorization: Bearer <MOOCFI_HANDOFF_SECRET>`, the secret mooc.fi mints inbound
52handoffs with. Body `{ token, client_ip }`. Answers 200 `{ upstream_id, moocfi_course_id,
53destination }`; 410 `spent` when the token was redeemed already or has expired; 410 `ip_mismatch`
54when `client_ip` is not the IP the token was minted from, which leaves the token unspent; 400
55`invalid` when the token is missing, malformed or unknown, or `client_ip` is not an IP; 401 for a
56wrong secret; 429 per student; 503 when handoffs are not configured.
57*/
58#[instrument(skip_all)]
59pub async fn redeem_outbound_handoff(
60    request: HttpRequest,
61    pool: web::Data<PgPool>,
62    app_conf: web::Data<ApplicationConfiguration>,
63    jwt_key: web::Data<JwtKey>,
64    payload: web::Json<RedeemOutboundHandoffRequest>,
65) -> ControllerResult<HttpResponse> {
66    let token = skip_authorize();
67    let Some(handoff_secret) = &app_conf.moocfi_configuration.handoff_secret else {
68        return token.authorized_ok(refusal(StatusCode::SERVICE_UNAVAILABLE, "not_configured"));
69    };
70    if !super::is_authorized(&request, handoff_secret) {
71        return token.authorized_ok(refusal(StatusCode::UNAUTHORIZED, "unauthorized"));
72    }
73    if super::is_over_global_cap() {
74        return token.authorized_ok(rate_limited(None));
75    }
76    let Some(handoff) = payload
77        .token
78        .as_ref()
79        .map(|handoff| handoff.expose_secret().trim())
80        .filter(|handoff| {
81            handoff.len() == ACCESS_TOKEN_LENGTH
82                && handoff.bytes().all(|b| b.is_ascii_alphanumeric())
83        })
84    else {
85        return token.authorized_ok(refusal(StatusCode::BAD_REQUEST, "invalid"));
86    };
87    let Some(client_ip) = payload
88        .client_ip
89        .as_deref()
90        .and_then(|client_ip| client_ip.trim().parse::<IpAddr>().ok())
91    else {
92        return token.authorized_ok(refusal(StatusCode::BAD_REQUEST, "invalid"));
93    };
94
95    let mut conn = pool.acquire().await?;
96    let digest = Sha256::digest(handoff.as_bytes());
97    let Some(student_id) =
98        models::moocfi_outbound_handoffs::get_student_id(&mut conn, &digest).await?
99    else {
100        return token.authorized_ok(refusal(StatusCode::BAD_REQUEST, "invalid"));
101    };
102    if let Err(retry_after) = PER_STUDENT_LIMITS.check_key(student_id.to_string()) {
103        return token.authorized_ok(rate_limited(Some(retry_after)));
104    }
105    match models::moocfi_outbound_handoffs::redeem(
106        &mut conn,
107        &digest,
108        &client_ip_hash(client_ip, &jwt_key),
109    )
110    .await?
111    {
112        Ok(redeemed) => {
113            info!(outcome = "redeemed", "Redeemed a handoff to mooc.fi");
114            token.authorized_ok(HttpResponse::Ok().json(RedeemedOutboundHandoff {
115                upstream_id: redeemed.upstream_id,
116                moocfi_course_id: redeemed.moocfi_course_id,
117                destination: redeemed.destination,
118            }))
119        }
120        Err(RedeemRefusal::IpMismatch) => {
121            token.authorized_ok(refusal(StatusCode::GONE, "ip_mismatch"))
122        }
123        Err(RedeemRefusal::Spent) => token.authorized_ok(refusal(StatusCode::GONE, "spent")),
124        Err(RedeemRefusal::Unknown) => {
125            token.authorized_ok(refusal(StatusCode::BAD_REQUEST, "invalid"))
126        }
127    }
128}
129
130/// A refused redeem, logged under `kind`, which the body's `type` repeats.
131fn refusal(status: StatusCode, kind: &'static str) -> HttpResponse {
132    warn!(outcome = kind, "Refused to redeem a handoff to mooc.fi");
133    HttpResponse::build(status).json(serde_json::json!({ "type": kind }))
134}
135
136fn rate_limited(retry_after: Option<std::time::Duration>) -> HttpResponse {
137    super::with_retry_after(
138        refusal(StatusCode::TOO_MANY_REQUESTS, "rate_limited"),
139        retry_after,
140    )
141}
142
143pub fn _add_routes(cfg: &mut ServiceConfig) {
144    cfg.route("/redeem", web::post().to(redeem_outbound_handoff));
