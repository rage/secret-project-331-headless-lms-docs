399pub async fn execute(
400    store: &MockSuotarStore,
401    pool: &PgPool,
402    command: MockSuotarCommand,
403) -> CommandResult {
404    let name = command.name().to_string();
405    match run(store, pool, command).await {
406        Ok(result) => CommandResult::Ok {
407            command: name,
408            result,
409        },
410        Err(error) => CommandResult::Error {
411            command: Some(name),
412            code: error.code,
413            message: error.message,
414        },
415    }
416}

418async fn run(store: &MockSuotarStore, pool: &PgPool, command: MockSuotarCommand) -> Outcome {
419    // `reset { world }` installs nothing: the next contract request builds the world lazily.
420    if let MockSuotarCommand::Reset {
421        scope: ResetScope::World,
422    } = &command
423    {
424        store.flush().await?;
425        return Ok(json!({ "flushed": true }));
426    }
427    if let MockSuotarCommand::PushWorld(push) = command {
428        let marker = default_world::db_generation_marker(pool).await;
429        let world = world_from_push(push);
430        let counts = json!({
431            "persons": world.persons.len(),
432            "courseUnits": world.course_units.len(),
433            "enrolments": world.enrolments.len(),
434            "attainments": world.attainments.len(),
435            "submissions": world.submissions.len(),
436            "productTokens": world.product_tokens.len(),
437        });
438        let generation = store.install_world(&world, marker.as_deref()).await?;
439        return Ok(json!({ "generation": generation, "counts": counts }));
440    }
441
442    let generation = current_generation(store, pool).await?;
443    match command {
444        MockSuotarCommand::Reset { scope } => reset(store, &generation, scope).await,
445        MockSuotarCommand::PushWorld(_) => unreachable!("handled above"),
446        MockSuotarCommand::UpsertPersons { persons } => {
447            upsert_command(
448                store,
449                &generation,
450                EntityHash::Persons,
451                "studentNumbers",
452                persons,
453                person_from,
454                |person| person.student_number.clone(),
455            )
456            .await
457        }
458        MockSuotarCommand::UpsertCourseUnits { course_units } => {
459            upsert_command(
460                store,
461                &generation,
462                EntityHash::CourseUnits,
463                "courseCodes",
464                course_units,
465                course_unit_from,
466                |unit| unit.course_code.clone(),
467            )
468            .await
469        }
470        MockSuotarCommand::UpsertEnrolments { enrolments } => {
471            upsert_command(
472                store,
473                &generation,
474                EntityHash::Enrolments,
475                "enrolmentIds",
476                enrolments,
477                enrolment_from,
478                |enrolment| enrolment.id.clone(),
479            )
480            .await
481        }
482        MockSuotarCommand::UpsertAttainments { attainments } => {
483            upsert_command(
484                store,
485                &generation,
486                EntityHash::Attainments,
487                "attainmentIds",
488                attainments,
489                attainment_from,
490                |attainment| attainment.id.clone(),
491            )
492            .await
493        }
494        MockSuotarCommand::UpsertProductAccessTokens { tokens } => {
495            upsert_command(
496                store,
497                &generation,
498                EntityHash::ProductTokens,
499                "productIds",
500                tokens,
501                product_token_from,
502                |token| token.open_university_product_id.clone(),
503            )
504            .await
505        }
506        MockSuotarCommand::DeletePersons { student_numbers } => {
507            delete_persons(store, &generation, &student_numbers).await
508        }
509        MockSuotarCommand::AllocatePerson(args) => allocate_person(store, &generation, args).await,
510        MockSuotarCommand::GenerateRoster {
511            course_code,
512            realisation_id,
513            count,
514            student_number_prefix,
515        } => {
516            generate_roster(
517                store,
518                &generation,
519                &course_code,
520                &realisation_id,
521                count,
522                student_number_prefix.as_deref(),
523            )
524            .await
525        }
526        MockSuotarCommand::SetPersonBehaviour {
527            student_number,
528            patch,
529        } => {
530            let mut person: MockPerson = store
531                .get_json(&generation, EntityHash::Persons, &student_number)
532                .await?
533                .ok_or_else(|| {
534                    CommandError::new(
535                        "unknownPerson",
536                        format!("No person `{student_number}` in the world."),
537                    )
538                })?;
539            if let Some(ripeness) = patch.ripeness {
540                person.behaviour.ripeness = Some(ripeness);
541            }
542            if let Some(detection) = patch.duplicate_detection {
543                person.behaviour.duplicate_detection = Some(detection);
544            }
545            if let Some(email) = patch.primary_email {
546                person.primary_email = email;
547            }
548            if let Some(email) = patch.secondary_email {
549                person.secondary_email = Some(email);
550            }
551            store
552                .upsert_json(
553                    &generation,
554                    EntityHash::Persons,
555                    &BTreeMap::from([(student_number.clone(), person)]),
556                )
557                .await?;
558            Ok(json!({ "studentNumber": student_number }))
559        }
560        MockSuotarCommand::SetCourseBehaviour { course_code, patch } => {
561            let mut unit: MockCourseUnit = store
562                .get_json(&generation, EntityHash::CourseUnits, &course_code)
563                .await?
564                .ok_or_else(|| {
565                    CommandError::new(
566                        "unknownCourseUnit",
567                        format!("No course unit `{course_code}` in the world."),
568                    )
569                })?;
570            if let Some(allowed) = patch.import_allowed {
571                unit.behaviour.import_allowed = allowed;
572            }
573            store
574                .upsert_json(
575                    &generation,
576                    EntityHash::CourseUnits,
577                    &BTreeMap::from([(course_code.clone(), unit)]),
578                )
579                .await?;
580            Ok(json!({ "courseCode": course_code }))
581        }
582        MockSuotarCommand::TransitionSubmission {
583            submitted_attainment_id,
584            to,
585        } => transition(store, &generation, &[submitted_attainment_id], to).await,
586        MockSuotarCommand::TransitionSubmissionsFor {
587            student_number,
588            course_code,
589            to,
590        } => {
591            let ids =
592                submission_ids_for(store, &generation, &student_number, course_code.as_deref())
593                    .await?;
594            transition(store, &generation, &ids, to).await
595        }
596        MockSuotarCommand::ListSubmissions(filter) => {
597            let submissions: BTreeMap<String, MockSubmission> =
598                store.all_json(&generation, EntityHash::Submissions).await?;
599            let matching: Vec<&MockSubmission> = submissions
600                .values()
601                .filter(|submission| {
602                    filter
603                        .student_number
604                        .as_ref()
605                        .is_none_or(|value| &submission.student_number == value)
606                        && filter
607                            .course_code
608                            .as_ref()
609                            .is_none_or(|value| &submission.course_code == value)
610                })
611                .collect();
612            Ok(json!({ "submissions": matching }))
613        }
614        MockSuotarCommand::ArmFault(spec) => arm_fault(store, &generation, spec).await,
615        MockSuotarCommand::DisarmFault { id } => {
616            store
617                .disarm_faults(&generation, std::slice::from_ref(&id))
618                .await?;
619            Ok(json!({ "disarmed": [id] }))
620        }
621        MockSuotarCommand::DisarmFaults { owner } => {
622            let resolved = resolve_owner(store, &generation, &owner).await?;
623            let ids: Vec<String> = store
624                .faults(&generation)
625                .await?
626                .into_iter()
627                .filter(|fault| fault.owner.as_ref().is_some_and(|o| overlaps(o, &resolved)))
628                .map(|fault| fault.id)
629                .collect();
630            store.disarm_faults(&generation, &ids).await?;
631            Ok(json!({ "disarmed": ids }))
632        }
633        MockSuotarCommand::ListFaults(filter) => {
634            let remaining = store.remaining_budgets(&generation).await?;
635            let faults: Vec<serde_json::Value> = store
636                .faults(&generation)
637                .await?
638                .into_iter()
639                .filter(|fault| filter.id.as_ref().is_none_or(|id| &fault.id == id))
640                .map(|fault| {
641                    let left = remaining.get(&fault.id).copied().unwrap_or(0);
642                    let spent = fault.lifetime.budget().is_some() && left <= 0;
643                    json!({ "fault": fault, "remaining": left, "spent": spent })
644                })
645                .collect();
646            Ok(json!({ "faults": faults }))
647        }
648        MockSuotarCommand::ExplainFault { fault, against } => {
649            explain_fault(store, &generation, fault, against).await
650        }
651        MockSuotarCommand::SetDefaults { patch } => {
652            let mut defaults = store.preamble(&generation).await?.defaults;
653            apply_defaults_patch(&mut defaults, patch);
654            store.set_defaults(&generation, &defaults).await?;
655            Ok(serde_json::to_value(&defaults).unwrap_or(serde_json::Value::Null))
656        }
657        MockSuotarCommand::ApplyScenario { name, args } => {
658            scenarios::apply(store, &generation, &name, args).await
659        }
660        MockSuotarCommand::ListCalls(filter) => list_calls(store, &generation, filter).await,
661    }
662}

664/// The shared body behind every `Upsert*` command. `key_of` reads the id off the built entity rather
665/// than the wire type, so a derived id is what comes back under `result_key`.
666async fn upsert_command<U, T: Serialize>(
667    store: &MockSuotarStore,
668    generation: &str,
669    hash: EntityHash,
670    result_key: &'static str,
671    items: Vec<U>,
672    build: impl Fn(U) -> T,
673    key_of: impl Fn(&T) -> String,
674) -> Outcome {
675    let entries: BTreeMap<String, T> = items
676        .into_iter()
677        .map(|item| {
678            let entity = build(item);
679            (key_of(&entity), entity)
680        })
681        .collect();
682    let mut result = serde_json::Map::new();
683    result.insert(
684        result_key.to_string(),
685        json!(entries.keys().collect::<Vec<_>>()),
686    );

708async fn reset(store: &MockSuotarStore, generation: &str, scope: ResetScope) -> Outcome {
709    match scope {
710        // `Reset { scope: World }` never reaches this match — `run()` intercepts it first.
711        ResetScope::World => unreachable!("world reset is handled in `run` before dispatch"),
712        ResetScope::Faults => {
713            store.clear_faults(generation).await?;
714            Ok(json!({ "cleared": "faults" }))
715        }
716        ResetScope::Calls => {
717            store.clear_hash(generation, EntityHash::Calls).await?;
718            Ok(json!({ "cleared": "calls" }))
719        }
720        ResetScope::Persons(scope) => {
721            let mut student_numbers = scope.student_numbers.unwrap_or_default();
722            if let Some(owner) = scope.owner {
723                let resolved = resolve_owner(store, generation, &owner).await?;
724                student_numbers.extend(resolved.student_numbers);
725            }
726            student_numbers.sort();
727            student_numbers.dedup();
728            delete_persons(store, generation, &student_numbers).await
729        }
730    }
731}

913async fn transition(
914    store: &MockSuotarStore,
915    generation: &str,
916    ids: &[String],
917    to: SubmissionTarget,
918) -> Outcome {
919    let now = Utc::now();
920    let mut touched = Vec::new();
921    let mut updated: BTreeMap<String, MockSubmission> = BTreeMap::new();
922    let mut new_attainments: BTreeMap<String, MockAttainment> = BTreeMap::new();
923    let defaults = store.preamble(generation).await?.defaults;
924
925    for id in ids {
926        let Some(mut submission): Option<MockSubmission> = store
927            .get_json(generation, EntityHash::Submissions, id)
928            .await?
929        else {
930            return Err(CommandError::new(
931                "unknownSubmission",
932                format!("No submission `{id}` in the world."),
933            ));
934        };
935        match to {
936            SubmissionTarget::Registered => {
937                let attainment_id = ids::final_attainment_id(id);
938                let attainment = MockAttainment::from_submission(
939                    &submission,
940                    &attainment_id,
941                    AttainmentState::Attained,
942                    &defaults,
943                    now,
944                );
945                new_attainments.insert(attainment_id.clone(), attainment);
946                submission.lifecycle = SubmissionLifecycle::Registered {
947                    attainment_id,
948                    registered_at: now,
949                };
950            }
951            SubmissionTarget::Misregistered => {
952                let attainment_id = ids::final_attainment_id(id);
953                let attainment = MockAttainment::from_submission(
954                    &submission,
955                    &attainment_id,
956                    AttainmentState::Misregistered,
957                    &defaults,
958                    now,
959                );
960                new_attainments.insert(attainment_id.clone(), attainment);
961                submission.lifecycle = SubmissionLifecycle::Misregistered {
962                    attainment_id,
963                    misregistered_at: now,
964                };
965            }
966            SubmissionTarget::NotRegistered => {
967                submission.lifecycle = SubmissionLifecycle::Pending {
968                    ripeness: Ripeness::Manual,
969                };
970            }
971            SubmissionTarget::TimedOutButLanded => {
972                submission.lifecycle = SubmissionLifecycle::TimedOutButLanded {
973                    ripeness: Ripeness::Manual,
974                };
975            }
976            SubmissionTarget::TimedOutNothingLanded => {
977                submission.lifecycle = SubmissionLifecycle::TimedOutNothingLanded;
978            }
979        }
980        touched.push(id.clone());
981        updated.insert(id.clone(), submission);
982    }
983
984    store
985        .upsert_json(generation, EntityHash::Submissions, &updated)
986        .await?;
987    store
988        .upsert_json(generation, EntityHash::Attainments, &new_attainments)
989        .await?;
990    store.reindex(generation).await?;
991    Ok(json!({
992        "submittedAttainmentIds": touched,
993        "attainmentIds": new_attainments.keys().collect::<Vec<_>>(),
994    }))
995}

1013async fn build_fault(
1014    store: &MockSuotarStore,
1015    generation: &str,
1016    spec: super::faults::FaultSpec,
1017) -> Result<(Fault, (SuotarEndpoint, Stage)), CommandError> {
1018    let predicates = spec.when.into_predicates();
1019    let validated = validate(&predicates, &spec.then, spec.proves_double_submission)
1020        .map_err(|problem| CommandError::new(&problem.code, problem.message))?;
1021    let owner = match predicates.iter().find_map(|predicate| match predicate {
1022        Predicate::Owner(owner) => Some(owner.clone()),
1023        _ => None,
1024    }) {
1025        Some(owner) => Some(resolve_owner(store, generation, &owner).await?),
1026        None => None,
1027    };
1028    let parallel_safe = predicates.iter().any(|predicate| {
1029        matches!(
1030            predicate,
1031            Predicate::Owner(_) | Predicate::StudentNumber(_) | Predicate::CourseCode(_)
1032        )
1033    });
1034    let seq = store.next_fault_seq(generation).await?;
1035    Ok((
1036        Fault {
1037            id: spec.id,
1038            seq,
1039            when: predicates,
1040            then: spec.then,
1041            lifetime: spec.lifetime,
1042            proves_double_submission: spec.proves_double_submission,
1043            owner,
1044            parallel_safe,
1045            armed_at: Utc::now(),
1046        },
1047        validated,
1048    ))

1051async fn explain_fault(
1052    store: &MockSuotarStore,
1053    generation: &str,
1054    spec: super::faults::FaultSpec,
1055    against: Option<HypotheticalRequest>,
1056) -> Outcome {
1057    let (fault, (endpoint, stage)) = build_fault(store, generation, spec).await?;
1058    let mut explanation = json!({
1059        "valid": true,
1060        "endpoint": endpoint,
1061        "stage": stage,
1062        "parallelSafe": fault.parallel_safe,
1063        "owner": fault.owner,
1064        "resolvableKeys": resolvable_keys(endpoint),
1065        "requestShaped": fault.then.is_request_shaped(),
1066    });
1067    if let Some(request) = against {
1068        let items: Vec<ItemAddress> = request
1069            .items
1070            .iter()
1071            .map(|item| ItemAddress {
1072                request_item_id: item.request_item_id.clone(),
1073                student_number: item.student_number.clone(),
1074                course_code: item.course_code.clone(),
1075                product_id: item.product_id.clone(),
1076                submitted_attainment_id: item.submitted_attainment_id.clone(),
1077            })
1078            .collect();
1079        let mut per_stage = serde_json::Map::new();
1080        for candidate in Stage::ALL {
1081            let outcome = if fault.then.is_request_shaped() {
1082                matches_request(&fault, request.endpoint, candidate, &items)
1083            } else {
1084                items
1085                    .iter()
1086                    .map(|item| matches_item(&fault, request.endpoint, candidate, item))
1087                    .find(FaultMatch::fires)
1088                    .unwrap_or(FaultMatch::Missed("no item matched"))
1089            };
1090            per_stage.insert(
1091                candidate.as_str().to_string(),
1092                match outcome {
1093                    FaultMatch::Fires => json!({ "fires": true }),
1094                    FaultMatch::Missed(predicate) => {
1095                        json!({ "fires": false, "failedPredicate": predicate })
1096                    }
1097                },
1098            );
1099        }
1100        if let Some(object) = explanation.as_object_mut() {
1101            object.insert("against".to_string(), serde_json::Value::Object(per_stage));
1102        }
1103    }
1104    Ok(explanation)
1105}

1107async fn resolve_owner(
1108    store: &MockSuotarStore,
1109    generation: &str,
1110    owner: &OwnerRef,
1111) -> Result<ResolvedOwner, CommandError> {
1112    let mut resolved = ResolvedOwner {
1113        user: owner.user.clone(),
1114        course: owner.course.clone(),
1115        ..Default::default()
1116    };
1117    for (half, prefix) in [
1118        (owner.user.as_ref(), "user"),
1119        (owner.course.as_ref(), "course"),
1120    ] {
1121        let Some(value) = half else { continue };
1122        let field = format!("{prefix}:{value}");
1123        let Some(keys): Option<OwnerKeys> = store.owner_keys(generation, &field).await? else {
1124            // A fault that can never match must not be armed silently.
1125            let known = store.known_owner_refs(generation).await?.join(", ");
1126            return Err(CommandError::new(
1127                "unknownOwner",
1128                format!("`{field}` names nobody in the world. It knows: {known}."),
1129            ));
1130        };
1131        if prefix == "user" {
1132            resolved.student_numbers = keys.student_numbers;
1133        } else {
1134            resolved.course_codes = keys.course_codes;
1135            resolved.product_ids = keys.product_ids;
1136        }
1137    }
1138    Ok(resolved)
1139}

1549    pool: web::Data<PgPool>,
1550) -> ControllerResult<HttpResponse> {

1575                    && preamble.db_generation == db_generation,
1576                "counts": counts,
1577                "defaults": preamble.defaults,
1578            })
1579        }
1580        // Never installs one: a health check that built a world could not report an empty one.
1581        None => json!({
1582            "enabled": true,
1583            "generation": serde_json::Value::Null,
1584            "dbGeneration": db_generation,
1585            "worldDbGeneration": serde_json::Value::Null,

1587            "counts": serde_json::Value::Null,

1638pub async fn command(
1639    app_conf: web::Data<ApplicationConfiguration>,
1640    store: web::Data<MockSuotarStore>,
1641    pool: web::Data<PgPool>,
1642    body: web::Bytes,
1643) -> ControllerResult<HttpResponse> {
1644    super::assert_enabled(&app_conf);
1645    let token = skip_authorize();
1646    let parsed: MockSuotarCommand = match serde_json::from_slice(&body) {
1647        Ok(parsed) => parsed,
1648        Err(error) => {
1649            return token.authorized_ok(HttpResponse::BadRequest().json(CommandResult::Error {
1650                command: None,
1651                code: "unknownCommand".to_string(),
1652                message: error.to_string(),
1653            }));
1654        }
1655    };
1656    let result = execute(&store, &pool, parsed).await;
1657    token.authorized_ok(match &result {
1658        CommandResult::Ok { .. } => HttpResponse::Ok().json(&result),
1659        CommandResult::NotImplemented { .. } => HttpResponse::NotImplemented().json(&result),
1660        CommandResult::Error { code, .. } if code == "internalError" => {
1661            HttpResponse::InternalServerError().json(&result)
1662        }
1663        CommandResult::Error { .. } => HttpResponse::BadRequest().json(&result),
1664    })
1665}
