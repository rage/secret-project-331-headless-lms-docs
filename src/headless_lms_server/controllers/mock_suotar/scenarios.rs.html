1//! Named scenarios: small compositions of the control primitives, exposed as one command.
2//!
3//! A scenario earns its place only by composing something data alone cannot express — an armed fault
4//! — or by being a hands-free dev demo; per-spec fixtures come from the seed instead.
5//!
6//! Each writes the course unit for its `courseCode` whole, so its caller has to own that course code,
7//! and returns the identifiers it minted plus the scope its rows are ticked with.
8

42pub const SCENARIOS: [&str; 5] = [
43    "happy-path",
44    "happy-path-auto",
45    "timeout-but-landed",
46    "timeout-nothing-landed",
47    "post-send-death",
48];
49
50pub async fn apply(
51    store: &MockSuotarStore,
52    generation: &str,
53    name: &str,
54    args: ScenarioArgs,
55) -> Result<serde_json::Value, CommandError> {
56    if !SCENARIOS.contains(&name) {
57        return Err(CommandError::new(
58            "unknownScenario",
59            format!("No scenario `{name}`. Known: {}.", SCENARIOS.join(", ")),
60        ));
61    }
62    let mut result = match name {
63        "happy-path" => plain(store, generation, &args, Ripeness::Manual).await?,
64        "happy-path-auto" => {
65            plain(
66                store,
67                generation,
68                &args,
69                Ripeness::AutoAfterVerifyCalls { calls: 1 },
70            )
71            .await?
72        }
73        "timeout-but-landed" => timeout(store, generation, &args, Stage::AfterWrite, true).await?,
74        "timeout-nothing-landed" => {
75            timeout(store, generation, &args, Stage::Resolve, false).await?
76        }
77        "post-send-death" => post_send_death(store, generation, &args).await?,
78        _ => unreachable!("checked against the catalogue above"),
79    };
80
81    // The scope comes from the owner the caller passed, not from the fixtures touched.
82    if let Some(object) = result.as_object_mut() {
83        object.insert("scenario".to_string(), json!(name));
84        match &args.owner {
85            Some(owner) if !owner.is_empty() => {
86                object.insert(
87                    "scope".to_string(),
88                    json!({ "courseSlug": owner.course, "userEmail": owner.user }),
89                );
90                object.insert("owner".to_string(), json!(owner));
91            }
92            _ => {
93                object.insert("scope".to_string(), serde_json::Value::Null);
94            }
95        }
96    }
97    Ok(result)
98}

100async fn plain(
101    store: &MockSuotarStore,
102    generation: &str,
103    args: &ScenarioArgs,
104    ripeness: Ripeness,
105) -> Result<serde_json::Value, CommandError> {
106    let realisation = ensure_course(store, generation, args).await?;
107    let student_number = put_person(store, generation, args, Some(ripeness)).await?;
108    let course_code = course_code(args)?;
109    let enrolment_id = put_enrolment(
110        store,
111        generation,
112        &student_number,
113        &course_code,
114        &realisation,
115    )
116    .await?;
117    Ok(json!({
118        "studentNumber": student_number,
119        "personId": ids::person_id(&student_number),
120        "enrolmentId": enrolment_id,
121        "realisationId": realisation.id,
122        "courseCode": course_code,
123        "kind": realisation.kind,
124    }))
125}

127async fn timeout(
128    store: &MockSuotarStore,
129    generation: &str,
130    args: &ScenarioArgs,
131    stage: Stage,
132    disclose: bool,
133) -> Result<serde_json::Value, CommandError> {
134    let mut base = plain(store, generation, args, Ripeness::Manual).await?;
135    let student_number = string_field(&base, "studentNumber")?;
136    // After the write, `sisuTimeout` leaves the world indistinguishable from a successful import; at
137    // `resolve` nothing landed.
138    let fault_id = format!("timeout-{student_number}");
139    arm(
140        store,
141        generation,
142        &fault_id,
143        vec![
144            Predicate::Endpoint(SuotarEndpoint::ImportAttainments),
145            Predicate::Stage(stage),
146            Predicate::StudentNumber(student_number.clone()),
147        ],
148        Effect::ItemLevel {
149            code: "sisuTimeout".to_string(),
150            message: None,
151            disclose_submitted_attainment_id: disclose,
152        },
153        Lifetime {
154            matching_items: Some(1),
155            ..Default::default()
156        },
157    )
158    .await?;
159    merge(
160        &mut base,
161        json!({ "faultId": fault_id, "discloseId": disclose }),
162    );
163    Ok(base)
164}

166async fn post_send_death(
167    store: &MockSuotarStore,
168    generation: &str,
169    args: &ScenarioArgs,
170) -> Result<serde_json::Value, CommandError> {
171    let mut base = plain(store, generation, args, Ripeness::Manual).await?;
172    let student_number = string_field(&base, "studentNumber")?;
173    let fault_id = format!("post-send-death-{student_number}");
174    arm(
175        store,
176        generation,
177        &fault_id,
178        vec![
179            Predicate::Endpoint(SuotarEndpoint::ImportAttainments),
180            Predicate::Stage(Stage::AfterWrite),
181            Predicate::StudentNumber(student_number),
182        ],
183        Effect::ConnectionReset,
184        Lifetime {
185            matching_calls: Some(1),
186            ..Default::default()
187        },
188    )
189    .await?;
190    merge(&mut base, json!({ "faultId": fault_id }));
191    Ok(base)
192}

194/// A five-credit pass/fail realisation with an acceptor: the scenarios differ in the fault they arm,
195/// not in their data.
196async fn ensure_course(
197    store: &MockSuotarStore,
198    generation: &str,
199    args: &ScenarioArgs,
200) -> Result<MockRealisation, CommandError> {
201    let course_code = course_code(args)?;
202    let kind = args.realisation_kind.unwrap_or(RealisationKind::Degree);
203    let now = Utc::now();
204    let realisation = MockRealisation {
205        id: ids::realisation_id(&course_code, kind),
206        name: LocalizedName {
207            fi: course_code.clone(),
208            sv: course_code.clone(),
209            en: course_code.clone(),
210        },
211        assessment_item_id: ids::assessment_item_id(&course_code, kind),
212        kind,
213        activity_period: DatePeriod {
214            start_date: (now - Duration::days(180)).date_naive(),
215            end_date: (now + Duration::days(180)).date_naive(),
216        },
217        grade_scale_id: PASS_FAIL_SCALE.to_string(),
218        credits: CreditRange { min: 5.0, max: 5.0 },
219        acceptor_person_id: Some(ACCEPTOR_PERSON_ID.to_string()),
220        open_university_product_id: None,
221    };
222
223    let mut unit: MockCourseUnit = store
224        .get_json(generation, EntityHash::CourseUnits, &course_code)
225        .await?
226        .unwrap_or_else(|| MockCourseUnit {
227            course_unit_id: ids::course_unit_id(&course_code),
228            name: LocalizedName {
229                fi: course_code.clone(),
230                sv: course_code.clone(),
231                en: course_code.clone(),
232            },
233            realisations: Vec::new(),
234            behaviour: Default::default(),
235            owner_course_slug: args.owner.as_ref().and_then(|owner| owner.course.clone()),
236            course_code: course_code.clone(),
237        });
238    unit.behaviour.import_allowed = true;
239    unit.realisations
240        .retain(|existing| existing.id != realisation.id);
241    unit.realisations.push(realisation.clone());
242    store
243        .upsert_json(
244            generation,
245            EntityHash::CourseUnits,
246            &BTreeMap::from([(course_code, unit)]),
247        )
248        .await?;
249    store.reindex(generation).await?;
250    Ok(realisation)
251}

253async fn put_person(
254    store: &MockSuotarStore,
255    generation: &str,
256    args: &ScenarioArgs,
257    ripeness: Option<Ripeness>,
258) -> Result<String, CommandError> {
259    let student_number = match &args.student_number {
260        Some(student_number) => student_number.clone(),
261        None => {
262            let sequence = store.next_person_seq(generation).await?;
263            format!("99{sequence:07}")
264        }
265    };
266    let person = MockPerson {
267        person_id: ids::person_id(&student_number),
268        first_names: args
269            .first_names
270            .clone()
271            .unwrap_or_else(|| "Zzyzx".to_string()),
272        last_name: args
273            .last_name
274            .clone()
275            .unwrap_or_else(|| "Scenario".to_string()),
276        primary_email: args
277            .primary_email
278            .clone()
279            .unwrap_or_else(|| format!("zzyzx.scenario.{student_number}@helsinki.example")),
280        secondary_email: args.secondary_email.clone(),
281        behaviour: PersonBehaviour {
282            ripeness,
283            duplicate_detection: None,
284        },
285        owner_user_email: args.owner.as_ref().and_then(|owner| owner.user.clone()),
286        student_number: student_number.clone(),
287    };
288    store
289        .upsert_json(
290            generation,
291            EntityHash::Persons,
292            &BTreeMap::from([(student_number.clone(), person)]),
293        )
294        .await?;
295    store.reindex(generation).await?;
296    Ok(student_number)
297}

299async fn put_enrolment(
300    store: &MockSuotarStore,
301    generation: &str,
302    student_number: &str,
303    course_code: &str,
304    realisation: &MockRealisation,
305) -> Result<String, CommandError> {
306    let now = Utc::now();
307    let enrolment_id = ids::enrolment_id(student_number, realisation.kind);
308    let enrolment = MockEnrolment {
309        id: enrolment_id.clone(),
310        student_number: student_number.to_string(),
311        course_code: course_code.to_string(),
312        realisation_id: realisation.id.clone(),
313        state: EnrolmentState::Enrolled,
314        study_right_id: ids::study_right_id(student_number, realisation.kind),
315        study_right_validity_period: DatePeriod {
316            start_date: (now - Duration::days(365)).date_naive(),
317            end_date: (now + Duration::days(365)).date_naive(),
318        },
319        enrolment_date_time: now,
320    };
321    store
322        .upsert_json(
323            generation,
324            EntityHash::Enrolments,
325            &BTreeMap::from([(enrolment_id.clone(), enrolment)]),
326        )
327        .await?;
328    store.reindex(generation).await?;
329    Ok(enrolment_id)
330}

332async fn arm(
333    store: &MockSuotarStore,
334    generation: &str,
335    id: &str,
336    when: Vec<Predicate>,
337    then: Effect,
338    lifetime: Lifetime,
339) -> Result<(), CommandError> {
340    arm_fault(
341        store,
342        generation,
343        FaultSpec {
344            id: id.to_string(),
345            when: WhenSpec::Predicates(when),
346            then,
347            lifetime,
348            proves_double_submission: false,
349        },
350    )
351    .await?;
352    Ok(())
353}

355fn course_code(args: &ScenarioArgs) -> Result<String, CommandError> {
356    args.course_code
357        .clone()
358        .ok_or_else(|| CommandError::new("missingArgument", "This scenario needs a courseCode."))
359}

361fn string_field(value: &serde_json::Value, field: &str) -> Result<String, CommandError> {
362    value
363        .get(field)
364        .and_then(|value| value.as_str())
365        .map(str::to_string)
366        .ok_or_else(|| CommandError::new("internalError", format!("scenario lost its {field}")))
367}

369fn merge(target: &mut serde_json::Value, extra: serde_json::Value) {
370    if let (Some(target), Some(extra)) = (target.as_object_mut(), extra.as_object()) {
371        for (key, value) in extra {
372            target.insert(key.clone(), value.clone());
373        }
374    }
