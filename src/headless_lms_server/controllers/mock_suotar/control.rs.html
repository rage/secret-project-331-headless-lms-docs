1//! The mock Suotar control surface: test-only routes that are not part of Suotar's API.
2//!
3//! A tick runs one iteration of one phase synchronously, because the real loops are long-running
4//! intervals in their own Deployments that a Playwright spec cannot wait out. Gated on
5//! `test_mode && test_suotar`, so no token; the client is hand-written in
6//! `system-tests/src/utils/suotarControl.ts`.

39#[derive(Debug, Serialize, Deserialize, PartialEq)]
40#[serde(rename_all = "camelCase", tag = "status")]
41pub enum PhaseTickResult {
42    #[serde(rename_all = "camelCase")]
43    Ran {
44        phase: String,
45        items_processed: i32,
46        items_failed: i32,
47        /// Set when the iteration ran but failed; already scrubbed by the phase.
48        error: Option<String>,
49    },
50    /// No implementation is registered for this phase in `run_phase_once` yet.
51    PhaseNotImplemented { phase: String },
52    /// The phase is paused, or the circuit breaker for this scope is open. Not a failure.
53    Skipped { phase: String, reason: String },
54    /// The scope names something this phase's claim query cannot narrow on.
55    ScopeNotSupported { phase: String },
56    #[serde(rename_all = "camelCase")]
57    UnknownPhase {
58        phase: Option<String>,
59        known_phases: Vec<String>,
60    },
61}
62
63impl PhaseTickResult {
64    fn of(phase: CreditRegistrationPhase, tick: PhaseTick) -> Self {
65        match tick {
66            PhaseTick::Ran(outcome) => Self::Ran {
67                phase: phase.as_str().to_string(),
68                items_processed: outcome.items_processed,
69                items_failed: outcome.items_failed,
70                error: outcome.error,
71            },
72            PhaseTick::Skipped(reason) => Self::Skipped {
73                phase: phase.as_str().to_string(),
74                reason: match reason {
75                    PhaseSkipReason::Paused => "paused".to_string(),
76                    PhaseSkipReason::CircuitBreakerOpen => "circuitBreakerOpen".to_string(),
77                },
78            },
79            PhaseTick::ScopeNotSupported => Self::ScopeNotSupported {
80                phase: phase.as_str().to_string(),
81            },
82            PhaseTick::NotImplemented => Self::PhaseNotImplemented {
83                phase: phase.as_str().to_string(),
84            },
85        }
86    }
87}

96/// Runs one iteration of one pipeline phase: 200 when it ran, 501 `phaseNotImplemented`, 400
97/// `unknownPhase`. An optional scope narrows the rows the iteration may claim; absent is unscoped,
98/// which is what production does.
99async fn run_tick(
100    app_conf: web::Data<ApplicationConfiguration>,
101    pool: web::Data<PgPool>,
102    suotar_client: web::Data<SuotarClient>,
103    query: web::Query<RunTickQuery>,
104) -> ControllerResult<HttpResponse> {
105    super::assert_enabled(&app_conf);
106    let token = skip_authorize();
107
108    let Some(phase) = query
109        .phase
110        .as_deref()
111        .and_then(CreditRegistrationPhase::from_phase_name)
112    else {
113        return token.authorized_ok(HttpResponse::BadRequest().json(
114            PhaseTickResult::UnknownPhase {
115                phase: query.phase.clone(),
116                known_phases: known_phase_names(),
117            },
118        ));
119    };
120
121    let scope = match resolve_scope(&pool, &query).await? {
122        Ok(scope) => scope,
123        // Never a silent fall-through to sweeping everything.
124        Err(unresolved) => {
125            return token.authorized_ok(HttpResponse::BadRequest().json(unresolved));
126        }
127    };
128
129    let ctx = tick_context(&app_conf, &pool, &suotar_client);
130    let result = PhaseTickResult::of(phase, run_phase_once(&ctx, phase, &scope).await?);
131    token.authorized_ok(match &result {
132        PhaseTickResult::Ran { .. } | PhaseTickResult::Skipped { .. } => {
133            HttpResponse::Ok().json(&result)
134        }
135        PhaseTickResult::ScopeNotSupported { .. } => HttpResponse::BadRequest().json(&result),
136        _ => HttpResponse::NotImplemented().json(&result),
137    })
138}

140/// The combined form of `run-tick`, walking the sequence in pipeline order. Always 200; each phase
141/// reports its own status. Takes no scope on purpose: a suite that only ever ticks scoped never
142/// exercises the sweep-everything behaviour production has.
143async fn run_registrar_tick(
144    app_conf: web::Data<ApplicationConfiguration>,
145    pool: web::Data<PgPool>,
146    suotar_client: web::Data<SuotarClient>,
147) -> ControllerResult<HttpResponse> {
148    super::assert_enabled(&app_conf);
149    let token = skip_authorize();
150
151    let scope = PhaseScope::default();
152    let ctx = tick_context(&app_conf, &pool, &suotar_client);
153    let mut phases = Vec::new();
154    for phase in CreditRegistrationPhase::REGISTRAR_TICK_SEQUENCE {
155        phases.push(PhaseTickResult::of(
156            phase,
157            run_phase_once(&ctx, phase, &scope).await?,
158        ));
159    }
160    token.authorized_ok(HttpResponse::Ok().json(RegistrarTickResult { phases }))
161}

163/// Attributed to the tick rather than to a worker, so the audit log says which traffic a test made.
164fn tick_context<'a>(
165    app_conf: &'a ApplicationConfiguration,
166    pool: &'a PgPool,
167    suotar_client: &'a SuotarClient,
168) -> PhaseContext<'a> {
169    PhaseContext {
170        pool,
171        suotar_client,
172        test_mode: app_conf.test_mode,
173        caller: "run-tick",
174        base_url: &app_conf.base_url,
175    }
176}

178/// The outer error is a real failure; the inner one is a scope half that names nothing.
179async fn resolve_scope(
180    pool: &PgPool,
181    query: &RunTickQuery,
182) -> anyhow::Result<Result<PhaseScope, UnresolvedScope>> {
183    let mut scope = PhaseScope {
184        course_id: query.course_id,
185        user_id: query.user_id,
186        credit_registration_ids: Vec::new(),
187    };
188    if let Some(slug) = &query.course_slug {
189        let mut conn = pool.acquire().await?;
190        let found = models::courses::get_active_course_id_by_slug(&mut conn, slug).await?;
191        match found {
192            Some(id) => scope.course_id = Some(id),
193            None => {
194                return Ok(Err(UnresolvedScope {
195                    status: "unresolvedScope".to_string(),
196                    half: "courseSlug".to_string(),
197                    value: slug.clone(),
198                }));
199            }
200        }
201    }
202    if let Some(email) = &query.user_email {
203        let mut conn = pool.acquire().await?;
204        let found =
205            models::user_details::get_active_user_id_by_email_case_insensitive(&mut conn, email)
206                .await?;
207        match found {
208            Some(id) => scope.user_id = Some(id),
209            None => {
210                return Ok(Err(UnresolvedScope {
211                    status: "unresolvedScope".to_string(),
212                    half: "userEmail".to_string(),
213                    value: email.clone(),
214                }));
215            }
216        }
217    }
218    if let Some(raw) = &query.credit_registration_ids {
219        for part in raw.split(',').filter(|part| !part.trim().is_empty()) {
220            match Uuid::parse_str(part.trim()) {
221                Ok(id) => scope.credit_registration_ids.push(id),
222                Err(_) => {
223                    return Ok(Err(UnresolvedScope {
224                        status: "unresolvedScope".to_string(),
225                        half: "creditRegistrationIds".to_string(),
226                        value: part.trim().to_string(),
227                    }));
228                }
229            }
230        }
231    }
232    Ok(Ok(scope))
233}

235fn known_phase_names() -> Vec<String> {
236    CreditRegistrationPhase::ALL
237        .iter()
238        .map(|phase| phase.as_str().to_string())
239        .collect()
240}

285    /// Registers the real controller tree so the test sees the same gate production does. The pool is
286    /// never connected, so only phases that answer before they would need one can be driven here.
287    async fn call_run_tick(
288        test_suotar: bool,
289        query: &str,
290    ) -> actix_web::dev::ServiceResponse<actix_web::body::BoxBody> {
291        let app_conf = Data::new(app_conf(test_suotar));
292        let pool = Data::new(
293            PgPool::connect_lazy("postgres://headless-lms@localhost:54328/headless_lms_dev")
294                .expect("a lazy pool only parses the url"),
295        );
296        let service = test::init_service(
297            App::new()
298                .app_data(pool)
299                .app_data(Data::new(SuotarClient::mock_for_test()))
300                .app_data(app_conf.clone())
301                .service(
302                    web::scope("/api/v0")
303                        .configure(|cfg| configure_controllers(cfg, app_conf.clone())),
304                ),
305        )
306        .await;
307        let req = test::TestRequest::post()
308            .uri(&format!("/api/v0/mock-suotar/control/run-tick{query}"))
309            .to_request();
310        test::call_service(&service, req).await
311    }
