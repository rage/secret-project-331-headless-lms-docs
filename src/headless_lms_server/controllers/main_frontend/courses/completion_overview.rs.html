18/// How students complete a course and get its credits, for the course Overview tab.
19#[derive(Debug, Serialize, ToSchema)]
20pub struct CourseCompletionOverview {
21    /// In module order.
22    pub modules: Vec<CourseModuleCompletionOverview>,
23    /// The exams linked to the course, which every module passed by an exam on this platform uses.
24    pub exams: CourseExamsOverview,
25    /// Whether the user can view the course's credit registrations; without it every module's
26    /// `not_yet_registered_count` is 0 and `credit_registration_pause_reason` is `None`.
27    pub can_view_registrations: bool,
28}
29
30/// One module's completion and credit settings with the facts that tell whether they work.
31#[derive(Debug, Serialize, ToSchema)]
32pub struct CourseModuleCompletionOverview {
33    pub course_module_id: Uuid,
34    /// `None` for the default module.
35    pub name: Option<String>,
36    pub order_number: i32,
37    pub completion_method: CompletionMethod,
38    /// By requirements, the points needed to pass; by an exam, to take the exam.
39    pub points_threshold: Option<i32>,
40    /// By requirements, the exercises attempted needed to pass; by an exam, to take the exam.
41    pub exercises_attempted_threshold: Option<i32>,
42    pub exam_requirement: ExamRequirement,
43    pub credit_registration_route: CreditRegistrationRoute,
44    pub uh_course_code: Option<String>,
45    pub ects_credits: Option<f32>,
46    pub certification_enabled: bool,
47    pub has_registration_link: bool,
48    pub has_completions: bool,
49    /// Credit registrations neither in Sisu yet nor given up on.
50    pub not_yet_registered_count: i64,
51    /// `None` until Sisu has been asked about the course code.
52    pub course_code_allowed: Option<bool>,
53    pub credit_registration_paused_at: Option<DateTime<Utc>>,
54    pub credit_registration_pause_reason: Option<String>,
55}
56
57/// When the course's linked exams are held and whether they can pass anyone automatically.
58#[derive(Debug, Serialize, ToSchema)]
59pub struct CourseExamsOverview {
60    pub count: usize,
61    /// The latest end of an exam that has ended.
62    pub last_ended_at: Option<DateTime<Utc>>,
63    /// The window of the exam that ends next among those not ended yet.
64    pub next_starts_at: Option<DateTime<Utc>>,
65    pub next_ends_at: Option<DateTime<Utc>>,
66    /// Whether at least one exam completes the students who reach its pass mark; otherwise every
67    /// exam is graded manually or has no pass mark, and the teacher adds the completions.
68    pub any_has_pass_mark: bool,
69}

71/**
72GET `/api/v0/main-frontend/courses/{course_id}/completion-overview`
73*/
74#[utoipa::path(
75    get,
76    path = "",
77    operation_id = "getCourseCompletionOverview",
78    tag = "courses",
79    params(("course_id" = Uuid, Path, description = "Course id")),
80    responses(
81        (status = 200, description = "How students complete the course and get its credits", body = CourseCompletionOverview)
82    )
83)]
84#[instrument(skip(pool))]
85async fn get_completion_overview(
86    course_id: web::Path<Uuid>,
87    pool: web::Data<PgPool>,
88    user: AuthUser,
89) -> ControllerResult<web::Json<CourseCompletionOverview>> {
90    let mut conn = pool.acquire().await?;
91    let user_roles = models::roles::get_roles(&mut conn, user.id).await?;
92    let token = authorize_with_fetched_list_of_roles(
93        &mut conn,
94        Act::ViewInternalCourseStructure,
95        Res::Course(*course_id),
96        &user_roles,
97    )
98    .await?;
99    let can_view_registrations = is_permitted(
100        &mut conn,
101        Act::ViewAndManageCreditRegistrations,
102        Res::Course(*course_id),
103        &user_roles,
104    )
105    .await?;
106
107    let setups =
108        models::course_modules::get_completion_setups_by_course_id(&mut conn, *course_id).await?;
109    let mut not_yet_registered_counts: HashMap<Uuid, i64> = HashMap::new();
110    if can_view_registrations {
111        let counts = models::credit_registrations::count_by_module_and_state_for_course(
112            &mut conn, *course_id, None,
113        )
114        .await?;
115        for group in &counts {
116            if !matches!(
117                stage_of(group),
118                StudentFacingCreditRegistrationStatus::Registered
119                    | StudentFacingCreditRegistrationStatus::NotRegistering
120            ) {
121                *not_yet_registered_counts
122                    .entry(group.course_module_id)
123                    .or_default() += group.count;
124            }
125        }
126    }
127    let exams = models::exams::get_summaries_by_course_id(&mut conn, *course_id).await?;
128
129    let modules = setups
130        .into_iter()
131        .map(|setup| {
132            let has_thresholds = matches!(
133                setup.completion_method,
134                CompletionMethod::ByRequirements | CompletionMethod::ByExam
135            );
136            CourseModuleCompletionOverview {
137                course_module_id: setup.id,
138                order_number: setup.order_number,
139                completion_method: setup.completion_method,
140                points_threshold: setup
141                    .automatic_completion_number_of_points_treshold
142                    .filter(|_| has_thresholds),
143                exercises_attempted_threshold: setup
144                    .automatic_completion_number_of_exercises_attempted_treshold
145                    .filter(|_| has_thresholds),
146                exam_requirement: setup.exam_requirement,
147                credit_registration_route: setup.credit_registration_route,
148                certification_enabled: setup.certification_enabled,
149                has_registration_link: is_non_blank(
150                    setup.completion_registration_link_override.as_deref(),
151                ),
152                has_completions: setup.has_completions,
153                not_yet_registered_count: not_yet_registered_counts
154                    .get(&setup.id)
155                    .copied()
156                    .unwrap_or(0),
157                course_code_allowed: setup.credit_registration_course_code_allowed,
158                credit_registration_paused_at: setup.credit_registration_paused_at,
159                credit_registration_pause_reason: setup
160                    .credit_registration_pause_reason
161                    .filter(|_| can_view_registrations),
162                name: setup.name,
163                uh_course_code: setup.uh_course_code,
164                ects_credits: setup.ects_credits,
165            }
166        })
167        .collect();
168
169    let now = Utc::now();
170    let last_ended_at = exams
171        .iter()
172        .filter_map(|exam| exam.ends_at)
173        .filter(|ends_at| *ends_at <= now)
174        .max();
175    let next = exams
176        .iter()
177        .filter(|exam| exam.ends_at.is_some_and(|ends_at| ends_at > now))
178        .min_by_key(|exam| exam.ends_at);
179
180    token.authorized_ok(web::Json(CourseCompletionOverview {
181        modules,
182        exams: CourseExamsOverview {
183            count: exams.len(),
184            last_ended_at,
185            next_starts_at: next.and_then(|exam| exam.starts_at),
186            next_ends_at: next.and_then(|exam| exam.ends_at),
187            any_has_pass_mark: exams.iter().any(|exam| exam.pass_mark().is_some()),
188        },
189        can_view_registrations,
190    }))
191}

194    cfg.route("", web::get().to(get_completion_overview));
