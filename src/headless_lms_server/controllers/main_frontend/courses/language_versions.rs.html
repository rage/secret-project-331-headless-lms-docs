83/// Sets `can_edit` on each course for the user with `roles`. Returns whether the user can edit
84/// all of them, which fixes across the language versions need.
85pub(crate) async fn fill_can_edit<'a>(
86    conn: &mut PgConnection,
87    roles: &[Role],
88    courses: impl IntoIterator<Item = &'a mut LanguageVersionCourse>,
89) -> Result<bool, ControllerError> {
90    let mut can_edit_all = true;
91    for course in courses {
92        course.can_edit = is_permitted(conn, Act::Edit, Res::Course(course.id), roles).await?;
93        can_edit_all &= course.can_edit;
94    }
95    Ok(can_edit_all)
96}

98/// The group's join candidates the user can edit, with `can_edit` set.
99async fn get_editable_join_candidates(
100    conn: &mut PgConnection,
101    roles: &[Role],
102    snapshot: &LanguageGroupSnapshot,
103    search: Option<&str>,
104) -> Result<Vec<JoinCandidate>, ControllerError> {
105    let mut candidates = join_candidates::get_join_candidates(conn, snapshot, search).await?;
106    fill_can_edit(conn, roles, candidates.iter_mut().map(|c| &mut c.course)).await?;
107    candidates.retain(|c| c.course.can_edit);
108    Ok(candidates)
109}

111/// Requires `Edit` on every live course in `course_id`'s language group and on
112/// `additional_course_id`: fixes change links or student data across all of them. Returns the
113/// group's id.
114pub(crate) async fn authorize_edit_on_language_versions(
115    conn: &mut PgConnection,
116    roles: &[Role],
117    course_id: Uuid,
118    additional_course_id: Option<Uuid>,
119) -> Result<(AuthorizationToken, Uuid), ControllerError> {
120    let token =
121        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(course_id), roles)
122            .await?;
123    let course_language_group_id =
124        models::courses::get_course_language_group_id(conn, course_id).await?;
125    let language_versions =
126        models::courses::get_by_course_language_group_ids(conn, &[course_language_group_id])
127            .await?;
128    for other_course_id in language_versions
129        .iter()
130        .map(|c| c.id)
131        .chain(additional_course_id)
132        .filter(|id| *id != course_id)
133    {
134        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(other_course_id), roles)
135            .await?;
136    }
137    Ok((token, course_language_group_id))
138}

147/// Authorizes a read: `Edit` on the course is enough, as `can_edit` per course tells which fixes
148/// the user may apply.
149async fn authorize_read(
150    conn: &mut PgConnection,
151    user_id: Uuid,
152    course_id: Uuid,
153) -> Result<LanguageVersionsRequest, ControllerError> {
154    let roles = fetch_user_roles(conn, Some(user_id)).await?;
155    let token =
156        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(course_id), &roles)
157            .await?;
158    let course_language_group_id =
159        models::courses::get_course_language_group_id(conn, course_id).await?;
160    Ok(LanguageVersionsRequest {
161        token,
162        course_language_group_id,
163        roles,
164    })
165}

184/// Loads the group with `can_edit` filled in for the user with `roles`; returns whether the user
185/// can edit every course.
186pub(crate) async fn load_snapshot(
187    conn: &mut PgConnection,
188    roles: &[Role],
189    course_language_group_id: Uuid,
190    candidate_course_ids: &[Uuid],
191    student_counts: StudentCounts,
192) -> Result<(LanguageGroupSnapshot, bool), ControllerError> {
193    let mut snapshot = snapshot::load_with_candidates(
194        conn,
195        course_language_group_id,
196        candidate_course_ids,
197        student_counts,
198    )
199    .await?;
200    let can_edit_all_courses = fill_can_edit(conn, roles, snapshot.courses.iter_mut()).await?;
201    Ok((snapshot, can_edit_all_courses))
202}

237/**
238GET `/api/v0/main-frontend/courses/:id/language-versions/report` - Checks the language versions for problems.
239
240Needs `Edit` on this course only; `can_edit` per course tells which fixes the user may apply.
241*/
242#[utoipa::path(
243    get,
244    path = "/report",
245    operation_id = "getLanguageVersionReport",
246    tag = "course-language-versions",
247    params(("course_id" = Uuid, Path, description = "Course id")),
248    responses((status = 200, description = "Problems found in the language versions", body = LanguageVersionReport))
249)]
250#[instrument(skip(pool))]
251async fn get_report(
252    pool: web::Data<PgPool>,
253    course_id: web::Path<Uuid>,
254    user: AuthUser,
255) -> ControllerResult<web::Json<LanguageVersionReport>> {
256    let mut conn = pool.acquire().await?;
257    let LanguageVersionsRequest {
258        token,
259        course_language_group_id,
260        roles,
261    } = authorize_read(&mut conn, user.id, *course_id).await?;
262    let (snapshot, can_edit_all_courses) = load_snapshot(
263        &mut conn,
264        &roles,
265        course_language_group_id,
266        &[],
267        StudentCounts::Include,
268    )
269    .await?;
270    let lineage = suggestions::get_copy_lineage(&mut conn, &snapshot).await?;
271    let dismissals =
272        dismissals::get_by_course_language_group_id(&mut conn, course_language_group_id).await?;
273    let join_candidates = get_editable_join_candidates(&mut conn, &roles, &snapshot, None).await?;
274    let problems = detection::detect_problems(&snapshot, &lineage, &dismissals, &join_candidates);
275
276    let courses = snapshot.courses;
277    let course_ids: Vec<Uuid> = courses.iter().map(|c| c.id).collect();
278    let with_open_changesets: HashSet<Uuid> =
279        agent_changesets::get_course_ids_with_open_changesets(&mut conn, &course_ids)
280            .await?
281            .into_iter()
282            .collect();
283    token.authorized_ok(web::Json(LanguageVersionReport {
284        course_language_group_id,
285        can_edit_all_courses,
286        courses_with_open_agent_changesets: course_ids
287            .into_iter()
288            .filter(|id| with_open_changesets.contains(id))
289            .collect(),
290        courses,
291        problems,
292        checked_at: Utc::now(),
293    }))
294}

336    get,
337    path = "/suggestions",
338    operation_id = "getLanguageVersionSuggestions",

340    params(("course_id" = Uuid, Path, description = "Course id"), SuggestionsQuery),
341    responses((status = 200, description = "Suggested links", body = LinkSuggestions))
342)]
343#[instrument(skip(pool))]
344async fn get_suggestions(
345    pool: web::Data<PgPool>,
346    course_id: web::Path<Uuid>,
347    query: MultiQuery<SuggestionsQuery>,
348    user: AuthUser,
349) -> ControllerResult<web::Json<LinkSuggestions>> {
350    let mut conn = pool.acquire().await?;
351    let LanguageVersionsRequest {
352        token,
353        course_language_group_id,

357    for candidate_course_id in &candidate_course_ids {
358        authorize_with_fetched_list_of_roles(

362            &roles,
363        )
364        .await?;
365    }
366    let (snapshot, _) = load_snapshot(
367        &mut conn,
368        &roles,
369        course_language_group_id,
370        &candidate_course_ids,
371        StudentCounts::Omit,
372    )
373    .await?;
374    let lineage = suggestions::get_copy_lineage(&mut conn, &snapshot).await?;
375    let res = suggestions::suggest_links(&snapshot, &lineage, *course_id, query.chapter_number)?;
376    token.authorized_ok(web::Json(res))
