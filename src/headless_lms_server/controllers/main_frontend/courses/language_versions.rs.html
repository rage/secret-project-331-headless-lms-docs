89/// Sets `can_edit` on each course for the user with `roles`. Returns whether the user can edit
90/// all of them, which fixes across the language versions need.
91pub(crate) async fn fill_can_edit<'a>(
92    conn: &mut PgConnection,
93    roles: &[Role],
94    courses: impl IntoIterator<Item = &'a mut LanguageVersionCourse>,
95) -> Result<bool, ControllerError> {
96    let mut can_edit_all = true;
97    for course in courses {
98        course.can_edit = is_permitted(conn, Act::Edit, Res::Course(course.id), roles).await?;
99        can_edit_all &= course.can_edit;
100    }
101    Ok(can_edit_all)
102}

104/// The group's join candidates the user can edit, with `can_edit` set.
105async fn get_editable_join_candidates(
106    conn: &mut PgConnection,
107    roles: &[Role],
108    snapshot: &LanguageGroupSnapshot,
109    search: Option<&str>,
110) -> Result<Vec<JoinCandidate>, ControllerError> {
111    let mut candidates = join_candidates::get_join_candidates(conn, snapshot, search).await?;
112    fill_can_edit(conn, roles, candidates.iter_mut().map(|c| &mut c.course)).await?;
113    candidates.retain(|c| c.course.can_edit);
114    Ok(candidates)
115}

117/// Requires `Edit` on every live course in `course_id`'s language group and on
118/// `additional_course_id`: fixes change links or student data across all of them. Returns the
119/// group's id.
120pub(crate) async fn authorize_edit_on_language_versions(
121    conn: &mut PgConnection,
122    roles: &[Role],
123    course_id: Uuid,
124    additional_course_id: Option<Uuid>,
125) -> Result<(AuthorizationToken, Uuid), ControllerError> {
126    let token =
127        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(course_id), roles)
128            .await?;
129    let course_language_group_id =
130        models::courses::get_course_language_group_id(conn, course_id).await?;
131    let language_versions =
132        models::courses::get_by_course_language_group_ids(conn, &[course_language_group_id])
133            .await?;
134    for other_course_id in language_versions
135        .iter()
136        .map(|c| c.id)
137        .chain(additional_course_id)
138        .filter(|id| *id != course_id)
139    {
140        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(other_course_id), roles)
141            .await?;
142    }
143    Ok((token, course_language_group_id))
144}

153/// Authorizes a read: `Edit` on the course is enough, as `can_edit` per course tells which fixes
154/// the user may apply.
155async fn authorize_read(
156    conn: &mut PgConnection,
157    user_id: Uuid,
158    course_id: Uuid,
159) -> Result<LanguageVersionsRequest, ControllerError> {
160    let roles = fetch_user_roles(conn, Some(user_id)).await?;

162        authorize_with_fetched_list_of_roles(conn, Act::Edit, Res::Course(course_id), &roles)

164    let course_language_group_id =
165        models::courses::get_course_language_group_id(conn, course_id).await?;
166    Ok(LanguageVersionsRequest {
167        token,
168        course_language_group_id,
169        roles,
170    })
171}

190/// Loads the group with `can_edit` filled in for the user with `roles`; returns whether the user
191/// can edit every course.
192pub(crate) async fn load_snapshot(
193    conn: &mut PgConnection,
194    roles: &[Role],
195    course_language_group_id: Uuid,
196    candidate_course_ids: &[Uuid],
197    student_counts: StudentCounts,
198) -> Result<(LanguageGroupSnapshot, bool), ControllerError> {
199    let mut snapshot = snapshot::load_with_candidates(
200        conn,
201        course_language_group_id,
202        candidate_course_ids,
203        student_counts,
204    )
205    .await?;
206    let can_edit_all_courses = fill_can_edit(conn, roles, snapshot.courses.iter_mut()).await?;
207    Ok((snapshot, can_edit_all_courses))
208}

244GET `/api/v0/main-frontend/courses/:id/language-versions/report` - Checks the language versions for problems.
245
246Needs `Edit` on this course only; `can_edit` per course tells which fixes the user may apply.
247*/
248#[utoipa::path(
249    get,
250    path = "/report",
251    operation_id = "getLanguageVersionReport",
252    tag = "course-language-versions",
253    params(("course_id" = Uuid, Path, description = "Course id")),
254    responses((status = 200, description = "Problems found in the language versions", body = LanguageVersionReport))
255)]
256#[instrument(skip(pool))]
257async fn get_report(
258    pool: web::Data<PgPool>,
259    course_id: web::Path<Uuid>,
260    user: AuthUser,
261) -> ControllerResult<web::Json<LanguageVersionReport>> {
262    let mut conn = pool.acquire().await?;
263    let LanguageVersionsRequest {
264        token,
265        course_language_group_id,
266        roles,
267    } = authorize_read(&mut conn, user.id, *course_id).await?;
268    let (snapshot, can_edit_all_courses) = load_snapshot(
269        &mut conn,
270        &roles,
271        course_language_group_id,
272        &[],
273        StudentCounts::Include,
274    )
275    .await?;
276    let lineage = suggestions::get_copy_lineage(&mut conn, &snapshot).await?;
277    let dismissals =
278        dismissals::get_by_course_language_group_id(&mut conn, course_language_group_id).await?;
279    let join_candidates = get_editable_join_candidates(&mut conn, &roles, &snapshot, None).await?;
280    let problems = detection::detect_problems(&snapshot, &lineage, &dismissals, &join_candidates);
281
282    let courses = snapshot.courses;
283    let course_ids: Vec<Uuid> = courses.iter().map(|c| c.id).collect();
284    let with_open_changesets: HashSet<Uuid> =
285        agent_changesets::get_course_ids_with_open_changesets(&mut conn, &course_ids)
286            .await?
287            .into_iter()
288            .collect();
289    token.authorized_ok(web::Json(LanguageVersionReport {
290        course_language_group_id,
291        can_edit_all_courses,
292        courses_with_open_agent_changesets: course_ids
293            .into_iter()
294            .filter(|id| with_open_changesets.contains(id))
295            .collect(),
296        courses,
297        problems,
298        checked_at: Utc::now(),
299    }))
300}

336/// The snapshot and copy history that suggestions for `course_id` are made from, after
337/// authorizing the read and every candidate course.
338async fn load_for_suggestions(
339    conn: &mut PgConnection,
340    user: &AuthUser,
341    course_id: Uuid,
342    query: &SuggestionsQuery,
343) -> Result<
344    (
345        AuthorizationToken,
346        LanguageGroupSnapshot,
347        suggestions::CopyLineage,
348    ),
349    ControllerError,
350> {
351    let LanguageVersionsRequest {
352        token,
353        course_language_group_id,
354        roles,
355    } = authorize_read(conn, user.id, course_id).await?;
356    let candidate_course_ids = query.candidate_course_ids.clone().unwrap_or_default();
357    for candidate_course_id in &candidate_course_ids {
358        authorize_with_fetched_list_of_roles(
359            conn,
360            Act::Edit,
361            Res::Course(*candidate_course_id),
362            &roles,
363        )
364        .await?;
365    }
366    let (snapshot, _) = load_snapshot(
367        conn,
368        &roles,
369        course_language_group_id,
370        &candidate_course_ids,
371        StudentCounts::Omit,
372    )
373    .await?;
374    let lineage = suggestions::get_copy_lineage(conn, &snapshot).await?;
375    Ok((token, snapshot, lineage))
376}

407/**
408GET `/api/v0/main-frontend/language-version-ai-suggestions/:id` - Like the language versions' `/suggestions`, with AI picks from titles, names and content.
409
410Outside `/courses` so that the ingress can give it a longer timeout: it waits for the AI. When the
411AI is unavailable, the heuristic suggestions come back with `ai_status` saying so.
412*/
413#[utoipa::path(
414    get,
415    path = "/{course_id}",
416    operation_id = "getLanguageVersionAiSuggestions",
417    tag = "course-language-versions",
418    params(("course_id" = Uuid, Path, description = "Course id"), SuggestionsQuery),
419    responses((status = 200, description = "Suggested links", body = LinkSuggestions))
420)]
421#[instrument(skip(pool, app_conf))]
422async fn get_ai_suggestions(
423    pool: web::Data<PgPool>,
424    app_conf: web::Data<ApplicationConfiguration>,
425    course_id: web::Path<Uuid>,
426    query: MultiQuery<SuggestionsQuery>,
427    user: AuthUser,
428) -> ControllerResult<web::Json<LinkSuggestions>> {
429    let mut conn = pool.acquire().await?;
430    let (token, snapshot, lineage) =
431        load_for_suggestions(&mut conn, &user, *course_id, &query).await?;
432    let task_lm = application_task_default_language_models::get_for_task(
433        &mut conn,
434        ApplicationTask::LanguageVersionMatching,
435    )
436    .await
437    .inspect_err(|e| warn!("No model for language version matching: {e}"))
438    .ok();
439    let mut tasks = ai_matching::ai_matching_tasks(&snapshot, &lineage, *course_id);
440    if let Some(chapter_number) = query.chapter_number {
441        tasks.retain(|t| t.chapter_number == Some(chapter_number));
442    }
443    let (picks, ai_status) = match task_lm {
444        None => (None, AiSuggestionStatus::Unavailable),
445        Some(_) if tasks.is_empty() => (None, AiSuggestionStatus::Used),
446        Some(task_lm) => {
447            ai_matching::fill_excerpts(&mut conn, &mut tasks).await?;
448            // The AI can take minutes; don't hold a connection meanwhile.
449            drop(conn);
450            let (picks, failed_count) =
451                language_version_matching::suggest_counterparts(&app_conf, &task_lm, &tasks).await;
452            let status = if failed_count == 0 {
453                AiSuggestionStatus::Used
454            } else if failed_count < tasks.len() {
455                AiSuggestionStatus::PartlyUnavailable
456            } else {
457                AiSuggestionStatus::Unavailable
458            };
459            (Some(picks), status)
460        }
461    };
462    let mut res = suggestions::suggest_links(
463        &snapshot,
464        &lineage,
465        *course_id,
466        query.chapter_number,
467        picks.as_ref(),
468    )?;
469    res.ai_status = ai_status;
470    token.authorized_ok(web::Json(res))
471}

872/**

890        .route("/dismissals", web::post().to(post_dismissal))
891        .route("/dismissals", web::delete().to(delete_dismissal))
892        .route("/join-candidates", web::get().to(get_join_candidates))
893        .route(
894            "/join-preview/{joined_course_id}",
895            web::get().to(get_join_preview),
896        )
897        .route("/join", web::post().to(post_join))
898        .route(
899            "/split-preview/{split_course_id}",
900            web::get().to(get_split_preview),
901        )
902        .route("/split", web::post().to(post_split))
903        .route("/counterparts", web::post().to(post_counterpart))
