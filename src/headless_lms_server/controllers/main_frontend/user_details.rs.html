57pub async fn get_user_details(
58    user: AuthUser,
59    pool: web::Data<PgPool>,
60    path: web::Path<(Uuid, Uuid)>,
61) -> ControllerResult<web::Json<UserDetail>> {
62    let (course_id, user_id) = path.into_inner();
63    let mut conn = pool.acquire().await?;
64
65    let token = authorize(
66        &mut conn,
67        Act::ViewUserProgressOrDetails,
68        Some(user.id),
69        Res::Course(course_id),
70    )
71    .await?;
72    let res =
73        models::user_details::get_user_details_by_user_id_for_course(&mut conn, user_id, course_id)
74            .await?;
75    token.authorized_ok(web::Json(res))
76}

78/**
79POST `/api/v0/main-frontend/user-details/user-by-courses` - Find user details by user id with multi-course permission check
80Returns user details if the user has permission to view user details through any of the specified courses
81*/
82#[instrument(skip(pool))]
83#[utoipa::path(
84    post,
85    path = "/user-by-courses",
86    operation_id = "getUserDetailsByCourses",
87    tag = "user-details",
88    request_body = UserDetailsRequest,
89    responses(
90        (status = 200, description = "User details", body = UserDetail),
91        (status = 403, description = "Forbidden", body = serde_json::Value),
92        (status = 404, description = "User details not found", body = serde_json::Value),
93        (status = 500, description = "Internal server error", body = serde_json::Value)
94    )
95)]
96pub async fn get_user_details_by_courses(
97    user: AuthUser,
98    pool: web::Data<PgPool>,
99    payload: web::Json<UserDetailsRequest>,
100) -> ControllerResult<web::Json<UserDetail>> {
101    let mut conn = pool.acquire().await?;
102
103    // Check if the user has permission to view user details through any of the provided courses
104    let mut token = None;
105    let mut authorized_course_ids = Vec::new();
106
107    if payload.course_ids.is_empty() {
108        // One can view the details though global permissions even though they have not started any course yet
109        token = Some(
110            authorize(
111                &mut conn,
112                Act::ViewUserProgressOrDetails,
113                Some(user.id),
114                Res::GlobalPermissions,
115            )
116            .await?,
117        );
118    } else {
119        for course_id in &payload.course_ids {
120            if let Ok(auth_token) = authorize(
121                &mut conn,
122                Act::ViewUserProgressOrDetails,
123                Some(user.id),
124                Res::Course(*course_id),
125            )
126            .await
127            {
128                if token.is_none() {
129                    token = Some(auth_token);
130                }
131                authorized_course_ids.push(*course_id);
132            }
133        }
134    }
135
136    let token = token.ok_or_else(|| {
137        ControllerError::new(
138            ControllerErrorType::Forbidden,
139            "No permission to view user details through any of the provided courses".to_string(),
140            None,
141        )
142    })?;
143
144    if !payload.course_ids.is_empty() {
145        let enrollments = models::course_instance_enrollments::get_by_user_id_and_course_ids(
146            &mut conn,
147            payload.user_id,
148            &authorized_course_ids,
149        )
150        .await?;
151        let roles = models::roles::get_by_user_id_and_course_ids(
152            &mut conn,
153            payload.user_id,
154            &authorized_course_ids,
155        )
156        .await?;
157        if enrollments.is_empty() && roles.is_empty() {
158            return Err(controller_err!(
159                Forbidden,
160                "Target user is not linked to an authorized course".to_string()
161            ));
162        }
163    }
164
165    // If we have permission, get the user details without course restriction
166    let res = models::user_details::get_user_details_by_user_id(&mut conn, payload.user_id).await?;
167    token.authorized_ok(web::Json(res))
168}

332#[utoipa::path(
333    get,
334    path = "/users-ip-country",
335    operation_id = "getUsersIpCountry",
336    tag = "user-details",
337    responses(
338        (status = 200, description = "Country inferred from request IP", body = String)
339    )
340)]
341pub async fn get_user_country_by_ip(
342    req: HttpRequest,
343    ip_to_country_mapper: web::Data<IpToCountryMapper>,
344) -> ControllerResult<String> {
345    let connection_info = req.connection_info();
346
347    let ip: Option<IpAddr> = connection_info
348        .realip_remote_addr()
349        .and_then(|ip| ip.parse::<IpAddr>().ok());
350
351    let country = ip
352        .and_then(|ip| ip_to_country_mapper.map_ip_to_country(&ip))
353        .map(|c| c.to_string())
354        .unwrap_or_default();
355
356    let token = skip_authorize();
357    token.authorized_ok(country)
358}

370/**
371POST `/api/v0/main-frontend/user-details/update-user-info` - Updates the users information such as email, name, country and email communication consent
372*/
373#[instrument(skip(pool, app_conf, tmc_client))]
374#[utoipa::path(
375    post,
376    path = "/update-user-info",
377    operation_id = "updateUserInfo",
378    tag = "user-details",
379    request_body = UserInfoPayload,
380    responses(
381        (status = 200, description = "Updated user details", body = UserDetail)
382    )
383)]
384pub async fn update_user_info(
385    user: AuthUser,
386    pool: web::Data<PgPool>,
387    payload: web::Json<UserInfoPayload>,
388    tmc_client: web::Data<TmcClient>,
389    app_conf: web::Data<ApplicationConfiguration>,
390) -> ControllerResult<web::Json<UserDetail>> {
391    let mut tx = pool.begin().await?;
392
393    let existing_user = models::user_details::get_user_details_by_user_id(&mut tx, user.id)
394        .await
395        .context("Failed to fetch existing user data")?;
396
397    let user = models::users::get_by_id(&mut tx, user.id)
398        .await
399        .context("Failed to fetch user")?;
400
401    let updated_user = models::user_details::update_user_info(
402        &mut tx,
403        user.id,
404        &payload.email,
405        &payload.first_name,
406        &payload.last_name,
407        &payload.country,
408        payload.email_communication_consent,
409    )
410    .await
411    .context("Failed to update database")?;
412
413    let email_changed = existing_user.email != payload.email;
414    let first_name_changed = existing_user.first_name != Some(payload.first_name.clone());
415    let last_name_changed = existing_user.last_name != Some(payload.last_name.clone());
416
417    if email_changed || first_name_changed || last_name_changed {
418        controllers::auth::update_user_information_to_tmc(
419            user.upstream_id,
420            payload.first_name.clone(),
421            payload.last_name.clone(),
422            email_changed.then(|| payload.email.clone()),
423            &tmc_client,
424            &app_conf,
425        )
426        .await?;
427    } else {
428        info!("User info unchanged, skipping update to TMC.");
429    }
430
431    tx.commit().await?;
432
433    if email_changed {
434        // After the commit: a rolled-back edit must not mail a code for an address the account lacks.
435        let mut conn = pool.acquire().await?;
436        domain::email_ownership_verification::queue_verification_email_best_effort(
437            &mut conn,
438            app_conf.enable_email_ownership_verification,
439            user.id,
440        )
441        .await;
442    }
443
444    let token = skip_authorize();
445    token.authorized_ok(web::Json(updated_user))
446}
