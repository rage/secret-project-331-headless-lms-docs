72/// The same, for one of the two terminal-state mails. No address: these go to the account's own,
73/// which the reader either owns or already sees.
74#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
75pub struct NotificationEmailStatus {
76    pub kind: CreditRegistrationNotificationKind,
77    pub email_send_status: EmailSendStatus,
78    pub sent_at: Option<DateTime<Utc>>,
79}
80
81impl NotificationEmailStatus {
82    /// The mail that belongs with the row's current state, or `None` for a state that has none: a
83    /// row shows at most one line, and an old action-needed mail on a since-registered row would
84    /// contradict the badge above it.
85    pub(crate) fn for_state(
86        state: CreditRegistrationState,
87        credit_registration_id: Uuid,
88        mails: &[RegistrationNotificationEmail],
89    ) -> Option<Self> {
90        let wanted = CreditRegistrationNotificationKind::for_state(state)?;
91        let mail = mails.iter().find(|mail| {
92            mail.credit_registration_id == credit_registration_id && mail.kind == wanted
93        })?;
94        Some(Self {
95            kind: mail.kind,
96            email_send_status: mail.send_status.email_send_status,
97            sent_at: mail.send_status.sent_at,
98        })
99    }
100}

390/// When the study registry may next be asked about this row, or `None` before it has been asked at
391/// all. The response tells the student when the button comes back.
392fn next_enrolment_recheck_allowed_at(
393    enrolment_checked_at: Option<DateTime<Utc>>,
394) -> Option<DateTime<Utc>> {
395    enrolment_checked_at
396        .map(|checked| checked + chrono::Duration::seconds(ENROLMENT_RECHECK_MIN_INTERVAL_SECS))
397}
398
399/// Whether we looked recently enough that looking again would tell the student nothing new.
400fn looked_for_enrolment_recently(enrolment_checked_at: Option<DateTime<Utc>>) -> bool {
401    next_enrolment_recheck_allowed_at(enrolment_checked_at)
402        .is_some_and(|allowed| allowed > Utc::now())
403}

405/// Records the student action and makes the row due for its next enrolment check.
406///
407/// Shared by the recheck button and by pressing Done, which differ only in what the event says and
408/// in how they decide the allowance is free.
409async fn start_enrolment_recheck(
410    conn: &mut PgConnection,
411    user_id: Uuid,
412    registration_id: Uuid,
413    message: &str,
414) -> Result<(), ControllerError> {
415    let mut tx = conn.begin().await?;
416    models::credit_registration_events::insert(
417        &mut tx,
418        &NewCreditRegistrationEvent {
419            actor_user_id: Some(user_id),
420            message: Some(message.to_string()),
421            ..NewCreditRegistrationEvent::new(
422                registration_id,
423                CreditRegistrationEventKind::StudentAction,
424            )
425        },
426    )
427    .await?;
428    models::credit_registrations::make_due_now_batch(&mut tx, &[registration_id]).await?;
429    recompute_preconditions(
430        &mut tx,
431        &RegistrationScope {
432            credit_registration_ids: vec![registration_id],
433            ..RegistrationScope::default()
434        },
435        PRECONDITIONS_LIMIT,
436    )
437    .await?;
438    tx.commit().await?;
439    Ok(())
440}

442/**
443POST `/api/v0/main-frontend/credit-registrations/my/{id}/recheck-enrolment` - Asks the pipeline to
444look for an enrolment again, for a row parked because the study registry had none.
445*/
446#[instrument(skip(pool))]
447#[utoipa::path(
448    post,
449    path = "/my/{id}/recheck-enrolment",
450    operation_id = "requestCreditRegistrationEnrolmentRecheck",
451    tag = "credit-registrations",
452    params(("id" = Uuid, Path, description = "Credit registration id")),
453    responses(
454        (status = 200, description = "Whether a recheck was started", body = RequestCreditRegistrationEnrolmentRecheckResult),
455        (status = 403, description = "Not the caller's registration"),
456        (status = 400, description = "The registration is not waiting for an enrolment")
457    )
458)]
459pub async fn request_credit_registration_enrolment_recheck(
460    user: AuthUser,
461    pool: web::Data<PgPool>,
462    id: web::Path<Uuid>,
463) -> ControllerResult<web::Json<RequestCreditRegistrationEnrolmentRecheckResult>> {
464    let mut conn = pool.acquire().await?;
465    let token = skip_authorize();
466
467    let registration = models::credit_registrations::get_by_id(&mut conn, *id).await?;
468    if registration.user_id != user.id {
469        return Err(controller_err!(
470            Forbidden,
471            "Not your registration.".to_string()
472        ));
473    }
474    if registration.state != CreditRegistrationState::NoUsableEnrolment {
475        return Err(controller_err!(
476            BadRequest,
477            "This registration is not waiting for an enrolment.".to_string()
478        ));
479    }
480
481    let next_allowed = next_enrolment_recheck_allowed_at(registration.enrolment_checked_at);
482    if next_allowed.is_some_and(|allowed| allowed > Utc::now()) {
483        return token.authorized_ok(web::Json(RequestCreditRegistrationEnrolmentRecheckResult {
484            recheck_started: false,
485            next_recheck_allowed_at: next_allowed,
486        }));
487    }
488
489    start_enrolment_recheck(
490        &mut conn,
491        user.id,
492        registration.id,
493        "The student asked us to look for an enrolment again.",
494    )
495    .await?;
496
497    token.authorized_ok(web::Json(RequestCreditRegistrationEnrolmentRecheckResult {
498        recheck_started: true,
499        next_recheck_allowed_at: None,
500    }))
501}

658/**
659POST `/api/v0/main-frontend/credit-registrations/student-number-verifications/{token}/claim` - Spends
660a mailed link and links the student number to the signed-in account.
661
662Any signed-in account may claim any valid token: holding it proves control of the Sisu-held mailbox,
663and the session says which of our accounts the person wants to use.
664*/
665#[instrument(skip(pool, path))]
666#[utoipa::path(
667    post,
668    path = "/student-number-verifications/{token}/claim",
669    operation_id = "claimStudentNumberVerificationToken",
670    tag = "credit-registrations",
671    params(("token" = String, Path, description = "The mailed verification token")),
672    responses(
673        (status = 200, description = "What the claim did", body = ClaimStudentNumberVerificationTokenResult),
674        (status = 404, description = "No such token")
675    )
676)]
677pub async fn claim_student_number_verification_token(
678    user: AuthUser,
679    pool: web::Data<PgPool>,
680    path: web::Path<String>,
681) -> ControllerResult<web::Json<ClaimStudentNumberVerificationTokenResult>> {
682    let mut conn = pool.acquire().await?;
683    let auth_token = skip_authorize();
684
685    let verification_token = get_token_or_404(&mut conn, &path).await?;
686    let refused = |outcome| ClaimStudentNumberVerificationTokenResult {
687        outcome,
688        student_number: None,
689        linked_course_name: None,
690        newly_unblocked_registration_count: 0,
691    };
692
693    if verification_token.used_at.is_some() {
694        return auth_token.authorized_ok(web::Json(refused(
695            ClaimStudentNumberVerificationTokenOutcome::AlreadyUsed,
696        )));
697    }
698    if verification_token.expires_at <= Utc::now() || verification_token.deleted_at.is_some() {
699        return auth_token.authorized_ok(web::Json(refused(
700            ClaimStudentNumberVerificationTokenOutcome::Expired,
701        )));
702    }
703    if find_conflicting_account(&mut conn, &verification_token, user.id).await? {
704        return auth_token.authorized_ok(web::Json(refused(
705            ClaimStudentNumberVerificationTokenOutcome::StudentNumberAlreadyLinkedToAnotherAccount,
706        )));
707    }
708
709    let course_name = course_name_of_token(&mut conn, &verification_token).await?;
710    let current_link = verified_student_numbers::get_by_user_id(&mut conn, user.id).await?;
711    let already_ours = current_link
712        .as_ref()
713        .is_some_and(|link| link.student_number == verification_token.student_number);
714
715    let mut tx = conn.begin().await?;
716    // The atomic single-use guard: two concurrent claims cannot both win here.
717    if !student_number_verification_tokens::claim(&mut tx, &verification_token.token, user.id)
718        .await?
719    {
720        tx.rollback().await?;
721        return auth_token.authorized_ok(web::Json(refused(
722            ClaimStudentNumberVerificationTokenOutcome::AlreadyUsed,
723        )));
724    }
725    if already_ours {
726        tx.commit().await?;
727        return auth_token.authorized_ok(web::Json(ClaimStudentNumberVerificationTokenResult {
728            outcome: ClaimStudentNumberVerificationTokenOutcome::AlreadyLinkedToThisAccount,
729            student_number: Some(verification_token.student_number),
730            linked_course_name: course_name,
731            newly_unblocked_registration_count: 0,
732        }));
733    }
734
735    // A student who changed programmes has a new number; the old link is retired, not deleted, so the
736    // audit trail survives.
737    let (_, newly_unblocked_registration_count) =
738        verified_student_numbers::replace_verified_student_number(
739            &mut tx,
740            current_link.map(|link| link.id),
741            &NewVerifiedStudentNumber {
742                user_id: user.id,
743                student_number: verification_token.student_number.clone(),
744                sisu_person_id: verification_token.sisu_person_id.clone(),
745                first_names: verification_token.first_names.clone(),
746                last_name: verification_token.last_name.clone(),
747                verified_via: StudentNumberVerificationMethod::EmailedLink,
748                verified_via_email: Some(verification_token.emailed_to.clone()),
749                verified_via_email_match_field: None,
750                account_email_verified_at: None,
751                linked_by_user_id: None,
752                link_reason: None,
753                verified_from_course_id: verification_token.course_id,
754            },
755            Some(user.id),
756            CreditRegistrationEventKind::StudentAction,
757            "The student linked a student number.",
758        )
759        .await?;
760    tx.commit().await?;
761
762    auth_token.authorized_ok(web::Json(ClaimStudentNumberVerificationTokenResult {
763        outcome: ClaimStudentNumberVerificationTokenOutcome::Linked,
764        student_number: Some(verification_token.student_number),
765        linked_course_name: course_name,
766        newly_unblocked_registration_count,
767    }))
768}

770/// Assembles the wire rows for one account, adding the enrolment link and the linking-mail status the
771/// ledger does not carry.
772async fn build_my_credit_registrations(
773    conn: &mut PgConnection,
774    user_id: Uuid,
775    filter: StudentRegistrationFilter,
776) -> Result<Vec<MyCreditRegistration>, ControllerError> {
777    let rows =
778        models::credit_registrations::get_student_facing_by_user_id(conn, user_id, filter).await?;
779
780    let ids: Vec<Uuid> = rows.iter().map(|row| row.id).collect();
781    let notification_mails = student_notifications::get_for_registrations(conn, &ids).await?;
782
783    let mut enrolment_links: HashMap<String, Option<String>> = HashMap::new();
784    let mut linking_mails: Option<LinkingMailCache> = None;
785    let mut res = Vec::with_capacity(rows.len());
786    for row in rows {
787        let state = row.state;
788        let status = StudentFacingCreditRegistrationStatus::of(
789            state,
790            row.preconditions(),
791            row.enrolment_resolved,
792        );
793        let enrolment_link = if status == StudentFacingCreditRegistrationStatus::NeedsEnrolment {
794            resolve_enrolment_link(conn, &row, &mut enrolment_links).await?
795        } else {
796            None
797        };
798        let linking_email = if status == StudentFacingCreditRegistrationStatus::NeedsStudentNumber {
799            resolve_linking_email(conn, user_id, &row, &mut linking_mails).await?
800        } else {
801            None
802        };
803        let notification_email =
804            NotificationEmailStatus::for_state(state, row.id, &notification_mails);
805        res.push(to_my_credit_registration(
806            row,
807            status,
808            enrolment_link,
809            linking_email,
810            notification_email,
811        ));
812    }
813    Ok(res)
814}

882/// An account's linking mails and their send status, fetched once per request rather than once per
883/// row: they are the same for every one of a student's rows.
884struct LinkingMailCache {
885    mails: Vec<CreditRegistrationAccountLinkingEmail>,
886    reports: HashMap<Uuid, EmailSendStatusReport>,
887}
888
889/// The latest linking mail for this account's Sisu person on this course. `None` for an account that
890/// was never linked: the mail is addressed to a Sisu person, not to an email address.
891async fn resolve_linking_email(
892    conn: &mut PgConnection,
893    user_id: Uuid,
894    row: &StudentCreditRegistration,
895    cache: &mut Option<LinkingMailCache>,
896) -> Result<Option<LinkingEmailStatus>, ControllerError> {
897    if cache.is_none() {
898        let mails =
899            match verified_student_numbers::get_latest_including_deleted_by_user_id(conn, user_id)
900                .await?
901            {
902                Some(link) => {
903                    credit_registration_account_linking_emails::get_by_sisu_person_id(
904                        conn,
905                        &link.sisu_person_id,
906                    )
907                    .await?
908                }
909                None => Vec::new(),
910            };
911        let ids: Vec<Uuid> = mails.iter().map(|mail| mail.id).collect();
912        let reports =
913            credit_registration_account_linking_emails::get_send_status_reports(conn, &ids).await?;
914        *cache = Some(LinkingMailCache { mails, reports });
915    }
916    let cache = cache.as_ref().ok_or_else(|| {
917        controller_err!(
918            InternalServerError,
919            "linking mail cache was not populated".to_string()
920        )
921    })?;
922    let Some(mail) = cache
923        .mails
924        .iter()
925        .find(|mail| mail.course_id == row.course_id)
926    else {
927        return Ok(None);
928    };
929    let Some(report) = cache.reports.get(&mail.id) else {
930        return Ok(None);
931    };
932    Ok(Some(LinkingEmailStatus {
933        email_send_status: report.email_send_status,
934        sent_at: report.sent_at,
935        emailed_to_masked: mask_email(&mail.emailed_to),
936    }))
937}

993/// Keeps the domain and drops the local part: enough to recognise which mailbox to open, not a new
994/// disclosure of an address. Teachers get the same masking; only admins see an address in full.

1191pub async fn confirm_my_enrolment(
1192    user: AuthUser,
1193    pool: web::Data<PgPool>,
1194    course_module_id: web::Path<Uuid>,
1195) -> ControllerResult<web::Json<MyEnrolmentRoute>> {
1196    let mut conn = pool.acquire().await?;
1197    let token = skip_authorize();
1198
1199    let (current, registration) =
1200        build_my_enrolment_route(&mut conn, user.id, *course_module_id).await?;
1201    if current.route.is_none() {
1202        return Err(controller_err!(
1203            BadRequest,
1204            "Answer where you enrol before confirming that you have.".to_string()
1205        ));
1206    }
1207    if !current.can_change {
1208        return Err(controller_err!(
1209            BadRequest,
1210            "Your enrolment has already been found.".to_string()
1211        ));
1212    }
1213    let answer = credit_registration_enrolment_routes::set_enrolment_confirmed(
1214        &mut conn,
1215        current.course_module_completion_id,
1216        true,
1217    )
1218    .await?;
1219    if let Some(registration) = registration {
1220        bring_enrolment_check_forward(&mut conn, user.id, &registration).await?;
1221    }
1222
1223    token.authorized_ok(web::Json(my_enrolment_route(
1224        current.course_module_completion_id,
1225        Some(answer),
1226        current.can_change,
1227    )))
1228}

1281/// again would tell the student nothing new.

1284/// allowance of its own, so pressing Done cannot be used to poll the study registry.
1285async fn bring_enrolment_check_forward(
1286    conn: &mut PgConnection,
1287    user_id: Uuid,
1288    registration: &StudentCreditRegistration,
1289) -> Result<(), ControllerError> {

1291        || looked_for_enrolment_recently(registration.enrolment_checked_at)

1296        conn,
1297        user_id,
1298        registration.id,
1299        "The student said they had enrolled.",
1300    )
1301    .await
1302}
