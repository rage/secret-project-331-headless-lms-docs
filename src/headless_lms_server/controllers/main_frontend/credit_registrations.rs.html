298POST `/api/v0/main-frontend/credit-registrations/my/{id}/recheck-enrolment` - Asks the pipeline to
299look for an enrolment again, for a row parked because the study registry had none.
300*/
301#[instrument(skip(pool))]
302#[utoipa::path(
303    post,
304    path = "/my/{id}/recheck-enrolment",
305    operation_id = "requestCreditRegistrationEnrolmentRecheck",
306    tag = "credit-registrations",
307    params(("id" = Uuid, Path, description = "Credit registration id")),
308    responses(
309        (status = 200, description = "Whether a recheck was started", body = RequestCreditRegistrationEnrolmentRecheckResult),
310        (status = 403, description = "Not the caller's registration"),
311        (status = 400, description = "The registration is not waiting for an enrolment")
312    )
313)]
314pub async fn request_credit_registration_enrolment_recheck(
315    user: AuthUser,
316    pool: web::Data<PgPool>,
317    id: web::Path<Uuid>,
318) -> ControllerResult<web::Json<RequestCreditRegistrationEnrolmentRecheckResult>> {
319    let mut conn = pool.acquire().await?;
320    let token = skip_authorize();
321
322    let registration = models::credit_registrations::get_by_id(&mut conn, *id).await?;
323    if registration.user_id != user.id {
324        return Err(controller_err!(
325            Forbidden,
326            "Not your registration.".to_string()
327        ));
328    }
329    if registration.state != CreditRegistrationState::NoUsableEnrolment {
330        return Err(controller_err!(
331            BadRequest,
332            "This registration is not waiting for an enrolment.".to_string()
333        ));
334    }
335
336    let next_allowed = registration
337        .enrolment_checked_at
338        .map(|checked| checked + chrono::Duration::seconds(ENROLMENT_RECHECK_MIN_INTERVAL_SECS));
339    if next_allowed.is_some_and(|allowed| allowed > Utc::now()) {
340        return token.authorized_ok(web::Json(RequestCreditRegistrationEnrolmentRecheckResult {
341            recheck_started: false,
342            next_recheck_allowed_at: next_allowed,
343        }));
344    }
345
346    let mut tx = conn.begin().await?;
347    models::credit_registration_events::insert(
348        &mut tx,
349        &NewCreditRegistrationEvent {
350            actor_user_id: Some(user.id),
351            message: Some("The student asked us to look for an enrolment again.".to_string()),
352            ..NewCreditRegistrationEvent::new(
353                registration.id,
354                CreditRegistrationEventKind::StudentAction,
355            )
356        },
357    )
358    .await?;
359    models::credit_registrations::make_due_now(&mut tx, registration.id).await?;
360    recompute_preconditions(
361        &mut tx,
362        &RegistrationScope {
363            credit_registration_ids: vec![registration.id],
364            ..RegistrationScope::default()
365        },
366        PRECONDITIONS_LIMIT,
367    )
368    .await?;
369    tx.commit().await?;
370
371    token.authorized_ok(web::Json(RequestCreditRegistrationEnrolmentRecheckResult {
372        recheck_started: true,
373        next_recheck_allowed_at: None,
374    }))
375}

504/**
505POST `/api/v0/main-frontend/credit-registrations/student-number-verifications/{token}/claim` - Spends
506a mailed link and links the student number to the signed-in account.
507
508Any signed-in account may claim any valid token: holding it proves control of the Sisu-held mailbox,
509and the session says which of our accounts the person wants to use.
510*/
511#[instrument(skip(pool, path))]
512#[utoipa::path(
513    post,
514    path = "/student-number-verifications/{token}/claim",
515    operation_id = "claimStudentNumberVerificationToken",
516    tag = "credit-registrations",
517    params(("token" = String, Path, description = "The mailed verification token")),
518    responses(
519        (status = 200, description = "What the claim did", body = ClaimStudentNumberVerificationTokenResult),
520        (status = 404, description = "No such token")
521    )
522)]
523pub async fn claim_student_number_verification_token(
524    user: AuthUser,
525    pool: web::Data<PgPool>,
526    path: web::Path<String>,
527) -> ControllerResult<web::Json<ClaimStudentNumberVerificationTokenResult>> {
528    let mut conn = pool.acquire().await?;
529    let auth_token = skip_authorize();
530
531    let verification_token = get_token_or_404(&mut conn, &path).await?;
532    let refused = |outcome| ClaimStudentNumberVerificationTokenResult {
533        outcome,
534        student_number: None,
535        linked_course_name: None,
536        newly_unblocked_registration_count: 0,
537    };
538
539    if verification_token.used_at.is_some() {
540        return auth_token.authorized_ok(web::Json(refused(
541            ClaimStudentNumberVerificationTokenOutcome::AlreadyUsed,
542        )));
543    }
544    if verification_token.expires_at <= Utc::now() || verification_token.deleted_at.is_some() {
545        return auth_token.authorized_ok(web::Json(refused(
546            ClaimStudentNumberVerificationTokenOutcome::Expired,
547        )));
548    }
549    if find_conflicting_account(&mut conn, &verification_token, user.id).await? {
550        return auth_token.authorized_ok(web::Json(refused(
551            ClaimStudentNumberVerificationTokenOutcome::StudentNumberAlreadyLinkedToAnotherAccount,
552        )));
553    }
554
555    let course_name = course_name_of_token(&mut conn, &verification_token).await?;
556    let current_link = verified_student_numbers::get_by_user_id(&mut conn, user.id).await?;
557    let already_ours = current_link
558        .as_ref()
559        .is_some_and(|link| link.student_number == verification_token.student_number);
560
561    let mut tx = conn.begin().await?;
562    // The atomic single-use guard: two concurrent claims cannot both win here.
563    if !student_number_verification_tokens::claim(&mut tx, &verification_token.token, user.id)
564        .await?
565    {
566        tx.rollback().await?;
567        return auth_token.authorized_ok(web::Json(refused(
568            ClaimStudentNumberVerificationTokenOutcome::AlreadyUsed,
569        )));
570    }
571    if already_ours {
572        tx.commit().await?;
573        return auth_token.authorized_ok(web::Json(ClaimStudentNumberVerificationTokenResult {
574            outcome: ClaimStudentNumberVerificationTokenOutcome::AlreadyLinkedToThisAccount,
575            student_number: Some(verification_token.student_number),
576            linked_course_name: course_name,
577            newly_unblocked_registration_count: 0,
578        }));
579    }
580
581    // A student who changed programmes has a new number; the old link is retired, not deleted, so the
582    // audit trail survives.
583    let (_, newly_unblocked_registration_count) =
584        verified_student_numbers::replace_verified_student_number(
585            &mut tx,
586            current_link.map(|link| link.id),
587            &NewVerifiedStudentNumber {
588                user_id: user.id,
589                student_number: verification_token.student_number.clone(),
590                sisu_person_id: verification_token.sisu_person_id.clone(),
591                first_names: verification_token.first_names.clone(),

647*/
648#[instrument(skip(pool, payload))]
649#[utoipa::path(

652    operation_id = "setMyCourseCreditRegistrationConsent",
653    tag = "credit-registrations",
654    params(("course_id" = Uuid, Path, description = "Course id")),
655    request_body = SetMyCourseCreditRegistrationConsentPayload,
656    responses(
657        (status = 200, description = "The recorded answer and what it unblocked", body = SetMyCourseCreditRegistrationConsentResult)
658    )
659)]
660pub async fn set_my_course_credit_registration_consent(
661    user: AuthUser,
662    pool: web::Data<PgPool>,
663    course_id: web::Path<Uuid>,
664    payload: web::Json<SetMyCourseCreditRegistrationConsentPayload>,
665) -> ControllerResult<web::Json<SetMyCourseCreditRegistrationConsentResult>> {
666    let mut conn = pool.acquire().await?;
667    let token = authorize_access_to_course_material(&mut conn, Some(user.id), *course_id).await?;
668
669    let waiting_before = count_in_state(
670        &mut conn,
671        user.id,
672        *course_id,
673        CreditRegistrationState::PendingConsent,
674    )
675    .await?;
676
677    let mut tx = conn.begin().await?;
678    let consent = course_credit_registration_consents::upsert(
679        &mut tx,
680        user.id,
681        *course_id,
682        payload.consent_given,
683    )
684    .await?;
685    apply_consent_change(&mut tx, user.id, *course_id).await?;
686    let waiting_after = count_in_state(
687        &mut tx,
688        user.id,
689        *course_id,
690        CreditRegistrationState::PendingConsent,
691    )
692    .await?;
693    tx.commit().await?;
694
695    token.authorized_ok(web::Json(SetMyCourseCreditRegistrationConsentResult {
696        consent_given: consent.consent_given,
697        consent_given_at: consent.consent_given_at,
698        consent_withdrawn_at: consent.consent_withdrawn_at,
699        // Withdrawal empties the same queue by blocking the rows, so counting the difference either
700        // way would report a withdrawal as having unblocked something.
701        newly_unblocked_registration_count: if payload.consent_given {
702            (waiting_before - waiting_after).max(0)
703        } else {
704            0
705        },
706    }))
707}

710GET `/api/v0/main-frontend/credit-registrations/my/consents` - One row per course the signed-in
711account is enrolled on that offers credit registration, asked or not.
712*/
713#[instrument(skip(pool))]
714#[utoipa::path(
715    get,
716    path = "/my/consents",
717    operation_id = "getMyCreditRegistrationConsents",
718    tag = "credit-registrations",
719    responses(
720        (status = 200, description = "The caller's per-course consents", body = Vec<MyCreditRegistrationConsent>)
721    )
722)]
723pub async fn get_my_credit_registration_consents(
724    user: AuthUser,
725    pool: web::Data<PgPool>,
726) -> ControllerResult<web::Json<Vec<MyCreditRegistrationConsent>>> {
727    let mut conn = pool.acquire().await?;
728    let token = skip_authorize();
729
730    let course_ids = models::course_modules::get_credit_registration_course_ids_for_enrolled_user(
731        &mut conn, user.id,
732    )
733    .await?;
734    let courses = models::courses::get_by_ids(&mut conn, &course_ids).await?;
735    let consents = course_credit_registration_consents::get_by_user_id(&mut conn, user.id).await?;
736    let registrations =
737        models::credit_registrations::get_student_facing_by_user_id(&mut conn, user.id, None)
738            .await?;
739
740    let mut res: Vec<MyCreditRegistrationConsent> = courses
741        .into_iter()
742        .map(|course| {
743            let consent = consents.iter().find(|row| row.course_id == course.id);
744            let live = registrations
745                .iter()
746                .filter(|row| row.course_id == course.id && row.superseded_by_id.is_none());
747            MyCreditRegistrationConsent {
748                course_id: course.id,
749                course_name: course.name,
750                consent_given: consent.map(|row| row.consent_given),
751                consent_given_at: consent.and_then(|row| row.consent_given_at),
752                consent_withdrawn_at: consent.and_then(|row| row.consent_withdrawn_at),
753                asked_at: consent.map(|row| row.asked_at),
754                registrable_completion_count: live
755                    .clone()
756                    .filter(|row| row.state == CreditRegistrationState::PendingConsent)
757                    .count() as i64,
758                registered_count: live.filter(|row| row.state.is_success()).count() as i64,
759            }
760        })
761        .collect();
762    res.sort_by(|a, b| a.course_name.cmp(&b.course_name));
763
764    token.authorized_ok(web::Json(res))

767/// Assembles the wire rows for one account, adding the enrolment link and the linking-mail status the
768/// ledger does not carry.
769async fn build_my_credit_registrations(
770    conn: &mut PgConnection,
771    user_id: Uuid,
772    course_module_id: Option<Uuid>,
773) -> Result<Vec<MyCreditRegistration>, ControllerError> {
774    let rows = models::credit_registrations::get_student_facing_by_user_id(
775        conn,
776        user_id,
777        course_module_id,
778    )
779    .await?;
780
781    let mut enrolment_links: HashMap<String, Option<String>> = HashMap::new();
782    let mut linking_mails: Option<LinkingMailCache> = None;
783    let mut res = Vec::with_capacity(rows.len());
784    for row in rows {
785        let status = StudentFacingCreditRegistrationStatus::of(row.state);
786        let enrolment_link = if status == StudentFacingCreditRegistrationStatus::NeedsEnrolment {
787            resolve_enrolment_link(conn, &row, &mut enrolment_links).await?
788        } else {
789            None
790        };
791        let linking_email = if status == StudentFacingCreditRegistrationStatus::NeedsStudentNumber {
792            resolve_linking_email(conn, user_id, &row, &mut linking_mails).await?
793        } else {
794            None
795        };
796        res.push(to_my_credit_registration(
797            row,
798            status,
799            enrolment_link,
800            linking_email,
801        ));
802    }
803    Ok(res)
804}

867/// An account's linking mails and their send status, fetched once per request rather than once per
868/// row: they are the same for every one of a student's rows.
869struct LinkingMailCache {
870    mails: Vec<CreditRegistrationAccountLinkingEmail>,
871    reports: HashMap<Uuid, EmailSendStatusReport>,
872}
873
874/// The latest linking mail for this account's Sisu person on this course. `None` for an account that
875/// was never linked: the mail is addressed to a Sisu person, not to an email address.
876async fn resolve_linking_email(
877    conn: &mut PgConnection,
878    user_id: Uuid,
879    row: &StudentCreditRegistration,
880    cache: &mut Option<LinkingMailCache>,
881) -> Result<Option<LinkingEmailStatus>, ControllerError> {
882    if cache.is_none() {
883        let mails =
884            match verified_student_numbers::get_latest_including_deleted_by_user_id(conn, user_id)
885                .await?
886            {
887                Some(link) => {
888                    credit_registration_account_linking_emails::get_by_sisu_person_id(
889                        conn,
890                        &link.sisu_person_id,
891                    )
892                    .await?
893                }
894                None => Vec::new(),
895            };
896        let ids: Vec<Uuid> = mails.iter().map(|mail| mail.id).collect();
897        let reports =
898            credit_registration_account_linking_emails::get_send_status_reports(conn, &ids).await?;
899        *cache = Some(LinkingMailCache { mails, reports });
900    }
901    let cache = cache.as_ref().ok_or_else(|| {
902        controller_err!(
903            InternalServerError,
904            "linking mail cache was not populated".to_string()
905        )
906    })?;
907    let Some(mail) = cache
908        .mails
909        .iter()
910        .find(|mail| mail.course_id == row.course_id)
911    else {
912        return Ok(None);
913    };
914    let Some(report) = cache.reports.get(&mail.id) else {
915        return Ok(None);
916    };
917    Ok(Some(LinkingEmailStatus {
918        email_send_status: report.email_send_status,
919        sent_at: report.sent_at,
920        emailed_to_masked: mask_email(&mail.emailed_to),
921    }))
922}

924async fn build_course_consent(
925    conn: &mut PgConnection,
926    user_id: Uuid,
927    course_id: Uuid,
928) -> Result<MyCourseCreditRegistrationConsent, ControllerError> {
929    let course = models::courses::get_course(conn, course_id).await?;
930    // This query carries the enable flag, so a course with no Suotar module never fetches its modules.
931    let enabled_configs: Vec<_> =
932        models::course_modules::get_credit_registration_configs_by_course_id(conn, course_id)
933            .await?
934            .into_iter()
935            .filter(|config| config.enable_credit_registration_via_suotar)
936            .collect();
937    let enabled_ids: Vec<Uuid> = enabled_configs
938        .iter()
939        .map(|config| config.course_module_id)
940        .collect();
941    let names: HashMap<Uuid, Option<String>> = if enabled_ids.is_empty() {
942        HashMap::new()
943    } else {
944        models::course_modules::get_by_ids(conn, &enabled_ids)
945            .await?
946            .into_iter()
947            .map(|module| (module.id, module.name))
948            .collect()
949    };
950    let modules: Vec<CreditRegistrationConsentModule> = enabled_configs
951        .into_iter()
952        .map(|config| CreditRegistrationConsentModule {
953            id: config.course_module_id,
954            name: names.get(&config.course_module_id).cloned().flatten(),
955            uh_course_code: config.uh_course_code,
956            ects_credits: config.ects_credits,
957        })
958        .collect();
959    let consent =
960        course_credit_registration_consents::get_by_user_and_course(conn, user_id, course_id)
961            .await?;
962    let registrable_completion_count = count_in_state(
963        conn,
964        user_id,
965        course_id,
966        CreditRegistrationState::PendingConsent,
967    )
968    .await?;
969
970    Ok(MyCourseCreditRegistrationConsent {
971        course_id,
972        course_name: course.name,
973        asked: consent.is_some(),
974        consent_given: consent.as_ref().map(|row| row.consent_given),
975        consent_given_at: consent.as_ref().and_then(|row| row.consent_given_at),
976        consent_withdrawn_at: consent.as_ref().and_then(|row| row.consent_withdrawn_at),
977        credit_registration_enabled_for_course: !modules.is_empty(),
978        modules,
979        registrable_completion_count,
980    })
981}

1023/// Whether the token's holder is already live on some other account of ours.

1026/// and gets a new number, so checking the number alone lets the claim through and then trips

1054/// Keeps the domain and drops the local part: enough to recognise which mailbox to open, not a new
1055/// disclosure of an address. Teachers get the same masking; only admins see an address in full.
1056pub(crate) fn mask_email(email: &str) -> String {
1057    match email.split_once('@') {
1058        Some((_, domain)) => format!("...@{domain}"),
1059        None => "...".to_string(),
1060    }
