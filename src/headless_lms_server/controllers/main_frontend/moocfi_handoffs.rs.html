1/*!
2Handlers for HTTP requests to `/api/v0/main-frontend/moocfi-handoffs`.
3
4The student's browser arrives at `redeem` by a top-level navigation from mooc.fi's credit registration
5page, or from Elements of AI's, carrying a handoff minted by
6[`crate::controllers::moocfi::credit_registration_handoffs`]. Every answer to the navigation is a 303,
7never JSON:
8
9- the right account signed in: on to the registration page;
10- nobody signed in, arriving same-site from mooc.fi's own page: signed in, then on to the
11  registration page;
12- another account signed in, or any other arrival: to the course's landing page
13  (`/completion-registration/mooc-fi/{moocfi_course_id}/handoff`), with a single-use continuation
14  in an HttpOnly cookie that `continue` completes once the student confirms. Nobody is signed in
15  without asking on a navigation another site could have started, which would allow login CSRF;
16- a handoff for an account with any courses role, and anything else: to the normal login, which
17  never signs anyone in.
18
19A handoff is single use: signing in or starting a continuation spends it. Sessions it signs in are
20delegated; see [`authentication::remember`].
21*/

49const CONTINUATION_COOKIE: &str = "moocfi_handoff_continuation";
50const CONTINUATION_COOKIE_PATH: &str = "/api/v0/main-frontend/moocfi-handoffs";

52/// Long enough to read the landing page, short enough that a shared computer's next user rarely
53/// finds one waiting.
54const CONTINUATION_LIFETIME: chrono::TimeDelta = chrono::TimeDelta::seconds(60);
55
56/// An expired handoff still gets the wrong-account explanation, and is charged to its student's
57/// bucket, only this soon after expiry; older ones could otherwise be replayed to lock them out.
58const RECENTLY_EXPIRED: chrono::TimeDelta = chrono::TimeDelta::minutes(1);
59
60/// Must match the reasons the landing page reads.
61const REASON_DIFFERENT_ACCOUNT: &str = "different_account";
62const REASON_CONFIRM: &str = "confirm";
63const REASON_RATE_LIMITED: &str = "rate_limited";

65/// A lecture hall shares one address behind a campus NAT, so per IP this is only a coarse cap.
66static PER_IP_LIMITS: Lazy<EndpointLimiters> = Lazy::new(|| {
67    EndpointLimiters::from_config(&RateLimitConfig {
68        per_minute: Some(600),
69        per_hour: Some(6000),
70        ..Default::default()
71    })
72});
73
74/// Keyed by the student the handoff is for, which is what the per-IP cap cannot tell apart.
75static PER_STUDENT_LIMITS: Lazy<EndpointLimiters> = Lazy::new(|| {
76    EndpointLimiters::from_config(&RateLimitConfig {
77        per_minute: Some(20),
78        per_hour: Some(200),
79        ..Default::default()
80    })

83/// How one redeem ended, for the logs. Never logged with the IP.
84#[derive(Debug, Clone, Copy)]
85enum RedeemOutcome {
86    SignedIn,
87    AlreadySignedIn,
88    Invalid,
89    Expired,
90    Spent,
91    IpMismatch,
92    RateLimited,
93    StaffAccount,
94    DifferentAccount,
95    DifferentAccountNotContinuable,
96    NeedsConfirmation,
97}

99impl RedeemOutcome {
100    fn as_str(self) -> &'static str {
101        match self {
102            Self::SignedIn => "signed_in",
103            Self::AlreadySignedIn => "already_signed_in",
104            Self::Invalid => "invalid",
105            Self::Expired => "expired",
106            Self::Spent => "spent",
107            Self::IpMismatch => "ip_mismatch",
108            Self::RateLimited => "rate_limited",
109            Self::StaffAccount => "staff_account",
110            Self::DifferentAccount => "different_account",
111            Self::DifferentAccountNotContinuable => "different_account_not_continuable",
112            Self::NeedsConfirmation => "needs_confirmation",
113        }
114    }
115}

117/**
118GET `/api/v0/main-frontend/moocfi-handoffs/redeem?token=…` - Signs in the student a mooc.fi handoff
119was minted for and sends them to the course's registration page, or to the landing page at
120`/completion-registration/mooc-fi/{moocfi_course_id}/handoff?reason=…` to confirm first.
121*/
122#[instrument(skip_all)]
123pub async fn redeem_moocfi_handoff(
124    request: HttpRequest,
125    session: Session,
126    pool: web::Data<PgPool>,
127    app_conf: web::Data<ApplicationConfiguration>,
128    jwt_key: web::Data<JwtKey>,
129    query: web::Query<RedeemQuery>,
130) -> ControllerResult<HttpResponse> {
131    let token = skip_authorize();
132    let client_ip = request_client_ip(&request);
133    // Only the signature first: a different account is told so even when the handoff has expired.
134    let Some(claim) = query
135        .token
136        .as_deref()
137        .and_then(|handoff| MoocfiHandoffClaim::read_ignoring_expiry(handoff, &jwt_key).ok())
138    else {
139        return token.authorized_ok(fall_back_to_login(RedeemOutcome::Invalid, None));
140    };
141    let moocfi_course_id = claim.moocfi_course_id;
142    let is_unexpired = claim.is_unexpired_or_expired_within(chrono::TimeDelta::zero());
143    let is_recent = claim.is_unexpired_or_expired_within(RECENTLY_EXPIRED);
144    // The binding that stops login CSRF: an attacker can only mint for their own IP.
145    let is_bound = client_ip.is_some_and(|client_ip| claim.is_bound_to(client_ip, &jwt_key));
146
147    let is_ip_limited =
148        client_ip.is_some_and(|client_ip| PER_IP_LIMITS.check_key(client_ip.to_string()).is_err());
149    let is_student_limited = is_bound
150        && is_recent
151        && PER_STUDENT_LIMITS
152            .check_key(claim.user_id.to_string())
153            .is_err();
154    if is_ip_limited || is_student_limited {
155        log_outcome(RedeemOutcome::RateLimited);
156        return token.authorized_ok(see_other(&landing_page_path(
157            moocfi_course_id,
158            REASON_RATE_LIMITED,
159        )));
160    }
161    if !is_recent {
162        return token.authorized_ok(fall_back_to_login(RedeemOutcome::Expired, Some(&claim)));
163    }
164
165    // The session cookie is SameSite=Strict, so a cross-site arrival never gets this far.
166    let session_user = session_user_id(&session);
167    if session_user == Some(claim.user_id) {
168        log_outcome(RedeemOutcome::AlreadySignedIn);
169        return token.authorized_ok(see_other(&registration_page_path(&claim)));
170    }
171    // Checked again here, not only at the mint: a role granted since must not be signed in either.
172    let mut conn = pool.acquire().await?;
173    if models::roles::has_any_role(&mut conn, claim.user_id).await? {
174        return token.authorized_ok(fall_back_to_login(
175            RedeemOutcome::StaffAccount,
176            Some(&claim),
177        ));
178    }
179    if session_user.is_some() {
180        if is_unexpired && is_bound && spend(&mut conn, &claim).await? {
181            log_outcome(RedeemOutcome::DifferentAccount);
182            return token.authorized_ok(
183                to_landing_with_continuation(
184                    &mut conn,
185                    &claim,
186                    REASON_DIFFERENT_ACCOUNT,
187                    &app_conf,
188                )
189                .await?,
190            );
191        }
192        log_outcome(RedeemOutcome::DifferentAccountNotContinuable);
193        return token.authorized_ok(see_other(&landing_page_path(
194            moocfi_course_id,
195            REASON_DIFFERENT_ACCOUNT,
196        )));
197    }
198    if !is_unexpired {
199        return token.authorized_ok(fall_back_to_login(RedeemOutcome::Expired, Some(&claim)));
200    }
201    if !is_bound {
202        return token.authorized_ok(fall_back_to_login(RedeemOutcome::IpMismatch, Some(&claim)));
203    }
204    if !spend(&mut conn, &claim).await? {
205        return token.authorized_ok(fall_back_to_login(RedeemOutcome::Spent, Some(&claim)));
206    }
207    if !is_from_moocfi_page(&request, &app_conf) {
208        log_outcome(RedeemOutcome::NeedsConfirmation);
209        return token.authorized_ok(
210            to_landing_with_continuation(&mut conn, &claim, REASON_CONFIRM, &app_conf).await?,
211        );
212    }
213
214    let user = match models::users::get_by_id(&mut conn, claim.user_id).await {
215        Ok(user) => user,
216        Err(error) if error.error_type() == &ModelErrorType::RecordNotFound => {
217            return token.authorized_ok(fall_back_to_login(RedeemOutcome::Invalid, Some(&claim)));
218        }
219        Err(error) => return Err(error.into()),
220    };
221    // Against session fixation: the signed-in session must not reuse an id issued before sign-in.
222    session.renew();
223    authentication::remember(&session, user)?;
224    log_outcome(RedeemOutcome::SignedIn);
225    token.authorized_ok(see_other(&registration_page_path(&claim)))
226}

257/**
258GET `/api/v0/main-frontend/moocfi-handoffs/pending?moocfi_course_id=…` - The handoff waiting for the
259student's confirmation on the course's landing page, or 404 when none is waiting for that course,
260it has expired or been used, or it was minted for another IP.
261*/
262#[instrument(skip_all)]
263#[utoipa::path(
264    get,
265    path = "/pending",
266    operation_id = "getPendingMoocfiHandoff",
267    tag = "moocfi-handoffs",
268    params(PendingMoocfiHandoffQuery),
269    responses(
270        (status = 200, description = "The waiting handoff", body = PendingMoocfiHandoff),
271        (status = 404, description = "No handoff is waiting")
272    )
273)]
274pub async fn get_pending_moocfi_handoff(
275    request: HttpRequest,
276    session: Session,
277    pool: web::Data<PgPool>,
278    jwt_key: web::Data<JwtKey>,
279    query: web::Query<PendingMoocfiHandoffQuery>,
280) -> ControllerResult<HttpResponse> {
281    let token = skip_authorize();
282    let mut conn = pool.acquire().await?;
283    let (digest, continuation) =
284        pending_continuation(&mut conn, &request, &jwt_key, query.moocfi_course_id).await?;
285    let email = models::user_details::get_user_details_by_user_id(&mut conn, continuation.user_id)
286        .await?
287        .email;
288    let signed_in_account = signed_in_account(&session, &continuation);
289    let pending = PendingMoocfiHandoff {
290        moocfi_course_id: continuation.moocfi_course_id,
291        email,
292        signed_in_account,
293        redirect_path: registration_page_path_of(
294            continuation.moocfi_course_id,
295            continuation.ui_language.as_deref(),
296        ),
297    };
298    if signed_in_account == SignedInAccount::Same {
299        // Nothing left to confirm, so nothing is left waiting for the browser's next user.
300        moocfi_handoff_continuations::take(&mut conn, &digest, query.moocfi_course_id).await?;
301        return token.authorized_ok(clearing_continuation(HttpResponse::Ok().json(pending)));
302    }
303    token.authorized_ok(HttpResponse::Ok().json(pending))
304}

321/**
322POST `/api/v0/main-frontend/moocfi-handoffs/continue` - Completes the waiting handoff: signs out
323whoever is signed in and signs in the handoff's account. Single use.
324
325Only from a same-origin request, and the continuation cookie is SameSite=Lax, so another site cannot
326post it. 404 when no handoff is waiting for the course; 400 when the signed-in account is no longer
327the one the student saw; 403 for an account with a courses role.
328*/
329#[instrument(skip_all)]
330#[utoipa::path(
331    post,
332    path = "/continue",
333    operation_id = "continueMoocfiHandoff",
334    tag = "moocfi-handoffs",
335    request_body = ContinueMoocfiHandoffPayload,
336    responses(
337        (status = 200, description = "Signed in; where to go next", body = ContinuedMoocfiHandoff),
338        (status = 404, description = "No handoff is waiting")
339    )
340)]
341pub async fn continue_moocfi_handoff(
342    request: HttpRequest,
343    session: Session,
344    pool: web::Data<PgPool>,
345    app_conf: web::Data<ApplicationConfiguration>,
346    jwt_key: web::Data<JwtKey>,
347    payload: web::Json<ContinueMoocfiHandoffPayload>,
348) -> ControllerResult<HttpResponse> {
349    let token = skip_authorize();
350    if !is_same_origin(&request, &app_conf) {
351        return Err(controller_err!(Forbidden, "Forbidden.".to_string()));
352    }
353    let mut conn = pool.acquire().await?;
354    let (digest, continuation) =
355        pending_continuation(&mut conn, &request, &jwt_key, payload.moocfi_course_id).await?;
356    if signed_in_account(&session, &continuation) != payload.expected_signed_in_account {
357        return Err(controller_err!(
358            BadRequest,
359            "The signed-in account has changed.".to_string()
360        ));
361    }
362    // The continuation outlives the handoff by minutes, long enough for a role to be granted.
363    if models::roles::has_any_role(&mut conn, continuation.user_id).await? {
364        return Err(controller_err!(Forbidden, "Forbidden.".to_string()));
365    }
366    // Spent in between by a concurrent continue.
367    let Some(continuation) =
368        moocfi_handoff_continuations::take(&mut conn, &digest, payload.moocfi_course_id).await?
369    else {
370        return Err(controller_err!(NotFound, NO_CONTINUATION.to_string()));
371    };
372    if session_user_id(&session) != Some(continuation.user_id) {
373        let user = models::users::get_by_id(&mut conn, continuation.user_id).await?;
374        session.clear();
375        session.renew();
376        authentication::remember(&session, user)?;
377    }
378    info!(outcome = "continued", "Continued a mooc.fi handoff");
379    token.authorized_ok(clearing_continuation(HttpResponse::Ok().json(
380        ContinuedMoocfiHandoff {
381            redirect_path: registration_page_path_of(
382                continuation.moocfi_course_id,
383                continuation.ui_language.as_deref(),
384            ),
385        },
386    )))
387}

389const NO_CONTINUATION: &str = "No sign-in is waiting.";

391/// The cookie's continuation for the course, if one is waiting for this request's IP, with the
392/// cookie token's digest that spends it.
393async fn pending_continuation(
394    conn: &mut PgConnection,
395    request: &HttpRequest,
396    jwt_key: &JwtKey,
397    moocfi_course_id: Uuid,
398) -> Result<(Vec<u8>, PendingContinuation), ControllerError> {
399    let not_found = || controller_err!(NotFound, NO_CONTINUATION.to_string());
400    let digest = request
401        .cookie(CONTINUATION_COOKIE)
402        .map(|cookie| Sha256::digest(cookie.value().as_bytes()).to_vec())
403        .ok_or_else(not_found)?;
404    let continuation = moocfi_handoff_continuations::get_pending(conn, &digest, moocfi_course_id)
405        .await?
406        .ok_or_else(not_found)?;
407    let is_bound = request_client_ip(request)
408        .is_some_and(|client_ip| client_ip_hash(client_ip, jwt_key) == continuation.client_ip_hash);
409    if !is_bound {
410        return Err(not_found());
411    }
412    Ok((digest, continuation))
413}

415fn signed_in_account(session: &Session, continuation: &PendingContinuation) -> SignedInAccount {
416    match session_user_id(session) {
417        Some(user_id) if user_id == continuation.user_id => SignedInAccount::Same,
418        Some(_) => SignedInAccount::Different,
419        None => SignedInAccount::None,
420    }
421}

430/// From a page of ours. Browsers without Fetch Metadata, such as Safari before 16.4, still send
431/// `Origin` on a POST.
432fn is_same_origin(request: &HttpRequest, app_conf: &ApplicationConfiguration) -> bool {
433    match sec_fetch_site(request) {
434        Some(site) => site == "same-origin",
435        None => request
436            .headers()
437            .get("Origin")
438            .and_then(|value| value.to_str().ok())
439            .is_some_and(|origin| {
440                url::Url::parse(&app_conf.base_url)
441                    .is_ok_and(|base_url| base_url.origin().ascii_serialization() == origin)
442            }),
443    }
444}

446/// A navigation from mooc.fi's own page. Same-site alone would admit any site under mooc.fi, so
447/// the `Referer` origin must also be mooc.fi's frontend. Same-origin is what dev's mock page sends.
448/// In test mode a missing `Sec-Fetch-Site` passes: Chromium sends none to the plain-http test site.
449fn is_from_moocfi_page(request: &HttpRequest, app_conf: &ApplicationConfiguration) -> bool {
450    let is_same_site = match sec_fetch_site(request) {
451        Some(site) => site == "same-site" || site == "same-origin",
452        None => app_conf.test_mode,
453    };
454    is_same_site
455        && request
456            .headers()
457            .get(header::REFERER)
458            .and_then(|value| value.to_str().ok())
459            .and_then(|referer| url::Url::parse(referer).ok())
460            .is_some_and(|referer| {
461                referer.origin().ascii_serialization()
462                    == app_conf.moocfi_configuration.frontend_origin()
463            })
464}

471async fn to_landing_with_continuation(
472    conn: &mut PgConnection,
473    claim: &MoocfiHandoffClaim,
474    reason: &str,
475    app_conf: &ApplicationConfiguration,
476) -> Result<HttpResponse, ControllerError> {
477    let continuation_token = models::library::oauth::tokens::generate_access_token();
478    moocfi_handoff_continuations::insert(
479        conn,
480        &Sha256::digest(continuation_token.as_bytes()),
481        &NewContinuation {
482            user_id: claim.user_id,
483            moocfi_course_id: claim.moocfi_course_id,
484            ui_language: claim.ui_language.as_deref(),
485            client_ip_hash: claim.client_ip_hash(),
486        },
487        CONTINUATION_LIFETIME,
488    )
489    .await?;
490    let mut response = redirect(&landing_page_path(claim.moocfi_course_id, reason));
491    response
492        .add_cookie(&continuation_cookie(continuation_token, app_conf))
493        .map_err(|error| {
494            ControllerError::new(
495                ControllerErrorType::InternalServerError,
496                "Failed to set the continuation cookie".to_string(),
497                Some(error.into()),
498            )
499        })?;
500    Ok(response)
501}

503/// Lax, unlike the Strict session cookie, so that it is set on a cross-site arrival and sent on the
504/// landing page's own requests, but never on another site's POST.
505fn continuation_cookie(value: String, app_conf: &ApplicationConfiguration) -> Cookie<'static> {
506    Cookie::build(CONTINUATION_COOKIE, value)
507        .path(CONTINUATION_COOKIE_PATH)
508        .http_only(true)
509        .secure(!app_conf.base_url.starts_with("http://"))
510        .same_site(SameSite::Lax)
511        .max_age(CookieDuration::seconds(CONTINUATION_LIFETIME.num_seconds()))
512        .finish()
513}
514
515/// Expires the continuation cookie. Sent wherever a session starts or ends, so a continuation
516/// left waiting never outlives the session it was shown beside.
517pub fn continuation_cookie_removal() -> Cookie<'static> {
518    let mut cookie = Cookie::build(CONTINUATION_COOKIE, "")
519        .path(CONTINUATION_COOKIE_PATH)
520        .finish();
521    cookie.make_removal();
522    cookie
523}
524
525fn clearing_continuation(mut response: HttpResponse) -> HttpResponse {
526    // Only fails on a cookie that cannot be a header value, which this one always can.
527    let _ = response.add_cookie(&continuation_cookie_removal());
528    response
529}

548fn fall_back_to_login(outcome: RedeemOutcome, claim: Option<&MoocfiHandoffClaim>) -> HttpResponse {
549    log_outcome(outcome);
550    let location = match claim {
551        Some(claim) => format!(
552            "/login?return_to={}",
553            pct_encode(&registration_page_path(claim))
554        ),
555        None => "/login".to_string(),
556    };
557    see_other(&location)
558}
