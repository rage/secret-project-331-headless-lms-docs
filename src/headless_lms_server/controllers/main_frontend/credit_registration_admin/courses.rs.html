49/// What the configuration check concluded about one module, freshly derived from the same facts and
50/// the same rule the `config-validation` phase uses, with the course code verdict Suotar gave that

62/// One row of the Courses tab: a course module, a mooc.fi code, or a mooc.fi target with no live

73    /// The course's name, or "mooc.fi: " and the target's name.
74    pub course_name: String,
75    pub course_module_id: Option<Uuid>,
76    /// The module's name, or the mooc.fi code's.
77    pub course_module_name: Option<String>,
78    /// The target's configuration, for a mooc.fi code only.
79    pub external_credit_target: Option<ExternalCreditTarget>,
80    /// The code's configuration; `None` on a module and on a target with no live codes.
81    pub external_credit_target_code: Option<ExternalCreditTargetCode>,
82    pub uh_course_code: Option<String>,
83    pub ects_credits: Option<f32>,

86    pub paused_at: Option<DateTime<Utc>>,
87    pub pause_reason: Option<String>,

100    /// the push path. For a mooc.fi code, exactly the completions it is responsible for, withdrawn
101    /// ones left out.
102    pub eligible_completion_count: i64,
103    pub registration_count: i64,
104    /// Registered, already in Sisu and better grade in Sisu: the denominator the failure rate is
105    /// taken against, with `failed_count`.
106    pub success_count: i64,
107    /// Of `success_count`, registered by us.
108    pub registered_count: i64,
109    /// Of `success_count`, already in Sisu or with a better grade there.
110    pub already_in_sisu_count: i64,
111    pub in_flight_count: i64,
112    pub failed_count: i64,
113    /// The row's registrations counted in Needs attention: the Registrations list with
114    /// `course_module_id` and `needs_attention=true` holds exactly these.
115    pub needs_attention_count: i64,
116    /// The module's "Where registrations stand", Not started left out; each count equals the
117    /// Registrations list filtered by its step, engagement and `course_module_id`.
118    pub where_registrations_stand: Vec<CreditRegistrationStepCount>,
119    pub last_registered_at: Option<DateTime<Utc>>,
120    pub top_error_code: Option<CreditRegistrationErrorCode>,
121}

252pub async fn get_credit_registration_course_summary(
253    user: AuthUser,
254    pool: web::Data<PgPool>,
255    course_or_target_id: web::Path<Uuid>,
256) -> ControllerResult<web::Json<AdminCreditRegistrationCourseSummary>> {
257    let mut conn = pool.acquire().await?;
258    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
259    let course_or_target_id = course_or_target_id.into_inner();
260
261    let summary = match get_course_or_target(&mut conn, course_or_target_id).await? {
262        CourseOrTarget::Course(course) => {
263            let credit_units = models::course_modules::get_by_course_id(&mut conn, course.id)
264                .await?
265                .iter()
266                .filter(|module| {
267                    models::course_modules::registers_credits(
268                        module.uh_course_code.as_deref(),
269                        module.credit_registration_route,
270                    )
271                })
272                .map(AdminCreditUnitCourseCode::from)
273                .collect();
274            AdminCreditRegistrationCourseSummary {
275                course_id: Some(course.id),
276                course_name: course.name,
277                credit_units,
278            }
279        }
280        CourseOrTarget::Target(target) => AdminCreditRegistrationCourseSummary {
281            course_id: None,
282            course_name: target.display_name(),
283            credit_units: external_credit_target_codes::get_by_target_id(&mut conn, target.id)
284                .await?
285                .iter()
286                .map(AdminCreditUnitCourseCode::from)
287                .collect(),
288        },
289    };
290
291    token.authorized_ok(web::Json(summary))
292}

308pub async fn get_credit_registration_stats_by_course(
309    user: AuthUser,
310    pool: web::Data<PgPool>,
311    app_conf: web::Data<ApplicationConfiguration>,
312) -> ControllerResult<web::Json<CreditRegistrationStatsByCourse>> {
313    let mut conn = pool.acquire().await?;
314    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
315
316    let overviews =
317        course_module_suotar_configurations::get_module_overviews(&mut conn, COURSES_LIMIT).await?;
318    let mut checks: HashMap<Uuid, CreditRegistrationCourseConfigCheck> =
319        get_config_facts_for_modules_with_course_code(&mut conn, None)
320            .await?
321            .into_iter()
322            .map(|facts| {
323                let check = check_module_config(&facts, CourseCodeVerdict::stored(&facts).as_ref());
324                (facts.course_module_id, config_check_stats(check))
325            })
326            .collect();
327    let mut totals: HashMap<Uuid, _> = credit_registrations::count_by_credit_unit(&mut conn)
328        .await?
329        .into_iter()
330        .map(|row| (row.credit_unit_id, row))
331        .collect();
332
333    let rules = attention_rules(&mut conn, &app_conf).await?;
334    let mut needs_attention: HashMap<Uuid, i64> = HashMap::new();
335    for row in credit_registrations::get_attention_items(&mut conn, &rules, None).await? {
336        if row.standing(&rules.blocking) == AttentionStanding::NeedsAttention {
337            *needs_attention.entry(row.credit_unit_id).or_insert(0) += 1;
338        }
339    }
340    let mut step_counts: HashMap<Uuid, Vec<StepCount>> = HashMap::new();
341    for row in credit_registrations::count_by_step_and_engagement(
342        &mut conn,
343        app_conf.suotar_configuration.account_linking_since,
344    )
345    .await?
346    {
347        step_counts.entry(row.credit_unit_id).or_default().push(row);
348    }
349
350    let mut unit_stats = |stats: &mut CreditRegistrationCourseStats| {
351        stats.needs_attention_count = needs_attention.remove(&stats.credit_unit_id).unwrap_or(0);
352        stats.where_registrations_stand =
353            sum_step_counts(step_counts.remove(&stats.credit_unit_id).iter().flatten());
354    };
355
356    let mut credit_units: Vec<CreditRegistrationCourseStats> = overviews
357        .into_iter()
358        .map(|overview| {
359            let module_id = overview.course_module_id;
360            let mut stats = to_course_stats(
361                overview,
362                checks.remove(&module_id).unwrap_or(NO_CHECK),
363                totals.remove(&module_id),
364            );
365            unit_stats(&mut stats);
366            stats
367        })
368        .collect();
369
370    let mut eligible_counts =
371        external_credit_target_codes::count_eligible_completions_by_code(&mut conn).await?;
372    let targets: HashMap<Uuid, ExternalCreditTarget> = external_credit_targets::get_all(&mut conn)
373        .await?
374        .into_iter()
375        .map(|target| (target.id, target))
376        .collect();
377    let codes = external_credit_target_codes::get_all(&mut conn).await?;
378    let targets_with_codes: HashSet<Uuid> = codes
379        .iter()
380        .map(|code| code.external_credit_target_id)
381        .collect();
382    for code in codes {
383        let Some(target) = targets.get(&code.external_credit_target_id) else {
384            continue;
385        };
386        // As the config-validation phase does, only codes of enabled targets are checked.
387        let check = if target.enabled_at.is_some() {
388            let facts = code.config_facts();
389            config_check_stats(check_target_code_config(
390                &facts,
391                CourseCodeVerdict::stored_for_target_code(&facts).as_ref(),
392            ))
393        } else {
394            NO_CHECK
395        };
396        let id = code.id;
397        let mut stats = target_code_course_stats(
398            target.clone(),
399            code,
400            check,
401            eligible_counts.remove(&id).unwrap_or(0),
402            totals.remove(&id),
403        );
404        unit_stats(&mut stats);
405        credit_units.push(stats);
406    }
407    // Listed so a target whose codes are all gone can still be opened to add one.
408    let mut codeless_targets: Vec<ExternalCreditTarget> = targets
409        .into_values()
410        .filter(|target| !targets_with_codes.contains(&target.id))
411        .collect();
412    codeless_targets.sort_by(|a, b| a.name.cmp(&b.name));
413    credit_units.extend(
414        codeless_targets
415            .into_iter()
416            .map(codeless_target_course_stats),
417    );
418
419    token.authorized_ok(web::Json(CreditRegistrationStatsByCourse {
420        misconfigured_count: credit_units
421            .iter()
422            .filter(|row| row.check.message.is_some())
423            .count() as i64,
424        credit_units,
425    }))
426}

428/**
429POST `/api/v0/main-frontend/credit-registration-admin/courses/{course_module_id}/pause` - Stops every
430phase from claiming this module's rows.
431
432Freezes the rows where they stand rather than cancelling them, so resuming picks up what was already
433in flight.
434*/
435#[instrument(skip(pool, payload))]
436#[utoipa::path(
437    post,
438    path = "/courses/{course_module_id}/pause",
439    operation_id = "adminPauseCourseModuleCreditRegistration",
440    tag = "credit-registration-admin",
441    params(("course_module_id" = Uuid, Path, description = "Course module id")),
442    request_body = AdminPauseCourseModulePayload,
443    responses(
444        (status = 200, description = "Paused"),
445        (status = 422, description = "No reason given, or the module has no Suotar configuration")
446    )
447)]
448pub async fn admin_pause_course_module_credit_registration(
449    user: AuthUser,
450    pool: web::Data<PgPool>,
451    course_module_id: web::Path<Uuid>,
452    payload: web::Json<AdminPauseCourseModulePayload>,
453) -> ControllerResult<web::Json<bool>> {
454    let mut conn = pool.acquire().await?;
455    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
456
457    let reason = required_reason(&payload.reason)?;
458    let module_id = *course_module_id;
459    require_suotar_configuration(&mut conn, module_id).await?;
460
461    let mut tx = conn.begin().await?;
462    course_module_suotar_configurations::set_paused(
463        &mut tx,
464        module_id,
465        Some(course_module_suotar_configurations::SuotarPause {
466            paused_at: Utc::now(),
467            paused_by_user_id: user.id,
468            reason: Some(reason),
469        }),
470    )
471    .await?;
472    record_module_action(
473        &mut tx,
474        CreditRegistrationAdminAction::PauseCourseModule,
475        module_id,
476        user.id,
477        Some(reason),
478    )
479    .await?;
480    tx.commit().await?;
481
482    token.authorized_ok(web::Json(true))
483}

529/// A module with no configuration row has nothing to pause: `set_paused` would update no rows and
530/// the call would report a pause that never happened.
531async fn require_suotar_configuration(
532    conn: &mut PgConnection,
533    course_module_id: Uuid,
534) -> Result<(), ControllerError> {
535    if course_module_suotar_configurations::exists(conn, course_module_id).await? {
536        return Ok(());
537    }
538    Err(controller_err!(
539        BadRequest,
540        "This course module has no credit registration configuration.".to_string()
541    ))
542}

581        course_module_id: Some(overview.course_module_id),
582        course_module_name: overview.course_module_name,

584        external_credit_target_code: None,
585        uh_course_code: overview.uh_course_code,
586        ects_credits: overview.ects_credits,
587        enrolment_link: overview.enrolment_link,
588        paused_at: overview.paused_at,
589        pause_reason: overview.pause_reason,
590        last_listed_at: overview.last_listed_at,
591        check,

966- Replaces the course codes of a mooc.fi target in one transaction: adds, updates and deletes
967codes so the target has exactly the ones sent. Deleting a code withdraws its completions not yet
968submitted.
969*/
970#[instrument(skip(pool, payload))]
971#[utoipa::path(
972    put,
973    path = "/courses/external-targets/{external_credit_target_id}/codes",
974    operation_id = "adminSaveExternalCreditTargetCodes",
975    tag = "credit-registration-admin",
976    params(("external_credit_target_id" = Uuid, Path, description = "mooc.fi target id")),
977    request_body = AdminExternalCreditTargetCodesPayload,
978    responses(
979        (status = 200, description = "Saved"),
980        (status = 422, description = "No reason given, an invalid configuration, a course code twice, or a prerequisite that is not another code sent, has a higher minimum tier, or leads round in a circle")
981    )
982)]
983pub async fn admin_save_external_credit_target_codes(
984    user: AuthUser,
985    pool: web::Data<PgPool>,
986    external_credit_target_id: web::Path<Uuid>,
987    payload: web::Json<AdminExternalCreditTargetCodesPayload>,
988) -> ControllerResult<web::Json<bool>> {
989    let mut conn = pool.acquire().await?;
990    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
991
992    let reason = required_reason(&payload.reason)?;
993    let target_id = *external_credit_target_id;
994    let codes = payload
995        .codes
996        .iter()
997        .map(|code| Ok((code.id, code_configuration(code)?)))
998        .collect::<Result<Vec<_>, ControllerError>>()?;
999    require_valid_prerequisites(&codes)?;
1000
1001    let mut tx = conn.begin().await?;
1002    external_credit_targets::get_by_id_for_update(&mut tx, target_id).await?;
1003    let live = external_credit_target_codes::get_by_target_id(&mut tx, target_id).await?;
1004    let live_by_id: HashMap<Uuid, &ExternalCreditTargetCode> =
1005        live.iter().map(|code| (code.id, code)).collect();
1006    let wanted: HashMap<Uuid, &ExternalCreditTargetCodeConfiguration> = codes
1007        .iter()
1008        .map(|(id, configuration)| (*id, configuration))
1009        .collect();
1010    let new_ids: Vec<Uuid> = codes
1011        .iter()
1012        .map(|(id, _)| *id)
1013        .filter(|id| !live_by_id.contains_key(id))
1014        .collect();
1015    if !external_credit_target_codes::get_by_ids_including_deleted(&mut tx, &new_ids)
1016        .await?
1017        .is_empty()
1018    {
1019        return Err(controller_err!(
1020            BadRequest,
1021            "A code sent belongs to another course or has been deleted.".to_string()
1022        ));
1023    }
1024
1025    // Prerequisites that change are cleared first and set last, so no write in between sees a
1026    // cycle, a dependent of a deleted code, or a prerequisite not added yet.
1027    for code in &live {
1028        let wanted_prerequisite = wanted
1029            .get(&code.id)
1030            .and_then(|configuration| configuration.prerequisite_code_id);
1031        if code.prerequisite_code_id.is_some() && code.prerequisite_code_id != wanted_prerequisite {
1032            external_credit_target_codes::clear_prerequisite(&mut tx, code.id).await?;
1033        }
1034    }
1035    let audit = AuditCodeChange {
1036        target_id,
1037        actor_user_id: user.id,
1038        reason,
1039    };
1040    for code in live.iter().filter(|code| !wanted.contains_key(&code.id)) {
1041        external_credit_target_codes::delete(&mut tx, code.id).await?;
1042        audit
1043            .record(&mut tx, serde_json::json!({ "change": "delete_code", "external_credit_target_code_id": code.id, "before": code }))
1044            .await?;
1045    }
1046    for (id, configuration) in codes.iter().filter(|(id, _)| new_ids.contains(id)) {
1047        let without_prerequisite = ExternalCreditTargetCodeConfiguration {
1048            prerequisite_code_id: None,
1049            ..configuration.clone()
1050        };
1051        external_credit_target_codes::insert(
1052            &mut tx,
1053            PKeyPolicy::Fixed(*id),
1054            target_id,
1055            &without_prerequisite,
1056        )
1057        .await?;
1058        audit
1059            .record(&mut tx, serde_json::json!({ "change": "create_code", "external_credit_target_code_id": id, "after": configuration }))
1060            .await?;
1061    }
1062    for (id, configuration) in &codes {
1063        let before = live_by_id.get(id);
1064        let is_new = before.is_none();
1065        if before.is_some_and(|code| code.configuration() == *configuration)
1066            || (is_new && configuration.prerequisite_code_id.is_none())
1067        {
1068            continue;
1069        }
1070        external_credit_target_codes::update(&mut tx, *id, configuration).await?;
1071        if let Some(before) = before {
1072            audit
1073                .record(&mut tx, serde_json::json!({ "change": "update_code", "external_credit_target_code_id": id, "before": before, "after": configuration }))
1074                .await?;
1075        }
1076    }
1077    tx.commit().await?;
1078
1079    token.authorized_ok(web::Json(true))
1080}

1082/// The audit record of one code change in [`admin_save_external_credit_target_codes`].
1083struct AuditCodeChange<'a> {
1084    target_id: Uuid,
1085    actor_user_id: Uuid,
1086    reason: &'a str,
1087}
1088
1089impl AuditCodeChange<'_> {
1090    async fn record(
1091        &self,
1092        tx: &mut PgConnection,
1093        details: serde_json::Value,
1094    ) -> Result<(), ControllerError> {
1095        record_target_action(
1096            tx,
1097            CreditRegistrationAdminAction::ConfigureExternalCreditTarget,
1098            self.target_id,
1099            self.actor_user_id,
1100            Some(self.reason),
1101            Some(details),
1102        )
1103        .await
1104    }
1105}

1107/// Refuses a code sent twice, and a prerequisite that is not another of `codes`, that a completion
1108/// earning the dependent may not earn, or that leads round in a circle.
1109fn require_valid_prerequisites(
1110    codes: &[(Uuid, ExternalCreditTargetCodeConfiguration)],
1111) -> Result<(), ControllerError> {
1112    let find = |id: Uuid| codes.iter().find(|(other, _)| *other == id);
1113    for (index, (id, configuration)) in codes.iter().enumerate() {
1114        if codes[..index].iter().any(|(other, _)| other == id) {
1115            return Err(controller_err!(
1116                BadRequest,
1117                "The same code was sent twice.".to_string()
1118            ));
1119        }
1120        let Some(prerequisite_code_id) = configuration.prerequisite_code_id else {
1121            continue;
1122        };
1123        let Some((_, prerequisite)) =
1124            find(prerequisite_code_id).filter(|_| prerequisite_code_id != *id)
1125        else {
1126            return Err(controller_err!(
1127                BadRequest,
1128                "The prerequisite must be another code of the same course.".to_string()
1129            ));
1130        };
1131        // A completion that earns the dependent must be able to earn the prerequisite too.
1132        if !external_credit_target_codes::is_tier_earned(
1133            prerequisite.minimum_tier,
1134            configuration.minimum_tier,
1135        ) {
1136            return Err(bad_request_with_reason(
1137                BadRequestReason::ExternalCreditTargetCodePrerequisiteTierAbove,
1138                "A prerequisite cannot need a higher tier than the code that waits for it."
1139                    .to_string(),
1140            ));
1141        }
1142        let mut next = Some(prerequisite_code_id);
1143        // Bounded: a cycle that misses this code is refused when one of its own codes is checked.
1144        for _ in 0..=codes.len() {
1145            let Some(step) = next else {
1146                break;
1147            };
1148            if step == *id {
1149                return Err(bad_request_with_reason(
1150                    BadRequestReason::ExternalCreditTargetCodePrerequisitesCircular,
1151                    "The prerequisites of this course go round in a circle.".to_string(),
1152                ));
1153            }
1154            next = find(step).and_then(|(_, code)| code.prerequisite_code_id);
1155        }
1156    }
1157    Ok(())
1158}

1160/// The payload as a code configuration to store, trimmed and checked.
1161fn code_configuration(
1162    payload: &AdminExternalCreditTargetCodePayload,
1163) -> Result<ExternalCreditTargetCodeConfiguration, ControllerError> {
1164    if !(payload.ects_credits.is_finite() && payload.ects_credits > 0.0) {
1165        return Err(controller_err!(
1166            BadRequest,
1167            "ECTS credits must be more than zero.".to_string()
1168        ));
1169    }
1170    if payload.minimum_tier.is_some_and(|tier| tier < 0) {
1171        return Err(controller_err!(
1172            BadRequest,
1173            "The minimum tier cannot be negative.".to_string()
1174        ));
1175    }
1176    let uh_course_code = required_text(&payload.uh_course_code, "The course code")?;
1177    let mut equivalent_course_codes: Vec<String> = Vec::new();
1178    for equivalent in &payload.equivalent_course_codes {
1179        let equivalent = equivalent.trim();
1180        if equivalent.is_empty()
1181            || equivalent.eq_ignore_ascii_case(&uh_course_code)
1182            || equivalent_course_codes
1183                .iter()
1184                .any(|kept| kept.eq_ignore_ascii_case(equivalent))
1185        {
1186            continue;
1187        }
1188        equivalent_course_codes.push(equivalent.to_string());
1189    }
1190    Ok(ExternalCreditTargetCodeConfiguration {
1191        name: required_text(&payload.name, "The name")?,
1192        uh_course_code,
1193        minimum_tier: payload.minimum_tier,
1194        ects_credits: payload.ects_credits,
1195        grading: payload.grading,
1196        enrolment_link: enrolment_link(payload.enrolment_link.as_deref())?,
1197        prerequisite_code_id: payload.prerequisite_code_id,
1198        equivalent_course_codes,
1199    })
1200}

1202/// `value` trimmed; refused when blank, naming `field` in the error.
1203fn required_text(value: &str, field: &str) -> Result<String, ControllerError> {
1204    let trimmed = value.trim();
1205    if trimmed.is_empty() {
1206        return Err(controller_err!(BadRequest, format!("{field} is required.")));
1207    }
1208    Ok(trimmed.to_string())
1209}

1211/// The link trimmed, `None` when blank. Students open it as a link, where a `javascript:` URL
1212/// would run as our page.
1213fn enrolment_link(link: Option<&str>) -> Result<Option<String>, ControllerError> {
1214    let Some(link) = link.map(str::trim).filter(|link| !link.is_empty()) else {
1215        return Ok(None);
1216    };
1217    if !url::Url::parse(link).is_ok_and(|url| url.scheme() == "https" && url.has_host()) {
1218        return Err(controller_err!(
1219            BadRequest,
1220            "The enrolment link must be a web address that starts with https://.".to_string()
1221        ));
1222    }
1223    Ok(Some(link.to_string()))
1224}

1226/// The payload as a configuration to store, trimmed and checked. `existing` keeps the time an
1227/// already enabled target was switched on.
1228fn target_configuration(
1229    payload: &AdminExternalCreditTargetPayload,
1230    existing: Option<&ExternalCreditTarget>,
1231) -> Result<ExternalCreditTargetConfiguration, ControllerError> {
1232    let language_code = required_text(&payload.language_code, "The language")?;
1233    if !is_ietf_language_code_like(&language_code) {
1234        return Err(controller_err!(
1235            BadRequest,
1236            "The language must be a language code such as fi or en-US.".to_string()
1237        ));
1238    }
1239    let moocfi_course_slug = required_text(&payload.moocfi_course_slug, "The mooc.fi course slug")?;
1240    if !is_plain_slug(&moocfi_course_slug) {
1241        return Err(controller_err!(
1242            BadRequest,
1243            "The mooc.fi course slug may only have letters, numbers, - and _.".to_string()
1244        ));
1245    }
1246    Ok(ExternalCreditTargetConfiguration {
1247        moocfi_course_id: payload.moocfi_course_id,
1248        moocfi_course_slug,
1249        name: required_text(&payload.name, "The name")?,
1250        language_code,
1251        delegated_at: payload.delegated_at,
1252        enabled_at: payload.enabled.then(|| {
1253            existing
1254                .and_then(|target| target.enabled_at)
1255                .unwrap_or_else(Utc::now)
1256        }),
1257    })
1258}
1259
1260/// Whether `slug` can be put in a mooc.fi path as it is.
1261fn is_plain_slug(slug: &str) -> bool {
1262    slug.len() <= 255
1263        && slug
1264            .bytes()
1265            .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_')
1266}
