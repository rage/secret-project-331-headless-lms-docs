20/// Suotar-enabled modules, comfortably above any real deployment's count. The only admin tab
21/// without a page or a limit of its own before this.
22const COURSES_LIMIT: i64 = 2_000;
23
24/// What the configuration check concluded about one module, freshly derived from the same facts and
25/// the same rule the `config-validation` phase uses.
26///
27/// `course_code_resolves` is `None` while no listing has been attempted: never checked is not the
28/// same as checked and failed, and the two must not render alike.
29#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
30pub struct CreditRegistrationCourseConfigCheck {
31    pub course_code_resolves: Option<bool>,
32    pub product_token_found: Option<bool>,
33    /// Every problem found, in one line. `None` means the module is fine.
34    pub message: Option<String>,
35}

37#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
38pub struct CreditRegistrationCourseStats {
39    pub course_id: Uuid,
40    pub course_name: String,
41    pub course_module_id: Uuid,
42    pub course_module_name: Option<String>,
43    pub uh_course_code: Option<String>,
44    pub ects_credits: Option<f32>,
45    pub open_university_product_id: Option<String>,
46    /// The module's override; `None` means the scale is derived from the completion.
47    pub grade_scale_id: Option<String>,
48    /// The old pull path is on as well, which would register the same completion twice.
49    pub old_flow_also_enabled: bool,
50    pub paused_at: Option<DateTime<Utc>>,
51    pub pause_reason: Option<String>,
52    pub active_realisation_count: i64,
53    pub last_listed_at: Option<DateTime<Utc>>,
54    /// What the current facts say. Recomputed on read, so a configuration fixed a minute ago no
55    /// longer shows as broken.
56    pub check: CreditRegistrationCourseConfigCheck,
57    /// When the `config-validation` phase last stamped its verdict on the row. `None` means never,
58    /// which the stored verdict beside it cannot express on its own.
59    pub config_checked_at: Option<DateTime<Utc>>,
60    /// The verdict as the phase stored it, which may be older than `check`.
61    pub stored_config_check_message: Option<String>,
62    /// Every passed, ECTS-eligible completion on the module, whichever path owns it. Wider than
63    /// what `materialize` takes, which is only the ones carrying `register_credits_via_suotar`, so
64    /// a module opted in mid-course keeps a permanent gap against `registration_count` for the
65    /// completions that predate the opt-in.
66    pub eligible_completion_count: i64,
67    pub registration_count: i64,
68    pub success_count: i64,
69    pub in_flight_count: i64,
70    pub failed_count: i64,
71    pub needs_admin_attention_count: i64,
72    pub last_registered_at: Option<DateTime<Utc>>,
73    pub top_error_code: Option<CreditRegistrationErrorCode>,
74}

93/**
94GET `/api/v0/main-frontend/credit-registration-admin/courses` - Every Suotar-enabled course module,
95its validated configuration and its volumes.
96*/
97#[instrument(skip(pool))]
98#[utoipa::path(
99    get,
100    path = "/courses",
101    operation_id = "getCreditRegistrationStatsByCourse",
102    tag = "credit-registration-admin",
103    responses(
104        (status = 200, description = "One row per enabled course module", body = CreditRegistrationStatsByCourse)
105    )
106)]
107pub async fn get_credit_registration_stats_by_course(
108    user: AuthUser,
109    pool: web::Data<PgPool>,
110) -> ControllerResult<web::Json<CreditRegistrationStatsByCourse>> {
111    let mut conn = pool.acquire().await?;
112    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
113
114    let overviews =
115        course_module_suotar_configurations::get_module_overviews(&mut conn, COURSES_LIMIT).await?;
116    let mut checks: HashMap<Uuid, CreditRegistrationCourseConfigCheck> =
117        get_config_facts_for_enabled_modules(&mut conn, None)
118            .await?
119            .into_iter()
120            .map(|facts| {
121                let check = check_module_config(&facts);
122                (
123                    facts.course_module_id,
124                    CreditRegistrationCourseConfigCheck {
125                        course_code_resolves: check.course_code_resolves,
126                        product_token_found: check.product_token_found,
127                        message: check.message,
128                    },
129                )
130            })
131            .collect();
132    let mut totals: HashMap<Uuid, _> = credit_registrations::count_by_module(&mut conn)
133        .await?
134        .into_iter()
135        .map(|row| (row.course_module_id, row))
136        .collect();
137
138    let modules: Vec<CreditRegistrationCourseStats> = overviews
139        .into_iter()
140        .map(|overview| {
141            let module_id = overview.course_module_id;
142            to_course_stats(
143                overview,
144                checks
145                    .remove(&module_id)
146                    .unwrap_or(CreditRegistrationCourseConfigCheck {
147                        course_code_resolves: None,
148                        product_token_found: None,
149                        message: None,
150                    }),
151                totals.remove(&module_id),
152            )
153        })
154        .collect();
155
156    token.authorized_ok(web::Json(CreditRegistrationStatsByCourse {
157        misconfigured_count: modules
158            .iter()
159            .filter(|row| row.check.message.is_some())
160            .count() as i64,
161        modules,
162    }))
163}

165/**
166POST `/api/v0/main-frontend/credit-registration-admin/courses/{course_module_id}/pause` - Stops every
167phase from claiming this module's rows.
168
169Freezes the rows where they stand rather than cancelling them, so resuming picks up what was already
170in flight.
171*/
172#[instrument(skip(pool, payload))]
173#[utoipa::path(
174    post,
175    path = "/courses/{course_module_id}/pause",
176    operation_id = "adminPauseCourseModuleCreditRegistration",
177    tag = "credit-registration-admin",
178    params(("course_module_id" = Uuid, Path, description = "Course module id")),
179    request_body = AdminPauseCourseModulePayload,
180    responses(
181        (status = 200, description = "Paused"),
182        (status = 422, description = "No reason given, or the module has no Suotar configuration")
183    )
184)]
185pub async fn admin_pause_course_module_credit_registration(
186    user: AuthUser,
187    pool: web::Data<PgPool>,
188    course_module_id: web::Path<Uuid>,
189    payload: web::Json<AdminPauseCourseModulePayload>,
190) -> ControllerResult<web::Json<bool>> {
191    let mut conn = pool.acquire().await?;
192    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
193
194    let reason = required_reason(&payload.reason)?;
195    let module_id = *course_module_id;
196    require_suotar_configuration(&mut conn, module_id).await?;
197
198    let mut tx = conn.begin().await?;
199    course_module_suotar_configurations::set_paused(
200        &mut tx,
201        module_id,
202        Some(course_module_suotar_configurations::SuotarPause {
203            paused_at: Utc::now(),
204            paused_by_user_id: user.id,
205            reason: Some(reason),
206        }),
207    )
208    .await?;
209    record_module_action(
210        &mut tx,
211        CreditRegistrationAdminAction::PauseCourseModule,
212        module_id,
213        user.id,
214        Some(reason),
215    )
216    .await?;
217    tx.commit().await?;
218
219    token.authorized_ok(web::Json(true))
220}

222/**
223POST `/api/v0/main-frontend/credit-registration-admin/courses/{course_module_id}/resume` - Lets the
224phases claim this module's rows again.
225*/
226#[instrument(skip(pool, payload))]
227#[utoipa::path(
228    post,
229    path = "/courses/{course_module_id}/resume",
230    operation_id = "adminResumeCourseModuleCreditRegistration",
231    tag = "credit-registration-admin",
232    params(("course_module_id" = Uuid, Path, description = "Course module id")),
233    request_body = AdminResumeCourseModulePayload,
234    responses(
235        (status = 200, description = "Resumed"),
236        (status = 422, description = "The module has no Suotar configuration")
237    )
238)]
239pub async fn admin_resume_course_module_credit_registration(
240    user: AuthUser,
241    pool: web::Data<PgPool>,
242    course_module_id: web::Path<Uuid>,
243    payload: web::Json<AdminResumeCourseModulePayload>,
244) -> ControllerResult<web::Json<bool>> {
245    let mut conn = pool.acquire().await?;
246    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
247
248    let module_id = *course_module_id;
249    require_suotar_configuration(&mut conn, module_id).await?;
250
251    let mut tx = conn.begin().await?;
252    course_module_suotar_configurations::set_paused(&mut tx, module_id, None).await?;
253    record_module_action(
254        &mut tx,
255        CreditRegistrationAdminAction::ResumeCourseModule,
256        module_id,
257        user.id,
258        payload.reason.as_deref(),
259    )
260    .await?;
261    tx.commit().await?;
262
263    token.authorized_ok(web::Json(true))
264}

266/// A module with no configuration row has nothing to pause: `set_paused` would update no rows and
267/// the call would report a pause that never happened.
268async fn require_suotar_configuration(
269    conn: &mut PgConnection,
270    course_module_id: Uuid,
271) -> Result<(), ControllerError> {
272    if course_module_suotar_configurations::exists(conn, course_module_id).await? {
273        return Ok(());
274    }
275    Err(controller_err!(
276        BadRequest,
277        "This course module has no credit registration configuration.".to_string()
278    ))
279}

281async fn record_module_action(
282    tx: &mut PgConnection,
283    action: CreditRegistrationAdminAction,
284    course_module_id: Uuid,
285    actor_user_id: Uuid,
286    reason: Option<&str>,
287) -> Result<(), ControllerError> {
288    models::credit_registration_admin_actions::record(
289        tx,
290        &NewCreditRegistrationAdminAction {
291            target_id: Some(course_module_id),
292            reason: reason.map(str::to_string),
293            ..NewCreditRegistrationAdminAction::new(
294                action,
295                CreditRegistrationAdminActionTarget::CourseModule,
296                actor_user_id,
297                GLOBAL_ADMIN_ROLE,
298            )
299        },
300    )
301    .await?;
302    Ok(())
303}

305fn to_course_stats(
306    overview: SuotarModuleOverview,
307    check: CreditRegistrationCourseConfigCheck,
308    totals: Option<credit_registrations::ModuleRegistrationTotals>,
309) -> CreditRegistrationCourseStats {
310    CreditRegistrationCourseStats {
311        course_id: overview.course_id,
312        course_name: overview.course_name,
313        course_module_id: overview.course_module_id,
314        course_module_name: overview.course_module_name,
315        uh_course_code: overview.uh_course_code,
316        ects_credits: overview.ects_credits,
317        open_university_product_id: overview.open_university_product_id,
318        grade_scale_id: overview.grade_scale_id,
319        old_flow_also_enabled: overview.old_flow_also_enabled,
320        paused_at: overview.paused_at,
321        pause_reason: overview.pause_reason,
322        active_realisation_count: overview.active_realisation_count,
323        last_listed_at: overview.last_listed_at,
324        check,
325        config_checked_at: overview.config_checked_at,
326        stored_config_check_message: overview.config_check_message,
327        eligible_completion_count: overview.eligible_completion_count,
328        registration_count: totals.as_ref().map_or(0, |row| row.total_count),
329        success_count: totals.as_ref().map_or(0, |row| row.success_count),
330        in_flight_count: totals.as_ref().map_or(0, |row| row.in_flight_count),
331        failed_count: totals.as_ref().map_or(0, |row| row.failed_count),
332        needs_admin_attention_count: totals
333            .as_ref()
334            .map_or(0, |row| row.needs_admin_attention_count),
335        last_registered_at: totals.as_ref().and_then(|row| row.last_registered_at),
336        top_error_code: totals.and_then(|row| row.top_error_code),
337    }
338}
