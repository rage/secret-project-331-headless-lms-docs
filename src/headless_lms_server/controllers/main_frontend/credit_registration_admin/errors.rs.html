21/// Rows per page of the attention queue when the caller names no limit.
22const ATTENTION_PAGE_SIZE: u32 = 50;
23const DEFAULT_ERROR_WINDOW_SECS: i64 = 24 * 60 * 60;
24const MAX_ERROR_WINDOW_SECS: i64 = 90 * 24 * 60 * 60;

26#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
27pub struct CreditRegistrationAttentionItem {
28    pub credit_registration_id: Uuid,
29    pub user_id: Uuid,
30    pub first_name: Option<String>,
31    pub last_name: Option<String>,
32    /// In full: this is the list support works from.
33    pub email: Option<String>,
34    pub course_id: Uuid,
35    pub course_name: String,
36    pub course_module_id: Uuid,
37    pub course_module_name: Option<String>,
38    pub state: CreditRegistrationState,
39    pub state_entered_at: DateTime<Utc>,
40    pub error_code: Option<CreditRegistrationErrorCode>,
41    pub attempt_count: i32,
42    pub next_attempt_at: DateTime<Utc>,
43    pub student_number: Option<String>,
44    /// Every detector that picked this row, so the table can group by any of them. Empty on a row
45    /// the pipeline flagged that no detector explains.
46    pub reasons: Vec<AttentionReason>,
47    /// The pipeline's cached "a human should look at this". A fact about the row, never a reason:
48    /// it says nothing about why, so it travels beside `reasons` rather than in them.
49    pub needs_admin_attention: bool,
50}

58#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
59pub struct CreditRegistrationAttentionItems {
60    /// The requested page of the queue.
61    pub items: Vec<CreditRegistrationAttentionItem>,
62    /// The whole queue, whatever this request filtered to: the canonical "needs a human" count, the
63    /// same number `/overview` reports and the tab badge shows.
64    pub total_count: i64,
65    /// Rows matching this request's narrowing, which is what `total_pages` pages through. Equal to
66    /// `total_count` when neither `reason` nor `without_reason` was given.
67    pub filtered_count: i64,
68    pub total_pages: u32,
69    /// Over the whole queue, not over the page or the filter, so the counts stay usable as facets.
70    pub counts_by_reason: Vec<CreditRegistrationAttentionReasonCount>,
71    /// Queue rows no detector picked, which the pipeline's flag alone put there. No `reason`
72    /// reaches them, so a surface that groups by reason has to offer `without_reason` beside the
73    /// reasons or leave this many rows unreachable.
74    pub flagged_without_reason_count: i64,
75}

153/**
154GET `/api/v0/main-frontend/credit-registration-admin/attention` - A page of the rows at least one
155detector wants a human to look at, with the detectors that picked each.
156
157Superseded attempts are outside every detector: acting on a replaced attempt is never right.
158`total_count` is the queue's length under the one definition of "needs a human"; `/overview`'s
159`needs_admin_attention_count` is the same number.
160*/
161#[instrument(skip(pool))]
162#[utoipa::path(
163    get,
164    path = "/attention",
165    operation_id = "getCreditRegistrationAttentionItems",
166    tag = "credit-registration-admin",
167    params(
168        ("page" = Option<u32>, Query, description = "Page number, from 1"),
169        ("limit" = Option<u32>, Query, description = "Rows per page"),
170        ("reason" = Option<Vec<AttentionReason>>, Query, description = "Only rows one of these detectors picked; repeat the parameter for several"),
171        ("without_reason" = Option<bool>, Query, description = "Only rows no detector picked, which the pipeline's flag alone put in the queue; selects nothing alongside reason"),
172        ("sort" = Option<String>, Query, description = "time_in_state, next_attempt or course")
173    ),
174    responses(
175        (status = 200, description = "A page of the rows needing a human, and how many for each reason", body = CreditRegistrationAttentionItems)
176    )
177)]
178pub async fn get_credit_registration_attention_items(
179    user: AuthUser,
180    pool: web::Data<PgPool>,
181    query: MultiQuery<AttentionQuery>,
182) -> ControllerResult<web::Json<CreditRegistrationAttentionItems>> {
183    let mut conn = pool.acquire().await?;
184    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
185
186    let pagination = parse_pagination(query.page, query.limit, ATTENTION_PAGE_SIZE)?;
187    let reasons: &[AttentionReason] = query.reason.as_deref().unwrap_or_default();
188    let only_without_reason = query.without_reason.unwrap_or(false);
189    let sort = match query.sort.as_deref() {
190        Some("next_attempt") => AttentionSort::NextAttempt,
191        Some("course") => AttentionSort::Course,
192        _ => AttentionSort::TimeInState,
193    };
194    let thresholds = stuck_thresholds();
195
196    let rows = credit_registrations::get_attention_items(
197        &mut conn,
198        &thresholds,
199        ATTENTION_TOO_MANY_ATTEMPTS,
200        credit_registrations::AttentionSelection {
201            reasons,
202            only_without_reason,
203            sort,
204            limit: pagination.limit(),
205            offset: pagination.offset(),
206        },
207    )
208    .await?;
209    let filtered_count = rows.first().map_or(0, |row| row.total_count);
210    // An unfiltered page already carries the whole queue's totals; only a narrowed or empty one
211    // needs them asked for separately.
212    let queue = match rows.first() {
213        Some(row) if reasons.is_empty() && !only_without_reason => Some(row.clone()),
214        _ => {
215            credit_registrations::count_needing_attention(
216                &mut conn,
217                &thresholds,
218                ATTENTION_TOO_MANY_ATTEMPTS,
219            )
220            .await?
221        }
222    };
223    let counts_by_reason = queue
224        .as_ref()
225        .map(AttentionRegistration::counts_by_reason)
226        .unwrap_or_default()
227        .into_iter()
228        .filter(|(_, count)| *count > 0)
229        .map(|(reason, count)| CreditRegistrationAttentionReasonCount { reason, count })
230        .collect();
231
232    token.authorized_ok(web::Json(CreditRegistrationAttentionItems {
233        items: rows.into_iter().map(to_attention_item).collect(),
234        total_count: queue.as_ref().map_or(0, |row| row.total_count),
235        filtered_count,
236        total_pages: pagination.total_pages(u32::try_from(filtered_count).unwrap_or(u32::MAX)),
237        counts_by_reason,
238        flagged_without_reason_count: queue
239            .as_ref()
240            .map_or(0, |row| row.flagged_without_reason_count),
241    }))
242}

244/**
245GET `/api/v0/main-frontend/credit-registration-admin/errors/by-code` - Error events per code over a
246window and the window before it, with the terminal verdicts of the same window beside them.
247
248Counts events, not rows: an error that happened really happened, whether or not a later attempt
249succeeded, and hiding it would hide the configuration bug that caused it.
250*/
251#[instrument(skip(pool))]
252#[utoipa::path(
253    get,
254    path = "/errors/by-code",
255    operation_id = "getCreditRegistrationErrorsByCode",
256    tag = "credit-registration-admin",
257    params(("window_secs" = Option<i64>, Query, description = "Window length in seconds; the same length before it is the comparison")),
258    responses(
259        (status = 200, description = "Per-code counts and the window's verdicts", body = CreditRegistrationErrorsByCode)
260    )
261)]
262pub async fn get_credit_registration_errors_by_code(
263    user: AuthUser,
264    pool: web::Data<PgPool>,
265    query: web::Query<ErrorWindowQuery>,
266) -> ControllerResult<web::Json<CreditRegistrationErrorsByCode>> {
267    let mut conn = pool.acquire().await?;
268    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
269
270    let window_secs = query
271        .window_secs
272        .unwrap_or(DEFAULT_ERROR_WINDOW_SECS)
273        .clamp(60, MAX_ERROR_WINDOW_SECS);
274    let codes =
275        credit_registration_events::get_error_code_counts_for_window(&mut conn, window_secs)
276            .await?
277            .into_iter()
278            .map(to_error_code_window)
279            .collect();
280    let totals = credit_registrations::count_terminal_outcomes_since(
281        &mut conn,
282        Utc::now() - chrono::Duration::seconds(window_secs),
283    )
284    .await?;
285
286    token.authorized_ok(web::Json(CreditRegistrationErrorsByCode {
287        window_secs,
288        codes,
289        verdicts: CreditRegistrationTerminalVerdicts {
290            registered_count: totals.registered_count,
291            duplicate_and_not_improved_count: totals.success_count - totals.registered_count,
292            failed_permanent_count: totals.failed_permanent_count,
293            cancelled_count: totals.cancelled_count,
294            total_count: totals.total_count,
295        },
296    }))
297}

299fn to_attention_item(row: AttentionRegistration) -> CreditRegistrationAttentionItem {
300    CreditRegistrationAttentionItem {
301        reasons: row.reasons(),
302        credit_registration_id: row.id,
303        user_id: row.user_id,
304        first_name: row.first_name,
305        last_name: row.last_name,
306        email: row.email,
307        course_id: row.course_id,
308        course_name: row.course_name,
309        course_module_id: row.course_module_id,
310        course_module_name: row.course_module_name,
311        state: row.state,
312        state_entered_at: row.state_entered_at,
313        error_code: row.error_code,
314        attempt_count: row.attempt_count,
315        next_attempt_at: row.next_attempt_at,
316        student_number: row.student_number,
317        needs_admin_attention: row.needs_admin_attention,
318    }
319}

321fn to_error_code_window(row: ErrorCodeWindowCounts) -> CreditRegistrationErrorCodeWindow {
322    CreditRegistrationErrorCodeWindow {
323        retryability: retryability(row.error_code),
324        error_code: row.error_code,
325        current_count: row.current_count,
326        previous_count: row.previous_count,
327        user_count: row.user_count,
328        course_count: row.course_count,
329        first_seen_at: row.first_seen_at,
330        last_seen_at: row.last_seen_at,
331        endpoints: row.endpoints,
332    }
333}
