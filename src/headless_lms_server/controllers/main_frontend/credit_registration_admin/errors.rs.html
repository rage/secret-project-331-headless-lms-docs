239/**
240GET `/api/v0/main-frontend/credit-registration-admin/attention` - The Needs attention tab: the rows
241that need a person by timeline phase, and the rows running late, explained by a blocking problem or
242recently dismissed.
243
244Superseded attempts are outside every detector: acting on a replaced attempt is never right.
245`total_count` is the one Needs attention count; `/overview`'s `needs_attention_count` is the same
246number.
247*/
248#[instrument(skip(pool, app_conf))]
249#[utoipa::path(
250    get,
251    path = "/attention",
252    operation_id = "getCreditRegistrationAttentionItems",
253    tag = "credit-registration-admin",
254    params(
255        ("limit" = Option<u32>, Query, description = "Rows listed per section; the counts cover every row"),
256        ("reason" = Option<Vec<AttentionReason>>, Query, description = "Only rows carrying one of these reasons; repeat the parameter for several"),
257        ("sort" = Option<String>, Query, description = "time_in_phase, next_attempt or course")
258    ),
259    responses(
260        (status = 200, description = "The Needs attention sections", body = CreditRegistrationAttentionItems)
261    )
262)]
263pub async fn get_credit_registration_attention_items(
264    user: AuthUser,
265    pool: web::Data<PgPool>,
266    query: MultiQuery<AttentionQuery>,
267    app_conf: web::Data<ApplicationConfiguration>,
268) -> ControllerResult<web::Json<CreditRegistrationAttentionItems>> {
269    let mut conn = pool.acquire().await?;
270    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
271
272    let section_size =
273        usize::try_from(parse_pagination(None, query.limit, ATTENTION_SECTION_SIZE)?.limit())
274            .unwrap_or(usize::MAX);
275    let reasons: &[AttentionReason] = query.reason.as_deref().unwrap_or_default();
276    let rules = attention_rules(&mut conn, &app_conf).await?;
277    let rows = credit_registrations::get_attention_items(&mut conn, &rules, None).await?;
278    let course_ids: Vec<Uuid> = rows
279        .iter()
280        .map(|row| row.course_id)
281        .collect::<BTreeSet<_>>()
282        .into_iter()
283        .collect();
284    let mut sends = manual_linking_email_sends_by_registration(&mut conn, &course_ids).await?;
285    let dismissed_recently = credit_registrations::get_dismissals_since(
286        &mut conn,
287        Utc::now() - chrono::Duration::days(RECENT_DISMISSAL_DAYS),
288        RECENT_DISMISSAL_LIMIT,
289    )
290    .await?
291    .into_iter()
292    .map(to_dismissal)
293    .collect();
294
295    let mut needing: Vec<CreditRegistrationAttentionItem> = Vec::new();
296    let mut running_late = Vec::new();
297    let mut explained_by_problem: Vec<CreditRegistrationBlockingProblemRows> = Vec::new();
298    for item in rows.into_iter().map(|row| {
299        let row_sends = sends.remove(&row.id).unwrap_or_default();
300        to_attention_item(row, &rules.blocking, row_sends)
301    }) {
302        match item.standing {
303            AttentionStanding::NeedsAttention => needing.push(item),
304            AttentionStanding::RunningLate => running_late.push(item),
305            AttentionStanding::ExplainedByProblem => {
306                let Some(problem) = item.blocking_problem.clone() else {
307                    continue;
308                };
309                match explained_by_problem
310                    .iter_mut()
311                    .find(|group| group.problem == problem)
312                {
313                    Some(group) => group.items.push(item),
314                    None => explained_by_problem.push(CreditRegistrationBlockingProblemRows {
315                        problem,
316                        total_count: 0,
317                        items: vec![item],
318                    }),
319                }
320            }
321            AttentionStanding::Dismissed => {}
322        }
323    }
324    for group in &mut explained_by_problem {
325        group.total_count = group.items.len() as i64;
326        group.items.truncate(section_size);
327    }
328    let running_late_count = running_late.len() as i64;
329    running_late.truncate(section_size);
330
331    let mut reason_counts: BTreeMap<AttentionReason, i64> = BTreeMap::new();
332    for reason in needing.iter().flat_map(|item| item.reasons.iter()) {
333        *reason_counts.entry(*reason).or_insert(0) += 1;
334    }
335    let counts_by_reason = reason_counts
336        .into_iter()
337        .map(|(reason, count)| CreditRegistrationAttentionReasonCount { reason, count })
338        .collect();
339    let total_count = needing.len() as i64;
340
341    let mut by_phase: BTreeMap<TimelinePhase, Vec<CreditRegistrationAttentionItem>> =
342        BTreeMap::new();
343    for item in needing.into_iter().filter(|item| {
344        reasons.is_empty() || item.reasons.iter().any(|reason| reasons.contains(reason))
345    }) {
346        by_phase.entry(item.phase).or_default().push(item);
347    }
348    let phases = by_phase
349        .into_iter()
350        .map(|(phase, mut items)| {
351            match query.sort.as_deref() {
352                Some("next_attempt") => items.sort_by_key(|item| item.next_attempt_at),
353                Some("course") => items.sort_by(|a, b| a.course_name.cmp(&b.course_name)),
354                _ => items.sort_by_key(|item| item.phase_started_at),
355            }
356            let total_count = items.len() as i64;
357            items.truncate(section_size);
358            CreditRegistrationAttentionPhaseRows {
359                phase,
360                total_count,
361                items,
362            }
363        })
364        .collect();
365
366    token.authorized_ok(web::Json(CreditRegistrationAttentionItems {
367        total_count,
368        counts_by_reason,
369        phases,
370        running_late,
371        running_late_count,
372        explained_by_problem,
373        dismissed_recently,
374    }))
375}

377/**
378POST `/api/v0/main-frontend/credit-registration-admin/registrations/{credit_registration_id}/dismiss-attention`
379- Takes a row off the Needs attention queue.
380
381The dismissal covers the reasons the row carries now; it comes back only when a different one
382fires. Leaves the pipeline's own flag alone.
383*/
384#[instrument(skip(pool, payload, app_conf))]
385#[utoipa::path(
386    post,
387    path = "/registrations/{credit_registration_id}/dismiss-attention",
388    operation_id = "adminDismissCreditRegistrationAttention",
389    tag = "credit-registration-admin",
390    params(("credit_registration_id" = Uuid, Path, description = "Credit registration id")),
391    request_body = AdminDismissAttentionPayload,
392    responses(
393        (status = 200, description = "The reasons dismissed", body = AdminDismissAttentionResult),
394        (status = 400, description = "No reason given, or nothing picks the row"),
395    )
396)]
397pub async fn admin_dismiss_credit_registration_attention(
398    user: AuthUser,
399    pool: web::Data<PgPool>,
400    credit_registration_id: web::Path<Uuid>,
401    payload: web::Json<AdminDismissAttentionPayload>,
402    app_conf: web::Data<ApplicationConfiguration>,
403) -> ControllerResult<web::Json<AdminDismissAttentionResult>> {
404    let mut conn = pool.acquire().await?;
405    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
406
407    let reason = required_reason(&payload.reason)?;
408    let id = *credit_registration_id;
409    let rules = attention_rules(&mut conn, &app_conf).await?;
410    let reasons = credit_registrations::get_attention_items(&mut conn, &rules, Some(&[id]))
411        .await?
412        .into_iter()
413        .next()
414        .map(|row| row.reasons)
415        .unwrap_or_default();
416    if reasons.is_empty() {
417        return Err(controller_err!(
418            BadRequest,
419            "Nothing puts this registration on the Needs attention queue.".to_string()
420        ));
421    }
422
423    let mut tx = conn.begin().await?;
424    credit_registrations::dismiss_attention(&mut tx, id, &reasons, user.id, reason).await?;
425    models::credit_registration_events::insert(
426        &mut tx,
427        &NewCreditRegistrationEvent {
428            actor_user_id: Some(user.id),
429            message: Some(reason.to_string()),
430            ..NewCreditRegistrationEvent::new(id, CreditRegistrationEventKind::AdminAction)
431        },
432    )
433    .await?;
434    models::credit_registration_admin_actions::record(
435        &mut tx,
436        &NewCreditRegistrationAdminAction {
437            target_id: Some(id),
438            reason: Some(reason.to_string()),
439            details: Some(serde_json::json!({ "attention_reasons": reasons })),
440            affected_row_count: Some(1),
441            ..NewCreditRegistrationAdminAction::new(
442                CreditRegistrationAdminAction::DismissAttention,
443                CreditRegistrationAdminActionTarget::CreditRegistration,
444                user.id,
445                GLOBAL_ADMIN_ROLE,
446            )
447        },
448    )
449    .await?;
450    tx.commit().await?;
451
452    token.authorized_ok(web::Json(AdminDismissAttentionResult {
453        dismissed_reasons: reasons,
454    }))
455}

457/**
458GET `/api/v0/main-frontend/credit-registration-admin/errors/by-code` - Error events per code over a
459window and the window before it, with the terminal verdicts of the same window beside them.
460
461Counts events, not rows: an error that happened really happened, whether or not a later attempt
462succeeded, and hiding it would hide the configuration bug that caused it.
463*/
464#[instrument(skip(pool))]
465#[utoipa::path(
466    get,
467    path = "/errors/by-code",
468    operation_id = "getCreditRegistrationErrorsByCode",
469    tag = "credit-registration-admin",
470    params(("window_secs" = Option<i64>, Query, description = "Window length in seconds; the same length before it is the comparison")),
471    responses(
472        (status = 200, description = "Per-code counts and the window's verdicts", body = CreditRegistrationErrorsByCode)
473    )
474)]
475pub async fn get_credit_registration_errors_by_code(
476    user: AuthUser,
477    pool: web::Data<PgPool>,
478    query: web::Query<ErrorWindowQuery>,
479) -> ControllerResult<web::Json<CreditRegistrationErrorsByCode>> {
480    let mut conn = pool.acquire().await?;
481    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
482
483    let window_secs = query
484        .window_secs
485        .unwrap_or(DEFAULT_ERROR_WINDOW_SECS)
486        .clamp(60, MAX_ERROR_WINDOW_SECS);
487    let live_counts = credit_registrations::count_by_error_code(&mut conn).await?;
488    let codes =
489        credit_registration_events::get_error_code_counts_for_window(&mut conn, window_secs)
490            .await?
491            .into_iter()
492            .map(|row| {
493                let live_count = live_counts
494                    .iter()
495                    .find(|live| live.error_code == row.error_code)
496                    .map_or(0, |live| live.live_count);
497                to_error_code_window(row, live_count)
498            })
499            .collect();
500    let totals = credit_registrations::count_terminal_outcomes_since(
501        &mut conn,
502        Utc::now() - chrono::Duration::seconds(window_secs),
503    )
504    .await?;
505
506    token.authorized_ok(web::Json(CreditRegistrationErrorsByCode {
507        window_secs,
508        codes,
509        verdicts: CreditRegistrationTerminalVerdicts {
510            registered_count: totals.registered_count,
511            duplicate_and_not_improved_count: totals.success_count - totals.registered_count,
512            failed_permanent_count: totals.failed_permanent_count,
513            cancelled_count: totals.cancelled_count,
514            total_count: totals.total_count,
515        },
516    }))
517}

519pub(super) fn to_attention_item(
520    row: AttentionRegistration,
521    problems: &BlockingProblems,
522    manual_linking_email_sends: Vec<AdminManualLinkingEmailSend>,
523) -> CreditRegistrationAttentionItem {
524    let position = row.position();
525    let standing = row.standing(problems);
526    let waits_on = WaitsOn::of(
527        position.step,
528        position.engagement,
529        standing == AttentionStanding::NeedsAttention,
530        row.needs_course_setup(),
531    );
532    CreditRegistrationAttentionItem {
533        blocking_problem: (standing == AttentionStanding::ExplainedByProblem)
534            .then(|| row.blocking_problem(problems))
535            .flatten(),
536        hand_actions: row
537            .resubmission_facts()
538            .hand_actions(ResubmissionStrictness::AnyExceptSubmissionUncertain),
539        timeline_step: position.step,
540        phase: position.step.phase(),
541        engagement: position.engagement,
542        waits_on,
543        standing,
544        credit_registration_id: row.id,
545        user_id: row.user_id,
546        first_name: row.first_name,
547        last_name: row.last_name,
548        email: row.email,
549        course_id: row.course_id,
550        course_name: row.course_name,
551        course_module_id: row.course_module_id,
552        course_module_name: row.course_module_name,
553        uh_course_code: row.uh_course_code,
554        state: row.state,
555        state_changed_at: row.state_changed_at,
556        phase_started_at: row.phase_started_at,
557        error_code: row.error_code,
558        attempt_count: row.attempt_count,
559        next_attempt_at: row.next_attempt_at,
560        student_number: expose_option(&row.student_number).map(str::to_owned),
561        reasons: row.reasons,
562        pressed_at: row.pressed_at,
563        enrolment_route: row.enrolment_route,
564        last_mailing_fetch_started_at: row.last_mailing_fetch_started_at,
565        is_enrolment_list_empty: row.is_enrolment_list_empty,
566        needs_admin_attention: row.needs_admin_attention,
567        manual_linking_email_sends,
568    }
569}
