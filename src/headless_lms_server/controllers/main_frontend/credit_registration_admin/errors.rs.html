20/// The `chatbot_syncer` precedent, and the point past which retrying is not the answer.
21const TOO_MANY_ATTEMPTS: i32 = 5;
22/// Bounds the attention table. A dashboard that renders ten thousand rows helps nobody, and the
23/// per-reason counts beside it say how much is left.
24const ATTENTION_LIMIT: i64 = 500;
25const DEFAULT_ERROR_WINDOW_SECS: i64 = 24 * 60 * 60;
26const MAX_ERROR_WINDOW_SECS: i64 = 90 * 24 * 60 * 60;

28/// Why a row is on the attention table. One row can carry several.
29#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, Hash, ToSchema)]
30#[serde(rename_all = "snake_case")]
31pub enum CreditRegistrationAttentionReason {
32    /// Past its state's threshold with the pipeline still owning it.
33    StuckInState,
34    PermanentError,
35    RetryWindowExpired,
36    Misregistered,
37    TooManyAttempts,
38    /// `submission_uncertain`: never retried automatically, and never in bulk.
39    OutcomeUncertain,
40    /// The pipeline itself asked for a human, e.g. because the completion drifted under the row.
41    FlaggedByPipeline,
42}

122/**
123GET `/api/v0/main-frontend/credit-registration-admin/thresholds` - Every number the alert rules and
124the stuck detectors use.
125
126The same values `/overview` embeds in its health block. Separate so a tab explaining "stuck after
1272 hours" can say so without reading the whole overview aggregate.
128*/
129#[instrument(skip(pool))]
130#[utoipa::path(
131    get,
132    path = "/thresholds",
133    operation_id = "getCreditRegistrationThresholds",
134    tag = "credit-registration-admin",
135    responses(
136        (status = 200, description = "The thresholds every rule and detector shares", body = CreditRegistrationAlertThresholds)
137    )
138)]
139pub async fn get_credit_registration_thresholds(
140    user: AuthUser,
141    pool: web::Data<PgPool>,
142) -> ControllerResult<web::Json<CreditRegistrationAlertThresholds>> {
143    let mut conn = pool.acquire().await?;
144    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
145    token.authorized_ok(web::Json(thresholds()))
146}

148/**
149GET `/api/v0/main-frontend/credit-registration-admin/attention` - The rows at least one detector
150wants a human to look at, with the detectors that picked each.
151
152Superseded attempts and rows abandoned by a consent withdrawal are outside every detector: neither
153is something a person can act on.
154*/
155#[instrument(skip(pool))]
156#[utoipa::path(
157    get,
158    path = "/attention",
159    operation_id = "getCreditRegistrationAttentionItems",
160    tag = "credit-registration-admin",
161    responses(
162        (status = 200, description = "Rows needing a human, and how many for each reason", body = CreditRegistrationAttentionItems)
163    )
164)]
165pub async fn get_credit_registration_attention_items(
166    user: AuthUser,
167    pool: web::Data<PgPool>,
168) -> ControllerResult<web::Json<CreditRegistrationAttentionItems>> {
169    let mut conn = pool.acquire().await?;
170    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
171
172    let items: Vec<CreditRegistrationAttentionItem> = credit_registrations::get_attention_items(
173        &mut conn,
174        &stuck_thresholds(),
175        TOO_MANY_ATTEMPTS,
176        ATTENTION_LIMIT,
177    )
178    .await?
179    .into_iter()
180    .map(to_attention_item)
181    .collect();
182
183    let counts_by_reason = ALL_ATTENTION_REASONS
184        .into_iter()
185        .map(|reason| CreditRegistrationAttentionReasonCount {
186            reason,
187            count: items
188                .iter()
189                .filter(|item| item.reasons.contains(&reason))
190                .count() as i64,
191        })
192        .filter(|row| row.count > 0)
193        .collect();
194
195    token.authorized_ok(web::Json(CreditRegistrationAttentionItems {
196        total_count: items.len() as i64,
197        items,
198        counts_by_reason,
199        max_items: ATTENTION_LIMIT,
200    }))
201}

203/**
204GET `/api/v0/main-frontend/credit-registration-admin/errors/by-code` - Error events per code over a
205window and the window before it, with the terminal verdicts of the same window beside them.
206
207Counts events, not rows: an error that happened really happened, whether or not a later attempt
208succeeded, and hiding it would hide the configuration bug that caused it.
209*/
210#[instrument(skip(pool))]
211#[utoipa::path(
212    get,
213    path = "/errors/by-code",
214    operation_id = "getCreditRegistrationErrorsByCode",
215    tag = "credit-registration-admin",
216    params(("window_secs" = Option<i64>, Query, description = "Window length in seconds; the same length before it is the comparison")),
217    responses(
218        (status = 200, description = "Per-code counts and the window's verdicts", body = CreditRegistrationErrorsByCode)
219    )
220)]
221pub async fn get_credit_registration_errors_by_code(
222    user: AuthUser,
223    pool: web::Data<PgPool>,
224    query: web::Query<ErrorWindowQuery>,
225) -> ControllerResult<web::Json<CreditRegistrationErrorsByCode>> {
226    let mut conn = pool.acquire().await?;
227    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
228
229    let window_secs = query
230        .window_secs
231        .unwrap_or(DEFAULT_ERROR_WINDOW_SECS)
232        .clamp(60, MAX_ERROR_WINDOW_SECS);
233    let codes =
234        credit_registration_events::get_error_code_counts_for_window(&mut conn, window_secs)
235            .await?
236            .into_iter()
237            .map(to_error_code_window)
238            .collect();
239    let totals = credit_registrations::count_terminal_outcomes_since(
240        &mut conn,
241        Utc::now() - chrono::Duration::seconds(window_secs),
242    )
243    .await?;
244
245    token.authorized_ok(web::Json(CreditRegistrationErrorsByCode {
246        window_secs,
247        codes,
248        verdicts: CreditRegistrationTerminalVerdicts {
249            registered_count: totals.registered_count,
250            duplicate_and_not_improved_count: totals.success_count - totals.registered_count,
251            failed_permanent_count: totals.failed_permanent_count,
252            cancelled_count: totals.cancelled_count,
253            abandoned_by_consent_withdrawal_count: totals.abandoned_count,
254            total_count: totals.total_count,
255        },
256    }))
257}

259const ALL_ATTENTION_REASONS: [CreditRegistrationAttentionReason; 7] = [
260    CreditRegistrationAttentionReason::StuckInState,
261    CreditRegistrationAttentionReason::PermanentError,
262    CreditRegistrationAttentionReason::RetryWindowExpired,
263    CreditRegistrationAttentionReason::Misregistered,
264    CreditRegistrationAttentionReason::TooManyAttempts,
265    CreditRegistrationAttentionReason::OutcomeUncertain,
266    CreditRegistrationAttentionReason::FlaggedByPipeline,
267];
268
269fn to_attention_item(row: AttentionRegistration) -> CreditRegistrationAttentionItem {
270    let flags = [
271        (
272            row.stuck_in_state,
273            CreditRegistrationAttentionReason::StuckInState,
274        ),
275        (
276            row.permanent_error,
277            CreditRegistrationAttentionReason::PermanentError,
278        ),
279        (
280            row.retry_window_expired,
281            CreditRegistrationAttentionReason::RetryWindowExpired,
282        ),
283        (
284            row.misregistered,
285            CreditRegistrationAttentionReason::Misregistered,
286        ),
287        (
288            row.too_many_attempts,
289            CreditRegistrationAttentionReason::TooManyAttempts,
290        ),
291        (
292            row.outcome_uncertain,
293            CreditRegistrationAttentionReason::OutcomeUncertain,
294        ),
295        (
296            row.flagged_by_pipeline,
297            CreditRegistrationAttentionReason::FlaggedByPipeline,
298        ),
299    ];
300    CreditRegistrationAttentionItem {
301        reasons: flags
302            .into_iter()
303            .filter_map(|(fired, reason)| fired.then_some(reason))
304            .collect(),
305        credit_registration_id: row.id,
306        user_id: row.user_id,
307        first_name: row.first_name,
308        last_name: row.last_name,
309        email: row.email,
310        course_id: row.course_id,
311        course_name: row.course_name,
312        course_module_id: row.course_module_id,
313        course_module_name: row.course_module_name,
314        state: row.state,
315        state_entered_at: row.state_entered_at,
316        error_code: row.error_code,
317        attempt_count: row.attempt_count,
318        next_attempt_at: row.next_attempt_at,
319        student_number: row.student_number,
320    }
321}

323fn to_error_code_window(row: ErrorCodeWindowCounts) -> CreditRegistrationErrorCodeWindow {
324    CreditRegistrationErrorCodeWindow {
325        retryability: retryability(row.error_code),
326        error_code: row.error_code,
327        current_count: row.current_count,
328        previous_count: row.previous_count,
329        user_count: row.user_count,
330        course_count: row.course_count,
331        first_seen_at: row.first_seen_at,
332        last_seen_at: row.last_seen_at,
333        endpoints: row.endpoints,
334    }
335}
