237/**
238GET `/api/v0/main-frontend/credit-registration-admin/attention` - The Needs attention tab: the rows
239that need a person by timeline phase, and the rows running late, explained by a blocking problem or
240recently dismissed.
241
242Superseded attempts are outside every detector: acting on a replaced attempt is never right.
243`total_count` is the one Needs attention count; `/overview`'s `needs_attention_count` is the same
244number.
245*/
246#[instrument(skip(pool, app_conf))]
247#[utoipa::path(
248    get,
249    path = "/attention",
250    operation_id = "getCreditRegistrationAttentionItems",
251    tag = "credit-registration-admin",
252    params(
253        ("limit" = Option<u32>, Query, description = "Rows listed per section; the counts cover every row"),
254        ("reason" = Option<Vec<AttentionReason>>, Query, description = "Only rows carrying one of these reasons; repeat the parameter for several"),
255        ("sort" = Option<String>, Query, description = "time_in_phase, next_attempt or course")
256    ),
257    responses(
258        (status = 200, description = "The Needs attention sections", body = CreditRegistrationAttentionItems)
259    )
260)]
261pub async fn get_credit_registration_attention_items(
262    user: AuthUser,
263    pool: web::Data<PgPool>,
264    query: MultiQuery<AttentionQuery>,
265    app_conf: web::Data<ApplicationConfiguration>,
266) -> ControllerResult<web::Json<CreditRegistrationAttentionItems>> {
267    let mut conn = pool.acquire().await?;
268    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
269
270    let section_size =
271        usize::try_from(parse_pagination(None, query.limit, ATTENTION_SECTION_SIZE)?.limit())
272            .unwrap_or(usize::MAX);
273    let reasons: &[AttentionReason] = query.reason.as_deref().unwrap_or_default();
274    let rules = attention_rules(&mut conn, &app_conf).await?;
275    let rows = credit_registrations::get_attention_items(&mut conn, &rules, None).await?;
276    let dismissed_recently = credit_registrations::get_dismissals_since(
277        &mut conn,
278        Utc::now() - chrono::Duration::days(RECENT_DISMISSAL_DAYS),
279        RECENT_DISMISSAL_LIMIT,
280    )
281    .await?
282    .into_iter()
283    .map(to_dismissal)
284    .collect();
285
286    let mut needing: Vec<CreditRegistrationAttentionItem> = Vec::new();
287    let mut running_late = Vec::new();
288    let mut explained_by_problem: Vec<CreditRegistrationBlockingProblemRows> = Vec::new();
289    for item in rows
290        .into_iter()
291        .map(|row| to_attention_item(row, &rules.blocking))
292    {
293        match item.standing {
294            AttentionStanding::NeedsAttention => needing.push(item),
295            AttentionStanding::RunningLate => running_late.push(item),
296            AttentionStanding::ExplainedByProblem => {
297                let Some(problem) = item.blocking_problem.clone() else {
298                    continue;
299                };
300                match explained_by_problem
301                    .iter_mut()
302                    .find(|group| group.problem == problem)
303                {
304                    Some(group) => group.items.push(item),
305                    None => explained_by_problem.push(CreditRegistrationBlockingProblemRows {
306                        problem,
307                        total_count: 0,
308                        items: vec![item],
309                    }),
310                }
311            }
312            AttentionStanding::Dismissed => {}
313        }
314    }
315    for group in &mut explained_by_problem {
316        group.total_count = group.items.len() as i64;
317        group.items.truncate(section_size);
318    }
319    let running_late_count = running_late.len() as i64;
320    running_late.truncate(section_size);
321
322    let mut reason_counts: BTreeMap<AttentionReason, i64> = BTreeMap::new();
323    for reason in needing.iter().flat_map(|item| item.reasons.iter()) {
324        *reason_counts.entry(*reason).or_insert(0) += 1;
325    }
326    let counts_by_reason = reason_counts
327        .into_iter()
328        .map(|(reason, count)| CreditRegistrationAttentionReasonCount { reason, count })
329        .collect();
330    let total_count = needing.len() as i64;
331
332    let mut by_phase: BTreeMap<TimelinePhase, Vec<CreditRegistrationAttentionItem>> =
333        BTreeMap::new();
334    for item in needing.into_iter().filter(|item| {
335        reasons.is_empty() || item.reasons.iter().any(|reason| reasons.contains(reason))
336    }) {
337        by_phase.entry(item.phase).or_default().push(item);
338    }
339    let phases = by_phase
340        .into_iter()
341        .map(|(phase, mut items)| {
342            match query.sort.as_deref() {
343                Some("next_attempt") => items.sort_by_key(|item| item.next_attempt_at),
344                Some("course") => items.sort_by(|a, b| a.course_name.cmp(&b.course_name)),
345                _ => items.sort_by_key(|item| item.phase_started_at),
346            }
347            let total_count = items.len() as i64;
348            items.truncate(section_size);
349            CreditRegistrationAttentionPhaseRows {
350                phase,
351                total_count,
352                items,
353            }
354        })
355        .collect();
356
357    token.authorized_ok(web::Json(CreditRegistrationAttentionItems {
358        total_count,
359        counts_by_reason,
360        phases,
361        running_late,
362        running_late_count,
363        explained_by_problem,
364        dismissed_recently,
365    }))
366}

368/**
369POST `/api/v0/main-frontend/credit-registration-admin/registrations/{credit_registration_id}/dismiss-attention`
370- Takes a row off the Needs attention queue.
371
372The dismissal covers the reasons the row carries now; it comes back only when a different one
373fires. Leaves the pipeline's own flag alone.
374*/
375#[instrument(skip(pool, payload, app_conf))]
376#[utoipa::path(
377    post,
378    path = "/registrations/{credit_registration_id}/dismiss-attention",
379    operation_id = "adminDismissCreditRegistrationAttention",
380    tag = "credit-registration-admin",
381    params(("credit_registration_id" = Uuid, Path, description = "Credit registration id")),
382    request_body = AdminDismissAttentionPayload,
383    responses(
384        (status = 200, description = "The reasons dismissed", body = AdminDismissAttentionResult),
385        (status = 400, description = "No reason given, or nothing picks the row"),
386    )
387)]
388pub async fn admin_dismiss_credit_registration_attention(
389    user: AuthUser,
390    pool: web::Data<PgPool>,
391    credit_registration_id: web::Path<Uuid>,
392    payload: web::Json<AdminDismissAttentionPayload>,
393    app_conf: web::Data<ApplicationConfiguration>,
394) -> ControllerResult<web::Json<AdminDismissAttentionResult>> {
395    let mut conn = pool.acquire().await?;
396    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
397
398    let reason = required_reason(&payload.reason)?;
399    let id = *credit_registration_id;
400    let rules = attention_rules(&mut conn, &app_conf).await?;
401    let reasons = credit_registrations::get_attention_items(&mut conn, &rules, Some(&[id]))
402        .await?
403        .into_iter()
404        .next()
405        .map(|row| row.reasons)
406        .unwrap_or_default();
407    if reasons.is_empty() {
408        return Err(controller_err!(
409            BadRequest,
410            "Nothing puts this registration on the Needs attention queue.".to_string()
411        ));
412    }
413
414    let mut tx = conn.begin().await?;
415    credit_registrations::dismiss_attention(&mut tx, id, &reasons, user.id, reason).await?;
416    models::credit_registration_events::insert(
417        &mut tx,
418        &NewCreditRegistrationEvent {
419            actor_user_id: Some(user.id),
420            message: Some(reason.to_string()),
421            ..NewCreditRegistrationEvent::new(id, CreditRegistrationEventKind::AdminAction)
422        },
423    )
424    .await?;
425    models::credit_registration_admin_actions::record(
426        &mut tx,
427        &NewCreditRegistrationAdminAction {
428            target_id: Some(id),
429            reason: Some(reason.to_string()),
430            details: Some(serde_json::json!({ "attention_reasons": reasons })),
431            affected_row_count: Some(1),
432            ..NewCreditRegistrationAdminAction::new(
433                CreditRegistrationAdminAction::DismissAttention,
434                CreditRegistrationAdminActionTarget::CreditRegistration,
435                user.id,
436                GLOBAL_ADMIN_ROLE,
437            )
438        },
439    )
440    .await?;
441    tx.commit().await?;
442
443    token.authorized_ok(web::Json(AdminDismissAttentionResult {
444        dismissed_reasons: reasons,
445    }))
446}

448/**
449GET `/api/v0/main-frontend/credit-registration-admin/errors/by-code` - Error events per code over a
450window and the window before it, with the terminal verdicts of the same window beside them.
451
452Counts events, not rows: an error that happened really happened, whether or not a later attempt
453succeeded, and hiding it would hide the configuration bug that caused it.
454*/
455#[instrument(skip(pool))]
456#[utoipa::path(
457    get,
458    path = "/errors/by-code",
459    operation_id = "getCreditRegistrationErrorsByCode",
460    tag = "credit-registration-admin",
461    params(("window_secs" = Option<i64>, Query, description = "Window length in seconds; the same length before it is the comparison")),
462    responses(
463        (status = 200, description = "Per-code counts and the window's verdicts", body = CreditRegistrationErrorsByCode)
464    )
465)]
466pub async fn get_credit_registration_errors_by_code(
467    user: AuthUser,
468    pool: web::Data<PgPool>,
469    query: web::Query<ErrorWindowQuery>,
470) -> ControllerResult<web::Json<CreditRegistrationErrorsByCode>> {
471    let mut conn = pool.acquire().await?;
472    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
473
474    let window_secs = query
475        .window_secs
476        .unwrap_or(DEFAULT_ERROR_WINDOW_SECS)
477        .clamp(60, MAX_ERROR_WINDOW_SECS);
478    let live_counts = credit_registrations::count_by_error_code(&mut conn).await?;
479    let codes =
480        credit_registration_events::get_error_code_counts_for_window(&mut conn, window_secs)
481            .await?
482            .into_iter()
483            .map(|row| {
484                let live_count = live_counts
485                    .iter()
486                    .find(|live| live.error_code == row.error_code)
487                    .map_or(0, |live| live.live_count);
488                to_error_code_window(row, live_count)
489            })
490            .collect();
491    let totals = credit_registrations::count_terminal_outcomes_since(
492        &mut conn,
493        Utc::now() - chrono::Duration::seconds(window_secs),
494    )
495    .await?;
496
497    token.authorized_ok(web::Json(CreditRegistrationErrorsByCode {
498        window_secs,
499        codes,
500        verdicts: CreditRegistrationTerminalVerdicts {
501            registered_count: totals.registered_count,
502            duplicate_and_not_improved_count: totals.success_count - totals.registered_count,
503            failed_permanent_count: totals.failed_permanent_count,
504            cancelled_count: totals.cancelled_count,
505            total_count: totals.total_count,
506        },
507    }))
508}

510pub(super) fn to_attention_item(
511    row: AttentionRegistration,
512    problems: &BlockingProblems,
513) -> CreditRegistrationAttentionItem {
514    let position = row.position();
515    let standing = row.standing(problems);
516    let waits_on = WaitsOn::of(
517        position.step,
518        position.engagement,
519        standing == AttentionStanding::NeedsAttention,
520        row.needs_course_setup(),
521    );
522    CreditRegistrationAttentionItem {
523        blocking_problem: (standing == AttentionStanding::ExplainedByProblem)
524            .then(|| row.blocking_problem(problems))
525            .flatten(),
526        hand_actions: row
527            .resubmission_facts()
528            .hand_actions(ResubmissionStrictness::AnyExceptSubmissionUncertain),
529        timeline_step: position.step,
530        phase: position.step.phase(),
531        engagement: position.engagement,
532        waits_on,
533        standing,
534        credit_registration_id: row.id,
535        user_id: row.user_id,
536        first_name: row.first_name,
537        last_name: row.last_name,
538        email: row.email,
539        course_id: row.course_id,
540        course_name: row.course_name,
541        course_module_id: row.course_module_id,
542        course_module_name: row.course_module_name,
543        uh_course_code: row.uh_course_code,
544        state: row.state,
545        state_changed_at: row.state_changed_at,
546        phase_started_at: row.phase_started_at,
547        error_code: row.error_code,
548        attempt_count: row.attempt_count,
549        next_attempt_at: row.next_attempt_at,
550        student_number: expose_option(&row.student_number).map(str::to_owned),
551        reasons: row.reasons,
552        pressed_at: row.pressed_at,
553        enrolment_route: row.enrolment_route,
554        last_mailing_fetch_started_at: row.last_mailing_fetch_started_at,
555        is_enrolment_list_empty: row.is_enrolment_list_empty,
556        unlinked_enrolled_before_count: row.unlinked_enrolled_before_count,
557        needs_admin_attention: row.needs_admin_attention,
558    }
559}
