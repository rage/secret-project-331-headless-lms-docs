45/// Every handler here gates on the same check; a submodule calls this instead of repeating it.
46async fn authorize_credit_registration_admin(
47    conn: &mut PgConnection,
48    user_id: Uuid,
49) -> Result<AuthorizationToken, ControllerError> {
50    authorize(
51        conn,
52        Act::Administrate,
53        Some(user_id),
54        Res::GlobalPermissions,
55    )
56    .await
57}

59/// Refuses an empty or whitespace reason. Every audited action names one.
60fn required_reason(reason: &str) -> Result<&str, ControllerError> {
61    let trimmed = reason.trim();
62    if trimmed.is_empty() {
63        return Err(controller_err!(
64            BadRequest,
65            "A reason is required.".to_string()
66        ));
67    }
68    Ok(trimmed)
69}

86/// Shared by the ledger detail view and the account-linking admin views: both show a person's linking
87/// mails alongside the token each one carries.
88async fn build_linking_emails(
89    conn: &mut PgConnection,
90    mails: Vec<CreditRegistrationAccountLinkingEmail>,
91) -> Result<Vec<AdminLinkingEmail>, ControllerError> {
92    let ids: Vec<Uuid> = mails.iter().map(|mail| mail.id).collect();
93    let reports =
94        credit_registration_account_linking_emails::get_send_status_reports(conn, &ids).await?;
95    let token_ids: Vec<Uuid> = mails
96        .iter()
97        .filter_map(|mail| mail.student_number_verification_token_id)
98        .collect();
99    let tokens = student_number_verification_tokens::get_by_ids(conn, &token_ids).await?;
100    Ok(mails
101        .into_iter()
102        .map(|mail| {
103            let token = mail
104                .student_number_verification_token_id
105                .and_then(|token_id| tokens.get(&token_id));
106            AdminLinkingEmail {
107                send_status: reports.get(&mail.id).cloned().unwrap_or_else(
108                    credit_registration_account_linking_emails::not_handed_over_yet,
109                ),
110                id: mail.id,
111                course_id: mail.course_id,
112                student_number: mail.student_number,
113                sisu_person_id: mail.sisu_person_id,
114                emailed_to: mail.emailed_to,
115                claimed_at: mail.sent_at,
116                token_claimed_by_user_id: token.and_then(|row| row.claimed_by_user_id),
117                token_used_at: token.and_then(|row| row.used_at),
118                token_expires_at: token.map(|row| row.expires_at),
119            }
120        })
121        .collect())
122}
