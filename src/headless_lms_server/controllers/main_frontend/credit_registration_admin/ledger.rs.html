284/// The states an admin may move a row to; everything else is the pipeline's to decide.
285#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
286#[serde(rename_all = "snake_case")]
287pub enum AdminCreditRegistrationStateMove {
288    /// Resubmit: the escape hatch out of `submission_uncertain`, and how a `misregistered` row is
289    /// tried again.
290    ReadyToSubmit,
291    Cancelled,
292}
293
294impl AdminCreditRegistrationStateMove {
295    fn to_state(self) -> CreditRegistrationState {
296        match self {
297            Self::ReadyToSubmit => CreditRegistrationState::ReadyToSubmit,
298            Self::Cancelled => CreditRegistrationState::Cancelled,
299        }
300    }
301}

303/// What one hand action does to a row: either a state move, or something that leaves the state
304/// alone. Kept apart because only the first is a transition, and only the first is refusable.
305#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
306#[serde(rename_all = "snake_case", tag = "kind")]
307pub enum AdminCreditRegistrationAction {
308    StateMove {
309        to_state: AdminCreditRegistrationStateMove,
310    },
311    /// Stops the row asking for a human.
312    ClearNeedsAdminAttention,
313    /// Makes the row due, so the phase owning its state claims it on the next pass instead of
314    /// waiting out a backoff of up to a day.
315    CheckNow,
316}
317
318impl AdminCreditRegistrationAction {
319    /// Why this action is refused on the row, or `None` if it may go ahead.
320    fn refusal(
321        self,
322        facts: &ResubmissionFacts,
323        strictness: ResubmissionStrictness,
324    ) -> Option<ResubmissionRefusal> {
325        match self {
326            Self::StateMove { to_state } => {
327                facts.admin_transition_refusal(to_state.to_state(), strictness)
328            }
329            Self::CheckNow => facts.check_now_refusal(),
330            // Even clearing a flag on a replaced attempt is an admin acting on the wrong row.
331            Self::ClearNeedsAdminAttention if facts.is_superseded => {
332                Some(ResubmissionRefusal::Superseded)
333            }
334            Self::ClearNeedsAdminAttention => None,
335        }
336    }
337}

461pub async fn list_credit_registrations_for_admin(
462    user: AuthUser,
463    pool: web::Data<PgPool>,
464    query: MultiQuery<ListCreditRegistrationsQuery>,
465    app_conf: web::Data<ApplicationConfiguration>,
466) -> ControllerResult<web::Json<Page<AdminCreditRegistrationRow>>> {
467    let mut conn = pool.acquire().await?;
468    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
469
470    let pagination = parse_pagination(query.page, query.limit, 50)?;
471    let search = non_empty(expose_option(&query.search));
472    let student_number = non_empty(expose_option(&query.student_number));
473    let sort = match query.sort.as_deref() {
474        Some("created") => AdminCreditRegistrationSort::Created,
475        Some("time_in_state") => AdminCreditRegistrationSort::TimeInState,
476        Some("attempts") => AdminCreditRegistrationSort::Attempts,
477        _ => AdminCreditRegistrationSort::LastActivity,
478    };
479    let only_needs_attention = query.needs_attention.unwrap_or(false);
480    let attention_reasons: &[AttentionReason] =
481        query.attention_reason.as_deref().unwrap_or_default();
482    let is_selected_by_attention = only_needs_attention || !attention_reasons.is_empty();
483    let whole_queue =
484        if is_selected_by_attention || sort == AdminCreditRegistrationSort::LastActivity {
485            Some(AttentionLookup::load(&mut conn, &app_conf, None).await?)
486        } else {
487            None
488        };
489    let needs_attention_ids = whole_queue
490        .as_ref()
491        .filter(|_| sort == AdminCreditRegistrationSort::LastActivity)
492        .map(|queue| queue.ids(|row| row.standing == AttentionStanding::NeedsAttention));
493    let selected_ids = whole_queue
494        .as_ref()
495        .filter(|_| is_selected_by_attention)
496        .map(|queue| {
497            queue.ids(|row| {
498                (!only_needs_attention || row.standing == AttentionStanding::NeedsAttention)
499                    && (attention_reasons.is_empty()
500                        || row
501                            .reasons
502                            .iter()
503                            .any(|reason| attention_reasons.contains(reason)))
504            })
505        });
506    let filters = AdminCreditRegistrationFilters {
507        states: query.state.as_deref(),
508        error_codes: query.error_code.as_deref(),
509        course_id: query.course_id,
510        course_module_id: query.course_module_id,
511        user_id: query.user_id,
512        student_number,
513        steps: query.step.as_deref().unwrap_or_default(),
514        engagements: query.engagement.as_deref().unwrap_or_default(),
515        hide_not_started: !query.include_not_started.unwrap_or(false),
516        account_linking_since: app_conf.suotar_configuration.account_linking_since,
517        first_ids: needs_attention_ids.as_deref(),
518        credit_registration_ids: selected_ids.as_deref(),
519        submitted_after: query.submitted_after,
520        submitted_before: query.submitted_before,
521        search,
522        search_id: search.and_then(|search| Uuid::parse_str(search).ok()),

849*/

851#[utoipa::path(
852    post,
853    path = "/registrations/bulk-transition",
854    operation_id = "adminBulkTransitionCreditRegistrations",
855    tag = "credit-registration-admin",
856    request_body = AdminBulkTransitionPayload,
857    responses(
858        (status = 200, description = "What each selected row did", body = AdminBulkTransitionResult),
859        (status = 422, description = "No reason given, or more ids than one call may take")
860    )
861)]
862pub async fn admin_bulk_transition_credit_registrations(
863    user: AuthUser,
864    pool: web::Data<PgPool>,
865    payload: web::Json<AdminBulkTransitionPayload>,
866) -> ControllerResult<web::Json<AdminBulkTransitionResult>> {
867    let mut conn = pool.acquire().await?;
868    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
869
870    let reason = required_reason(&payload.reason)?;
871    if payload.credit_registration_ids.len() as i64 > MAX_ROWS_PER_BULK_TRANSITION {
872        return Err(controller_err!(
873            BadRequest,
874            format!("At most {MAX_ROWS_PER_BULK_TRANSITION} registrations per call.")
875        ));
876    }
877
878    // A selection built by clicking can name the same row twice, and reporting the duplicate as a
879    // registration that does not exist would send an admin looking for a deleted row.
880    let ids: Vec<Uuid> = payload
881        .credit_registration_ids
882        .iter()
883        .copied()
884        .collect::<HashSet<_>>()
885        .into_iter()
886        .collect();
887
888    let mut tx = conn.begin().await?;
889    // Locked, and read inside the transaction: each row's refusal is judged here and acted on below,
890    // so a row the pipeline moves in between would make `apply_transition` refuse it and take every
891    // row already applied down with it.
892    let rows = credit_registrations::get_by_ids_for_update(&mut tx, &ids).await?;
893
894    let mut applied_count = 0;
895    let mut due_now_ids = Vec::new();
896    let mut skipped: HashMap<ResubmissionRefusal, i64> = HashMap::new();
897    for row in &rows {
898        match payload.action.refusal(
899            &row.resubmission_facts(),
900            ResubmissionStrictness::AnyExceptSubmissionUncertain,
901        ) {
902            Some(refusal) => *skipped.entry(refusal).or_insert(0) += 1,
903            None => {
904                let applied =
905                    apply_transition(&mut tx, row, payload.action, user.id, reason).await?;
906                if applied.needs_due_now {
907                    due_now_ids.push(row.id);
908                }
909                applied_count += 1;
910            }
911        }
912    }
913    // Batched rather than one `UPDATE` per row inside the loop above: the row transition needs its
914    // own audit event per row, but making it due now does not.
915    credit_registrations::make_due_now_batch(
916        &mut tx,
917        &due_now_ids,
918        EnrolmentCheckSource::AdminRequest,
919    )
920    .await?;
921    let mut skipped: Vec<AdminBulkTransitionSkipCount> = skipped
922        .into_iter()
923        .map(|(refusal, count)| AdminBulkTransitionSkipCount { refusal, count })
924        .collect();
925    skipped.sort_by_key(|skip| std::cmp::Reverse(skip.count));
926
927    models::credit_registration_admin_actions::record(
928        &mut tx,
929        &NewCreditRegistrationAdminAction {
930            reason: Some(reason.to_string()),
931            details: Some(serde_json::json!({
932                "action": payload.action,
933                "credit_registration_ids": payload.credit_registration_ids,
934                "skipped": skipped,
935            })),
936            affected_row_count: Some(applied_count),
937            ..NewCreditRegistrationAdminAction::new(
938                CreditRegistrationAdminAction::TransitionItem,
939                CreditRegistrationAdminActionTarget::CreditRegistration,
940                user.id,
941                GLOBAL_ADMIN_ROLE,
942            )
943        },
944    )
945    .await?;
946    tx.commit().await?;
947
948    token.authorized_ok(web::Json(AdminBulkTransitionResult {
949        applied_count: i64::from(applied_count),
950        skipped,
951        not_found_count: ids.len() as i64 - rows.len() as i64,
952        max_rows_per_call: MAX_ROWS_PER_BULK_TRANSITION,
953    }))
954}

1023/// What one hand action did to its row.
1024struct AppliedHandAction {
1025    outcome: AdminTransitionOutcome,
1026    /// Where the row ended up.
1027    state: CreditRegistrationState,
1028    needs_admin_attention: bool,
1029    /// The caller must still make the row due now.
1030    needs_due_now: bool,
1031}
1032
1033/// Applies one hand action in the caller's transaction.
1034///
1035/// The caller has already asked `admin_transition_refusal` whether this row may take the move,
1036/// because what a refusal is reported as differs per caller. Making the row due is left to the
1037/// caller too, rather than done here, so the bulk caller can batch it over every row it applies
1038/// instead of one `UPDATE` per row.
1039async fn apply_transition(
1040    tx: &mut PgConnection,
1041    row: &credit_registrations::CreditRegistration,
1042    action: AdminCreditRegistrationAction,
1043    actor_user_id: Uuid,
1044    reason: &str,
1045) -> Result<AppliedHandAction, ControllerError> {
1046    let id = row.id;
1047    Ok(match action {
1048        AdminCreditRegistrationAction::ClearNeedsAdminAttention => {
1049            if !row.needs_admin_attention {
1050                AppliedHandAction {
1051                    outcome: AdminTransitionOutcome::NoChange,
1052                    state: row.state,
1053                    needs_admin_attention: false,
1054                    needs_due_now: false,
1055                }
1056            } else {
1057                credit_registrations::set_needs_admin_attention(
1058                    tx,
1059                    id,
1060                    credit_registrations::AdminAttention::Clear,
1061                )
1062                .await?;
1063                insert_admin_action_event(tx, id, actor_user_id, reason).await?;
1064                AppliedHandAction {
1065                    outcome: AdminTransitionOutcome::Applied,
1066                    state: row.state,
1067                    needs_admin_attention: false,
1068                    needs_due_now: false,
1069                }
1070            }
1071        }
1072        AdminCreditRegistrationAction::CheckNow => {
1073            insert_admin_action_event(tx, id, actor_user_id, reason).await?;
1074            AppliedHandAction {
1075                outcome: AdminTransitionOutcome::Applied,
1076                state: row.state,
1077                needs_admin_attention: row.needs_admin_attention,
1078                needs_due_now: true,
1079            }
1080        }
1081        AdminCreditRegistrationAction::StateMove { to_state } => {
1082            let to_state = to_state.to_state();
1083            let after = credit_registrations::transition(
1084                tx,
1085                id,
1086                &Transition {
1087                    needs_admin_attention: Some(credit_registrations::AdminAttention::Clear),
1088                    event_kind: CreditRegistrationEventKind::AdminAction,
1089                    event_message: Some(reason.to_string()),
1090                    actor_user_id: Some(actor_user_id),
1091                    // Refuses to overwrite a row the pipeline (or another admin) has moved on since
1092                    // `row` was read. The bulk caller reads its rows locked, so only the single-row
1093                    // path can actually trip this.
1094                    expected_from_state: Some(row.state),
1095                    ..Transition::by_hand(to_state)
1096                },
1097            )
1098            .await?;
1099            // Nothing else brings the row forward, so without a due-now the resubmit would sit out
1100            // the backoff whatever failed last set.
1101            AppliedHandAction {
1102                outcome: AdminTransitionOutcome::Applied,
1103                state: after.state,
1104                needs_admin_attention: after.needs_admin_attention,
1105                needs_due_now: !after.state.is_terminal(),
1106            }
1107        }
1108    })
1109}

1153/// The Needs attention queue as one request saw it, for marking rows of the ledger.
1154struct AttentionLookup {
1155    /// Queue order, which `ids` preserves.
1156    rows: Vec<AttentionLookupRow>,
1157    index_by_id: HashMap<Uuid, usize>,
1158}
1159
1160struct AttentionLookupRow {
1161    id: Uuid,
1162    reasons: Vec<AttentionReason>,
1163    standing: AttentionStanding,
1164    blocking_problem: Option<BlockingProblem>,
1165    dismissed_reasons: Option<Vec<AttentionReason>>,
1166    dismissed_at: Option<DateTime<Utc>>,
1167    dismissed_by_user_id: Option<Uuid>,
1168    dismissed_by_first_name: Option<String>,
1169    dismissed_by_last_name: Option<String>,
1170    dismissal_reason: Option<String>,
1171}
1172
1173impl AttentionLookup {
1174    /// `only_ids` narrows the lookup to those rows; `None` loads the whole queue.
1175    async fn load(
1176        conn: &mut PgConnection,
1177        app_conf: &ApplicationConfiguration,
1178        only_ids: Option<&[Uuid]>,
1179    ) -> Result<Self, ControllerError> {
1180        let rules = attention_rules(conn, app_conf).await?;
1181        let rows = credit_registrations::get_attention_items(conn, &rules, only_ids)
1182            .await?
1183            .into_iter()
1184            .map(|row| {
1185                let standing = row.standing(&rules.blocking);
1186                AttentionLookupRow {
1187                    id: row.id,
1188                    standing,
1189                    blocking_problem: (standing == AttentionStanding::ExplainedByProblem)
1190                        .then(|| row.blocking_problem(&rules.blocking))
1191                        .flatten(),
1192                    reasons: row.reasons,
1193                    dismissed_reasons: row.dismissed_reasons,
1194                    dismissed_at: row.dismissed_at,
1195                    dismissed_by_user_id: row.dismissed_by_user_id,
1196                    dismissed_by_first_name: row.dismissed_by_first_name,
1197                    dismissed_by_last_name: row.dismissed_by_last_name,
1198                    dismissal_reason: row.dismissal_reason,
1199                }
1200            })
1201            .collect::<Vec<_>>();
1202        let index_by_id = rows
1203            .iter()
1204            .enumerate()
1205            .map(|(index, row)| (row.id, index))
1206            .collect();
1207        Ok(Self { rows, index_by_id })
1208    }
1209
1210    fn ids(&self, keep: impl Fn(&AttentionLookupRow) -> bool) -> Vec<Uuid> {
1211        self.rows
1212            .iter()
1213            .filter(|row| keep(row))
1214            .map(|row| row.id)
1215            .collect()
1216    }
1217
1218    fn get(&self, id: Uuid) -> Option<&AttentionLookupRow> {
1219        self.index_by_id.get(&id).map(|&index| &self.rows[index])
1220    }
1221}
