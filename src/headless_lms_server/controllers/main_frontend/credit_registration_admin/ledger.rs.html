131/// What an admin may move a row to; everything else is the pipeline's to decide.
132#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
133#[serde(rename_all = "snake_case")]
134pub enum AdminCreditRegistrationTransitionTarget {
135    /// Resubmit: the escape hatch out of `submission_uncertain`, and how a `misregistered` row is
136    /// tried again.
137    ReadyToSubmit,
138    Cancelled,
139    /// Leaves the state alone and stops the row asking for a human.
140    ClearNeedsAdminAttention,
141    /// Leaves the state alone and makes the row due, so the phase owning its state claims it on the
142    /// next pass instead of waiting out a backoff of up to a day.
143    CheckNow,
144}

168/// `serde_urlencoded` reads a single occurrence of a key as a scalar, not a one-element sequence, so
169/// a `Vec` field otherwise refuses a query string that repeats the parameter zero or one times.
170fn one_or_many<'de, D, T>(deserializer: D) -> Result<Option<Vec<T>>, D::Error>
171where
172    D: serde::Deserializer<'de>,
173    T: Deserialize<'de>,
174{
175    #[derive(Deserialize)]
176    #[serde(untagged)]
177    enum OneOrMany<T> {
178        One(T),
179        Many(Vec<T>),
180    }
181    Ok(
182        Option::<OneOrMany<T>>::deserialize(deserializer)?.map(|repr| match repr {
183            OneOrMany::One(value) => vec![value],
184            OneOrMany::Many(values) => values,
185        }),
186    )
187}

239pub async fn list_credit_registrations_for_admin(
240    user: AuthUser,
241    pool: web::Data<PgPool>,
242    query: web::Query<ListCreditRegistrationsQuery>,
243) -> ControllerResult<web::Json<AdminCreditRegistrationsPage>> {
244    let mut conn = pool.acquire().await?;
245    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
246
247    let pagination = Pagination::new(query.page.unwrap_or(1), query.limit.unwrap_or(50))
248        .map_err(|e| controller_err!(BadRequest, e.to_string()))?;
249    let search = query
250        .search
251        .as_deref()
252        .map(str::trim)
253        .filter(|search| !search.is_empty());
254    let student_number = query
255        .student_number
256        .as_deref()
257        .map(str::trim)
258        .filter(|number| !number.is_empty());
259    let filters = AdminCreditRegistrationFilters {
260        states: query.state.as_deref(),
261        error_codes: query.error_code.as_deref(),
262        course_id: query.course_id,
263        course_module_id: query.course_module_id,
264        user_id: query.user_id,
265        student_number,
266        needs_admin_attention: query.needs_admin_attention.unwrap_or(false),
267        submitted_after: query.submitted_after,
268        submitted_before: query.submitted_before,
269        search,
270        search_id: search.and_then(|search| Uuid::parse_str(search).ok()),
271        include_superseded: query.include_superseded.unwrap_or(false),
272    };
273    let sort = match query.sort.as_deref() {
274        Some("created") => AdminCreditRegistrationSort::Created,
275        Some("time_in_state") => AdminCreditRegistrationSort::TimeInState,
276        Some("attempts") => AdminCreditRegistrationSort::Attempts,
277        _ => AdminCreditRegistrationSort::LastActivity,
278    };
279
280    let total_count = credit_registrations::count_admin_facing(&mut conn, &filters).await?;
281    let data = credit_registrations::get_admin_facing(
282        &mut conn,
283        &filters,
284        sort,
285        pagination.limit(),
286        pagination.offset(),
287    )
288    .await?
289    .into_iter()
290    .map(to_admin_row)
291    .collect();
292
293    token.authorized_ok(web::Json(AdminCreditRegistrationsPage {
294        data,
295        total_count,
296        total_pages: pagination.total_pages(u32::try_from(total_count).unwrap_or(u32::MAX)),
297    }))
298}

317pub async fn get_credit_registration_for_admin(
318    user: AuthUser,
319    pool: web::Data<PgPool>,
320    credit_registration_id: web::Path<Uuid>,
321) -> ControllerResult<web::Json<AdminCreditRegistrationDetails>> {
322    let mut conn = pool.acquire().await?;
323    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
324
325    let id = *credit_registration_id;
326    let registration = one_admin_row(&mut conn, id)
327        .await?
328        .ok_or_else(|| controller_err!(NotFound, "Not found.".to_string()))?;
329    let attempts = credit_registrations::get_admin_facing(
330        &mut conn,
331        &AdminCreditRegistrationFilters {
332            user_id: Some(registration.user_id),
333            course_id: Some(registration.course_id),
334            // No field for this predicate; search_id already matches it (it is also matched against
335            // cr.id and cr.user_id, which cannot collide with a completion id).
336            search_id: Some(registration.course_module_completion_id),
337            include_superseded: true,
338            ..AdminCreditRegistrationFilters::default()
339        },
340        AdminCreditRegistrationSort::Created,
341        i64::from(u8::MAX),
342        0,
343    )
344    .await?
345    .into_iter()
346    .filter(|row| row.course_module_completion_id == registration.course_module_completion_id)
347    .map(to_admin_row)
348    .collect();
349
350    let events: Vec<AdminCreditRegistrationEvent> =
351        models::credit_registration_events::get_by_registration_id(&mut conn, id)
352            .await?
353            .into_iter()
354            .map(|event| AdminCreditRegistrationEvent {
355                id: event.id,
356                created_at: event.created_at,
357                kind: event.kind,
358                from_state: event.from_state,
359                to_state: event.to_state,
360                error_code: event.error_code,
361                message: event.message,
362                actor_user_id: event.actor_user_id,
363                suotar_api_call_id: event.suotar_api_call_id,
364                details: event.details,
365            })
366            .collect();
367    let suotar_api_calls =
368        suotar_api_calls::get_by_credit_registration_id(&mut conn, id, i64::from(u8::MAX))
369            .await?
370            .into_iter()
371            .map(to_admin_api_call)
372            .collect();
373    let actions = models::credit_registration_admin_actions::get_by_target(
374        &mut conn,
375        CreditRegistrationAdminActionTarget::CreditRegistration,
376        id,
377    )
378    .await?;
379
380    let sisu_person_id = match &registration.sisu_person_id {
381        Some(person_id) => Some(person_id.clone()),
382        None => verified_student_numbers::get_latest_including_deleted_by_user_id(
383            &mut conn,
384            registration.user_id,
385        )
386        .await?
387        .map(|link| link.sisu_person_id),
388    };
389    let linking_emails = match sisu_person_id {
390        Some(person_id) => {
391            let mails = credit_registration_account_linking_emails::get_by_sisu_person_id(
392                &mut conn, &person_id,
393            )
394            .await?;
395            build_linking_emails(&mut conn, mails).await?
396        }
397        None => Vec::new(),
398    };
399    let consent = course_credit_registration_consents::get_by_user_and_course(
400        &mut conn,
401        registration.user_id,
402        registration.course_id,
403    )
404    .await?;
405
406    token.authorized_ok(web::Json(AdminCreditRegistrationDetails {
407        registration: to_admin_row(registration),
408        attempts,
409        events,
410        suotar_api_calls,
411        actions,
412        linking_emails,
413        consent_given: consent.as_ref().map(|row| row.consent_given),
414        consent_withdrawn_at: consent.and_then(|row| row.consent_withdrawn_at),
415    }))
416}

418/**
419POST `/api/v0/main-frontend/credit-registration-admin/registrations/{credit_registration_id}/transition`
420- Moves one row by hand.
421
422The escape hatch out of `submission_uncertain`, which the pipeline never leaves on its own because
423re-importing could put a second attainment on a real transcript. Resubmitting re-checks consent, because
424a `misregistered` row sits outside the automatic machinery that would otherwise have checked it.
425*/
426#[instrument(skip(pool, payload))]
427#[utoipa::path(
428    post,
429    path = "/registrations/{credit_registration_id}/transition",
430    operation_id = "adminTransitionCreditRegistration",
431    tag = "credit-registration-admin",
432    params(("credit_registration_id" = Uuid, Path, description = "Credit registration id")),
433    request_body = AdminTransitionCreditRegistrationPayload,
434    responses(
435        (status = 200, description = "What the transition did", body = AdminTransitionCreditRegistrationResult),
436        (status = 422, description = "No reason given"),
437        (status = 404, description = "No such registration")
438    )
439)]
440pub async fn admin_transition_credit_registration(
441    user: AuthUser,
442    pool: web::Data<PgPool>,
443    credit_registration_id: web::Path<Uuid>,
444    payload: web::Json<AdminTransitionCreditRegistrationPayload>,
445) -> ControllerResult<web::Json<AdminTransitionCreditRegistrationResult>> {
446    let mut conn = pool.acquire().await?;
447    let token = authorize_credit_registration_admin(&mut conn, user.id).await?;
448
449    let reason = required_reason(&payload.reason)?;
450    let id = *credit_registration_id;
451    let row = credit_registrations::get_by_id(&mut conn, id).await?;
452    if row.superseded_by_id.is_some() {
453        return Err(controller_err!(
454            BadRequest,
455            "This attempt has been replaced by a later one. Act on the later one.".to_string()
456        ));
457    }
458
459    if payload.to_state == AdminCreditRegistrationTransitionTarget::ReadyToSubmit {
460        let consent = course_credit_registration_consents::get_by_user_and_course(
461            &mut conn,
462            row.user_id,
463            row.course_id,
464        )
465        .await?;
466        // consent_withdrawn_at survives a later consent as an audit trail, so consent_given alone is
467        // what counts as consented, here and in the precondition engine.
468        let consented = consent.is_some_and(|c| c.consent_given);
469        if !consented {
470            return token.authorized_ok(web::Json(AdminTransitionCreditRegistrationResult {
471                outcome: AdminTransitionOutcome::RefusedWithoutConsent,
472                state: row.state,
473                needs_admin_attention: row.needs_admin_attention,
474            }));
475        }
476    }
477
478    let mut tx = conn.begin().await?;
479    let (outcome, after_state, needs_admin_attention) = match payload.to_state {
480        AdminCreditRegistrationTransitionTarget::ClearNeedsAdminAttention => {
481            if !row.needs_admin_attention {
482                (AdminTransitionOutcome::NoChange, row.state, false)
483            } else {
484                credit_registrations::set_needs_admin_attention(&mut tx, id, false).await?;
485                insert_admin_action_event(&mut tx, id, user.id, reason).await?;
486                (AdminTransitionOutcome::Applied, row.state, false)
487            }
488        }
489        AdminCreditRegistrationTransitionTarget::CheckNow => {
490            credit_registrations::make_due_now(&mut tx, id).await?;
491            insert_admin_action_event(&mut tx, id, user.id, reason).await?;
492            (
493                AdminTransitionOutcome::Applied,
494                row.state,
495                row.needs_admin_attention,
496            )
497        }
498        target => {
499            let to_state = match target {
500                AdminCreditRegistrationTransitionTarget::ReadyToSubmit => {
501                    CreditRegistrationState::ReadyToSubmit
502                }
503                _ => CreditRegistrationState::Cancelled,
504            };
505            let after = credit_registrations::transition(
506                &mut tx,
507                id,
508                &Transition {
509                    needs_admin_attention: Some(false),
510                    event_kind: CreditRegistrationEventKind::AdminAction,
511                    event_message: Some(reason.to_string()),
512                    actor_user_id: Some(user.id),
513                    // `row` was read before this transaction started; refuse to overwrite if the
514                    // pipeline (or another admin) has since moved the row on.
515                    expected_from_state: Some(row.state),
516                    ..Transition::to(to_state)
517                },
518            )
519            .await?;
520            // Nothing else brings the row forward, so without this the resubmit sits out the backoff
521            // whatever failed last set.
522            if !after.state.is_terminal() {
523                credit_registrations::make_due_now(&mut tx, id).await?;
524            }
525            (
526                AdminTransitionOutcome::Applied,
527                after.state,
528                after.needs_admin_attention,
529            )
530        }
531    };
532    models::credit_registration_admin_actions::record(
533        &mut tx,
534        &NewCreditRegistrationAdminAction {
535            target_id: Some(id),
536            reason: Some(reason.to_string()),
537            before_state: Some(row.state),
538            after_state: Some(after_state),
539            details: Some(serde_json::json!({ "outcome": outcome })),
540            affected_row_count: Some(1),
541            ..NewCreditRegistrationAdminAction::new(
542                CreditRegistrationAdminAction::TransitionItem,
543                CreditRegistrationAdminActionTarget::CreditRegistration,
544                user.id,
545                GLOBAL_ADMIN_ROLE,
546            )
547        },
548    )
549    .await?;
550    tx.commit().await?;
551
552    token.authorized_ok(web::Json(AdminTransitionCreditRegistrationResult {
553        outcome,
554        state: after_state,
555        needs_admin_attention,
556    }))
557}

559/// Records an admin action against the row's timeline without moving its state, for the two
560/// transitions that only clear a flag or reschedule the row.
561async fn insert_admin_action_event(
562    tx: &mut PgConnection,
563    id: Uuid,
564    actor_user_id: Uuid,
565    reason: &str,
566) -> Result<(), ControllerError> {
567    models::credit_registration_events::insert(
568        tx,
569        &models::credit_registration_events::NewCreditRegistrationEvent {
570            actor_user_id: Some(actor_user_id),
571            message: Some(reason.to_string()),
572            ..models::credit_registration_events::NewCreditRegistrationEvent::new(
573                id,
574                CreditRegistrationEventKind::AdminAction,
575            )
576        },
577    )
578    .await?;
579    Ok(())
580}
