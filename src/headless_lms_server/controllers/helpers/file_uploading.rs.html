48/** Tracks uploaded object paths for cleanup when the batch fails. */
49pub struct ExerciseServiceUploadCleanup {
50    pub path: String,
51}

53/// Deletes the objects an upload has already stored, unless the upload got far enough to
54/// [`disarm`](Self::disarm) it.
55///
56/// Objects reach the store before their rows are committed, so an upload that fails part way has to
57/// delete them itself; the reaper only finds objects through the rows that were never written. On
58/// the handler's own error path call [`clean_up`](Self::clean_up) and await it. The `Drop` path is
59/// the backstop for the case that has no error path at all: the client aborting the multipart body,
60/// which drops the handler future at an await point.
61pub struct UploadCleanup {
62    pub uploaded_paths: Vec<ExerciseServiceUploadCleanup>,
63    file_store: web::Data<dyn FileStore>,
64    armed: bool,
65}
66
67impl UploadCleanup {
68    pub fn new(file_store: web::Data<dyn FileStore>) -> Self {
69        Self {
70            uploaded_paths: Vec::new(),
71            file_store,
72            armed: true,
73        }
74    }
75
76    /// Deletes what has been stored so far and disarms, so the `Drop` backstop cannot delete the
77    /// same objects again.
78    pub async fn clean_up(&mut self) {
79        self.armed = false;
80        for uploaded in std::mem::take(&mut self.uploaded_paths) {
81            if let Err(delete_error) = self.file_store.delete(Path::new(&uploaded.path)).await {
82                error!(
83                    "Failed to delete file '{}' during cleanup: {delete_error}",
84                    uploaded.path
85                );
86            }
87        }
88    }
89
90    /// Call only once the uploads are recorded, with no await in between: an await would give the
91    /// runtime a chance to drop the handler and delete objects that already have rows.
92    pub fn disarm(&mut self) {
93        self.armed = false;
94    }
95}
96
97impl Drop for UploadCleanup {
98    fn drop(&mut self) {
99        if !self.armed || self.uploaded_paths.is_empty() {
100            return;
101        }
102        let uploaded_paths = std::mem::take(&mut self.uploaded_paths);
103        let file_store = self.file_store.clone();
104        // `drop` cannot await, so the deletes run detached and outlive this request.
105        actix_web::rt::spawn(async move {
106            for uploaded in uploaded_paths {
107                if let Err(delete_error) = file_store.delete(Path::new(&uploaded.path)).await {
108                    error!(
109                        "Failed to delete file '{}' during cleanup: {delete_error}",
110                        uploaded.path
111                    );
112                }
113            }
114        });
115    }
116}

126/// Processes an upload from an exercise service, an exercise iframe or a native client.
127/// This function assumes that any permission checks have already been made.
128///
129/// `path_prefix` only namespaces the stored objects; it carries no authorization meaning.
130pub async fn process_exercise_service_upload(
131    conn: &mut PgConnection,
132    path_prefix: &str,
133    payload: Multipart,
134    file_store: &dyn FileStore,
135    uploaded_paths: &mut Vec<ExerciseServiceUploadCleanup>,
136    uploader: Option<Uuid>,
137    base_url: &str,
138) -> Result<Vec<ExerciseServiceUpload>, ControllerError> {
139    let streamed =
140        stream_exercise_service_upload(path_prefix, payload, file_store, uploaded_paths, base_url)
141            .await?;
142    let mut tx = conn.begin().await?;
143    let uploads = record_exercise_service_upload(&mut tx, streamed, uploader).await?;
144    tx.commit().await?;
145    Ok(uploads)
146}

148/// The parts of an upload that have reached the object store but have no database row yet.
149pub struct StreamedExerciseServiceUpload {
150    parts: Vec<ExerciseServiceUploadMetadata>,
151}
152
153/// Streams every multipart part to the object store, issuing no statements at all.
154///
155/// Split from the row inserts on purpose: the limits here are byte-based, not time-based, so a
156/// handler that opened a transaction first would hold a connection `idle in transaction` for as
157/// long as the client cares to trickle 100 MiB, and enough concurrent slow uploads would exhaust
158/// the pool and hold back the vacuum xmin horizon.
159pub async fn stream_exercise_service_upload(
160    path_prefix: &str,
161    mut payload: Multipart,
162    file_store: &dyn FileStore,
163    uploaded_paths: &mut Vec<ExerciseServiceUploadCleanup>,
164    base_url: &str,
165) -> Result<StreamedExerciseServiceUpload, ControllerError> {
166    let mut parts = Vec::new();
167    let mut ids = HashSet::new();
168    let batch_bytes = Arc::new(AtomicU64::new(0));
169    while let Some(item) = payload.next().await {
170        let field = item.map_err(|err| {
171            controller_err!(
172                BadRequest,
173                format!("Failed to read multipart field: {}", err),
174                anyhow::anyhow!("Multipart error: {}", err)
175            )
176        })?;
177        validate_exercise_upload_file_count(parts.len())?;
178        let field_name = {
179            let name_ref = field.name().ok_or_else(|| {
180                controller_err!(
181                    BadRequest,
182                    "Tried to upload a multipart field without a field name or field ID"
183                        .to_string()
184                )
185            })?;
186            name_ref.to_string()
187        };
188
189        validate_exercise_upload_id(&field_name, &mut ids)?;
190        let filename = validate_exercise_upload_filename(
191            field
192                .content_disposition()
193                .and_then(|disposition| disposition.get_filename()),
194        )?
195        .to_string();
196
197        let random_filename = generate_random_string(32);
198        let path = format!("{path_prefix}/{random_filename}");
199        uploaded_paths.push(ExerciseServiceUploadCleanup { path: path.clone() });
200        let mime_type = field
201            .content_type()
202            .map(ToString::to_string)
203            .unwrap_or_default();
204        let (stream, stream_error, uploaded_bytes) =
205            limited_exercise_upload_stream(field, batch_bytes.clone());
206        let upload_result = file_store
207            .upload_stream(Path::new(&path), stream, &mime_type)
208            .await;
209        if let Some(error) = stream_error
210            .lock()
211            .unwrap_or_else(|poisoned| poisoned.into_inner())
212            .take()
213        {
214            return Err(error);
215        }
216        upload_result?;
217        let url = format!("{base_url}/api/v0/files/{path}");
218        parts.push(ExerciseServiceUploadMetadata {
219            path,
220            filename,
221            mime_type,
222            size_bytes: uploaded_bytes.load(Ordering::SeqCst) as i64,
223            url,
224        });
225    }
226    validate_exercise_upload_not_empty(parts.len())?;
227    Ok(StreamedExerciseServiceUpload { parts })
228}

230/// Records the `file_uploads` rows for an already-streamed upload.
231///
232/// Takes the caller's transaction so that a caller which has further rows to write — the answer
233/// upload routes bind each upload to an exercise and user — lands them together with these.
234pub async fn record_exercise_service_upload(
235    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
236    streamed: StreamedExerciseServiceUpload,
237    uploader: Option<Uuid>,
238) -> Result<Vec<ExerciseServiceUpload>, ControllerError> {
239    let StreamedExerciseServiceUpload { parts } = streamed;
240    let mut uploads = Vec::with_capacity(parts.len());
241    for part in parts {
242        let file_upload_id = models::file_uploads::insert(
243            tx,
244            &part.filename,
245            &part.path,
246            &part.mime_type,
247            uploader,
248            Some(part.size_bytes),
249        )
250        .await?;
251        uploads.push(ExerciseServiceUpload {
252            entry: ExerciseServiceUploadResultEntry {
253                id: file_upload_id,
254                url: part.url,
255            },
256            name: part.filename,
257            mime: part.mime_type,
258            size_bytes: part.size_bytes,
259        });
260    }
261    Ok(uploads)
262}

264/// Returns a side channel for typed upload-limit errors because the stream yields `anyhow` errors,
265/// and the running byte count of this part. Both are only meaningful once `upload_stream` has
266/// returned: the count is still climbing while the stream is being consumed.
267fn limited_exercise_upload_stream(
268    field: mp::Field,
269    batch_bytes: Arc<AtomicU64>,
270) -> (
271    GenericPayload,
272    Arc<Mutex<Option<ControllerError>>>,
273    Arc<AtomicU64>,
274) {
275    let per_file_bytes = Arc::new(AtomicU64::new(0));
276    let stream_error = Arc::new(Mutex::new(None));
277    let payload_stream_error = stream_error.clone();
278    let payload_per_file_bytes = per_file_bytes.clone();
279    let stream = Box::pin(futures::stream::try_unfold(
280        (
281            field,
282            payload_per_file_bytes,
283            batch_bytes,
284            payload_stream_error,
285        ),
286        |(mut field, per_file_bytes, batch_bytes, stream_error)| async move {
287            let Some(chunk) = field.next().await else {
288                return Ok(None);
289            };
290            let chunk = chunk.map_err(|error| anyhow::Error::msg(error.to_string()))?;
291            if let Err(error) =
292                consume_exercise_upload_bytes(&per_file_bytes, &batch_bytes, chunk.len())
293            {
294                *stream_error
295                    .lock()
296                    .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(error);
297                return Err(anyhow::Error::msg("Exercise upload exceeds the size limit"));
298            }
299            Ok(Some((
300                chunk,
301                (field, per_file_bytes, batch_bytes, stream_error),
302            )))
303        },
304    ));
305    (stream, stream_error, per_file_bytes)
306}

308fn validate_exercise_upload_file_count(files_received: usize) -> Result<(), ControllerError> {
309    if files_received < EXERCISE_UPLOAD_MAX_FILES {
310        return Ok(());
311    }
312    Err(controller_err!(
313        BadRequest,
314        format!("A maximum of {EXERCISE_UPLOAD_MAX_FILES} files can be uploaded at once")
315    ))
316}

318fn validate_exercise_upload_id(
319    field_name: &str,
320    ids: &mut HashSet<String>,
321) -> Result<(), ControllerError> {
322    Uuid::parse_str(field_name).map_err(|_| {
323        controller_err!(
324            BadRequest,
325            "Each exercise upload field name must be a UUID".to_string()
326        )
327    })?;
328    if ids.insert(field_name.to_string()) {
329        return Ok(());
330    }
331    Err(controller_err!(
332        BadRequest,
333        "Duplicate exercise upload field id".to_string()
334    ))
335}

358fn consume_exercise_upload_bytes(
359    per_file_bytes: &AtomicU64,
360    batch_bytes: &AtomicU64,
361    chunk_len: usize,
362) -> Result<(), ControllerError> {
363    let chunk_len = u64::try_from(chunk_len).map_err(|error| {
364        controller_err!(
365            BadRequest,
366            "exercise upload chunk length overflow".to_string(),
367            error
368        )
369    })?;
370    let file_total = per_file_bytes.fetch_add(chunk_len, Ordering::Relaxed) + chunk_len;
371    let batch_total = batch_bytes.fetch_add(chunk_len, Ordering::Relaxed) + chunk_len;
372    if file_total <= EXERCISE_UPLOAD_MAX_FILE_BYTES
373        && batch_total <= EXERCISE_UPLOAD_MAX_BATCH_BYTES
374    {
375        return Ok(());
376    }
377    Err(controller_err!(
378        BadRequest,
379        "Exercise upload exceeds the 100 MiB per-file or batch limit"
380    ))
381}

454/// Processes an upload from CMS.
455pub async fn upload_file_from_cms(
456    headers: &HeaderMap,
457    mut payload: Multipart,
458    store_kind: StoreKind,
459    file_store: &dyn FileStore,
460    conn: &mut PgConnection,
461    user: AuthUser,
462) -> Result<PathBuf, ControllerError> {
463    let file_payload = payload.next().await.ok_or_else(|| {
464        ControllerError::new(ControllerErrorType::BadRequest, "Missing form data", None)
465    })?;
466    match file_payload {
467        Ok(field) => {
468            upload_field_from_cms(headers, field, store_kind, file_store, conn, user).await
469        }
470        Err(err) => Err(ControllerError::new(
471            ControllerErrorType::InternalServerError,
472            err.to_string(),
473            None,
474        )),
475    }
476}

478/// Processes an upload from CMS.
479pub async fn upload_field_from_cms(
480    headers: &HeaderMap,
481    field: Field,
482    store_kind: StoreKind,
483    file_store: &dyn FileStore,
484    conn: &mut PgConnection,
485    user: AuthUser,
486) -> Result<PathBuf, ControllerError> {
487    validate_media_headers(headers, &user, conn).await?;
488    let path = match field.content_type().map(|ct| ct.type_()) {
489        Some(mime::AUDIO) => generate_audio_path(&field, store_kind)?,
490        Some(mime::IMAGE) => generate_image_path(&field, store_kind)?,
491        _ => generate_file_path(&field, store_kind)?,
492    };
493    upload_field_to_storage(conn, &path, field, file_store, Some(user)).await?;
494    Ok(path)
495}

497/// Processes an upload for an organization's image.
498pub async fn upload_image_for_organization(
499    headers: &HeaderMap,
500    mut payload: Multipart,
501    organization: &DatabaseOrganization,
502    file_store: &Arc<dyn FileStore>,
503    user: AuthUser,
504    conn: &mut PgConnection,
505) -> Result<PathBuf, ControllerError> {
506    validate_media_headers(headers, &user, conn).await?;
507    let next_payload: Result<Field, mp::MultipartError> =
508        payload.next().await.ok_or_else(|| {
509            ControllerError::new(ControllerErrorType::BadRequest, "Missing form data", None)
510        })?;
511    match next_payload {
512        Ok(field) => {
513            let path: PathBuf = match field.content_type().map(|ct| ct.type_()) {
514                Some(mime::IMAGE) => {
515                    generate_image_path(&field, StoreKind::Organization(organization.id))
516                }
517                Some(unsupported) => Err(ControllerError::new(
518                    ControllerErrorType::BadRequest,
519                    format!("Unsupported image Mime type: {}", unsupported),
520                    None,
521                )),
522                None => Err(ControllerError::new(
523                    ControllerErrorType::BadRequest,
524                    "Missing image Mime type",
525                    None,
526                )),
527            }?;
528            upload_field_to_storage(conn, &path, field, file_store.as_ref(), Some(user)).await?;
529            Ok(path)
530        }
531        Err(err) => Err(ControllerError::new(
532            ControllerErrorType::InternalServerError,
533            err.to_string(),
534            None,
535        )),
536    }
537}

539// These limits must match the limits in CMS/src/services/backend/media/uploadMediaToServer.ts
540// If you modify these, update the TypeScript file as well.
541// Note: The nginx ingress also has a limit on max request size (see kubernetes/base/ingress.yml)
542const FILE_SIZE_LIMITS: &[(mime::Name, i32)] = &[
543    // 10 MB for images
544    (mime::IMAGE, 10 * 1024 * 1024),
545    // 100 MB for audio
546    (mime::AUDIO, 100 * 1024 * 1024),
547    // 100 MB for video
548    (mime::VIDEO, 100 * 1024 * 1024),
549    // 25 MB for documents/other files
550    (mime::APPLICATION, 25 * 1024 * 1024),
551];
552// 10 MB default fallback
553const DEFAULT_FILE_SIZE_LIMIT: i32 = 10 * 1024 * 1024;
554
555fn get_size_limit_for_mime(mime_type: Option<mime::Name>) -> i32 {
556    mime_type
557        .and_then(|mime| FILE_SIZE_LIMITS.iter().find(|(m, _)| *m == mime))
558        .map(|(_, size)| *size)
559        .unwrap_or(DEFAULT_FILE_SIZE_LIMIT)
560}

562/// Uploads the data from the multipart `field` to the given `path` in file storage.
563async fn upload_field_to_storage(
564    conn: &mut PgConnection,
565    path: &Path,
566    field: mp::Field,
567    file_store: &dyn FileStore,
568    uploader: Option<AuthUser>,
569) -> Result<(), ControllerError> {
570    // Check file size limit based on mime type
571    let mime_type = field.content_type().map(|ct| ct.type_());
572    let size_limit = get_size_limit_for_mime(mime_type);
573
574    // Get size from content disposition if available
575    // Note: This does not enforce the size of the file since the client can lie about the content length
576    if let Some(content_disposition) = field.content_disposition()
577        && let Some(size_str) = content_disposition
578            .parameters
579            .iter()
580            .find_map(|p| p.as_unknown("size"))
581        && let Ok(size) = size_str.parse::<u64>()
582        && size > size_limit as u64
583    {
584        return Err(ControllerError::new(
585            ControllerErrorType::BadRequest,
586            format!(
587                "File size {} exceeds limit of {} bytes for type {}",
588                size,
589                size_limit,
590                mime_type.map_or("unknown".to_string(), |m| m.to_string())
591            ),
592            None,
593        ));
594    }
595
596    // TODO: convert archives into a uniform format
597    let mime_type = field
598        .content_type()
599        .map(|ct| ct.to_string())
600        .unwrap_or_default();
601
602    let name = {
603        let name_ref = field.name().ok_or_else(|| {
604            ControllerError::new(
605                ControllerErrorType::BadRequest,
606                "Tried to upload a file without a file name".to_string(),
607                None,
608            )
609        })?;
610        name_ref.to_string()
611    };
612
613    let contents = Box::pin(field.map_err(|orig| anyhow::Error::msg(orig.to_string())));
614
615    upload_file_to_storage(
616        conn,
617        path,
618        &name,
619        &mime_type,
620        contents,
621        file_store,
622        uploader.map(|u| u.id),
623    )
624    .await?;
625    Ok(())
626}

684fn make_filename_safe(path: &PathBuf) -> PathBuf {
685    let mut path_buf = path.to_owned();
686    let random_string = Alphanumeric.sample_string(&mut rand::rng(), 25);
687    path_buf.set_file_name(random_string);
688    if let Some(ext) = path.extension() {
689        // For convenience, we'll keep the original extension in most cases. We'll just filter out any potentially problematic characters.
690        let ext = ext
691            .to_str()
692            .unwrap_or("")
693            .chars()
694            .filter(|c| c.is_alphanumeric())
695            .collect::<String>();
696        path_buf.set_extension(ext);
697    }
698    path_buf
699}

711/// Generates a path for an audio file with the appropriate extension.
712fn generate_audio_path(field: &Field, store_kind: StoreKind) -> Result<PathBuf, ControllerError> {
713    let extension = match field
714        .content_type()
715        .map(|ct| ct.to_string())
716        .unwrap_or_default()
717        .as_str()
718    {
719        "audio/aac" => ".aac",
720        "audio/mpeg" => ".mp3",
721        "audio/ogg" => ".oga",
722        "audio/opus" => ".opus",
723        "audio/wav" => ".wav",
724        "audio/webm" => ".weba",
725        "audio/midi" => ".mid",
726        "audio/x-midi" => ".mid",
727        unsupported => {
728            return Err(ControllerError::new(
729                ControllerErrorType::BadRequest,
730                format!("Unsupported audio Mime type: {}", unsupported),
731                None,
732            ));
733        }
734    };
735    let mut file_name = generate_random_string(30);
736    file_name.push_str(extension);
737    let path = path(&file_name, FileType::Audio, store_kind);
738    Ok(path)

741/// Generates a path for a generic file with the appropriate extension based on its filename.
742fn generate_file_path(field: &Field, store_kind: StoreKind) -> Result<PathBuf, ControllerError> {
743    let field_content = field.content_disposition().ok_or_else(|| {
744        ControllerError::new(
745            ControllerErrorType::BadRequest,
746            "No content disposition in uploaded file".to_string(),
747            None,
748        )
749    })?;
750    let field_content_name = field_content.get_filename().ok_or_else(|| {
751        ControllerError::new(
752            ControllerErrorType::BadRequest,
753            "Missing file name in content-disposition",
754            None,
755        )
756    })?;
757
758    let mut file_name = generate_random_string(30);
759    let uploaded_file_extension = get_extension_from_filename(field_content_name);
760    if let Some(extension) = uploaded_file_extension {
761        file_name.push_str(format!(".{}", extension).as_str());
762    }
763
764    let path = path(&file_name, FileType::File, store_kind);
765    Ok(path)
766}

801/// Generates a path for an audio file with the appropriate extension.
802async fn validate_media_headers(
803    headers: &HeaderMap,
804    user: &AuthUser,
805    conn: &mut PgConnection,
806) -> ControllerResult<()> {
807    let content_type = headers.get(header::CONTENT_TYPE).ok_or_else(|| {
808        ControllerError::new(
809            ControllerErrorType::BadRequest,
810            "Please provide a Content-Type header",
811            None,
812        )
813    })?;
814    let content_type_string = String::from_utf8_lossy(content_type.as_bytes()).to_string();
815
816    if !content_type_string.contains("multipart/form-data") {
817        return Err(ControllerError::new(
818            ControllerErrorType::BadRequest,
819            format!("Unsupported type: {}", content_type_string),
820            None,
821        ));
822    }
823
824    let content_length = headers.get(header::CONTENT_LENGTH).ok_or_else(|| {
825        ControllerError::new(
826            ControllerErrorType::BadRequest,
827            "Please provide a Content-Length in header",
828            None,
829        )
830    })?;
831    let content_length_number = String::from_utf8_lossy(content_length.as_bytes())
832        .to_string()
833        .parse::<i32>()
834        .map_err(|original_err| {
835            ControllerError::new(
836                ControllerErrorType::InternalServerError,
837                original_err.to_string(),
838                Some(original_err.into()),
839            )
840        })?;
841
842    let mime_type = headers
843        .get("X-File-Type")
844        .map(|h| h.to_str().unwrap_or("application/octet-stream"))
845        .unwrap_or("application/octet-stream")
846        .split('/')
847        .next()
848        .map(|s| match s {
849            "image" => mime::IMAGE,
850            "audio" => mime::AUDIO,
851            "video" => mime::VIDEO,
852            "application" => mime::APPLICATION,
853            _ => mime::APPLICATION,
854        });
855    let size_limit = get_size_limit_for_mime(mime_type);
856
857    // Note: This does not enforce the size of the file since the client can lie about the content length
858    if content_length_number > size_limit {
859        return Err(ControllerError::new(
860            ControllerErrorType::BadRequest,
861            format!(
862                "File size {} exceeds limit of {} bytes for type {}",
863                content_length_number,
864                size_limit,
865                mime_type.map_or("unknown".to_string(), |m| m.to_string())
866            ),
867            None,
868        ));
869    }
870
871    let token = authorize(conn, Act::Teach, Some(user.id), Res::AnyCourse).await?;
872    token.authorized_ok(())
873}

881fn path(file_name: &str, file_type: FileType, store_kind: StoreKind) -> PathBuf {
882    let (base_dir, base_id) = match store_kind {
883        StoreKind::Organization(id) => ("organization", id),
884        StoreKind::Course(id) => ("course", id),
885        StoreKind::Exam(id) => ("exam", id),
886    };
887    let file_type_subdir = match file_type {
888        FileType::Image => "images",
889        FileType::Audio => "audios",
890        FileType::File => "files",
891    };
892    [base_dir, &base_id.to_string(), file_type_subdir, file_name]
893        .iter()
894        .collect()
895}
