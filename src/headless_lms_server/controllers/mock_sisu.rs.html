20/// Each mock code has a current version (`-1`) and an ended one (`-2`).
21async fn mock_sisu_id_query(
22    query_params: web::Query<MockSisuRequest>,
23    app_conf: web::Data<ApplicationConfiguration>,
24) -> ControllerResult<String> {
25    assert!(app_conf.test_mode && app_conf.test_sisu);
26    let code = query_params.code_query.as_str();
27    let search_results = match MOCK_COURSE_CODES.iter().position(|c| *c == code) {
28        Some(index) => (1..=2)
29            .map(|version| SearchResult {
30                id: format!("mock-id-{}-{version}", index + 1),
31                code: Some(code.to_string()),
32            })
33            .collect(),
34        None => vec![],
35    };
36    let res = serde_json::to_string(&CourseUnitSearchResults { search_results })?;
37    let token = skip_authorize();
38    token.authorized_ok(res)
39}

41async fn mock_sisu_course_info(
42    app_conf: web::Data<ApplicationConfiguration>,
43    id: web::Path<String>,
44) -> ControllerResult<HttpResponse> {
45    assert!(app_conf.test_mode && app_conf.test_sisu);
46    let token = skip_authorize();
47    let Some((index, version)) = id
48        .strip_prefix("mock-id-")
49        .and_then(|rest| rest.split_once('-'))
50        .and_then(|(index, version)| Some((index.parse::<usize>().ok()?, version)))
51        .filter(|(index, version)| {
52            (1..=MOCK_COURSE_CODES.len()).contains(index) && matches!(*version, "1" | "2")
53        })
54    else {
55        return token.authorized_ok(HttpResponse::NotFound().finish());
56    };
57    let validity_period = if version == "1" {
58        SisuCourseInfoValidityPeriod {
59            start_date: Some("2020-08-01".to_string()),
60            end_date: None,
61        }
62    } else {
63        SisuCourseInfoValidityPeriod {
64            start_date: Some("2015-08-01".to_string()),
65            end_date: Some("2020-08-01".to_string()),
66        }
67    };
68    let course_unit = mock_course_unit(
69        id.as_str(),
70        MOCK_COURSE_CODES[index - 1],
71        &format!("Mock Docker {index}"),
72        validity_period,
73    );
74    token.authorized_ok(HttpResponse::Ok().json(course_unit))
75}

139async fn mock_sisu_person(

145    let (first_name, last_name, email, title) = match id.as_str() {

155    token.authorized_ok(HttpResponse::Ok().json(SisuPerson {
156        id: id.into_inner(),
157        first_name: Some(first_name.to_string()),
158        last_name: Some(last_name.to_string()),
159        email_address: Some(email.to_string()),
160        titles: vec![Additional {
161            en: Some(title.to_string()),
162            fi: None,

164        }],

168async fn mock_sisu_organisation(
169    app_conf: web::Data<ApplicationConfiguration>,
170    id: web::Path<String>,
171) -> ControllerResult<HttpResponse> {
172    assert!(app_conf.test_mode && app_conf.test_sisu);
173    let token = skip_authorize();
174    if id.as_str() != MOCK_ORGANISATION_ID {
175        return token.authorized_ok(HttpResponse::NotFound().finish());
176    }
177    token.authorized_ok(HttpResponse::Ok().json(json!({
178        "id": MOCK_ORGANISATION_ID,
179        "name": { "en": "Mock Degree Programme" },
180    })))
181}
