5/**
6POST `/api/v0/exercise-services/grading/grading-update/:submission_id`
7
8Receives a grading update from an exercise service.
9*/
10#[utoipa::path(
11    post,
12    path = "/grading-update/{submission_id}",
13    operation_id = "updateExerciseTaskGrading",
14    tag = "exercise-services-grading",
15    params(
16        ("submission_id" = Uuid, Path, description = "Exercise task submission id")
17    ),
18    request_body = ExerciseTaskGradingResult,
19    responses(
20        (status = 200, description = "Grading update applied")
21    )
22)]
23#[instrument(skip(pool))]
24async fn grading_update(
25    submission_id: web::Path<Uuid>,
26    grading_result: web::Json<ExerciseTaskGradingResult>,
27    grading_update_claim: GradingUpdateClaim,
28    pool: web::Data<PgPool>,
29) -> ControllerResult<web::Json<()>> {
30    // accessed from exercise services, can't authenticate using login,
31    // the upload claim is used to verify requests instead
32    let token = skip_authorize();
33    let grading_result = grading_result.into_inner();
34
35    // Ensure that the claim is valid for this specific submission
36    verify_claim_matches_submission(*submission_id, grading_update_claim.submission_id())?;
37
38    let mut conn = pool.acquire().await?;
39    apply_grading_update(&mut conn, *submission_id, &grading_result).await?;
40
41    token.authorized_ok(web::Json(()))
42}
43
44/// Rejects a grading update whose signed claim authorizes a different submission than the one
45/// addressed by the URL path.
46fn verify_claim_matches_submission(
47    submission_id: Uuid,
48    claim_submission_id: Uuid,
49) -> Result<(), ControllerError> {
50    if submission_id != claim_submission_id {
51        return Err(controller_err!(
52            BadRequest,
53            "Grading upload claim didn't match the submission id".to_string()
54        ));
55    }
56    Ok(())
57}
58
59/// Applies a grading result to the submission's existing grading.
60async fn apply_grading_update(
61    conn: &mut PgConnection,
62    submission_id: Uuid,
63    grading_result: &ExerciseTaskGradingResult,
64) -> Result<(), ControllerError> {
65    let submission = models::exercise_task_submissions::get_submission(conn, submission_id).await?;
66    let slide =
67        models::exercise_slides::get_exercise_slide(conn, submission.exercise_slide_id).await?;
68    let grading =
69        models::exercise_task_gradings::get_by_exercise_task_submission_id(conn, submission_id)
70            .await?
71            .ok_or_else(|| {
72                controller_err!(
73                    BadRequest,
74                    "No existing grading for the submission found".to_string()
75                )
76            })?;
77    let exercise = models::exercises::get_by_id(conn, slide.exercise_id).await?;
78    models::exercise_task_gradings::update_grading(conn, &grading, grading_result, &exercise)
79        .await?;
80    Ok(())
81}

120    /// An exercise service must not be able to use a claim signed for one submission to
121    /// overwrite the grading of another; the path id and the claim's id have to agree.
122    #[test]
123    fn claim_for_another_submission_is_rejected() {
124        let submission_id = Uuid::new_v4();
125        let other_submission_id = Uuid::new_v4();
126        let err = verify_claim_matches_submission(submission_id, other_submission_id)
127            .expect_err("a claim for another submission must be rejected");
128        assert_eq!(err.status_code(), StatusCode::UNPROCESSABLE_ENTITY);
129        let value = error_body(err);
130        assert_eq!(value["type"], "validation_error");
131        assert!(
132            value["message"]
133                .as_str()
134                .unwrap_or_default()
135                .contains("didn't match the submission id"),
136            "unexpected message: {}",
137            value["message"]
138        );
139    }

141    #[test]
142    fn claim_for_the_same_submission_is_accepted() {
143        let submission_id = Uuid::new_v4();
144        assert!(verify_claim_matches_submission(submission_id, submission_id).is_ok());
145    }

248    /// Decodes a controller error's JSON response body.
249    fn error_body(err: ControllerError) -> serde_json::Value {
250        let response = err.error_response();
251        let bytes = actix_web::body::to_bytes(response.into_body())
252            .now_or_never()
253            .expect("body resolves immediately")
254            .expect("body bytes");
255        serde_json::from_slice(&bytes).expect("json")
256    }
