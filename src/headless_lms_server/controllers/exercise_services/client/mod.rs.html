1/*!
2Handlers for `/api/v0/exercise-services/client`.
3
4A generic native-client API for exercise services: over plain HTTP it plays the role an
5exercise service's in-browser IFrame plays on the web (download a stub, edit locally,
6submit, poll grading, review old submissions). Specs and answers stay opaque plugin-owned
7blobs the host only forwards.
8
9Which services this API serves is not hardcoded: an exercise service is served exactly when it
10declares `supports_native_client` in its service info. The course/exercise queries filter on that
11capability, and submit rejects a task whose service lacks it.
12*/

108/// Refuses a request over the user's quota with 429 `rate_limited`. The quota is counted per server
109/// process, so with several replicas a user gets that many times the quota at most.
110fn check_user_quota(limits: &EndpointLimiters, user_id: Uuid) -> Result<(), ControllerError> {
111    limits
112        .check_key(user_id.to_string())
113        .map_err(|retry_after| {
114            controller_err!(
115                TooManyRequests(Some(retry_after)),
116                "Too many requests; try again later.".to_string()
117            )
118        })
119}

121/// Header a client sends to advertise its version, e.g. `0.39.4`.
122const CLIENT_VERSION_HEADER: &str = "X-Client-Version";
123
124/// Rejects clients older than `minimum` with `426 Upgrade Required`; a `None` minimum
125/// passes everything. When a minimum is set, a missing or unparseable client version
126/// counts as obsolete.
127fn check_client_version(
128    client_version: Option<&str>,
129    minimum: Option<Version>,
130) -> Result<(), ControllerError> {
131    let Some(minimum) = minimum else {
132        return Ok(());
133    };
134    if client_version
135        .and_then(Version::parse)
136        .is_some_and(|client| client >= minimum)
137    {
138        return Ok(());
139    }
140    Err(controller_err!(
141        UpgradeRequired,
142        format!("This client is obsolete; the minimum supported version is {minimum}.")
143    ))
144}

146/// Extractor guarding every client route: reads `X-Client-Version` and rejects clients older than
147/// [`ApplicationConfiguration::exercise_services_client_minimum_version`] before the handler runs.
148/// Yields no data; its presence applies the check.
149#[derive(Debug)]
150pub struct SupportedClient;
151
152impl FromRequest for SupportedClient {
153    type Error = ControllerError;
154    type Future = Ready<Result<Self, ControllerError>>;
155
156    fn from_request(req: &HttpRequest, _payload: &mut actix_http::Payload) -> Self::Future {
157        let Some(app_conf) = req.app_data::<web::Data<ApplicationConfiguration>>() else {
158            return ready(Err(controller_err!(
159                InternalServerError,
160                "Missing application configuration".to_string()
161            )));
162        };
163        let client_version = req
164            .headers()
165            .get(CLIENT_VERSION_HEADER)
166            .and_then(|value| value.to_str().ok());
167        ready(
168            check_client_version(
169                client_version,
170                app_conf.exercise_services_client_minimum_version,
171            )
172            .map(|()| Self),
173        )
174    }
175}

398#[instrument(skip(pool, file_store, app_conf))]
399async fn get_course_exercises(
400    pool: web::Data<PgPool>,
401    user: UserFromOAuthToken,
402    course: web::Path<Uuid>,
403    file_store: web::Data<dyn FileStore>,
404    app_conf: web::Data<ApplicationConfiguration>,
405    _client: SupportedClient,
406) -> ControllerResult<web::Json<Vec<api::ExerciseSlide>>> {
407    let mut conn = pool.acquire().await?;
408    let token = authorize_course_access(&mut conn, user.id, *course).await?;
409
410    let capable_slugs = native_client_capable_slugs(&mut conn).await?;
411    let mut slides = Vec::new();
412    let open_chapters = course_chapters(&mut conn, user.id, *course).await?.open;
413
414    let course = models::courses::get_course(&mut conn, *course).await?;
415    let locking_statuses = chapter_locking_statuses(&mut conn, user.id, &course).await?;
416    let organization =
417        models::organizations::get_organization(&mut conn, course.organization_id).await?;
418    let pages = models::pages::get_pages_by_course_id(&mut conn, course.id).await?;
419    let page_order_numbers: HashMap<Uuid, i32> = pages
420        .iter()
421        .map(|page| (page.id, page.order_number))
422        .collect();
423    let page_url_paths: HashMap<Uuid, String> = pages
424        .into_iter()
425        .map(|page| (page.id, page.url_path))
426        .collect();
427    let mut open_chapter_exercises: Vec<_> =
428        models::exercises::get_exercises_by_course_id(&mut conn, course.id)
429            .await?
430            .into_iter()
431            .filter(|e| {
432                e.chapter_id
433                    .map(|ci| open_chapters.contains_key(&ci))
434                    .unwrap_or_default()
435            })
436            .collect();
437    sort_in_material_order(
438        &mut open_chapter_exercises,
439        &open_chapters,
440        &page_order_numbers,
441    );
442    for open_exercise in open_chapter_exercises {
443        let (slide, _) = models::exercises::get_or_select_exercise_slide(
444            &mut conn,
445            Some(user.id),
446            &open_exercise,
447            models_requests::fetch_service_info,
448            file_store.as_ref(),
449            app_conf.as_ref(),
450        )
451        .await?;
452        // The per-user "reveal once solved" gate lives in `get_exercise`, so this list view
453        // never reveals model solutions.
454        let tasks = client_tasks_from_slide(slide.exercise_tasks, &capable_slugs, false);
455        if !tasks.is_empty() {
456            let chapter = open_exercise
457                .chapter_id
458                .and_then(|id| open_chapters.get(&id));
459            let deadline = open_exercise.effective_deadline(chapter.and_then(|c| c.deadline));
460            slides.push(api::ExerciseSlide {
461                slide_id: slide.id,
462                exercise_id: open_exercise.id,
463                course_id: course.id,
464                exam_id: None,
465                exercise_name: open_exercise.name,
466                exercise_order_number: open_exercise.order_number,
467                deadline,
468                tasks,
469                page_url: page_url_paths
470                    .get(&open_exercise.page_id)
471                    .and_then(|url_path| {
472                        course_page_url(
473                            &app_conf.base_url,
474                            &organization.slug,
475                            &course.slug,
476                            url_path,
477                        )
478                    }),
479                chapter: chapter.map(|c| exercise_chapter(c, &locking_statuses)),
480            });
481        }
482    }
483
484    token.authorized_ok(web::Json(slides))
485}

487/// Sorts a course's exercises as its material lists them: by chapter, then page, then position on
488/// the page. `chapters` holds the chapters of the exercises being sorted.
489fn sort_in_material_order(
490    exercises: &mut [models::exercises::Exercise],
491    chapters: &HashMap<Uuid, DatabaseChapter>,
492    page_order_numbers: &HashMap<Uuid, i32>,
493) {
494    exercises.sort_by_key(|exercise| {
495        (
496            exercise
497                .chapter_id
498                .and_then(|id| chapters.get(&id))
499                .map(|chapter| chapter.chapter_number),
500            page_order_numbers.get(&exercise.page_id).copied(),
501            exercise.order_number,
502            exercise.id,
503        )
504    });
505}

531#[instrument(skip(pool, file_store, app_conf))]
532async fn get_exercise(
533    pool: web::Data<PgPool>,
534    user: UserFromOAuthToken,
535    exercise_id: web::Path<Uuid>,
536    file_store: web::Data<dyn FileStore>,
537    app_conf: web::Data<ApplicationConfiguration>,
538    _client: SupportedClient,
539) -> ControllerResult<web::Json<api::ExerciseSlide>> {
540    let mut conn = pool.acquire().await?;
541    let token = authorize(
542        &mut conn,
543        Act::View,
544        Some(user.id),
545        Res::Exercise(*exercise_id),
546    )
547    .await
548    .map_err(keyed_no_access)?;
549
550    let exercise = models::exercises::get_by_id(&mut conn, *exercise_id).await?;
551    if let Some(exam_id) = exercise.exam_id {
552        // Checked before selecting a slide, which would be stored for the user.
553        let exam = answerable_exam_or_not_found(&mut conn, exam_id, user.id).await?;
554        let capable_slugs = native_client_capable_slugs(&mut conn).await?;
555        let slide = exam_exercise_slide(
556            &mut conn,
557            user.id,
558            &exam,
559            exercise,
560            &capable_slugs,
561            file_store.as_ref(),
562            app_conf.as_ref(),
563        )
564        .await?
565        .ok_or_else(|| {
566            controller_err!(
567                NotFound,
568                "This exercise has a task this client cannot answer, so it must be answered in the exam".to_string()
569            )
570        })?;
571        return token.authorized_ok(web::Json(slide));
572    }
573    if let Some(course_id) = exercise.course_id {
574        authorize_access_to_exercise_chapter(&mut conn, user.id, course_id, exercise.chapter_id)
575            .await?;
576    }
577    let (exercise_slide, course_or_exam_id) = models::exercises::get_or_select_exercise_slide(
578        &mut conn,
579        Some(user.id),
580        &exercise,
581        models_requests::fetch_service_info,
582        file_store.as_ref(),
583        app_conf.as_ref(),
584    )
585    .await?;
586    let course_id = match course_or_exam_id {
587        Some(CourseOrExamId::Course(course_id)) => course_id,
588        Some(CourseOrExamId::Exam(_)) => {
589            // Exam exercises are out of scope for this API; report not found rather than
590            // misdescribing it as an enrollment problem.
591            return Err(controller_err!(
592                NotFound,
593                "This exercise belongs to an exam, which the client API does not serve".to_string()
594            ));
595        }
596        None => {
597            return Err(not_enrolled_error());
598        }
599    };
600
601    let user_exercise_state = models::user_exercise_states::get_user_exercise_state_if_exists(
602        &mut conn,
603        user.id,
604        exercise.id,
605        CourseOrExamId::Course(course_id),
606    )
607    .await?;
608    let score_given = user_exercise_state
609        .and_then(|s| s.score_given)
610        .unwrap_or(0.0);
611    let slide_submission_counts =
612        models::exercise_slide_submissions::get_exercise_slide_submission_counts_for_exercise_user(
613            &mut conn,
614            exercise.id,
615            CourseOrExamId::Course(course_id),
616            user.id,
617        )
618        .await?;
619    let slide_submission_count = slide_submission_counts
620        .get(&exercise_slide.id)
621        .copied()
622        .unwrap_or(0);
623    let chapter = match exercise.chapter_id {
624        Some(chapter_id) => Some(models::chapters::get_chapter(&mut conn, chapter_id).await?),
625        None => None,
626    };
627    let deadline = exercise.effective_deadline(chapter.as_ref().and_then(|c| c.deadline));
628    let reveal_model_solution = crate::domain::exercises::reveals_model_solution(
629        &mut conn,
630        &exercise,
631        deadline,
632        score_given,
633        slide_submission_count,
634    )
635    .await?
636        || is_course_staff(&mut conn, user.id, course_id).await?;
637
638    // Without the capability filter this endpoint would hand out task ids belonging to services
639    // that cannot serve a native client, which submit would then have to reject.
640    let capable_slugs = native_client_capable_slugs(&mut conn).await?;
641    let tasks = client_tasks_from_slide(
642        exercise_slide.exercise_tasks,
643        &capable_slugs,
644        reveal_model_solution,
645    );
646    if tasks.is_empty() {
647        return Err(controller_err!(
648            NotFound,
649            "This exercise has a task this client cannot answer, so it must be answered in the course material".to_string()
650        ));
651    }
652
653    let course = models::courses::get_course(&mut conn, course_id).await?;
654    let organization =
655        models::organizations::get_organization(&mut conn, course.organization_id).await?;
656    let page = models::pages::get_page(&mut conn, exercise.page_id).await?;
657    let locking_statuses = chapter_locking_statuses(&mut conn, user.id, &course).await?;
658    token.authorized_ok(web::Json(api::ExerciseSlide {
659        slide_id: exercise_slide.id,
660        exercise_id: exercise.id,
661        course_id,
662        exam_id: None,
663        exercise_name: exercise.name,
664        exercise_order_number: exercise.order_number,
665        deadline,
666        tasks,
667        page_url: course_page_url(
668            &app_conf.base_url,
669            &organization.slug,
670            &course.slug,
671            &page.url_path,
672        ),
673        chapter: chapter
674            .as_ref()
675            .map(|c| exercise_chapter(c, &locking_statuses)),
676    }))
677}
