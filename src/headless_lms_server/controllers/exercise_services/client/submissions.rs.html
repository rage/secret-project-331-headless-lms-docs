3/**
4 * POST /api/v0/exercise-services/client/exercises/:id/files
5 *
6 * Stores files for a later submission to this exercise. Every multipart field name must be a
7 * UUID the client picks; the host assigns the ids a submit request then names. Uploads are
8 * bound to this exercise and user, and unreferenced ones are reaped, so a client should upload
9 * immediately before submitting.
10 *
11 * Gated on the caller being able to answer the exercise at all, so stored objects cannot be
12 * accumulated past a deadline or a closed exam. The per-slide try limit is only checked across the
13 * whole exercise here, because this route is bound to an exercise rather than a slide. Files for a
14 * snapshot (`?for=snapshot`) skip the deadline, try-limit and chapter-lock checks, as the
15 * snapshot route does.
16 */
17#[utoipa::path(
18    post,
19    path = "/exercises/{id}/files",
20    operation_id = "uploadClientExerciseFiles",
21    tag = "exercise-services-client",
22    security(("bearer_auth" = [])),
23    params(
24        ("id" = Uuid, Path, description = "Exercise id"),
25        UploadQuery,
26        ("X-Client-Version" = Option<String>, Header, description = "Optional client version; obsolete clients get 426")
27    ),
28    request_body(
29        content = String,
30        content_type = "multipart/form-data",
31        description = "One file part per file, each field name a distinct client-chosen UUID and each part carrying a file name"
32    ),
33    responses(
34        (status = 200, description = "The stored files, in the order the parts were sent", body = api::UploadedFiles),
35        (status = 401, description = "The bearer token is missing or was rejected", body = crate::domain::error::ApiErrorResponse),
36        (status = 403, description = "The token lacks the `exercise-services` scope (message_key `insufficient_scope`), the user may not view this exercise (`no_access`), its chapter has not opened yet (`chapter_not_open_yet`), chapter locking keeps it closed (`chapter_not_unlocked_yet`, `chapter_completed_and_locked`), or the exercise is locked for the user (`exercise_locked`); neither lock is checked for `?for=snapshot`", body = crate::domain::error::ApiErrorResponse),
37        (status = 404, description = "No exercise with the given id exists", body = crate::domain::error::ApiErrorResponse),
38        (status = 422, description = "The user is not enrolled to this exercise's course, or for `?for=snapshot` never was (message_key `not_enrolled`), the exercise can no longer be answered because its deadline has passed (`deadline_passed`) or every slide is out of tries (`out_of_tries`) (neither checked for `?for=snapshot`), or the multipart body violates the field-name, file-count or size rules", body = crate::domain::error::ApiErrorResponse),
39        (status = 426, description = "The client is obsolete and must be upgraded", body = crate::domain::error::ApiErrorResponse),
40        (status = 429, description = "The user has uploaded too many files recently (message_key `rate_limited`)", body = crate::domain::error::ApiErrorResponse)
41    )

44#[instrument(skip(pool, file_store, payload, app_conf))]
45pub(super) async fn upload_exercise_files(
46    pool: web::Data<PgPool>,
47    file_store: web::Data<dyn FileStore>,
48    exercise_id: web::Path<Uuid>,
49    query: web::Query<UploadQuery>,
50    payload: Multipart,
51    user: UserFromOAuthToken,
52    app_conf: web::Data<ApplicationConfiguration>,
53    _client: SupportedClient,
54) -> ControllerResult<web::Json<api::UploadedFiles>> {
55    let mut conn = pool.acquire().await?;
56    let token = authorize(
57        &mut conn,
58        Act::View,
59        Some(user.id),
60        Res::Exercise(*exercise_id),
61    )
62    .await
63    .map_err(keyed_no_access)?;
64
65    let exercise = models::exercises::get_by_id(&mut conn, *exercise_id).await?;
66    let is_for_snapshot = query.r#for == Some(UploadFor::Snapshot);
67    let owner = verify_can_answer(&mut conn, user.id, &exercise, is_for_snapshot).await?;
68    if !is_for_snapshot {
69        domain::exercises::verify_user_can_answer_exercise(&mut conn, user.id, &exercise).await?;
70    }
71    check_user_quota(&UPLOAD_LIMITS, user.id)?;
72
73    let mut cleanup = file_uploading::UploadCleanup::new(file_store.clone());
74    let stored = store_client_uploads(
75        &mut conn,
76        &file_uploading::AnswerUploadDestination {
77            owner,
78            exercise_id: exercise.id,
79            user_id: user.id,
80        },
81        payload,
82        file_store.as_ref(),
83        &mut cleanup.uploaded_paths,
84        &app_conf,
85    )
86    .await;
87    let uploads = match stored {
88        Ok(uploads) => uploads,
89        Err(error) => {
90            // A commit failure inside `store_client_uploads` surfaces here too: up to 100 MiB is
91            // already in the store with no `file_uploads` row, so nothing the reaper can find.
92            cleanup.clean_up().await;
93            return Err(error);
94        }
95    };
96    cleanup.disarm();
97
98    let data_files = uploads
99        .into_iter()
100        .map(|upload| api::AnswerFile {
101            id: upload.entry.id,
102            name: upload.name,
103            mime: upload.mime,
104            size_bytes: Some(upload.size_bytes),
105            // These files are not part of an answer yet; a submit decides their order.
106            order_number: None,
107            url: upload.entry.url,
108        })
109        .collect();
110    token.authorized_ok(web::Json(api::UploadedFiles { data_files }))
111}

129/// Stores the multipart parts and binds them to the exercise and user, so that a failure to
130/// record the binding cannot leave uploads the reaper is unable to find.
131///
132/// The transaction opens only after the last byte has been streamed: the multipart body has no
133/// time limit, so opening it first would pin a pool connection `idle in transaction` for the whole
134/// upload.
135pub(super) async fn store_client_uploads(
136    conn: &mut PgConnection,
137    destination: &file_uploading::AnswerUploadDestination,
138    payload: Multipart,
139    file_store: &dyn FileStore,
140    uploaded_paths: &mut Vec<file_uploading::ExerciseServiceUploadCleanup>,
141    app_conf: &ApplicationConfiguration,
142) -> Result<Vec<file_uploading::ExerciseServiceUpload>, ControllerError> {
143    let streamed = file_uploading::stream_exercise_service_upload(
144        file_uploading::UploadPathScheme::Answer(destination),
145        payload,
146        file_store,
147        uploaded_paths,
148        app_conf,
149    )
150    .await?;
151
152    let mut tx = conn.begin().await?;
153    let uploads = file_uploading::record_exercise_service_upload(
154        &mut tx,
155        streamed,
156        Some(destination.user_id),
157    )
158    .await?;
159    let file_upload_ids: Vec<Uuid> = uploads.iter().map(|u| u.entry.id).collect();
160    models::exercise_answer_uploads::insert_many(
161        &mut tx,
162        destination.exercise_id,
163        destination.user_id,
164        &file_upload_ids,
165        models::exercise_answer_uploads::AnswerUploadOrigin::NativeClient,
166    )
167    .await?;
168    tx.commit().await?;
169    Ok(uploads)
170}

201pub(super) async fn submit_exercise(
202    pool: web::Data<PgPool>,
203    file_store: web::Data<dyn FileStore>,
204    jwt_key: web::Data<JwtKey>,
205    exercise_id: web::Path<Uuid>,
206    submission: web::Json<api::ExerciseSlideSubmission>,
207    user: UserFromOAuthToken,
208    app_conf: web::Data<ApplicationConfiguration>,
209    _client: SupportedClient,
210) -> ControllerResult<web::Json<api::ExerciseTaskSubmissionResult>> {
211    let mut conn = pool.acquire().await?;
212    let token = authorize(
213        &mut conn,
214        Act::View,
215        Some(user.id),
216        Res::Exercise(*exercise_id),
217    )
218    .await
219    .map_err(keyed_no_access)?;
220
221    let submission = submission.into_inner();
222    let exercise = models::exercises::get_by_id(&mut conn, *exercise_id).await?;
223    verify_can_answer(&mut conn, user.id, &exercise, false).await?;
224    let exercise_slide =
225        models::exercise_slides::get_exercise_slide(&mut conn, submission.exercise_slide_id)
226            .await?;
227    let exercise_task =
228        models::exercise_tasks::get_exercise_task_by_id(&mut conn, submission.exercise_task_id)
229            .await?;
230
231    verify_slide_and_task_belong(
232        exercise.id,
233        exercise_slide.id,
234        exercise_slide.exercise_id,
235        exercise_task.id,
236        exercise_task.exercise_slide_id,
237    )?;
238    let capable_slugs = native_client_capable_slugs(&mut conn).await?;
239    verify_task_is_client_capable(
240        exercise_task.id,
241        &exercise_task.exercise_type,
242        &capable_slugs,
243    )?;
244    verify_slide_is_client_answerable(&mut conn, exercise_slide.id, &capable_slugs).await?;
245    check_user_quota(&SUBMIT_LIMITS, user.id)?;
246
247    let result = domain::exercises::process_submission(
248        &mut conn,
249        user.id,
250        exercise,
251        &StudentExerciseSlideSubmission {
252            exercise_slide_id: submission.exercise_slide_id,
253            exercise_task_submissions: vec![StudentExerciseTaskSubmission {
254                exercise_task_id: submission.exercise_task_id,
255                answer_kind: submission.answer_kind.map(model_answer_kind),
256                data_json: submission.data_json,
257                data_files: submission.data_files,
258            }],
259        },
260        jwt_key.into_inner(),
261        file_store.as_ref(),
262        app_conf.as_ref(),
263    )
264    .await?;
265
266    // one task submission in, so exactly one result out
267    let task_submission = result
268        .exercise_task_submission_results
269        .into_iter()
270        .next()
271        .ok_or_else(|| {
272            controller_err!(
273                InternalServerError,
274                "Failed to find exercise task submission id".to_string()
275            )
276        })?;
277
278    token.authorized_ok(web::Json(api::ExerciseTaskSubmissionResult {
279        task_submission_id: task_submission.submission.id,
280        slide_submission_id: task_submission.submission.exercise_slide_submission_id,
281    }))
282}

412#[instrument(skip(pool, file_store, app_conf))]
413pub(super) async fn get_submission_grading(
414    pool: web::Data<PgPool>,
415    submission_id: web::Path<Uuid>,
416    user: UserFromOAuthToken,
417    file_store: web::Data<dyn FileStore>,
418    app_conf: web::Data<ApplicationConfiguration>,
419    _client: SupportedClient,
420) -> ControllerResult<web::Json<api::ExerciseTaskSubmissionStatus>> {
421    let mut conn = pool.acquire().await?;
422    let submission = models::exercise_task_submissions::get_by_id(
423        &mut conn,
424        *submission_id,
425        file_store.as_ref(),
426        app_conf.as_ref(),
427    )
428    .await?;
429    let slide_submission = models::exercise_slide_submissions::get_by_id(
430        &mut conn,
431        submission.exercise_slide_submission_id,
432    )
433    .await?;
434    verify_submission_owner(
435        slide_submission.user_id,
436        user.id,
437        "Cannot view another user's submission grading".to_string(),
438    )?;
439    let token = skip_authorize();
440    if let Some(exam_id) = slide_submission.exam_id
441        && exam_hides_gradings(&mut conn, exam_id, user.id).await?
442    {
443        return token.authorized_ok(web::Json(api::ExerciseTaskSubmissionStatus::GradingHidden));
444    }
445
446    let grading = models::exercise_task_gradings::get_by_exercise_task_submission_id(
447        &mut conn,
448        *submission_id,
449    )
450    .await?;
451    let Some(grading) = grading else {
452        return token.authorized_ok(web::Json(api::ExerciseTaskSubmissionStatus::NoGradingYet));
453    };
454    let (exercise_progress, submission_page_url) = match slide_submission.course_id {
455        Some(course_id) => {
456            let exercise =
457                models::exercises::get_by_id(&mut conn, slide_submission.exercise_id).await?;
458            let course = models::courses::get_course(&mut conn, course_id).await?;
459            let state = models::user_exercise_states::get_user_exercise_state_if_exists(
460                &mut conn,
461                user.id,
462                exercise.id,
463                CourseOrExamId::Course(course_id),
464            )
465            .await?;
466            let organization =
467                models::organizations::get_organization(&mut conn, course.organization_id).await?;
468            let page = models::pages::get_page(&mut conn, exercise.page_id).await?;
469            // The course material scrolls to the element named by the exercise id.
470            let submission_page_url = course_page_url(
471                &app_conf.base_url,
472                &organization.slug,
473                &course.slug,
474                &page.url_path,
475            )
476            .map(|url| format!("{url}#{}", exercise.id));
477            (
478                Some(Box::new(
479                    exercise_progress(&mut conn, user.id, &exercise, &course, state.as_ref())
480                        .await?,
481                )),
482                submission_page_url,
483            )
484        }
485        None => (None, None),
486    };
487    token.authorized_ok(web::Json(api::ExerciseTaskSubmissionStatus::Grading {
488        grading_progress: map_grading_progress(grading.grading_progress),
489        score_given: grading.score_given,
490        grading_started_at: grading.grading_started_at,
491        grading_completed_at: grading.grading_completed_at,
492        feedback_json: grading.feedback_json,
493        feedback_text: grading.feedback_text,
494        exercise_progress,
495        submission_page_url,
496    }))
497}

505        GradingProgress::Pending => api::GradingProgress::Pending,

531        (status = 404, description = "No exercise with the given id exists", body = crate::domain::error::ApiErrorResponse),
532        (status = 426, description = "The client is obsolete and must be upgraded", body = crate::domain::error::ApiErrorResponse)

535#[instrument(skip(pool))]
536pub(super) async fn get_exercise_submissions(
537    pool: web::Data<PgPool>,
538    exercise_id: web::Path<Uuid>,
539    user: UserFromOAuthToken,
540    _client: SupportedClient,
541) -> ControllerResult<web::Json<Vec<api::ExerciseSlideSubmissionListItem>>> {
542    let mut conn = pool.acquire().await?;
543    // The query is scoped to the current user, so no further authorization is needed.
544    let token = skip_authorize();
545    let grading_is_hidden = match models::exercises::get_by_id(&mut conn, *exercise_id)
546        .await?
547        .exam_id
548    {
549        Some(exam_id) => exam_hides_gradings(&mut conn, exam_id, user.id).await?,
550        None => false,
551    };
552
553    // One query for the whole listing, gradings joined in: an editor slide has a single task, so
554    // its grading is the slide's grading.
555    let submissions =
556        models::exercise_slide_submissions::get_users_submissions_for_exercise_with_gradings(
557            &mut conn,
558            user.id,
559            *exercise_id,
560        )
561        .await?;
562
563    let snapshots = models::exercise_answer_snapshots::get_by_user_and_exercise(
564        &mut conn,
565        user.id,
566        *exercise_id,
567    )
568    .await?;
569
570    let listed_ids: Vec<Uuid> = submissions
571        .iter()
572        .map(|submission| submission.id)
573        .chain(snapshots.iter().map(|snapshot| snapshot.id))
574        .collect();
575    let shared_ids =
576        models::exercise_slide_submission_shares::shared_among(&mut conn, user.id, &listed_ids)
577            .await?;
578
579    let mut items = submission_list_items(submissions, snapshots, &shared_ids);
580    if grading_is_hidden {
581        for item in &mut items {
582            item.score_given = None;
583            item.grading_progress = None;
584        }
585    }
586    token.authorized_ok(web::Json(items))
587}

589/// Submissions and snapshots in one list, newest first.
590pub(super) fn submission_list_items(
591    submissions: Vec<UserExerciseSlideSubmissionWithGrading>,
592    snapshots: Vec<ExerciseAnswerSnapshot>,
593    shared_ids: &HashSet<Uuid>,
594) -> Vec<api::ExerciseSlideSubmissionListItem> {
595    let submissions =
596        submissions
597            .into_iter()
598            .map(|submission| api::ExerciseSlideSubmissionListItem {
599                id: submission.id,
600                exercise_id: submission.exercise_id,
601                created_at: submission.created_at,
602                score_given: submission.score_given,
603                grading_progress: submission.grading_progress.map(map_grading_progress),
604                kind: api::SubmissionKind::Submission,
605                is_shared: shared_ids.contains(&submission.id),
606            });
607    let snapshots = snapshots
608        .into_iter()
609        .map(|snapshot| api::ExerciseSlideSubmissionListItem {
610            id: snapshot.id,
611            exercise_id: snapshot.exercise_id,
612            created_at: snapshot.created_at,
613            score_given: None,
614            grading_progress: None,
615            kind: match snapshot.purpose {
616                SnapshotPurpose::Paste => api::SubmissionKind::Paste,
617                SnapshotPurpose::Backup => api::SubmissionKind::Backup,
618            },
619            is_shared: shared_ids.contains(&snapshot.id),
620        });
621    let mut items: Vec<_> = submissions.chain(snapshots).collect();
622    items.sort_by(|a, b| b.created_at.cmp(&a.created_at).then(b.id.cmp(&a.id)));
623    items
624}

646        (status = 403, description = "Cannot download another user's submission (message_key `no_access`)", body = crate::domain::error::ApiErrorResponse),
647        (status = 404, description = "No submission with the given id exists", body = crate::domain::error::ApiErrorResponse),
648        (status = 426, description = "The client is obsolete and must be upgraded", body = crate::domain::error::ApiErrorResponse)
649    )
650)]
651#[instrument(skip(pool, file_store, app_conf))]
652pub(super) async fn download_submission(
653    pool: web::Data<PgPool>,
654    file_store: web::Data<dyn FileStore>,
655    submission_id: web::Path<Uuid>,
656    user: UserFromOAuthToken,
657    app_conf: web::Data<ApplicationConfiguration>,
658    _client: SupportedClient,
659) -> ControllerResult<web::Json<api::SubmissionFiles>> {
660    let mut conn = pool.acquire().await?;
661    if let Some(snapshot) =
662        models::exercise_answer_snapshots::try_get_by_id(&mut conn, *submission_id).await?
663    {
664        verify_submission_owner(
665            snapshot.user_id,
666            user.id,
667            "Cannot download another user's submission".to_string(),
668        )?;
669        let token = skip_authorize();
670        let files = models::exercise_answer_snapshots::get_files(&mut conn, snapshot.id).await?;
671        return token.authorized_ok(web::Json(submission_files_response(
672            files,
673            file_store.as_ref(),
674            app_conf.as_ref(),
675        )?));
676    }
677    let slide_submission =
678        models::exercise_slide_submissions::get_by_id(&mut conn, *submission_id).await?;
679    verify_submission_owner(
680        slide_submission.user_id,
681        user.id,
682        "Cannot download another user's submission".to_string(),
683    )?;
684    let token = skip_authorize();
685
686    let task_submissions = models::exercise_task_submissions::get_by_exercise_slide_submission_id(
687        &mut conn,
688        *submission_id,
689        file_store.as_ref(),
690        app_conf.as_ref(),
691    )
692    .await?;
693    // Resolved from the host's own file records, never from the exercise service's answer: the
694    // answer is an opaque plugin-owned blob and reading it would tie this endpoint to one plugin's
695    // shape. Both native-client and IFrame-made file answers are recorded here at submit time, so
696    // both origins are served by this one read.
697    let task_submission_ids: Vec<Uuid> = task_submissions.iter().map(|ts| ts.id).collect();
698    let files = models::exercise_task_submission_files::get_by_task_submission_ids(
699        &mut conn,
700        &task_submission_ids,
701    )
702    .await?;
703
704    token.authorized_ok(web::Json(submission_files_response(
705        files,
706        file_store.as_ref(),
707        app_conf.as_ref(),
708    )?))
709}

763            .map(Into::into)

767                    name: file.name,
768                    mime: file.mime,
769                    size_bytes: file.size_bytes,
770                    order_number: Some(file.order_number),
771                    url: file_store.get_claimed_download_url(file.file_upload_id, app_conf)?,
772                })
773            })
774            .collect::<UtilResult<_>>()?,
775    })

800        (status = 426, description = "The client is obsolete and must be upgraded", body = crate::domain::error::ApiErrorResponse)
801    )

803#[instrument(skip(pool, app_conf))]
804pub(super) async fn share_submission(
805    pool: web::Data<PgPool>,
806    submission_id: web::Path<Uuid>,
807    user: UserFromOAuthToken,
808    app_conf: web::Data<ApplicationConfiguration>,
809    _client: SupportedClient,
810) -> ControllerResult<web::Json<api::PasteResult>> {
811    let mut conn = pool.acquire().await?;
812    let share = if let Some(snapshot) =
813        models::exercise_answer_snapshots::try_get_by_id(&mut conn, *submission_id).await?
814    {
815        verify_submission_owner(
816            snapshot.user_id,
817            user.id,
818            "Cannot share another user's submission".to_string(),
819        )?;
820        // Backups are pruned past the cap, which would break their links, and are not rate limited.
821        if snapshot.purpose != SnapshotPurpose::Paste {
822            return Err(bad_request_with_reason(
823                BadRequestReason::BackupNotShareable,
824                "Only a paste can be shared, not a backup.".to_string(),
825            ));
826        }
827        domain::exercise_services::submission_sharing::share_snapshot(
828            &mut conn,
829            snapshot.id,
830            user.id,
831        )
832        .await?
833    } else {
834        let slide_submission =
835            models::exercise_slide_submissions::get_by_id(&mut conn, *submission_id).await?;
836        verify_submission_owner(
837            slide_submission.user_id,
838            user.id,
839            "Cannot share another user's submission".to_string(),
840        )?;
841        verify_not_exam_submission(&slide_submission)?;
842        domain::exercise_services::submission_sharing::share_submission(
843            &mut conn,
844            *submission_id,
845            user.id,
846        )
847        .await?
848    };
849    let token = skip_authorize();
850    let paste_url = format!(
851        "{}/shared-submissions/{}",
852        app_conf.base_url.trim_end_matches('/'),
853        share.id
854    );
855
856    token.authorized_ok(web::Json(api::PasteResult { paste_url }))
857}
