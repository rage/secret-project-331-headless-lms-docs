45/// Add controllers from all the submodules.
46pub fn configure_controllers(
47    cfg: &mut ServiceConfig,
48    app_conf: web::Data<ApplicationConfiguration>,
49) {
50    cfg.service(web::scope("/course-material").configure(course_material::_add_routes))
51        .service(web::scope("/cms").configure(cms::_add_routes))
52        .service(web::scope("/files").configure(files::_add_routes))
53        .service(web::scope("/main-frontend").configure(main_frontend::_add_routes))
54        .service(web::scope("/auth").configure(auth::_add_routes))
55        .service(web::scope("/errors").configure(errors::_add_routes))
56        .service(web::scope("/study-registry").configure(study_registry::_add_routes))
57        .service(web::scope("/exercise-services").configure(exercise_services::_add_routes))
58        .service(
59            web::scope("/other-domain-redirects").configure(other_domain_redirects::_add_routes),
60        )
61        .service(web::scope("/health").configure(health::_add_routes))
62        .service(web::scope("/tmc-server").configure(tmc_server::_add_routes))
63        .service({
64            let [per_grant, per_user] = mcp_rate_limits(app_conf.test_mode);
65            // Registered inside out: authentication runs first and gives the limits their key.
66            web::scope("/mcp")
67                .wrap(per_grant)
68                .wrap(per_user)
69                .wrap(actix_web::middleware::from_fn(
70                    headless_lms_mcp::http::auth::authenticate,
71                ))
72                .configure(headless_lms_mcp::http::configure)
73        })
74        .default_service(web::to(not_found));
75    if app_conf.test_chatbot && app_conf.test_mode {
76        cfg.service(web::scope("/mock-azure").configure(mock_azure::_add_routes))
77            .service(
78                web::scope("/mock-document-storage").configure(mock_document_storage::_add_routes),
79            );
80    }
81    if app_conf.test_sisu && app_conf.test_mode {
82        cfg.service(web::scope("/mock-sisu").configure(mock_sisu::_add_routes));
83    }
84    if app_conf.test_suotar && app_conf.test_mode {
85        cfg.service(mock_suotar::scope());
86    }
87}

89/// The MCP server's limits, per grant and per user. A user's grants together get twice what one
90/// grant does, so one runaway agent cannot use up the quota of the user's other agent.
91fn mcp_rate_limits(test_mode: bool) -> [RateLimit; 2] {
92    let multiplier = if test_mode { 100 } else { 1 };
93    let limit = |per_minute: u64, per_hour: u64, key| {
94        RateLimit::new(RateLimitConfig {
95            per_minute: Some(per_minute * multiplier),
96            per_hour: Some(per_hour * multiplier),
97            ..Default::default()
98        })
99        .keyed_by(key)
100    };
101    [
102        limit(120, 3000, RateLimitKey::Grant),
103        limit(240, 6000, RateLimitKey::User),
104    ]
105}

107async fn not_found(req: HttpRequest) -> HttpResponse {
