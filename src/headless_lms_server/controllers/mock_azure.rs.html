116// GET /api/v0/mock_azure/api/projects/test/openai/v1/responses
117// POST /api/v0/mock_azure/api/projects/test/openai/v1/responses
118async fn mock_azure_chat_responses(
119    app_conf: web::Data<ApplicationConfiguration>,
120    payload: web::Json<AzureCompletionRequest>,
121) -> ControllerResult<String> {
122    assert!(app_conf.test_chatbot && app_conf.test_mode);
123    let message_suggestion_user_prompt = USER_PROMPT;
124
125    let message_kind = &payload
126        .base
127        .input
128        .last()
129        .ok_or(ControllerError::new(
130            ControllerErrorType::BadRequest,
131            "No messages in request, there should be at least one.",
132            None,
133        ))?
134        .message_type;
135    let message = match message_kind {
136        InputItem::Message {
137            role: _role,
138            content,
139        } => Ok(content.to_owned()),
140        _ => Err(ControllerError::new(
141            ControllerErrorType::BadRequest,
142            "The request had a tool call or tool response. This shouldn't happen when using message suggestion LLM.",
143            None,
144        )),
145    }?.get_content_text();
146
147    let suggest_prompt_match = message
148        .matches(message_suggestion_user_prompt)
149        .collect::<Vec<&str>>();
150    let cms_suggest_match = message.contains(USER_PROMPT_PREFIX);
151    let description_suggestion_match = message.contains(DESCRIPTION_USER_PROMPT);
152    let res: String = if !suggest_prompt_match.is_empty() {
153        SUGGESTION.to_string()
154    } else if cms_suggest_match {
155        CMS_SUGGESTION.to_string()
156    } else if description_suggestion_match {
157        DESCRIPTION_SUGGESTION.to_string()
158    } else {
159        get_response(app_conf.base_url.clone())?
160    };
161    let token = skip_authorize();
162    token.authorized_ok(res)
163}

176        usage: EmbeddingResponseUsage {
177            prompt_tokens: payload.input.len() as i32,
178            total_tokens: payload.input.len() as i32,
179        },
180        data: payload
181            .input
182            .iter()
183            .enumerate()
184            .map(|(index, _)| Embedding {
185                index: index as i32,
186                embedding: vec![0.0; 1536],
187                object: "embedding".to_string(),

189            .collect(),
190    };
191    let res = serde_json::to_string(&mock_response)?;
192    let token = skip_authorize();
193    token.authorized_ok(res)
194}
