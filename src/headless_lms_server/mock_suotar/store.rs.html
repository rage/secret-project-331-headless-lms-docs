157    async fn conn(&self) -> anyhow::Result<ConnectionManager> {
158        let manager = self
159            .connection
160            .get_or_try_init(|| async {
161                let config = ConnectionManagerConfig::new()
162                    .set_response_timeout(Some(MOCK_REDIS_RESPONSE_TIMEOUT));
163                ConnectionManager::new_with_config(self.client.clone(), config).await
164            })
165            .await
166            .context("the mock Suotar could not reach Redis")?;
167        Ok(manager.clone())
168    }

170    fn cached_generation(&self) -> Option<String> {
171        self.generation.read().ok().and_then(|g| g.clone())
172    }

174    fn cache_generation(&self, generation: Option<String>) {
175        if let Ok(mut cached) = self.generation.write() {
176            *cached = generation;
177        }
178    }

180    pub async fn live_generation(&self) -> anyhow::Result<Option<String>> {
181        if let Some(generation) = self.cached_generation() {
182            return Ok(Some(generation));
183        }
184        let mut conn = self.conn().await?;
185        let generation: Option<String> = conn.get(GENERATION_KEY).await?;
186        self.cache_generation(generation.clone());
187        Ok(generation)
188    }

190    /// Flips the generation pointer last, so no request sees a half-installed world and a push needs
191    /// nothing cleared before it.
192    pub async fn install_world(
193        &self,
194        world: &World,
195        db_generation: Option<&str>,
196    ) -> anyhow::Result<String> {
197        let previous = self.live_generation().await?;
198        let mut conn = self.conn().await?;
199        let sequence: i64 = conn.incr(GENERATION_SEQ_KEY, 1).await?;
200        let generation = format!("g{sequence}");
201
202        let mut pipe = redis::pipe();
203        pipe.atomic();
204        pipe.hset(
205            key(&generation, META),
206            "defaults",
207            serde_json::to_string(&world.defaults)?,
208        )
209        .ignore();
210        pipe.hset(
211            key(&generation, META),
212            "installedAt",
213            Utc::now().to_rfc3339(),
214        )
215        .ignore();
216        if let Some(db_generation) = db_generation {
217            pipe.hset(key(&generation, META), "dbGeneration", db_generation)
218                .ignore();
219        }
220        write_entity_hash(&mut pipe, &generation, PERSONS, &world.persons)?;
221        write_entity_hash(&mut pipe, &generation, COURSE_UNITS, &world.course_units)?;
222        write_entity_hash(&mut pipe, &generation, ENROLMENTS, &world.enrolments)?;
223        write_entity_hash(&mut pipe, &generation, ATTAINMENTS, &world.attainments)?;
224        write_entity_hash(&mut pipe, &generation, SUBMISSIONS, &world.submissions)?;
225        write_entity_hash(
226            &mut pipe,
227            &generation,
228            SISU_VIOLATIONS,
229            &world.sisu_violations,
230        )?;
231        write_derived_indexes(&mut pipe, &generation, world)?;
232        pipe.set(GENERATION_KEY, &generation).ignore();
233        pipe.query_async::<()>(&mut conn).await?;
234
235        self.cache_generation(Some(generation.clone()));
236
237        if let Some(previous) = previous.filter(|previous| previous != &generation) {
238            let mut cleanup = redis::pipe();
239            for name in PREFIXED_KEYS {
240                cleanup.del(key(&previous, name)).ignore();
241            }
242            cleanup.query_async::<()>(&mut conn).await?;
243        }
244        Ok(generation)
245    }

247    /// Serialised so a burst of first requests against an empty index does not each mint a generation.
248    pub async fn install_if_absent(
249        &self,
250        world: &World,
251        db_generation: Option<&str>,
252    ) -> anyhow::Result<String> {
253        let _guard = self.install_lock.lock().await;
254        self.cache_generation(None);
255        if let Some(generation) = self.live_generation().await?
256            && self.has_world(&generation).await?
257        {
258            return Ok(generation);
259        }
260        self.install_world(world, db_generation).await
261    }

277    pub async fn preamble(&self, generation: &str) -> anyhow::Result<Preamble> {
278        let mut conn = self.conn().await?;
279        let (meta, faults, remaining): (
280            HashMap<String, String>,
281            HashMap<String, String>,
282            HashMap<String, i64>,
283        ) = redis::pipe()
284            .hgetall(key(generation, META))
285            .hgetall(key(generation, FAULTS))
286            .hgetall(key(generation, FAULTS_REMAINING))
287            .query_async(&mut conn)
288            .await?;
289
290        let defaults = match meta.get("defaults") {
291            Some(raw) => {
292                serde_json::from_str(raw).context("stored world defaults are unreadable")?
293            }
294            None => WorldDefaults::default(),
295        };
296        let mut faults: Vec<Fault> = faults
297            .values()
298            .map(|raw| serde_json::from_str::<Fault>(raw))
299            .collect::<Result<_, _>>()
300            .context("a stored fault is unreadable")?;
301        faults.sort_by_key(|fault| fault.seq);
302
303        Ok(Preamble {
304            generation: generation.to_string(),
305            defaults_present: meta.contains_key("defaults"),
306            defaults,
307            db_generation: meta.get("dbGeneration").cloned(),
308            faults,
309            remaining,
310        })
311    }

331    /// Two pipelined round trips whatever the batch size: the keyed hashes, then the entities they
332    /// point at.
333    pub async fn load_for_person_course(
334        &self,
335        generation: &str,
336        student_numbers: &[String],
337        course_codes: &[String],
338    ) -> anyhow::Result<WorkingSet> {
339        let mut conn = self.conn().await?;
340        let person_course_keys: Vec<String> = student_numbers
341            .iter()
342            .flat_map(|student_number| {
343                course_codes
344                    .iter()
345                    .map(move |course_code| person_course_key(student_number, course_code))
346            })
347            .collect();
348
349        let mut first = redis::pipe();
350        push_hmget(&mut first, &key(generation, PERSONS), student_numbers);
351        push_hmget(&mut first, &key(generation, COURSE_UNITS), course_codes);
352        push_hmget(
353            &mut first,
354            &key(generation, IDX_ENROLMENTS_BY_PERSON),
355            student_numbers,
356        );
357        push_hmget(
358            &mut first,
359            &key(generation, IDX_ATTAINMENTS_BY_PERSON_COURSE),
360            &person_course_keys,
361        );
362        push_hmget(
363            &mut first,
364            &key(generation, IDX_SUBMISSIONS_BY_PERSON_COURSE),
365            &person_course_keys,
366        );
367        push_hmget(
368            &mut first,
369            &key(generation, SISU_VIOLATIONS),
370            &person_course_keys,
371        );
372        let (
373            persons,
374            course_units,
375            enrolments_by_person,
376            attainment_ids,
377            submission_ids,
378            sisu_violations,
379        ): (Fields, Fields, Fields, Fields, Fields, Fields) = first.query_async(&mut conn).await?;
380
381        let mut working = WorkingSet {
382            persons: zip_json(student_numbers, persons)?,
383            course_units: zip_json(course_codes, course_units)?,
384            enrolments_by_person: zip_json(student_numbers, enrolments_by_person)?,
385            attainments_by_person_course: zip_json(&person_course_keys, attainment_ids)?,
386            submissions_by_person_course: zip_json(&person_course_keys, submission_ids)?,
387            sisu_violations: zip_json(&person_course_keys, sisu_violations)?,
388            ..Default::default()
389        };
390
391        let enrolment_ids = flatten(working.enrolments_by_person.values());
392        let attainment_ids = flatten(working.attainments_by_person_course.values());
393        let submission_ids = flatten(working.submissions_by_person_course.values());
394
395        let mut second = redis::pipe();
396        push_hmget(&mut second, &key(generation, ENROLMENTS), &enrolment_ids);
397        push_hmget(&mut second, &key(generation, ATTAINMENTS), &attainment_ids);
398        push_hmget(&mut second, &key(generation, SUBMISSIONS), &submission_ids);
399        let (enrolments, attainments, submissions): (Fields, Fields, Fields) =
400            second.query_async(&mut conn).await?;
401
402        working.enrolments = zip_json(&enrolment_ids, enrolments)?;
403        working.attainments = zip_json(&attainment_ids, attainments)?;
404        working.submissions = zip_json(&submission_ids, submissions)?;
405        Ok(working)
406    }

471    /// The one write of a request: changed entities plus its call-log entry, in one atomic pipeline.
472    pub async fn commit(
473        &self,
474        generation: &str,
475        working: &WorkingSet,
476        call: &RecordedCall,
477        call_log_capacity: usize,
478    ) -> anyhow::Result<()> {
479        let mut conn = self.conn().await?;
480        let mut pipe = redis::pipe();
481        pipe.atomic();
482        for write in &working.writes {
483            match write {
484                WorldWrite::UpsertSubmission(id) => {
485                    let submission = working
486                        .submissions
487                        .get(id)
488                        .ok_or_else(|| anyhow!("write names a submission the working set lost"))?;
489                    pipe.hset(
490                        key(generation, SUBMISSIONS),
491                        id,
492                        serde_json::to_string(submission)?,
493                    )
494                    .ignore();
495                }
496                WorldWrite::IndexSubmission {
497                    student_number,
498                    course_code,
499                } => {
500                    let field = person_course_key(student_number, course_code);
501                    let ids = working
502                        .submissions_by_person_course
503                        .get(&field)
504                        .cloned()
505                        .unwrap_or_default();
506                    pipe.hset(
507                        key(generation, IDX_SUBMISSIONS_BY_PERSON_COURSE),
508                        field,
509                        serde_json::to_string(&ids)?,
510                    )
511                    .ignore();
512                }
513            }
514        }
515        pipe.lpush(key(generation, CALLS), serde_json::to_string(call)?)
516            .ignore();
517        pipe.ltrim(
518            key(generation, CALLS),
519            0,
520            call_log_capacity.saturating_sub(1) as isize,
521        )
522        .ignore();
523        pipe.query_async::<()>(&mut conn).await?;
524        Ok(())
525    }

553    pub async fn upsert_json<T: Serialize>(
554        &self,
555        generation: &str,
556        hash: EntityHash,
557        entries: &BTreeMap<String, T>,
558    ) -> anyhow::Result<()> {
559        if entries.is_empty() {
560            return Ok(());
561        }
562        let mut conn = self.conn().await?;
563        let mut pipe = redis::pipe();
564        pipe.atomic();
565        for (field, value) in entries {
566            pipe.hset(
567                key(generation, hash.name()),
568                field,
569                serde_json::to_string(value)?,
570            )
571            .ignore();
572        }
573        pipe.query_async::<()>(&mut conn).await?;
574        Ok(())
575    }

684    /// Every read of the call log is bounded: a list has no index, so a filter is always a scan.
685    pub async fn recent_calls(
686        &self,
687        generation: &str,
688        limit: usize,

690        if limit == 0 {
691            return Ok(Vec::new());
692        }
693        let mut conn = self.conn().await?;
694        let raw: Vec<String> = conn
695            .lrange(key(generation, CALLS), 0, limit as isize - 1)
696            .await?;
697        raw.iter()
698            .map(|entry| {
699                serde_json::from_str(entry).context("a stored call-log entry is unreadable")
700            })
701            .collect()

704    pub async fn counts(&self, generation: &str) -> anyhow::Result<WorldCounts> {
705        let mut conn = self.conn().await?;
706        let (
707            persons,
708            course_units,
709            enrolments,
710            attainments,
711            submissions,
712            sisu_violations,
713            call_log_len,
714        ): (usize, usize, usize, usize, usize, usize, usize) = redis::pipe()
715            .hlen(key(generation, PERSONS))
716            .hlen(key(generation, COURSE_UNITS))
717            .hlen(key(generation, ENROLMENTS))
718            .hlen(key(generation, ATTAINMENTS))
719            .hlen(key(generation, SUBMISSIONS))
720            .hlen(key(generation, SISU_VIOLATIONS))
721            .llen(key(generation, CALLS))
722            .query_async(&mut conn)
723            .await?;
724        let faults = self.faults(generation).await?;
725        let remaining = self.remaining_budgets(generation).await?;
726        let spent = faults
727            .iter()
728            .filter(|fault| {
729                fault.lifetime.budget().is_some()
730                    && remaining.get(&fault.id).copied().unwrap_or(0) <= 0
731            })
732            .count();
733        Ok(WorldCounts {
734            persons,
735            course_units,
736            enrolments,
737            attainments,
738            submissions,
739            sisu_violations,
740            faults_armed: faults.len() - spent,
741            faults_spent: spent,
742            call_log_len,
743        })
744    }

778    /// Rebuilds every derived index from the stored entities, so an upsert cannot leave one behind.
779    pub async fn reindex(&self, generation: &str) -> anyhow::Result<()> {
780        let world = World {
781            defaults: WorldDefaults::default(),
782            persons: self.all_json(generation, EntityHash::Persons).await?,
783            course_units: self.all_json(generation, EntityHash::CourseUnits).await?,
784            enrolments: self.all_json(generation, EntityHash::Enrolments).await?,
785            attainments: self.all_json(generation, EntityHash::Attainments).await?,
786            submissions: self.all_json(generation, EntityHash::Submissions).await?,
787            sisu_violations: self
788                .all_json(generation, EntityHash::SisuViolations)
789                .await?,
790        };
791
792        let mut conn = self.conn().await?;
793        let mut pipe = redis::pipe();
794        pipe.atomic();
795        for name in [
796            IDX_ENROLMENTS_BY_PERSON,
797            IDX_ENROLMENTS_BY_REALISATION,
798            IDX_ATTAINMENTS_BY_PERSON_COURSE,
799            IDX_SUBMISSIONS_BY_PERSON_COURSE,
800            IDX_OWNER_KEYS,
801        ] {
802            pipe.del(key(generation, name)).ignore();
803        }
804        write_derived_indexes(&mut pipe, generation, &world)?;
805        pipe.query_async::<()>(&mut conn).await?;
806        Ok(())
807    }

810/// Hashes the command surface reads and writes by name.
811#[derive(Debug, Clone, Copy, PartialEq, Eq)]
812pub enum EntityHash {
813    Persons,
814    CourseUnits,
815    Enrolments,
816    Attainments,
817    Submissions,
818    SisuViolations,
819    Faults,
820    OwnerKeys,
821    Calls,
822}
823
824impl EntityHash {
825    fn name(self) -> &'static str {
826        match self {
827            Self::Persons => PERSONS,
828            Self::CourseUnits => COURSE_UNITS,
829            Self::Enrolments => ENROLMENTS,
830            Self::Attainments => ATTAINMENTS,
831            Self::Submissions => SUBMISSIONS,
832            Self::SisuViolations => SISU_VIOLATIONS,
833            Self::Faults => FAULTS,
834            Self::OwnerKeys => IDX_OWNER_KEYS,
835            Self::Calls => CALLS,
836        }
837    }
838}

850fn write_entity_hash<T: Serialize>(
851    pipe: &mut redis::Pipeline,
852    generation: &str,
853    name: &str,

863fn write_derived_indexes(
864    pipe: &mut redis::Pipeline,
865    generation: &str,
866    world: &World,
867) -> anyhow::Result<()> {
868    let mut by_person: BTreeMap<String, Vec<String>> = BTreeMap::new();
869    let mut by_realisation: BTreeMap<String, Vec<String>> = BTreeMap::new();
870    for enrolment in world.enrolments.values() {
871        by_person
872            .entry(enrolment.student_number.clone())
873            .or_default()
874            .push(enrolment.id.clone());
875        by_realisation
876            .entry(enrolment.realisation_id.clone())
877            .or_default()
878            .push(enrolment.id.clone());
879    }
880    let mut attainments_by_person_course: BTreeMap<String, Vec<String>> = BTreeMap::new();
881    for attainment in world.attainments.values() {
882        attainments_by_person_course
883            .entry(person_course_key(
884                &attainment.student_number,
885                &attainment.course_code,
886            ))
887            .or_default()
888            .push(attainment.id.clone());
889    }
890    let mut submissions_by_person_course: BTreeMap<String, Vec<String>> = BTreeMap::new();
891    for submission in world.submissions.values() {
892        submissions_by_person_course
893            .entry(person_course_key(
894                &submission.student_number,
895                &submission.course_code,
896            ))
897            .or_default()
898            .push(submission.submitted_attainment_id.clone());
899    }
900
901    let mut owner_keys: BTreeMap<String, OwnerKeys> = BTreeMap::new();
902    for person in world.persons.values() {
903        if let Some(email) = &person.owner_user_email {
904            owner_keys
905                .entry(format!("user:{email}"))
906                .or_default()
907                .student_numbers
908                .push(person.student_number.clone());
909        }
910    }
911    for unit in world.course_units.values() {
912        if let Some(slug) = &unit.owner_course_slug {
913            owner_keys
914                .entry(format!("course:{slug}"))
915                .or_default()
916                .course_codes
917                .push(unit.course_code.clone());
918        }
919    }
920
921    for (name, index) in [
922        (IDX_ENROLMENTS_BY_PERSON, by_person),
923        (IDX_ENROLMENTS_BY_REALISATION, by_realisation),
924        (
925            IDX_ATTAINMENTS_BY_PERSON_COURSE,
926            attainments_by_person_course,
927        ),
928        (
929            IDX_SUBMISSIONS_BY_PERSON_COURSE,
930            submissions_by_person_course,
931        ),
932    ] {
933        for (field, ids) in index {
934            pipe.hset(key(generation, name), field, serde_json::to_string(&ids)?)
935                .ignore();
936        }
937    }
938    for (field, keys) in owner_keys {
939        pipe.hset(
940            key(generation, IDX_OWNER_KEYS),
941            field,
942            serde_json::to_string(&keys)?,
943        )
944        .ignore();
945    }
946    Ok(())
947}

949/// Always `HMGET`, so the reply is an array whatever the field count — a single-field `HGET` answers
950/// with a bare string, and no fields at all is a protocol error.
951fn hmget_cmd(hash: &str, fields: &[String]) -> redis::Cmd {
952    let mut cmd = redis::cmd("HMGET");
953    cmd.arg(hash);
954    if fields.is_empty() {
955        cmd.arg("");
956    }
957    for field in fields {
958        cmd.arg(field);
959    }
960    cmd
961}
962
963fn push_hmget(pipe: &mut redis::Pipeline, hash: &str, fields: &[String]) {
964    pipe.add_command(hmget_cmd(hash, fields));
965}
966
967async fn hmget_json<T: DeserializeOwned>(
968    conn: &mut ConnectionManager,
969    hash: &str,
970    fields: &[String],
971) -> anyhow::Result<BTreeMap<String, T>> {
972    if fields.is_empty() {
973        return Ok(BTreeMap::new());
974    }
975    let values: Vec<Option<String>> = hmget_cmd(hash, fields).query_async(conn).await?;
976    zip_json(fields, values)
977}
978
979fn zip_json<T: DeserializeOwned>(
980    fields: &[String],
981    values: Vec<Option<String>>,
982) -> anyhow::Result<BTreeMap<String, T>> {
983    let mut out = BTreeMap::new();
984    for (field, value) in fields.iter().zip(values) {
985        if let Some(value) = value {
986            out.insert(
987                field.clone(),
988                serde_json::from_str(&value)
989                    .with_context(|| format!("stored value for `{field}` is unreadable"))?,
990            );
991        }
992    }
993    Ok(out)
994}

996fn flatten<'a, I: Iterator<Item = &'a Vec<String>>>(lists: I) -> Vec<String> {
997    lists
998        .flat_map(|list| list.iter().cloned())
999        .unique()
1000        .collect()
