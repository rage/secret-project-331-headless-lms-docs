1//! Removes files uploaded through the exercise-service upload route that never reached a saved
2//! spec.
3//!
4//! Nothing else can reclaim them: the host stores specs as opaque blobs, so a stored file's only
5//! reference may sit inside content it cannot parse. What makes a file safe here is a declaration
6//! — `exercise_task_spec_files` for a live spec, `page_history_spec_files` for every snapshot a
7//! restore could bring back. Because history is kept, this reaper mostly collects uploads a
8//! teacher abandoned before saving, not files dropped from a spec that was once saved.
9//!
10//! The binding row is soft-deleted rather than removed, leaving an audit trail of what was
11//! reclaimed.
12

25const MAX_CONCURRENT_REAPS: usize = 8;
26
27pub async fn main() -> anyhow::Result<()> {
28    dotenv().ok();
29    ProgramConfig::ensure_default_rust_log_for_workers();
30    setup_tracing()?;
31    let database_url = ProgramConfig::database_url_with_default();
32    let base_url = ProgramConfig::required("BASE_URL")?;
33    let file_store = setup_file_store(&FileStoreRuntimeConfig::try_from_env()?, &base_url).await;
34    let db_pool = PgPool::connect(&database_url).await?;
35    reap(&db_pool, file_store.as_ref()).await
36}
37
38async fn reap(pool: &PgPool, file_store: &dyn FileStore) -> anyhow::Result<()> {
39    let mut conn = pool.acquire().await?;
40    let reapable = models::exercise_spec_uploads::get_reapable(&mut conn).await?;
41    drop(conn);
42    info!("Reaping {} abandoned spec uploads.", reapable.len());
43
44    let mut reaped = 0;
45    let mut skipped = 0;
46    let mut failed = 0;
47    let mut results = stream::iter(reapable)
48        .map(|upload| async move {
49            let file_upload_id = upload.file_upload_id;
50            let result = match pool.acquire().await {
51                Ok(mut conn) => reap_one(&mut conn, file_store, &upload).await,
52                Err(err) => Err(err.into()),
53            };
54            (file_upload_id, result)
55        })
56        .buffer_unordered(MAX_CONCURRENT_REAPS);
57    while let Some((file_upload_id, result)) = results.next().await {
58        match result {
59            Ok(true) => reaped += 1,
60            Ok(false) => skipped += 1,
61            Err(err) => {
62                failed += 1;
63                error!("Failed to reap spec upload {}: {:#?}", file_upload_id, err);
64            }
65        }
66    }
67    info!(
68        "Abandoned spec uploads reaped. Succeeded: {reaped}, skipped: {skipped}, failed: {failed}."
69    );
70    // The CronJob's exit status is the only signal anyone watches, so a run where every delete
71    // failed must not look green.
72    if failed > 0 {
73        anyhow::bail!(
74            "Failed to reap {failed} of {} spec uploads.",
75            reaped + failed
76        );
77    }
78    Ok(())
79}

81/// Retires the record, removes the object, and only then soft-deletes the `file_uploads` row.
82/// `Ok(false)` means a save came to declare the file after `get_reapable` listed it, so it is no
83/// longer reapable.
84///
85/// Deleting the `file_uploads` row last is what makes a failed object delete recoverable:
86/// `get_reapable` still sees the row and retries it on a later run, instead of orphaning the
87/// object forever.
88async fn reap_one(
89    conn: &mut PgConnection,
90    file_store: &dyn FileStore,
91    upload: &models::exercise_spec_uploads::ReapableUpload,
92) -> anyhow::Result<bool> {
93    if !models::exercise_spec_uploads::mark_reaped(conn, upload.id).await? {
94        return Ok(false);
95    }
96    match file_store.delete(Path::new(&upload.path)).await {
97        Ok(()) => {}
98        Err(err) if matches!(err.error_type(), UtilErrorType::FileNotFound) => {
99            warn!(
100                "Spec upload {} was already gone from the file store at {}.",
101                upload.file_upload_id, upload.path
102            );
103        }
104        Err(err) => return Err(err.into()),
105    }
106    // `optional` tolerates a file a previous interrupted run already soft-deleted, without
107    // swallowing real database errors.
108    models::file_uploads::delete_and_fetch_path(conn, upload.file_upload_id)
109        .await
110        .optional()?;
111    Ok(true)
112}
