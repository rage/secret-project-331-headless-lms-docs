18/// Seeds the guide course that the course design assistant searches, and makes its chatbot the
19/// assistant.
20pub async fn seed_course_design_guide(
21    app_config: &ApplicationConfiguration,
22    course_id: Uuid,
23    common_course_data: CommonCourseData,
24) -> Result<Uuid> {
25    let CommonCourseData {
26        db_pool,
27        organization_id: org,
28        teacher_user_id,
29        ..
30    } = common_course_data;
31    let mut conn = db_pool.acquire().await?;
32    let cx = SeedContext {
33        teacher: teacher_user_id,
34        org,
35        base_course_ns: course_id,
36    };
37    info!("Inserting the course design guide course");
38
39    let llm = chatbot_configurations_models::get_default(&mut conn).await?;
40    let assistant_configuration_id = cx.v5(b"chatbot_config:course_designer_assistant");
41
42    let course = CourseBuilder::new("MOOC design guide", "mooc-design-guide")
43        .desc("Guidance for teams designing a MOOC with the course design planner.")
44        .chatbot(true)
45        .course_id(course_id)
46        .instance(CourseInstanceConfig {
47            name: None,
48            description: None,
49            support_email: None,
50            teacher_in_charge_name: "admin".to_string(),
51            teacher_in_charge_email: "admin@example.com".to_string(),
52            opening_time: None,
53            closing_time: None,
54            instance_id: Some(cx.v5(b"instance:default")),
55        })
56        .chatbot_config(NewChatbotConf {
57            course_id: Some(course_id),
58            chatbotconf_id: Some(assistant_configuration_id),
59            enabled_to_students: false,
60            chatbot_name: "Course design assistant".to_string(),
61            prompt: "You help teams design MOOCs. Base your advice on the guide course and on the team's plan, and keep it practical and brief.".to_string(),
62            initial_message: "Hi! I can review your plan, answer questions about MOOC design and suggest changes for you to approve.".to_string(),
63            use_azure_search: true,
64            enabled_tool_categories: vec![
65                ToolCategory::CourseMaterial,
66                ToolCategory::Interaction,
67                ToolCategory::CoursePlanning,
68            ],
69            model_id: llm.id,
70            ..Default::default()
71        })
72        .role(teacher_user_id, UserRole::Teacher)
73        .module(
74            ModuleBuilder::new()
75                .order(0)
76                .chapter(
77                    ChapterBuilder::new(1, "What a MOOC is")
78                        .fixed_ids(cx.v5(b"chapter:1"), cx.v5(b"chapter:1:front-page"))
79                        .pages([
80                            guide_page(&cx, "/chapter-1/what-is-a-mooc", "What is a MOOC?", WHAT_IS_A_MOOC),
81                            guide_page(&cx, "/chapter-1/addie", "The MOOC development process", ADDIE),
82                            guide_page(&cx, "/chapter-1/course-structure", "Course structure on mooc.fi", COURSE_STRUCTURE),
83                            guide_page(&cx, "/chapter-1/chapter-and-page-structure", "Chapter and page structure", CHAPTER_AND_PAGE_STRUCTURE),
84                        ]),
85                )
86                .chapter(
87                    ChapterBuilder::new(2, "Designing the course")
88                        .fixed_ids(cx.v5(b"chapter:2"), cx.v5(b"chapter:2:front-page"))
89                        .pages([
90                            guide_page(&cx, "/chapter-2/learning-goals", "Learning goals and learning objectives", LEARNING_GOALS),
91                            guide_page(&cx, "/chapter-2/core-content", "Course core content and structure", CORE_CONTENT),
92                            guide_page(&cx, "/chapter-2/content-formats", "Content formats", CONTENT_FORMATS),
93                            guide_page(&cx, "/chapter-2/exercises", "Exercises and student workload", EXERCISES),
94                            guide_page(&cx, "/chapter-2/style-guide", "Style guide", STYLE_GUIDE),
95                        ]),
96                )
97                .chapter(
98                    ChapterBuilder::new(3, "Implementing and evaluating the course")
99                        .fixed_ids(cx.v5(b"chapter:3"), cx.v5(b"chapter:3:front-page"))
100                        .pages([
101                            guide_page(&cx, "/chapter-3/implementation", "Implementation", IMPLEMENTATION),
102                            guide_page(&cx, "/chapter-3/evaluation", "Evaluation", EVALUATION),
103                        ]),
104                ),
105        );
106    let (course, _default_instance, _last_module) = course.seed(&mut conn, app_config, &cx).await?;
107    chatbot_configurations::set_course_designer_assistant(&mut conn, assistant_configuration_id)
108        .await?;
109    Ok(course.id)
110}

113    Heading(&'static str),
114    Paragraph(&'static str),
115    BulletList(&'static [&'static str]),
116}

120/// A guide page whose block ids derive from its path, so reseeding keeps them stable.
121fn guide_page(cx: &SeedContext, path: &str, title: &str, blocks: &[GuideBlock]) -> PageBuilder {
122    let id = |suffix: String| cx.v5(format!("page:{path}:{suffix}").as_bytes());
123    PageBuilder::new(path, title).blocks(blocks.iter().enumerate().map(|(index, block)| {
124        match block {
125            Heading(text) => heading(text, id(index.to_string()), 2),
126            Paragraph(text) => paragraph(text, id(index.to_string())),
127            BulletList(items) => list(
128                id(index.to_string()),
129                false,
130                items
131                    .iter()
132                    .enumerate()
133                    .map(|(item_index, item)| list_item(id(format!("{index}:{item_index}")), item))
134                    .collect::<Vec<GutenbergBlock>>(),
135            ),
136        }
137    }))
138}
