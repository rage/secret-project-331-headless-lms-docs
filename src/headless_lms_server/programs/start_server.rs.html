22/// The entrypoint to the server.
23pub async fn main() -> anyhow::Result<()> {
24    dotenv().ok();
25    setup_tracing()?;
26
27    // Required by rustls 0.23 so kube-client can build TLS configs.
28    ring::default_provider()
29        .install_default()
30        .expect("failed to install rustls ring crypto provider");
31
32    let runtime_config = ServerRuntimeConfig::try_from_env()?;
33    let private_cookie_key = runtime_config.private_cookie_key.clone();
34    let test_mode = runtime_config.test_mode;
35    let allow_no_https_for_development = runtime_config.allow_no_https_for_development;
36    let host = runtime_config.host.clone();
37    let port = runtime_config.port.clone();
38    set_server_runtime_config(runtime_config.clone())?;
39
40    if test_mode {
41        info!("***********************************");
42        info!("*  Starting backend in test mode  *");
43        info!("***********************************");
44    }
45    let server_config = ServerConfigBuilder::from_runtime_config(&runtime_config)
46        .await
47        .expect("Failed to create server config builder from runtime config")
48        .build()
49        .await
50        .expect("Failed to create server config");
51    let mut server = HttpServer::new(move || {
52        let server_config = server_config.clone();
53        App::new()
54            .configure(move |config| config::configure(config, server_config))
55            .wrap(
56                SessionMiddleware::builder(
57                    CookieSessionStore::default(),
58                    Key::from(private_cookie_key.expose_secret().as_bytes()),
59                )
60                .cookie_name("session".to_string())
61                .cookie_secure(!allow_no_https_for_development)
62                .cookie_same_site(SameSite::Strict) // Default api can only be accessed from the main website. Public api will be less strict on this.
63                .cookie_http_only(true) // Cookie is inaccessible from javascript for security
64                .cookie_path("/api".to_string()) // browser won't send the cookie unless this path exists in the request url
65                .cookie_content_security(CookieContentSecurity::Private)
66                .session_lifecycle(SessionLifecycle::PersistentSession(
67                    PersistentSession::default()
68                        .session_ttl(actix_web::cookie::time::Duration::days(100))
69                        .session_ttl_extension_policy(TtlExtensionPolicy::OnEveryRequest),
70                ))
71                .build(),
72            )
73            .wrap(
74                Logger::new(
75                    "Completed %{request_line}xi %s %b bytes - %D ms, request_id=%{request-id}o",
76                )
77                .custom_request_replace("request_line", redacted_request_line),
78            )
79    })
80    // Must outlive ingress-nginx's 60 s upstream keepalive, or nginx reuses a connection we are
81    // closing and answers the (unretried) POST with a 502.
82    .keep_alive(Duration::from_secs(75));
83
84    // this will enable us to keep application running during recompile: systemfd --no-pid -s http::5000 -- cargo watch -x run
85    let mut listenfd = ListenFd::from_env();
86    server = match listenfd.take_tcp_listener(0)? {
87        Some(listener) => server.listen(listener)?,
88        None => {
89            let bind_address = format!("{}:{}", host, port);
90            info!("Binding to address: {}", bind_address);
91            server.bind(bind_address)?
92        }
93    };
94
95    info!("Starting server.");
96    server.run().await?;
97
98    Ok(())
99}

101/// The request line for the access log with query values dropped (names kept) and the
102/// student-number linking token masked: student numbers travel in search query parameters and the
103/// token claims one.
104fn redacted_request_line(req: &ServiceRequest) -> String {
105    let mut path = String::with_capacity(req.path().len());
106    let mut is_token_segment = false;
107    for (i, segment) in req.path().split('/').enumerate() {
108        if i > 0 {
109            path.push('/');
110        }
111        path.push_str(if is_token_segment { "{token}" } else { segment });
112        // The route segment the credit registration controller mounts the token routes under.
113        is_token_segment = segment == "student-number-verifications";
114    }
115    let query = req.query_string();
116    if !query.is_empty() {
117        path.push('?');
118        for (i, pair) in query.split('&').enumerate() {
119            if i > 0 {
120                path.push('&');
121            }
122            path.push_str(pair.split('=').next().unwrap_or_default());
123        }
124    }
125    format!("{} {} {:?}", req.method(), path, req.version())
126}
