1//! The alert rules the credit registration dashboard renders.
2//!
3//! An alert carries identifiers and numbers, never prose: the study registry's own error text is
4//! written for an integrator and is not translated, so the frontend renders one key per alert id
5//! with these values interpolated. The thresholds travel with the alerts, not hardcoded twice.

21/// Within this much of the past, one rejected credential is enough.
22const CREDENTIAL_REJECTION_WINDOW_SECS: i64 = 60 * 60;
23const UNREACHABLE_WINDOW_SECS: i64 = 15 * 60;
24/// Below this the run is a bad minute rather than an outage.
25const UNREACHABLE_CONSECUTIVE_FAILURES: i64 = 3;
26const SISU_OUTAGE_WINDOW_SECS: i64 = 15 * 60;
27/// Below this many items the share below is one bad batch, not a signal.
28const SISU_OUTAGE_MIN_ITEMS: i64 = 10;
29const SISU_OUTAGE_FAILURE_SHARE_PERCENT: i64 = 30;
30const STUCK_THRESHOLDS: StuckThresholds = StuckThresholds {
31    ready_to_submit_secs: 2 * 60 * 60,
32    submitting_secs: 15 * 60,
33    awaiting_verification_secs: 24 * 60 * 60,
34    failed_retryable_secs: 3 * 24 * 60 * 60,
35};
36/// Above this many stuck rows the backlog stops being something to look at tomorrow.
37const STUCK_CRITICAL_COUNT: i64 = 50;
38const LINKING_MAIL_WINDOW_SECS: i64 = 7 * 24 * 60 * 60;
39/// A phase is late once this many of its own intervals have passed without a heartbeat.
40/// `pub(crate)` because the dashboard's phase rows apply the same threshold server-side.
41pub(crate) const PHASE_HEARTBEAT_INTERVAL_MULTIPLIER: i32 = 2;
42/// Failures in a row before a phase counts as broken rather than unlucky.
43pub(crate) const PHASE_CONSECUTIVE_FAILURE_LIMIT: i32 = 5;
44/// A phase that owns a nonempty queue and has not succeeded within this many of its own intervals
45/// is running without getting anywhere, which no failure count catches.
46const PHASE_SUCCESS_INTERVAL_MULTIPLIER: i32 = 10;
47/// The window every "in the last day" rule shares.
48const TERMINAL_WINDOW_SECS: i64 = 24 * 60 * 60;
49const PERMANENT_FAILURE_COUNT: i64 = 20;
50const PERMANENT_FAILURE_RATE_PERCENT: i64 = 10;
51/// A reversal is always worth saying; this many at once is an incident.
52const MISREGISTRATION_CRITICAL_COUNT: i64 = 5;
53/// Linking mails one hour may hand over before the volume itself is the problem.
54const LINKING_MAIL_HOURLY_CAP: i64 = 500;
55/// Queued work that makes a day without a single completion mean something.
56const IDLE_QUEUE_DEPTH: i64 = 20;
57/// How long a completion may sit outside the ledger before `materialize` is the suspect rather
58/// than the clock.
59const NEVER_ENTERED_MIN_AGE_SECS: i64 = 6 * 60 * 60;
60/// Bounds the anti-join behind that rule; a bigger backlog reports as this many.
61const NEVER_ENTERED_SAMPLE_LIMIT: i64 = 100;
62const LATENCY_WINDOW_SECS: i64 = 7 * 24 * 60 * 60;
63/// Under this the registry is quick enough that a doubling says nothing.
64const LATENCY_REGRESSION_FLOOR_SECS: i64 = 6 * 60 * 60;
65const LATENCY_REGRESSION_FACTOR: i64 = 2;
66/// One person the registry names differently from the account whose address matched is worth a
67/// look: it is the only signal we get that a university address was reissued.
68const FAST_TRACK_NAME_MISMATCH_COUNT: i64 = 1;

100#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
101pub struct CreditRegistrationAlert {
102    pub id: CreditRegistrationAlertId,
103    pub severity: CreditRegistrationAlertSeverity,
104    /// How many rows, calls or phases the rule found.
105    pub count: i64,
106    /// What `count` is out of, where the rule measured one. Not a threshold: thresholds are the
107    /// same for every evaluation and travel separately.
108    pub total: Option<i64>,
109    /// When it last happened, where the rule has an instant to point at.
110    pub at: Option<DateTime<Utc>>,
111    /// An identifier the operator can act on — a phase name, a ledger state, a mail domain. Never a
112    /// sentence, and never anything the study registry wrote.
113    pub subject: Option<String>,
114}

190/// Runs every rule and ranks what it found.
191///
192/// `stuck` and `depths` are passed in because the caller already reads both aggregates, the two
193/// most expensive reads in the request. `depths` is the live count per state, superseded rows
194/// excluded, as [`credit_registrations::count_by_state`] returns it.
195pub async fn evaluate(
196    conn: &mut PgConnection,
197    stuck: &[StuckRegistrationCount],
198    depths: &[(CreditRegistrationState, i64)],
199) -> ModelResult<CreditRegistrationHealth> {
200    let now = Utc::now();
201    let mut alerts = Vec::new();
202
203    let credentials = suotar_api_calls::count_credential_rejections_since(
204        conn,
205        now - chrono::Duration::seconds(CREDENTIAL_REJECTION_WINDOW_SECS),
206    )
207    .await?;
208    if credentials.count > 0 {
209        alerts.push(CreditRegistrationAlert {
210            id: CreditRegistrationAlertId::CredentialsRejected,
211            severity: CreditRegistrationAlertSeverity::Critical,
212            count: credentials.count,
213            total: None,
214            at: credentials.last_at,
215            subject: None,
216        });
217    }
218
219    let unreachable = suotar_api_calls::count_unreachable_run_since(
220        conn,
221        now - chrono::Duration::seconds(UNREACHABLE_WINDOW_SECS),
222    )
223    .await?;
224    if unreachable.count >= UNREACHABLE_CONSECUTIVE_FAILURES {
225        alerts.push(CreditRegistrationAlert {
226            id: CreditRegistrationAlertId::StudyRegistryUnreachable,
227            severity: CreditRegistrationAlertSeverity::Critical,
228            count: unreachable.count,
229            total: None,
230            at: unreachable.last_at,
231            subject: None,
232        });
233    }
234
235    if let Some(alert) = sisu_outage_alert(conn, now).await? {
236        alerts.push(alert);
237    }
238    if let Some(alert) = stuck_alert(stuck) {
239        alerts.push(alert);
240    }
241    if let Some(alert) = linking_mail_alert(conn, now).await? {
242        alerts.push(alert);
243    }
244    if let Some(alert) = linking_mail_rate_alert(conn, now).await? {
245        alerts.push(alert);
246    }
247    alerts.extend(phase_alerts(conn, now, depths).await?);
248    alerts.extend(terminal_outcome_alerts(conn, now, depths).await?);
249    if let Some(alert) = course_configuration_alert(conn).await? {
250        alerts.push(alert);
251    }
252    if let Some(alert) = never_entered_alert(conn).await? {
253        alerts.push(alert);
254    }
255    if let Some(alert) = latency_regression_alert(conn, now).await? {
256        alerts.push(alert);
257    }
258    if let Some(alert) = fast_track_name_mismatch_alert(conn).await? {
259        alerts.push(alert);
260    }
261
262    alerts.sort_by_key(|alert| {
263        (
264            std::cmp::Reverse(alert.severity),
265            alert.id != CreditRegistrationAlertId::CredentialsRejected,
266        )
267    });
268    let status = match alerts.iter().map(|alert| alert.severity).max() {
269        Some(CreditRegistrationAlertSeverity::Critical) => HealthStatus::Error,
270        Some(CreditRegistrationAlertSeverity::Warning) => HealthStatus::Warning,
271        Some(CreditRegistrationAlertSeverity::Info) | None => HealthStatus::Healthy,
272    };
273
274    Ok(CreditRegistrationHealth {
275        status,
276        alerts,
277        thresholds: thresholds(),
278    })
279}

281/// The share of recent items the study registry blamed on Sisu. Our only proxy for Sisu's uptime,
282/// which is why it is a rule of its own rather than part of the request-level one above.
283async fn sisu_outage_alert(
284    conn: &mut PgConnection,
285    now: DateTime<Utc>,
286) -> ModelResult<Option<CreditRegistrationAlert>> {
287    let totals = credit_registration_events::count_item_outcomes_since(
288        conn,
289        now - chrono::Duration::seconds(SISU_OUTAGE_WINDOW_SECS),
290    )
291    .await?;
292    if totals.item_count < SISU_OUTAGE_MIN_ITEMS
293        || totals.sisu_unavailable_count * 100
294            < totals.item_count * SISU_OUTAGE_FAILURE_SHARE_PERCENT
295    {
296        return Ok(None);
297    }
298    Ok(Some(CreditRegistrationAlert {
299        id: CreditRegistrationAlertId::SisuUnavailable,
300        severity: CreditRegistrationAlertSeverity::Critical,
301        count: totals.sisu_unavailable_count,
302        total: Some(totals.item_count),
303        at: totals.last_sisu_unavailable_at,
304        subject: None,
305    }))
306}

308/// Rows the pipeline should have moved on by now. `abandoned_by_consent_withdrawal` is terminal, so
309/// it is outside this by construction rather than by a filter that could be forgotten.
310fn stuck_alert(stuck: &[StuckRegistrationCount]) -> Option<CreditRegistrationAlert> {
311    let total: i64 = stuck.iter().map(|row| row.count).sum();
312    if total == 0 {
313        return None;
314    }
315    let severe: i64 = stuck.iter().map(|row| row.severely_stuck_count).sum();
316    let worst = stuck.iter().max_by_key(|row| row.count);
317    let severity = if total > STUCK_CRITICAL_COUNT || severe > 0 {
318        CreditRegistrationAlertSeverity::Critical
319    } else {
320        CreditRegistrationAlertSeverity::Warning
321    };
322    Some(CreditRegistrationAlert {
323        id: CreditRegistrationAlertId::StuckRegistrations,
324        severity,
325        count: total,
326        total: None,
327        at: worst.and_then(|row| row.oldest_state_entered_at),
328        subject: worst.map(|row| state_name(row.state)),
329    })
330}

332/// Linking mails we could not hand over at all. The recipient domain rides along because an
333/// undeliverable host is the usual cause.
334async fn linking_mail_alert(
335    conn: &mut PgConnection,
336    now: DateTime<Utc>,
337) -> ModelResult<Option<CreditRegistrationAlert>> {
338    let since = now - chrono::Duration::seconds(LINKING_MAIL_WINDOW_SECS);
339    let totals =
340        credit_registration_account_linking_emails::get_send_status_totals_since(conn, since, now)
341            .await?;
342    if totals.send_failed == 0 {
343        return Ok(None);
344    }
345    let top_domain = credit_registration_account_linking_emails::get_send_failure_domains_since(
346        conn, since, now,
347    )
348    .await?
349    .into_iter()
350    .next()
351    .map(|row| row.domain);
352    Ok(Some(CreditRegistrationAlert {
353        id: CreditRegistrationAlertId::LinkingMailSendFailed,
354        severity: CreditRegistrationAlertSeverity::Warning,
355        count: totals.send_failed,
356        total: Some(totals.mails_in_window),
357        at: totals.last_send_failed_at,
358        subject: top_domain,
359    }))
360}

362/// The volume guard: how many people we mailed in the last hour against what an hour should hold.
363/// Counts addresses, which is what the per-person caps govern.
364async fn linking_mail_rate_alert(
365    conn: &mut PgConnection,
366    now: DateTime<Utc>,
367) -> ModelResult<Option<CreditRegistrationAlert>> {
368    let sent = credit_registration_account_linking_emails::count_sent_since(
369        conn,
370        now - chrono::Duration::hours(1),
371    )
372    .await?;
373    if sent <= LINKING_MAIL_HOURLY_CAP {
374        return Ok(None);
375    }
376    let severity = if sent > LINKING_MAIL_HOURLY_CAP * 2 {
377        CreditRegistrationAlertSeverity::Critical
378    } else {
379        CreditRegistrationAlertSeverity::Warning
380    };
381    Ok(Some(CreditRegistrationAlert {
382        id: CreditRegistrationAlertId::LinkingMailRateCapExceeded,
383        severity,
384        count: sent,
385        total: Some(LINKING_MAIL_HOURLY_CAP),
386        at: Some(now),
387        subject: None,
388    }))
389}

391/// What the phase table says about itself: phases that stopped reporting, and phases that report
392/// but get nowhere.
393///
394/// A phase that has never heartbeated is deliberately outside both: a freshly migrated database has
395/// no heartbeats at all, and that would keep the banner permanently red.
396async fn phase_alerts(
397    conn: &mut PgConnection,
398    now: DateTime<Utc>,
399    depths: &[(CreditRegistrationState, i64)],
400) -> ModelResult<Vec<CreditRegistrationAlert>> {
401    let phases = credit_registration_phase_state::get_all(conn).await?;
402    let mut stale: Vec<&str> = Vec::new();
403    let mut failing: Vec<&str> = Vec::new();
404    let mut paused = 0;
405    let mut last_paused_at = None;
406    for phase in &phases {
407        if let Some(paused_at) = phase.paused_at {
408            paused += 1;
409            last_paused_at = last_paused_at.max(Some(paused_at));
410            continue;
411        }
412        let interval = i64::from(phase.expected_interval_secs);
413        if let Some(last_heartbeat_at) = phase.last_heartbeat_at
414            && (now - last_heartbeat_at).num_seconds()
415                > interval * i64::from(PHASE_HEARTBEAT_INTERVAL_MULTIPLIER)
416        {
417            stale.push(&phase.phase);
418        }
419        let owns_work =
420            crate::domain::credit_registration_phases::CreditRegistrationPhase::from_phase_name(
421                &phase.phase,
422            )
423            .is_some_and(|known| owned_depth(known, depths) > 0);
424        let unproductive = owns_work
425            && phase.last_success_at.is_some_and(|last_success_at| {
426                (now - last_success_at).num_seconds()
427                    > interval * i64::from(PHASE_SUCCESS_INTERVAL_MULTIPLIER)
428            });
429        if phase.consecutive_failures >= PHASE_CONSECUTIVE_FAILURE_LIMIT || unproductive {
430            failing.push(&phase.phase);
431        }
432    }
433
434    let mut alerts = Vec::new();
435    if !stale.is_empty() {
436        alerts.push(CreditRegistrationAlert {
437            id: CreditRegistrationAlertId::PhaseHeartbeatStale,
438            severity: CreditRegistrationAlertSeverity::Critical,
439            count: stale.len() as i64,
440            total: Some(phases.len() as i64),
441            at: None,
442            subject: stale.first().map(|phase| (*phase).to_string()),
443        });
444    }
445    if !failing.is_empty() {
446        alerts.push(CreditRegistrationAlert {
447            id: CreditRegistrationAlertId::PhaseFailing,
448            severity: CreditRegistrationAlertSeverity::Critical,
449            count: failing.len() as i64,
450            total: Some(phases.len() as i64),
451            at: None,
452            subject: failing.first().map(|phase| (*phase).to_string()),
453        });
454    }
455    if paused > 0 && paused == phases.len() {
456        alerts.push(CreditRegistrationAlert {
457            id: CreditRegistrationAlertId::PipelinePausedGlobally,
458            severity: CreditRegistrationAlertSeverity::Info,
459            count: paused as i64,
460            total: Some(phases.len() as i64),
461            at: last_paused_at,
462            subject: None,
463        });
464    }
465    Ok(alerts)
466}

468/// The three rules read off the last day's terminal outcomes: failures piling up, reversals, and a
469/// pipeline that finished nothing while holding work.
470async fn terminal_outcome_alerts(
471    conn: &mut PgConnection,
472    now: DateTime<Utc>,
473    depths: &[(CreditRegistrationState, i64)],
474) -> ModelResult<Vec<CreditRegistrationAlert>> {
475    let since = now - chrono::Duration::seconds(TERMINAL_WINDOW_SECS);
476    let totals = credit_registrations::count_terminal_outcomes_since(conn, since).await?;
477    let mut alerts = Vec::new();
478
479    let rate_broken = totals.total_count >= PERMANENT_FAILURE_COUNT
480        && totals.failed_permanent_count * 100
481            > totals.total_count * PERMANENT_FAILURE_RATE_PERCENT;
482    if totals.failed_permanent_count >= PERMANENT_FAILURE_COUNT || rate_broken {
483        alerts.push(CreditRegistrationAlert {
484            id: CreditRegistrationAlertId::PermanentFailuresAccumulating,
485            severity: CreditRegistrationAlertSeverity::Warning,
486            count: totals.failed_permanent_count,
487            total: Some(totals.total_count),
488            at: Some(now),
489            subject: None,
490        });
491    }
492
493    let misregistered = credit_registrations::count_entered_state_since(
494        conn,
495        CreditRegistrationState::Misregistered,
496        since,
497    )
498    .await?;
499    if misregistered > 0 {
500        alerts.push(CreditRegistrationAlert {
501            id: CreditRegistrationAlertId::MisregistrationsDetected,
502            severity: if misregistered >= MISREGISTRATION_CRITICAL_COUNT {
503                CreditRegistrationAlertSeverity::Critical
504            } else {
505                CreditRegistrationAlertSeverity::Warning
506            },
507            count: misregistered,
508            total: None,
509            at: Some(now),
510            subject: None,
511        });
512    }
513
514    let queued = depth_of(depths, CreditRegistrationState::ReadyToSubmit)
515        + depth_of(depths, CreditRegistrationState::AwaitingVerification);
516    if totals.total_count == 0 && queued > IDLE_QUEUE_DEPTH {
517        alerts.push(CreditRegistrationAlert {
518            id: CreditRegistrationAlertId::PipelineIdle,
519            severity: CreditRegistrationAlertSeverity::Warning,
520            count: queued,
521            total: None,
522            at: Some(now),
523            subject: None,
524        });
525    }
526    Ok(alerts)
527}

546/// Completions old enough that `materialize` has had every chance and still has no ledger row for
547/// them. Sampled rather than counted, so the anti-join stops early on a large backlog.
548async fn never_entered_alert(
549    conn: &mut PgConnection,
550) -> ModelResult<Option<CreditRegistrationAlert>> {
551    let found = get_unmaterialised_eligible_completions(
552        conn,
553        NEVER_ENTERED_MIN_AGE_SECS,
554        NEVER_ENTERED_SAMPLE_LIMIT,
555    )
556    .await?;
557    if found.is_empty() {
558        return Ok(None);
559    }
560    Ok(Some(CreditRegistrationAlert {
561        id: CreditRegistrationAlertId::CompletionsNeverEntered,
562        severity: CreditRegistrationAlertSeverity::Warning,
563        count: found.len() as i64,
564        total: Some(NEVER_ENTERED_SAMPLE_LIMIT),
565        at: found.first().map(|row| row.created_at),
566        subject: None,
567    }))
568}

570/// How long the study registry is taking to confirm, this week against last. `count` is this
571/// week's p95 in seconds and `total` last week's, so the banner can name both.
572async fn latency_regression_alert(
573    conn: &mut PgConnection,
574    now: DateTime<Utc>,
575) -> ModelResult<Option<CreditRegistrationAlert>> {
576    let window = chrono::Duration::seconds(LATENCY_WINDOW_SECS);
577    let current =
578        credit_registrations::get_registration_latency_between(conn, now - window, now).await?;
579    let (Some(current_p95), true) = (current.p95_confirmation_secs, current.registered_count > 0)
580    else {
581        return Ok(None);
582    };
583    if current_p95 < LATENCY_REGRESSION_FLOOR_SECS {
584        return Ok(None);
585    }
586    let previous = credit_registrations::get_registration_latency_between(
587        conn,
588        now - window * 2,
589        now - window,
590    )
591    .await?;
592    let Some(previous_p95) = previous
593        .p95_confirmation_secs
594        .filter(|_| previous.registered_count > 0)
595    else {
596        return Ok(None);
597    };
598    if current_p95 <= previous_p95 * LATENCY_REGRESSION_FACTOR {
599        return Ok(None);
600    }
601    Ok(Some(CreditRegistrationAlert {
602        id: CreditRegistrationAlertId::ConfirmationLatencyRegressed,
603        severity: CreditRegistrationAlertSeverity::Info,
604        count: current_p95,
605        total: Some(previous_p95),
606        at: Some(now),
607        subject: None,
608    }))
609}

631fn depth_of(depths: &[(CreditRegistrationState, i64)], state: CreditRegistrationState) -> i64 {
632    depths
633        .iter()
634        .find(|(row_state, _)| *row_state == state)
635        .map_or(0, |(_, count)| *count)
636}
637
638fn owned_depth(
639    phase: crate::domain::credit_registration_phases::CreditRegistrationPhase,
640    depths: &[(CreditRegistrationState, i64)],
641) -> i64 {
642    phase
643        .owned_states()
644        .iter()
645        .map(|state| depth_of(depths, *state))
646        .sum()
647}

649/// The state's own wire name, taken from its serialisation so the two cannot drift.
650fn state_name(state: CreditRegistrationState) -> String {
651    serde_json::to_value(state)
652        .ok()
653        .and_then(|value| value.as_str().map(str::to_string))
654        .unwrap_or_default()
655}
