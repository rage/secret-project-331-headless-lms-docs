1//! The alert rules the credit registration dashboard renders.
2//!
3//! An alert carries identifiers and numbers, never prose: the study registry's own error text is
4//! written for an integrator and is not translated, so the frontend renders one key per alert id
5//! with these values interpolated. The thresholds travel with the alerts, not hardcoded twice.

17/// Within this much of the past, one rejected credential is enough.
18const CREDENTIAL_REJECTION_WINDOW_SECS: i64 = 60 * 60;
19const UNREACHABLE_WINDOW_SECS: i64 = 15 * 60;
20/// Below this the run is a bad minute rather than an outage.
21const UNREACHABLE_CONSECUTIVE_FAILURES: i64 = 3;
22const STUCK_THRESHOLDS: StuckThresholds = StuckThresholds {
23    ready_to_submit_secs: 2 * 60 * 60,
24    submitting_secs: 15 * 60,
25    awaiting_verification_secs: 24 * 60 * 60,
26    failed_retryable_secs: 3 * 24 * 60 * 60,
27};
28/// Above this many stuck rows the backlog stops being something to look at tomorrow.
29const STUCK_CRITICAL_COUNT: i64 = 50;
30const LINKING_MAIL_WINDOW_SECS: i64 = 7 * 24 * 60 * 60;
31/// A phase is late once this many of its own intervals have passed without a heartbeat.
32/// `pub(crate)` because the dashboard's phase rows apply the same threshold server-side.
33pub(crate) const PHASE_HEARTBEAT_INTERVAL_MULTIPLIER: i32 = 2;

35#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
36#[serde(rename_all = "snake_case")]
37pub enum CreditRegistrationAlertId {
38    CredentialsRejected,
39    StudyRegistryUnreachable,
40    StuckRegistrations,
41    LinkingMailSendFailed,
42    PhaseHeartbeatStale,
43}

46#[serde(rename_all = "snake_case")]
47pub enum CreditRegistrationAlertSeverity {
48    Warning,
49    Critical,
50}

52#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
53pub struct CreditRegistrationAlert {
54    pub id: CreditRegistrationAlertId,
55    pub severity: CreditRegistrationAlertSeverity,
56    /// How many rows, calls or phases the rule found.
57    pub count: i64,
58    /// When it last happened, where the rule has an instant to point at.
59    pub at: Option<DateTime<Utc>>,
60    /// An identifier the operator can act on — a phase name, a ledger state, a mail domain. Never a
61    /// sentence, and never anything the study registry wrote.
62    pub subject: Option<String>,
63}

88pub fn thresholds() -> CreditRegistrationAlertThresholds {
89    CreditRegistrationAlertThresholds {
90        credential_rejection_window_secs: CREDENTIAL_REJECTION_WINDOW_SECS,
91        unreachable_window_secs: UNREACHABLE_WINDOW_SECS,
92        unreachable_consecutive_failures: UNREACHABLE_CONSECUTIVE_FAILURES,
93        stuck_ready_to_submit_secs: STUCK_THRESHOLDS.ready_to_submit_secs,
94        stuck_submitting_secs: STUCK_THRESHOLDS.submitting_secs,
95        stuck_awaiting_verification_secs: STUCK_THRESHOLDS.awaiting_verification_secs,
96        stuck_failed_retryable_secs: STUCK_THRESHOLDS.failed_retryable_secs,
97        stuck_critical_count: STUCK_CRITICAL_COUNT,
98        linking_mail_window_secs: LINKING_MAIL_WINDOW_SECS,
99        phase_heartbeat_interval_multiplier: PHASE_HEARTBEAT_INTERVAL_MULTIPLIER,
100    }

107/// Runs every rule and ranks what it found. `stuck` is passed in because the caller already reads
108/// that aggregate, the most expensive read in the request.
109pub async fn evaluate(
110    conn: &mut PgConnection,
111    stuck: &[StuckRegistrationCount],
112) -> ModelResult<CreditRegistrationHealth> {
113    let now = Utc::now();
114    let mut alerts = Vec::new();
115
116    let credentials = suotar_api_calls::count_credential_rejections_since(
117        conn,
118        now - chrono::Duration::seconds(CREDENTIAL_REJECTION_WINDOW_SECS),
119    )
120    .await?;
121    if credentials.count > 0 {
122        alerts.push(CreditRegistrationAlert {
123            id: CreditRegistrationAlertId::CredentialsRejected,
124            severity: CreditRegistrationAlertSeverity::Critical,
125            count: credentials.count,
126            at: credentials.last_at,
127            subject: None,
128        });
129    }
130
131    let unreachable = suotar_api_calls::count_unreachable_run_since(
132        conn,
133        now - chrono::Duration::seconds(UNREACHABLE_WINDOW_SECS),
134    )
135    .await?;
136    if unreachable.count >= UNREACHABLE_CONSECUTIVE_FAILURES {
137        alerts.push(CreditRegistrationAlert {
138            id: CreditRegistrationAlertId::StudyRegistryUnreachable,
139            severity: CreditRegistrationAlertSeverity::Critical,
140            count: unreachable.count,
141            at: unreachable.last_at,
142            subject: None,
143        });
144    }
145
146    if let Some(alert) = stuck_alert(stuck) {
147        alerts.push(alert);
148    }
149    if let Some(alert) = linking_mail_alert(conn, now).await? {
150        alerts.push(alert);
151    }
152    if let Some(alert) = phase_heartbeat_alert(conn, now).await? {
153        alerts.push(alert);
154    }
155
156    alerts.sort_by_key(|alert| {
157        (
158            std::cmp::Reverse(alert.severity),
159            alert.id != CreditRegistrationAlertId::CredentialsRejected,
160        )
161    });
162    let status = match alerts.iter().map(|alert| alert.severity).max() {
163        Some(CreditRegistrationAlertSeverity::Critical) => HealthStatus::Error,
164        Some(CreditRegistrationAlertSeverity::Warning) => HealthStatus::Warning,
165        None => HealthStatus::Healthy,
166    };
167
168    Ok(CreditRegistrationHealth {
169        status,
170        alerts,
171        thresholds: thresholds(),
172    })
173}

175/// Rows the pipeline should have moved on by now. `abandoned_by_consent_withdrawal` is terminal, so
176/// it is outside this by construction rather than by a filter that could be forgotten.
177fn stuck_alert(stuck: &[StuckRegistrationCount]) -> Option<CreditRegistrationAlert> {
178    let total: i64 = stuck.iter().map(|row| row.count).sum();
179    if total == 0 {
180        return None;
181    }
182    let severe: i64 = stuck.iter().map(|row| row.severely_stuck_count).sum();
183    let worst = stuck.iter().max_by_key(|row| row.count);
184    let severity = if total > STUCK_CRITICAL_COUNT || severe > 0 {
185        CreditRegistrationAlertSeverity::Critical
186    } else {
187        CreditRegistrationAlertSeverity::Warning
188    };
189    Some(CreditRegistrationAlert {
190        id: CreditRegistrationAlertId::StuckRegistrations,
191        severity,
192        count: total,
193        at: worst.and_then(|row| row.oldest_state_entered_at),
194        subject: worst.map(|row| state_name(row.state)),
195    })
196}

198/// Linking mails we could not hand over at all. The recipient domain rides along because an
199/// undeliverable host is the usual cause.
200async fn linking_mail_alert(
201    conn: &mut PgConnection,
202    now: DateTime<Utc>,
203) -> ModelResult<Option<CreditRegistrationAlert>> {
204    let since = now - chrono::Duration::seconds(LINKING_MAIL_WINDOW_SECS);
205    let totals =
206        credit_registration_account_linking_emails::get_send_status_totals_since(conn, since, now)
207            .await?;
208    if totals.send_failed == 0 {
209        return Ok(None);
210    }
211    let top_domain = credit_registration_account_linking_emails::get_send_failure_domains_since(
212        conn, since, now,
213    )
214    .await?
215    .into_iter()
216    .next()
217    .map(|row| row.domain);
218    Ok(Some(CreditRegistrationAlert {
219        id: CreditRegistrationAlertId::LinkingMailSendFailed,
220        severity: CreditRegistrationAlertSeverity::Warning,
221        count: totals.send_failed,
222        at: totals.last_send_failed_at,
223        subject: top_domain,
224    }))
225}

227/// A phase that reported and then stopped. Deliberately not "has never reported": the unimplemented
228/// phases never heartbeat and a freshly migrated database has none at all, so that would keep the
229/// banner permanently red.
230async fn phase_heartbeat_alert(
231    conn: &mut PgConnection,
232    now: DateTime<Utc>,
233) -> ModelResult<Option<CreditRegistrationAlert>> {
234    let phases = credit_registration_phase_state::get_all(conn).await?;
235    let mut stale: Vec<&str> = Vec::new();
236    for phase in &phases {
237        if phase.paused_at.is_some() {
238            continue;
239        }
240        let Some(last_heartbeat_at) = phase.last_heartbeat_at else {
241            continue;
242        };
243        let allowed = i64::from(phase.expected_interval_secs)
244            * i64::from(PHASE_HEARTBEAT_INTERVAL_MULTIPLIER);
245        if (now - last_heartbeat_at).num_seconds() > allowed {
246            stale.push(&phase.phase);
247        }
248    }
249    if stale.is_empty() {
250        return Ok(None);
251    }
252    Ok(Some(CreditRegistrationAlert {
253        id: CreditRegistrationAlertId::PhaseHeartbeatStale,
254        severity: CreditRegistrationAlertSeverity::Critical,
255        count: stale.len() as i64,
256        at: None,
257        subject: stale.first().map(|phase| (*phase).to_string()),
258    }))
259}

261/// The state's own wire name, taken from its serialisation so the two cannot drift.
262fn state_name(state: credit_registrations::CreditRegistrationState) -> String {
263    serde_json::to_value(state)
264        .ok()
265        .and_then(|value| value.as_str().map(str::to_string))
266        .unwrap_or_default()
267}
