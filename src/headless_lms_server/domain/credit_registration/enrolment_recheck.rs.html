15/// Whether a row's student or a teacher may ask for an enrolment check right now, which is when
16/// their buttons show.
17pub fn can_request_enrolment_recheck(
18    state: CreditRegistrationState,
19    enrolment_check_requested_at: Option<DateTime<Utc>>,
20    enrolment_checked_at: Option<DateTime<Utc>>,
21) -> bool {
22    state == CreditRegistrationState::NoUsableEnrolment
23        && !enrolment_checks::is_check_request_limited(
24            enrolment_check_requested_at,
25            enrolment_checked_at,
26            Utc::now(),
27        )
28}

30/// A row waiting for an enrolment to check, and the completion it registers.
31#[derive(Debug, Clone, Copy)]
32pub struct RecheckTarget {
33    pub registration_id: Uuid,
34    pub course_module_completion_id: Uuid,
35}

37/// Asks for an enrolment check of a row waiting for one, and records who asked.
38///
39/// Shared by the student's recheck button, pressing Done and the teacher's recheck, which differ in
40/// `source` and the event. Every kind shares one limit on asking; see
41/// [`enrolment_checks::request_check`]. The completion's check signal is recorded unless the row
42/// turned out not to be waiting, even for a request refused as too soon.
43pub async fn start_enrolment_recheck(
44    conn: &mut PgConnection,
45    actor_user_id: Uuid,
46    target: RecheckTarget,
47    source: EnrolmentCheckSource,
48    event_kind: CreditRegistrationEventKind,
49    message: &str,
50) -> Result<CheckRequestOutcome, ControllerError> {
51    let registration_id = target.registration_id;
52    let mut tx = conn.begin().await?;
53    let outcome =
54        enrolment_checks::request_check(&mut tx, registration_id, source, Utc::now()).await?;
55    if outcome == CheckRequestOutcome::NotWaiting {
56        tx.commit().await?;
57        return Ok(outcome);
58    }
59    credit_registration_enrolment_check_signals::record_check_request(
60        &mut tx,
61        target.course_module_completion_id,
62        source,
63    )
64    .await?;
65    if outcome == CheckRequestOutcome::TooSoon {
66        tx.commit().await?;
67        return Ok(outcome);
68    }
69    models::credit_registration_events::insert(
70        &mut tx,
71        &NewCreditRegistrationEvent {
72            actor_user_id: Some(actor_user_id),
73            message: Some(message.to_string()),
74            ..NewCreditRegistrationEvent::new(registration_id, event_kind)
75        },
76    )
77    .await?;
78    tx.commit().await?;
79    Ok(outcome)
80}
