1//! Storing ungraded answer snapshots (pastes and backups) for a native client.
2
3use crate::domain::exercise_services::answer_uploads;
4use crate::domain::rate_limit_middleware_builder::{EndpointLimiters, RateLimitConfig};
5use crate::prelude::*;
6use models::exercise_answer_snapshots::{
7    ExerciseAnswerSnapshot, NewExerciseAnswerSnapshot, SnapshotPurpose,
8};
9use std::sync::LazyLock;
10
11/// How many backups of one exercise a user keeps; storing another deletes the oldest. Pastes are
12/// not counted, since each one backs a link the user has handed out.
13pub const MAX_BACKUPS_PER_USER_AND_EXERCISE: i64 = 20;
14
15/// Pastes a user may store of one exercise per minute. Backups need no limit of their own: each
16/// one past [`MAX_BACKUPS_PER_USER_AND_EXERCISE`] deletes the oldest, so their storage stays
17/// bounded.
18const PASTES_PER_MINUTE: u64 = 3;
19/// Pastes a user may store of one exercise per hour.
20const PASTES_PER_HOUR: u64 = 20;
21
22static PASTE_LIMITS: LazyLock<EndpointLimiters> = LazyLock::new(|| {
23    EndpointLimiters::from_config(&RateLimitConfig {
24        per_minute: Some(PASTES_PER_MINUTE),
25        per_hour: Some(PASTES_PER_HOUR),
26        ..Default::default()
27    })
28});
29
30/// The answer a snapshot stores. The caller has established which course exercise, slide and task
31/// it is for and that the user may view them.
32pub struct SnapshotAnswer {
33    pub user_id: Uuid,
34    pub course_id: Uuid,
35    pub exercise_id: Uuid,
36    pub exercise_slide_id: Uuid,
37    pub exercise_task_id: Uuid,
38    pub purpose: SnapshotPurpose,
39    pub data_json: Option<serde_json::Value>,
40    pub file_upload_ids: Vec<Uuid>,
41}
42
43/// Stores a snapshot of the named uploads, then trims the user's backups of the exercise to
44/// [`MAX_BACKUPS_PER_USER_AND_EXERCISE`]. A paste over the per-user, per-exercise limit is
45/// refused with 429.
46///
47/// Applies the same upload checks as a file submit, but nothing a submit's acceptance depends on:
48/// no deadline, try limit or chapter lock, because a snapshot is never graded.
49pub async fn store_snapshot(
50    conn: &mut PgConnection,
51    answer: SnapshotAnswer,
52) -> Result<ExerciseAnswerSnapshot, ControllerError> {
53    answer_uploads::verify_answer_names_uploads(&answer.file_upload_ids)?;
54
55    let mut tx = conn.begin().await?;
56    answer_uploads::lock_and_verify_uploads_are_usable(
57        &mut tx,
58        answer.exercise_id,
59        answer.user_id,
60        &answer.file_upload_ids,
61    )
62    .await?;
63    // Last before the insert, so that a request refused for another reason uses up no quota.
64    if answer.purpose == SnapshotPurpose::Paste
65        && let Err(retry_after) =
66            PASTE_LIMITS.check_key(format!("{}:{}", answer.user_id, answer.exercise_id))
67    {
68        return Err(controller_err!(
69            TooManyRequests(Some(retry_after)),
70            "Too many pastes of this exercise; try again later.".to_string()
71        ));
72    }
73    let snapshot = models::exercise_answer_snapshots::insert(
74        &mut tx,
75        NewExerciseAnswerSnapshot {
76            user_id: answer.user_id,
77            course_id: answer.course_id,
78            exercise_id: answer.exercise_id,
79            exercise_slide_id: answer.exercise_slide_id,
80            exercise_task_id: answer.exercise_task_id,
81            purpose: answer.purpose,
82            data_json: answer.data_json,
83        },
84        &answer.file_upload_ids,
85    )
86    .await?;
87    if answer.purpose == SnapshotPurpose::Backup {
88        models::exercise_answer_snapshots::delete_backups_beyond(
89            &mut tx,
90            answer.user_id,
91            answer.exercise_id,
92            MAX_BACKUPS_PER_USER_AND_EXERCISE,
93        )
94        .await?;
95    }
96    tx.commit().await?;
97    Ok(snapshot)
98}
