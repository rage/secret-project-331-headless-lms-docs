1//! Storing ungraded answer snapshots (pastes and backups) for a native client.
2
3use crate::domain::exercise_services::answer_uploads;
4use crate::domain::rate_limit_middleware_builder::{EndpointLimiters, RateLimitConfig};
5use crate::prelude::*;
6use models::exercise_answer_snapshots::{
7    ExerciseAnswerSnapshot, NewExerciseAnswerSnapshot, SnapshotPurpose,
8};
9use std::sync::LazyLock;
10
11/// How many backups of one exercise a user keeps; storing another deletes the oldest. Pastes are
12/// not counted, since each one backs a link the user has handed out.
13pub const MAX_BACKUPS_PER_USER_AND_EXERCISE: i64 = 20;
14
15/// Pastes a user may keep of one exercise. Old pastes are never deleted to make room, since each
16/// backs a link the user has handed out, so further pastes are refused instead.
17pub const MAX_PASTES_PER_USER_AND_EXERCISE: i64 = 200;
18
19/// Pastes a user may store of one exercise per minute. Backups need no limit of their own: each
20/// one past [`MAX_BACKUPS_PER_USER_AND_EXERCISE`] deletes the oldest, so their storage stays
21/// bounded.
22const PASTES_PER_MINUTE: u64 = 3;
23/// Pastes a user may store of one exercise per hour.
24const PASTES_PER_HOUR: u64 = 20;
25
26static PASTE_LIMITS: LazyLock<EndpointLimiters> = LazyLock::new(|| {
27    EndpointLimiters::from_config(&RateLimitConfig {
28        per_minute: Some(PASTES_PER_MINUTE),
29        per_hour: Some(PASTES_PER_HOUR),
30        ..Default::default()
31    })
32});
33
34/// The answer a snapshot stores. The caller has established which course exercise, slide and task
35/// it is for and that the user may view them.
36pub struct SnapshotAnswer {
37    pub user_id: Uuid,
38    pub course_id: Uuid,
39    pub exercise_id: Uuid,
40    pub exercise_slide_id: Uuid,
41    pub exercise_task_id: Uuid,
42    pub purpose: SnapshotPurpose,
43    pub data_json: Option<serde_json::Value>,
44    pub file_upload_ids: Vec<Uuid>,
45}
46
47/// Stores a snapshot of the named uploads, then trims the user's backups of the exercise to
48/// [`MAX_BACKUPS_PER_USER_AND_EXERCISE`]. A paste over the per-user, per-exercise limit is
49/// refused with 429.
50///
51/// Applies the same upload checks as a file submit, but nothing a submit's acceptance depends on:
52/// no deadline, try limit or chapter lock, because a snapshot is never graded.
53pub async fn store_snapshot(
54    conn: &mut PgConnection,
55    answer: SnapshotAnswer,
56) -> Result<ExerciseAnswerSnapshot, ControllerError> {
57    answer_uploads::verify_answer_names_uploads(&answer.file_upload_ids)?;
58
59    let mut tx = conn.begin().await?;
60    answer_uploads::lock_and_verify_uploads_are_usable(
61        &mut tx,
62        answer.exercise_id,
63        answer.user_id,
64        &answer.file_upload_ids,
65    )
66    .await?;
67    if answer.purpose == SnapshotPurpose::Paste
68        && models::exercise_answer_snapshots::count_pastes(
69            &mut tx,
70            answer.user_id,
71            answer.exercise_id,
72        )
73        .await?
74            >= MAX_PASTES_PER_USER_AND_EXERCISE
75    {
76        return Err(controller_err!(
77            TooManyRequests(None),
78            "You have stored the maximum number of pastes of this exercise.".to_string()
79        ));
80    }
81    // Last before the insert, so that a request refused for another reason uses up no quota.
82    if answer.purpose == SnapshotPurpose::Paste
83        && let Err(retry_after) =
84            PASTE_LIMITS.check_key(format!("{}:{}", answer.user_id, answer.exercise_id))
85    {
86        return Err(controller_err!(
87            TooManyRequests(Some(retry_after)),
88            "Too many pastes of this exercise; try again later.".to_string()
89        ));
90    }
91    let snapshot = models::exercise_answer_snapshots::insert(
92        &mut tx,
93        NewExerciseAnswerSnapshot {
94            user_id: answer.user_id,
95            course_id: answer.course_id,
96            exercise_id: answer.exercise_id,
97            exercise_slide_id: answer.exercise_slide_id,
98            exercise_task_id: answer.exercise_task_id,
99            purpose: answer.purpose,
100            data_json: answer.data_json,
101        },
102        &answer.file_upload_ids,
103    )
104    .await?;
105    if answer.purpose == SnapshotPurpose::Backup {
106        models::exercise_answer_snapshots::delete_backups_beyond(
107            &mut tx,
108            answer.user_id,
109            answer.exercise_id,
110            MAX_BACKUPS_PER_USER_AND_EXERCISE,
111        )
112        .await?;
113    }
114    tx.commit().await?;
115    Ok(snapshot)
116}
