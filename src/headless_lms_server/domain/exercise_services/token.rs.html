22/// Authenticated user extracted from a courses.mooc.fi OAuth 2.0 access token.
23///
24/// The client sends an opaque access token issued by this backend's own OAuth
25/// provider (via the device-authorization flow) as `Authorization: Bearer <token>`.
26/// The token is hashed to a digest, looked up in `oauth_access_tokens`, gated on
27/// the `exercise-services` scope, and mapped to the local user that owns it.
28///
29/// The exact error bodies and status codes are a contract with the tmc-langs client,
30/// which keys its error mapping on them:
31///  - a missing / invalid / expired / revoked token, a token whose client may
32///    not use Bearer tokens, a sender-constrained (DPoP) token, or a token whose
33///    user no longer exists all yield `401` with an `unauthorized` body;
34///  - a valid token that lacks the `exercise-services` scope yields `403` with a
35///    `forbidden` body.
36#[derive(Debug, Clone)]
37pub struct UserFromOAuthToken(User);

52/// Builds the `401 unauthorized` error the langs client expects for any rejected token.
53fn unauthorized(message: &str) -> ControllerError {
54    controller_err!(Unauthorized, message.to_string())
55}
56
57/// Builds the `403 forbidden` error the langs client expects for a token missing the scope.
58fn forbidden(message: &str) -> ControllerError {
59    controller_err!(Forbidden, message.to_string())
60}

62/// The response to a token `resolve_bearer_token` rejected.
63///
64/// A database failure is a `500`, never a `401`: langs turns a `401` into
65/// refresh-then-DELETE-credentials, so collapsing a transient DB blip into `401` would force-log-out
66/// every user for the duration of the outage.
67fn bearer_error(err: BearerTokenError) -> ControllerError {
68    match err {
69        BearerTokenError::Invalid(message) => unauthorized(message),
70        BearerTokenError::Database(err) => {
71            let source: anyhow::Error = err.into();
72            controller_err!(
73                InternalServerError,
74                "A database error occurred while validating the access token.".to_string(),
75                source
76            )
77        }
78    }
79}

81/// Resolve an opaque Bearer access token to the local user that owns it, requiring the
82/// `exercise-services` scope.
83///
84/// Pure DB work (no actix, no cache) so it can be unit-tested directly. Also returns the token's
85/// `expires_at`, so the caller can cap cache TTL to it.
86async fn resolve_oauth_user(
87    conn: &mut PgConnection,
88    token: &SecretString,
89    token_hmac_key: &SecretString,
90) -> Result<(User, DateTime<Utc>), ControllerError> {
91    let token = resolve_bearer_token(conn, token.expose_secret(), token_hmac_key)
92        .await
93        .map_err(bearer_error)?;
94    if !token
95        .scopes
96        .iter()
97        .any(|scope| scope == EXERCISE_SERVICES_SCOPE)
98    {
99        return Err(forbidden(
100            "The access token does not grant the required exercise-services scope.",
101        ));
102    }
103    Ok((token.user, token.expires_at))
104}

106impl FromRequest for UserFromOAuthToken {
107    type Error = ControllerError;
108    type Future = LocalBoxFuture<'static, Result<Self, ControllerError>>;
109
110    fn from_request(req: &HttpRequest, _payload: &mut actix_http::Payload) -> Self::Future {
111        let app_data = (|| -> Result<_, ControllerError> {
112            let pool = req
113                .app_data::<web::Data<PgPool>>()
114                .ok_or_else(|| {
115                    controller_err!(InternalServerError, "Missing database pool".to_string())
116                })?
117                .clone();
118            let app_conf = req
119                .app_data::<web::Data<ApplicationConfiguration>>()
120                .ok_or_else(|| {
121                    controller_err!(
122                        InternalServerError,
123                        "Missing application configuration".to_string()
124                    )
125                })?
126                .clone();
127            let cache = req
128                .app_data::<web::Data<Cache>>()
129                .ok_or_else(|| controller_err!(InternalServerError, "Missing cache".to_string()))?
130                .clone();
131            Ok((pool, app_conf, cache))
132        })();
133
134        let auth_header = req
135            .headers()
136            .get(header::AUTHORIZATION)
137            .and_then(|value| value.to_str().ok())
138            .and_then(bearer_token)
139            .map(|token| SecretString::new(token.into()));
140
141        async move {
142            let (pool, app_conf, cache) = app_data?;
143            let Some(token) = auth_header else {
144                return Err(unauthorized("Missing bearer token"));
145            };
146            let mut conn = pool.acquire().await?;
147
148            // In test/dev mode a small set of fixed tokens map straight to seeded users so
149            // tests can skip the device flow. Anything else falls through to the real
150            // OAuth-token path below.
151            if app_conf.test_mode {
152                warn!("Test mode is on: fixed test tokens map directly to seeded users.");
153                if let Some(user) =
154                    authentication::authenticate_test_token(&mut conn, &token, &app_conf)
155                        .await
156                        .map_err(as_controller_error(
157                            ControllerErrorType::Unauthorized,
158                            "Could not find user for test token".to_string(),
159                        ))?
160                {
161                    return Ok(Self(user));
162                }
163            }
164
165            let token_hmac_key = &app_conf.oauth_server_configuration.oauth_token_hmac_key;
166            let digest = token_digest_sha256(token.expose_secret(), token_hmac_key);
167
168            // A cache hit skips `resolve_oauth_user`, and with it the token-validity,
169            // bearer_allowed, scope and soft-delete checks, for the cache TTL — see
170            // `MAX_CACHE_TTL` for why that bound is the security-relevant number. The TTL is
171            // additionally clamped to the token's own remaining lifetime so a cached mapping
172            // never outlives the token.
173            let user = match load_user(&cache, &digest, token_hmac_key).await {
174                Some(user) => user,
175                None => {
176                    let (user, expires_at) =
177                        resolve_oauth_user(&mut conn, &token, token_hmac_key).await?;
178                    let ttl = cache_ttl_for_token(expires_at, Utc::now());
179                    cache_user(&cache, &digest, token_hmac_key, &user, ttl).await;
180                    user
181                }
182            };
183
184            Ok(Self(user))
185        }
186        .boxed_local()
187    }
188}

190/// Domain separator for the cache-key KDF. Changing it invalidates every cache entry.
191const CACHE_KEY_CONTEXT: &str = "headless-lms exercise-services token cache v1";
192
193/// Cache key for a token, derived from its `oauth_access_tokens.digest` rather than the token
194/// plaintext: bulk revocation only ever holds digests, so a plaintext-derived key could not
195/// be evicted. Keyed (not a bare hash) so a leaked Redis dump is inert and a guessed digest
196/// cannot be confirmed offline.
197fn digest_to_cache_key(digest: &Digest, token_hmac_key: &SecretString) -> String {
198    let subkey = blake3::derive_key(CACHE_KEY_CONTEXT, token_hmac_key.expose_secret().as_bytes());
199    format!(
200        "user:{}",
201        blake3::keyed_hash(&subkey, digest.as_slice()).to_hex()
202    )
203}
204
205/// Upper bound on cache TTL, independent of the token's own expiry.
206///
207/// A cache hit skips every authorization check in `resolve_oauth_user`, so this bounds how long a
208/// change goes unnoticed when nothing evicts for it. Revocation, refresh-family reuse, consent
209/// withdrawal and user deletion all evict; these do not:
210///  - a grant narrowed by a new consent, which keeps its tokens and only caps their scopes;
211///  - a hard `DELETE FROM users`/`oauth_clients`, whose `ON DELETE CASCADE` into
212///    `oauth_access_tokens` runs no Rust;
213///  - `oauth_clients` changes (`bearer_allowed = false`, soft delete, narrowed `scopes`), which
214///    have no Rust mutator to evict from.
215///
216/// For those this bound is the only guard, which is why it is minutes and not an hour.
217const MAX_CACHE_TTL: Duration = Duration::from_secs(15 * 60);
218
219/// `min(MAX_CACHE_TTL, expires_at - now)`, floored at zero — a flat TTL could otherwise
220/// outlive a short-lived token.
221fn cache_ttl_for_token(expires_at: DateTime<Utc>, now: DateTime<Utc>) -> Duration {
222    let until_expiry = (expires_at - now).to_std().unwrap_or(Duration::ZERO);
223    until_expiry.min(MAX_CACHE_TTL)
224}

226pub async fn cache_user(
227    cache: &Cache,
228    digest: &Digest,
229    token_hmac_key: &SecretString,
230    user: &User,
231    ttl: Duration,
232) {
233    cache
234        .cache_json(digest_to_cache_key(digest, token_hmac_key), user, ttl)
235        .await;
236}

238pub async fn load_user(
239    cache: &Cache,
240    digest: &Digest,
241    token_hmac_key: &SecretString,
242) -> Option<User> {
243    cache
244        .get_json(digest_to_cache_key(digest, token_hmac_key))
245        .await
246}

248/// Evicts a cached user mapping. Call this right after revoking a token so it can't
249/// keep authenticating from a stale cache hit for the rest of its TTL.
250///
251/// Entries written before this keying scheme shipped are unreachable and only age out
252/// within `MAX_CACHE_TTL`.
253pub(crate) async fn invalidate_cached_user(
254    cache: &Cache,
255    digest: &Digest,
256    token_hmac_key: &SecretString,
257) {
258    cache
259        .invalidate(digest_to_cache_key(digest, token_hmac_key))
260        .await;
261}
262
263/// Evicts every mapping a batch revocation invalidated — a refresh family, a withdrawn consent, a
264/// deleted user. Same contract as `invalidate_cached_user`, for the callers that revoke more than
265/// one token at a time.
266pub(crate) async fn invalidate_cached_users(
267    cache: &Cache,
268    digests: &[Digest],
269    token_hmac_key: &SecretString,
270) {
271    for digest in digests {
272        invalidate_cached_user(cache, digest, token_hmac_key).await;
273    }
274}

276/// Soft-deletes a user and evicts every cached mapping their access tokens had.
277///
278/// The only supported way to delete a user: `models::users::delete_user` returns the digests of the
279/// tokens it hard-deleted, and a caller that drops them leaves a banned account authenticating from
280/// a cache hit for the rest of `MAX_CACHE_TTL`. Both callers — self-service account deletion and the
281/// `sync_tmc_users` batch — go through here so neither can forget.
282pub async fn delete_user_and_invalidate_cached_tokens(
283    conn: &mut PgConnection,
284    cache: &Cache,
285    token_hmac_key: &SecretString,
286    user_id: uuid::Uuid,
287) -> Result<(), models::ModelError> {
288    let revoked = models::users::delete_user(conn, user_id).await?;
289    invalidate_cached_users(cache, &revoked, token_hmac_key).await;
290    Ok(())
291}

343    /// Insert an access token, and the grant it comes from, for the given user/client and return
344    /// its plaintext.
345    async fn insert_token(
346        conn: &mut PgConnection,
347        client: &OAuthClient,
348        user_id: uuid::Uuid,
349        scopes: &[String],
350        token_type: TokenType,
351        expires_at: chrono::DateTime<Utc>,
352    ) -> String {
353        let plaintext = generate_access_token();
354        let digest = token_digest_sha256(&plaintext, &hmac_key());
355        let dpop_jkt = match token_type {
356            TokenType::Bearer => None,
357            // A JWK SHA-256 thumbprint is 43 base64url chars (DB CHECK requires 43..=128).
358            TokenType::DPoP => Some("0123456789abcdefghijklmnopqrstuvwxyzABCDEFG"),
359        };
360        OAuthAccessToken::insert(
361            conn,
362            NewAccessTokenParams {
363                digest: &digest,
364                user_id: Some(user_id),
365                client_id: client.id,
366                scopes,
367                audience: None,
368                token_type,
369                dpop_jkt,
370                metadata: serde_json::Map::new(),
371                expires_at,
372            },
373        )
374        .await
375        .unwrap();
376        OAuthUserClientScopes::insert(conn, user_id, client.id, scopes)
377            .await
378            .unwrap();
379        plaintext
380    }

646    /// A token that authenticated once (populating the cache) must not keep
647    /// authenticating from that entry after it's revoked. Drives the underlying
648    /// functions directly rather than the actix extractor or `/revoke` handler,
649    /// since those need a `PgPool`/running `App` this crate's tests don't set up.
650    ///
651    /// Needs a real Redis; a no-op (not a failure) when `REDIS_URL` isn't set,
652    /// matching `headless_lms_utils::cache`'s own test convention.
653    #[actix_web::test]
654    async fn revoking_an_access_token_evicts_its_cached_user() {
655        let Some(cache) = connected_test_cache().await else {
656            return;
657        };
658
659        insert_data!(:tx, :user);
660        let client = insert_client(tx.as_mut(), &[EXERCISE_SERVICES_SCOPE.to_string()], true).await;
661        let plaintext = insert_token(
662            tx.as_mut(),
663            &client,
664            user,
665            &[EXERCISE_SERVICES_SCOPE.to_string()],
666            TokenType::Bearer,
667            Utc::now() + ChronoDuration::hours(1),
668        )
669        .await;
670        let token = secret(&plaintext);
671
672        // Populate the cache the way the extractor would on a first request.
673        let (resolved, expires_at) = resolve_oauth_user(tx.as_mut(), &token, &hmac_key())
674            .await
675            .expect("token should resolve before revocation");
676        let ttl = cache_ttl_for_token(expires_at, Utc::now());
677        let digest = token_digest_sha256(&plaintext, &hmac_key());
678        cache_user(&cache, &digest, &hmac_key(), &resolved, ttl).await;
679        assert!(
680            load_user(&cache, &digest, &hmac_key()).await.is_some(),
681            "user should be cached after the first successful resolution"
682        );
683
684        // Revoke it the way the /revoke controller's access-token path does:
685        // hard-delete the DB row, then evict the cache entry.
686        OAuthAccessToken::revoke_by_digest(
687            tx.as_mut(),
688            token_digest_sha256(&plaintext, &hmac_key()),
689        )
690        .await
691        .expect("revoke_by_digest should succeed");
692        invalidate_cached_user(&cache, &digest, &hmac_key()).await;
693
694        // The very next lookup must miss the cache (not just eventually,
695        // once the TTL naturally expires).
696        assert!(
697            load_user(&cache, &digest, &hmac_key()).await.is_none(),
698            "a revoked token's cache entry must not survive revocation"
699        );
700
701        // And re-resolving from the DB must now fail too, confirming this
702        // isn't passing only because the cache was never populated.
703        let err = resolve_oauth_user(tx.as_mut(), &token, &hmac_key())
704            .await
705            .expect_err("a revoked token must not resolve from the DB either");
706        assert_eq!(err.status_code(), StatusCode::UNAUTHORIZED);
707    }
