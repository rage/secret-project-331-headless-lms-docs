23/// Authenticated user extracted from a courses.mooc.fi OAuth 2.0 access token.
24///
25/// The client sends an opaque access token issued by this backend's own OAuth
26/// provider (via the device-authorization flow) as `Authorization: Bearer <token>`.
27/// The token is hashed to a digest, looked up in `oauth_access_tokens`, gated on
28/// the `exercise-services` scope, and mapped to the local user that owns it.
29///
30/// The exact error bodies and status codes are a contract with the tmc-langs client,
31/// which keys its error mapping on them:
32///  - a missing / invalid / expired / revoked token, a token whose client may
33///    not use Bearer tokens, a sender-constrained (DPoP) token, or a token whose
34///    user no longer exists all yield `401` with an `unauthorized` body;
35///  - a valid token that lacks the `exercise-services` scope yields `403` with a
36///    `forbidden` body.
37#[derive(Debug, Clone)]
38pub struct UserFromOAuthToken(User);

53/// Builds the `401 unauthorized` error the langs client expects for any rejected token.
54fn unauthorized(message: &str) -> ControllerError {
55    controller_err!(Unauthorized, message.to_string())
56}
57
58/// Builds the `403 insufficient_scope` error for a token missing the scope.
59fn forbidden(message: &str) -> ControllerError {
60    controller_err!(
61        ForbiddenWithReason(crate::domain::error::ForbiddenReason::InsufficientScope),
62        message.to_string()
63    )
64}

66/// The response to a token `resolve_bearer_token` rejected.
67///
68/// A database failure is a `500`, never a `401`: langs turns a `401` into
69/// refresh-then-DELETE-credentials, so collapsing a transient DB blip into `401` would force-log-out
70/// every user for the duration of the outage.
71fn bearer_error(err: BearerTokenError) -> ControllerError {
72    match err {
73        BearerTokenError::Invalid(message) => unauthorized(message),
74        BearerTokenError::Database(err) => {
75            let source: anyhow::Error = err.into();
76            controller_err!(
77                InternalServerError,
78                "A database error occurred while validating the access token.".to_string(),
79                source
80            )
81        }
82    }
83}

85/// Resolve an opaque Bearer access token to the local user that owns it, requiring the
86/// `exercise-services` scope, and record its grant as used.
87///
88/// Pure DB work (no actix, no cache) so it can be unit-tested directly. Also returns the token's
89/// `expires_at`, so the caller can cap cache TTL to it. A cache hit skips this, so the grant's
90/// `last_used_at` lags real use by up to the cache TTL.
91async fn resolve_oauth_user(
92    conn: &mut PgConnection,
93    token: &SecretString,
94    token_hmac_key: &SecretString,
95) -> Result<(User, DateTime<Utc>), ControllerError> {
96    let token = resolve_bearer_token(conn, token.expose_secret(), token_hmac_key)
97        .await
98        .map_err(bearer_error)?;
99    if !token
100        .scopes
101        .iter()
102        .any(|scope| scope == EXERCISE_SERVICES_SCOPE)
103    {
104        return Err(forbidden(
105            "The access token does not grant the required exercise-services scope.",
106        ));
107    }
108    // Only the account page reads it, so a failed write must not fail the request.
109    if let Err(err) =
110        OAuthUserClientScopes::mark_grant_used(conn, token.grant.id, token.grant.last_used_at).await
111    {
112        warn!(
113            "Could not record the use of OAuth grant {}: {err}",
114            token.grant.id
115        );
116    }
117    Ok((token.user, token.expires_at))
118}

120impl FromRequest for UserFromOAuthToken {
121    type Error = ControllerError;
122    type Future = LocalBoxFuture<'static, Result<Self, ControllerError>>;
123
124    fn from_request(req: &HttpRequest, _payload: &mut actix_http::Payload) -> Self::Future {
125        let app_data = (|| -> Result<_, ControllerError> {
126            let pool = req
127                .app_data::<web::Data<PgPool>>()
128                .ok_or_else(|| {
129                    controller_err!(InternalServerError, "Missing database pool".to_string())
130                })?
131                .clone();
132            let app_conf = req
133                .app_data::<web::Data<ApplicationConfiguration>>()
134                .ok_or_else(|| {
135                    controller_err!(
136                        InternalServerError,
137                        "Missing application configuration".to_string()
138                    )
139                })?
140                .clone();
141            let cache = req
142                .app_data::<web::Data<Cache>>()
143                .ok_or_else(|| controller_err!(InternalServerError, "Missing cache".to_string()))?
144                .clone();
145            Ok((pool, app_conf, cache))
146        })();
147
148        let auth_header = req
149            .headers()
150            .get(header::AUTHORIZATION)
151            .and_then(|value| value.to_str().ok())
152            .and_then(bearer_token)
153            .map(|token| SecretString::new(token.into()));
154
155        async move {
156            let (pool, app_conf, cache) = app_data?;
157            let Some(token) = auth_header else {
158                return Err(unauthorized("Missing bearer token"));
159            };
160            let mut conn = pool.acquire().await?;
161
162            // In test/dev mode a small set of fixed tokens map straight to seeded users so
163            // tests can skip the device flow. Anything else falls through to the real
164            // OAuth-token path below.
165            if app_conf.test_mode {
166                warn!("Test mode is on: fixed test tokens map directly to seeded users.");
167                if let Some(user) =
168                    authentication::authenticate_test_token(&mut conn, &token, &app_conf)
169                        .await
170                        .map_err(as_controller_error(
171                            ControllerErrorType::Unauthorized,
172                            "Could not find user for test token".to_string(),
173                        ))?
174                {
175                    return Ok(Self(user));
176                }
177            }
178
179            let token_hmac_key = &app_conf.oauth_server_configuration.oauth_token_hmac_key;
180            let digest = token_digest_sha256(token.expose_secret(), token_hmac_key);
181
182            // A cache hit skips `resolve_oauth_user`, and with it the token-validity,
183            // bearer_allowed, scope and soft-delete checks, for the cache TTL — see
184            // `MAX_CACHE_TTL` for why that bound is the security-relevant number. The TTL is
185            // additionally clamped to the token's own remaining lifetime so a cached mapping
186            // never outlives the token.
187            let user = match load_user(&cache, &digest, token_hmac_key).await {
188                Some(user) => user,
189                None => {
190                    let (user, expires_at) =
191                        resolve_oauth_user(&mut conn, &token, token_hmac_key).await?;
192                    let ttl = cache_ttl_for_token(expires_at, Utc::now());
193                    cache_user(&cache, &digest, token_hmac_key, &user, ttl).await;
194                    user
195                }
196            };
197
198            Ok(Self(user))
199        }
200        .boxed_local()
201    }
202}

204/// Domain separator for the cache-key KDF. Changing it invalidates every cache entry.
205const CACHE_KEY_CONTEXT: &str = "headless-lms exercise-services token cache v1";
206
207/// Cache key for a token, derived from its `oauth_access_tokens.digest` rather than the token
208/// plaintext: bulk revocation only ever holds digests, so a plaintext-derived key could not
209/// be evicted. Keyed (not a bare hash) so a leaked Redis dump is inert and a guessed digest
210/// cannot be confirmed offline.
211fn digest_to_cache_key(digest: &Digest, token_hmac_key: &SecretString) -> String {
212    let subkey = blake3::derive_key(CACHE_KEY_CONTEXT, token_hmac_key.expose_secret().as_bytes());
213    format!(
214        "user:{}",
215        blake3::keyed_hash(&subkey, digest.as_slice()).to_hex()
216    )
217}
218
219/// Upper bound on cache TTL, independent of the token's own expiry.
220///
221/// A cache hit skips every authorization check in `resolve_oauth_user`, so this bounds how long a
222/// change goes unnoticed when nothing evicts for it. Revocation, refresh-family reuse, consent
223/// withdrawal and user deletion all evict; these do not:
224///  - a grant narrowed by a new consent, which keeps its tokens and only caps their scopes;
225///  - a hard `DELETE FROM users`/`oauth_clients`, whose `ON DELETE CASCADE` into
226///    `oauth_access_tokens` runs no Rust;
227///  - `oauth_clients` changes (`bearer_allowed = false`, soft delete, narrowed `scopes`), which
228///    have no Rust mutator to evict from.
229///
230/// For those this bound is the only guard, which is why it is minutes and not an hour.
231const MAX_CACHE_TTL: Duration = Duration::from_secs(15 * 60);
232
233/// `min(MAX_CACHE_TTL, expires_at - now)`, floored at zero — a flat TTL could otherwise
234/// outlive a short-lived token.
235fn cache_ttl_for_token(expires_at: DateTime<Utc>, now: DateTime<Utc>) -> Duration {
236    let until_expiry = (expires_at - now).to_std().unwrap_or(Duration::ZERO);
237    until_expiry.min(MAX_CACHE_TTL)
238}

240pub async fn cache_user(
241    cache: &Cache,
242    digest: &Digest,
243    token_hmac_key: &SecretString,
244    user: &User,
245    ttl: Duration,
246) {
247    cache
248        .cache_json(digest_to_cache_key(digest, token_hmac_key), user, ttl)
249        .await;
250}
251
252pub async fn load_user(
253    cache: &Cache,
254    digest: &Digest,
255    token_hmac_key: &SecretString,
256) -> Option<User> {
257    cache
258        .get_json(digest_to_cache_key(digest, token_hmac_key))
259        .await
260}

262/// Evicts a cached user mapping. Call this right after revoking a token so it can't
263/// keep authenticating from a stale cache hit for the rest of its TTL.
264///
265/// Entries written before this keying scheme shipped are unreachable and only age out
266/// within `MAX_CACHE_TTL`.
267pub(crate) async fn invalidate_cached_user(
268    cache: &Cache,
269    digest: &Digest,
270    token_hmac_key: &SecretString,
271) {
272    cache
273        .invalidate(digest_to_cache_key(digest, token_hmac_key))
274        .await;
275}
276
277/// Evicts every mapping a batch revocation invalidated — a refresh family, a withdrawn consent, a
278/// deleted user. Same contract as `invalidate_cached_user`, for the callers that revoke more than
279/// one token at a time.
280pub(crate) async fn invalidate_cached_users(
281    cache: &Cache,
282    digests: &[Digest],
283    token_hmac_key: &SecretString,
284) {
285    for digest in digests {
286        invalidate_cached_user(cache, digest, token_hmac_key).await;
287    }
288}
289
290/// Soft-deletes a user and evicts every cached mapping their access tokens had.
291///
292/// The only supported way to delete a user: `models::users::delete_user` returns the digests of the
293/// tokens it hard-deleted, and a caller that drops them leaves a banned account authenticating from
294/// a cache hit for the rest of `MAX_CACHE_TTL`. Both callers — self-service account deletion and the
295/// `sync_tmc_users` batch — go through here so neither can forget.
296pub async fn delete_user_and_invalidate_cached_tokens(
297    conn: &mut PgConnection,
298    cache: &Cache,
299    token_hmac_key: &SecretString,
300    user_id: uuid::Uuid,
301) -> Result<(), models::ModelError> {
302    let revoked = models::users::delete_user(conn, user_id).await?;
303    invalidate_cached_users(cache, &revoked, token_hmac_key).await;
304    Ok(())
305}

356    /// Insert an access token, and the grant it comes from, for the given user/client and return
357    /// its plaintext.
358    async fn insert_token(
359        conn: &mut PgConnection,
360        client: &OAuthClient,
361        user_id: uuid::Uuid,
362        scopes: &[String],
363        token_type: TokenType,
364        expires_at: chrono::DateTime<Utc>,
365    ) -> String {
366        let plaintext = generate_access_token();
367        let digest = token_digest_sha256(&plaintext, &hmac_key());
368        let dpop_jkt = match token_type {
369            TokenType::Bearer => None,
370            // A JWK SHA-256 thumbprint is 43 base64url chars (DB CHECK requires 43..=128).
371            TokenType::DPoP => Some("0123456789abcdefghijklmnopqrstuvwxyzABCDEFG"),
372        };
373        OAuthAccessToken::insert(
374            conn,
375            NewAccessTokenParams {
376                digest: &digest,
377                user_id: Some(user_id),
378                client_id: client.id,
379                scopes,
380                audience: None,
381                token_type,
382                dpop_jkt,
383                metadata: serde_json::Map::new(),
384                expires_at,
385                refresh_token_family_id: None,
386            },
387        )
388        .await
389        .unwrap();
390        OAuthUserClientScopes::insert(conn, user_id, client.id, scopes)
391            .await
392            .unwrap();
393        plaintext
394    }

695    /// A token that authenticated once (populating the cache) must not keep
696    /// authenticating from that entry after it's revoked. Drives the underlying
697    /// functions directly rather than the actix extractor or `/revoke` handler,
698    /// since those need a `PgPool`/running `App` this crate's tests don't set up.
699    ///
700    /// Needs a real Redis; a no-op (not a failure) when `REDIS_URL` isn't set,
701    /// matching `headless_lms_utils::cache`'s own test convention.
702    #[actix_web::test]
703    async fn revoking_an_access_token_evicts_its_cached_user() {
704        let Some(cache) = connected_test_cache().await else {
705            return;
706        };
707
708        insert_data!(:tx, :user);
709        let client = insert_client(tx.as_mut(), &[EXERCISE_SERVICES_SCOPE.to_string()], true).await;
710        let plaintext = insert_token(
711            tx.as_mut(),
712            &client,
713            user,
714            &[EXERCISE_SERVICES_SCOPE.to_string()],
715            TokenType::Bearer,
716            Utc::now() + ChronoDuration::hours(1),
717        )
718        .await;
719        let token = secret(&plaintext);
720
721        // Populate the cache the way the extractor would on a first request.
722        let (resolved, expires_at) = resolve_oauth_user(tx.as_mut(), &token, &hmac_key())
723            .await
724            .expect("token should resolve before revocation");
725        let ttl = cache_ttl_for_token(expires_at, Utc::now());
726        let digest = token_digest_sha256(&plaintext, &hmac_key());
727        cache_user(&cache, &digest, &hmac_key(), &resolved, ttl).await;
728        assert!(
729            load_user(&cache, &digest, &hmac_key()).await.is_some(),
730            "user should be cached after the first successful resolution"
731        );
732
733        // Revoke it the way the /revoke controller's access-token path does:
734        // hard-delete the DB row, then evict the cache entry.
735        OAuthAccessToken::revoke_by_digest(
736            tx.as_mut(),
737            token_digest_sha256(&plaintext, &hmac_key()),
738        )
739        .await
740        .expect("revoke_by_digest should succeed");
741        invalidate_cached_user(&cache, &digest, &hmac_key()).await;
742
743        // The very next lookup must miss the cache (not just eventually,
744        // once the TTL naturally expires).
745        assert!(
746            load_user(&cache, &digest, &hmac_key()).await.is_none(),
747            "a revoked token's cache entry must not survive revocation"
748        );
749
750        // And re-resolving from the DB must now fail too, confirming this
751        // isn't passing only because the cache was never populated.
752        let err = resolve_oauth_user(tx.as_mut(), &token, &hmac_key())
753            .await
754            .expect_err("a revoked token must not resolve from the DB either");
755        assert_eq!(err.status_code(), StatusCode::UNAUTHORIZED);
756    }

821    /// path both the self-service account deletion and the `sync_tmc_users` batch take; the batch
822    /// used to drop the digests and rely on the TTL alone.
823    ///
824    /// Needs a real Redis; a no-op (not a failure) when `REDIS_URL` isn't set.
