1//! Recording the files a submission was made from when the submitter named none.
2//!
3//! A native client uploads its files first and names them in the submit, so the host knows them
4//! without reading the answer. An answer made in an exercise service's IFrame names nothing: its
5//! files are inside the service's own answer blob, which the host must not interpret. The service
6//! itself can, so the host asks it — over the `answer_files_endpoint_path` it declares — and stores
7//! what comes back as ordinary `file_uploads`. Both origins therefore end up in
8//! `exercise_task_submission_files`, which is the only thing `download_submission` reads.
9
10use crate::domain::models_requests;
11use crate::prelude::*;
12use base64::Engine;
13use headless_lms_utils::strings::generate_random_string;
14use std::path::Path;
15
16/// Object-store namespace for files the host extracted from a service's answer. Kept apart from
17/// the native client's uploads because nothing binds these to a pending submit — they are recorded
18/// against a submission from the moment they exist, so the client-upload reaper must not see them.
19const ANSWER_FILE_PATH_PREFIX: &str = "exercise-answer-files";
20
21/// Ceiling on what one answer may materialize, mirroring the native client's per-batch upload
22/// limit. Without it a service could turn one submit into an unbounded number of object writes.
23const MAX_ANSWER_FILES: usize = 10;
24const MAX_ANSWER_FILE_BYTES: usize = 100 * 1024 * 1024;
25
26/// Creates a submission of an answer that names no host-stored uploads, recording the files the
27/// answer consists of against it.
28///
29/// The counterpart of the exercise-services client API's submit, which records the uploads the
30/// client named instead. Both leave the same rows behind, which is what makes
31/// `GET submissions/{id}/download` return the same shape for either origin.
32pub async fn submit_recording_answer_files(
33    conn: &mut PgConnection,
34    user_id: Uuid,
35    exercise: models::exercises::Exercise,
36    submission: &models::library::grading::StudentExerciseSlideSubmission,
37    jwt_key: std::sync::Arc<crate::domain::models_requests::JwtKey>,
38    file_store: &dyn FileStore,
39) -> Result<models::library::grading::StudentExerciseSlideSubmissionResult, ControllerError> {
40    // Ahead of the transaction because it makes an HTTP hop per task, and the exercise service is
41    // free to take its time.
42    let mut materialized: Vec<MaterializedAnswerFiles> =
43        Vec::with_capacity(submission.exercise_task_submissions.len());
44    for task_submission in &submission.exercise_task_submissions {
45        let exercise_task =
46            models::exercise_tasks::get_exercise_task_by_id(conn, task_submission.exercise_task_id)
47                .await?;
48        let files = match materialize_answer_files(
49            conn,
50            &exercise_task,
51            &task_submission.data_json,
52            file_store,
53        )
54        .await
55        {
56            Ok(files) => files,
57            Err(error) => {
58                discard_materialized(file_store, &materialized).await;
59                return Err(error);
60            }
61        };
62        materialized.push(MaterializedAnswerFiles {
63            exercise_task_id: task_submission.exercise_task_id,
64            files,
65        });
66    }
67
68    // The submission and the record of which files it was made from must land together: a
69    // submission without that record is one `download_submission` can never serve.
70    let mut tx = conn.begin().await?;
71    let result = create_and_record(
72        &mut tx,
73        user_id,
74        exercise,
75        submission,
76        jwt_key,
77        &materialized,
78    )
79    .await;
80    let result = match result {
81        Ok(result) => result,
82        Err(error) => {
83            // A failed grading hop discards its own writes and records a rejected submission
84            // instead; committing is what keeps that audit row rather than rolling it back too.
85            if let Err(commit_error) = tx.commit().await {
86                error!("Failed to commit after a failed submission: {commit_error}");
87            }
88            discard_materialized(file_store, &materialized).await;
89            return Err(error);
90        }
91    };
92    tx.commit().await?;
93    Ok(result)
94}

96async fn create_and_record(
97    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
98    user_id: Uuid,
99    exercise: models::exercises::Exercise,
100    submission: &models::library::grading::StudentExerciseSlideSubmission,
101    jwt_key: std::sync::Arc<crate::domain::models_requests::JwtKey>,
102    materialized: &[MaterializedAnswerFiles],
103) -> Result<models::library::grading::StudentExerciseSlideSubmissionResult, ControllerError> {
104    let result = crate::domain::exercises::process_submission(
105        &mut *tx, user_id, exercise, submission, jwt_key,
106    )
107    .await?;
108    for entry in materialized {
109        if entry.files.is_empty() {
110            continue;
111        }
112        let created = result
113            .exercise_task_submission_results
114            .iter()
115            .find(|result| result.submission.exercise_task_id == entry.exercise_task_id)
116            .ok_or_else(|| {
117                controller_err!(
118                    InternalServerError,
119                    format!(
120                        "No submission was created for exercise task {}",
121                        entry.exercise_task_id
122                    )
123                )
124            })?;
125        record_answer_files(tx, created.submission.id, user_id, &entry.files).await?;
126    }
127    Ok(result)
128}

130async fn discard_materialized(
131    file_store: &dyn FileStore,
132    materialized: &[MaterializedAnswerFiles],
133) {
134    for entry in materialized {
135        discard_written_answer_files(file_store, &entry.files).await;
136    }
137}

139/// A file written to the object store but not yet recorded, so a caller that fails can delete it.
140pub struct WrittenAnswerFile {
141    pub path: String,
142    pub name: String,
143    pub mime: String,
144}
145
146/// The files of one task's answer, ready to be recorded once the submission row exists.
147pub struct MaterializedAnswerFiles {
148    pub exercise_task_id: Uuid,
149    pub files: Vec<WrittenAnswerFile>,
150}
151
152/// Asks the task's exercise service which files the answer consists of and writes them to the
153/// object store. Returns an empty list when the service declares no such endpoint, which is the
154/// one case in which an IFrame-made submission stays undownloadable.
155///
156/// Writes objects but no rows: the caller records the `file_uploads` inside the transaction that
157/// creates the submission, so a rollback cannot leave a file pointing at no submission. The
158/// returned paths are the caller's to delete if it does roll back.
159pub async fn materialize_answer_files(
160    conn: &mut PgConnection,
161    exercise_task: &models::exercise_tasks::ExerciseTask,
162    answer: &serde_json::Value,
163    file_store: &dyn FileStore,
164) -> Result<Vec<WrittenAnswerFile>, ControllerError> {
165    let exercise_service = models::exercise_services::get_exercise_service_by_exercise_type(
166        conn,
167        &exercise_task.exercise_type,
168    )
169    .await?;
170    let service_info =
171        models::exercise_service_info::get_service_info(conn, exercise_service.id).await?;
172    let Some(url) =
173        models::exercise_services::get_internal_answer_files_url(&exercise_service, &service_info)?
174    else {
175        return Ok(Vec::new());
176    };
177
178    let reported =
179        models_requests::post_answer_files_request(url, exercise_task.public_spec.as_ref(), answer)
180            .await?;
181    if reported.len() > MAX_ANSWER_FILES {
182        return Err(controller_err!(
183            InternalServerError,
184            format!(
185                "The exercise service '{}' reported {} files for one answer, more than the {MAX_ANSWER_FILES} allowed",
186                exercise_task.exercise_type,
187                reported.len()
188            )
189        ));
190    }
191
192    let mut written = Vec::with_capacity(reported.len());
193    let mut total_bytes = 0usize;
194    for file in reported {
195        let contents = base64::engine::general_purpose::STANDARD
196            .decode(&file.data)
197            .map_err(|error| {
198                controller_err!(
199                    InternalServerError,
200                    format!(
201                        "The exercise service '{}' reported file '{}' with data that is not base64",
202                        exercise_task.exercise_type, file.name
203                    ),
204                    anyhow::Error::from(error)
205                )
206            })?;
207        total_bytes += contents.len();
208        if total_bytes > MAX_ANSWER_FILE_BYTES {
209            return Err(controller_err!(
210                InternalServerError,
211                format!(
212                    "The exercise service '{}' reported more than {MAX_ANSWER_FILE_BYTES} bytes of files for one answer",
213                    exercise_task.exercise_type
214                )
215            ));
216        }
217        let mime = file
218            .mime
219            .filter(|mime| !mime.is_empty())
220            .unwrap_or_else(|| "application/octet-stream".to_string());
221        let path = format!("{ANSWER_FILE_PATH_PREFIX}/{}", generate_random_string(32));
222        file_store.upload(Path::new(&path), contents, &mime).await?;
223        written.push(WrittenAnswerFile {
224            path,
225            name: file.name,
226            mime,
227        });
228    }
229    Ok(written)
230}

232/// Records already-written answer files against a task submission, in the order the service
233/// reported them. Runs in the caller's transaction so the rows and the submission land together.
234pub async fn record_answer_files(
235    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
236    exercise_task_submission_id: Uuid,
237    uploader: Uuid,
238    files: &[WrittenAnswerFile],
239) -> Result<(), ControllerError> {
240    let mut file_upload_ids = Vec::with_capacity(files.len());
241    for file in files {
242        file_upload_ids.push(
243            models::file_uploads::insert(tx, &file.name, &file.path, &file.mime, Some(uploader))
244                .await?,
245        );
246    }
247    models::exercise_task_submission_files::insert_many(
248        tx,
249        exercise_task_submission_id,
250        &file_upload_ids,
251    )
252    .await?;
253    Ok(())
254}

256/// Deletes objects written for a submission that never landed. Reported, not propagated: the
257/// caller is already failing for a reason the user needs to see.
258pub async fn discard_written_answer_files(file_store: &dyn FileStore, files: &[WrittenAnswerFile]) {
259    for file in files {
260        if let Err(error) = file_store.delete(Path::new(&file.path)).await {
261            error!(
262                "Failed to delete the answer file {} of a submission that was not created: {error}",
263                file.path
264            );
265        }
266    }
267}
