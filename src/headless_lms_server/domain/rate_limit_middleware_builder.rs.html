38/// Whose requests share one quota.
39#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
40pub enum RateLimitKey {
41    #[default]
42    ClientIp,
43    /// The signed-in user, falling back to the client IP for anonymous requests. For endpoints
44    /// that act on the caller's own data, where users behind one NAT address must not share a quota.
45    User,
46}
47
48/// Request header whose value, when equal to `RATELIMIT_PROTECTION_SAFE_API_KEY`, skips a
49/// [`RateLimit::bypassable`] limiter. tmc-server sends it, and we send it to tmc-server.
50pub const RATELIMIT_PROTECTION_SAFE_API_KEY_HEADER: &str = "RATELIMIT-PROTECTION-SAFE-API-KEY";
51
52/// Which key, if any, lets a request skip the limiter.
53#[derive(Clone)]
54enum Bypass {
55    Off,
56    RuntimeConfigKey,
57    #[cfg(test)]
58    Key(SecretString),
59}
60
61type Key = String;
62type Limiter = RateLimiter<Key, DefaultKeyedStateStore<Key>, DefaultClock>;

64#[derive(Clone)]
65struct EndpointLimiters {
66    month: Option<Arc<Limiter>>,
67    day: Option<Arc<Limiter>>,
68    hour: Option<Arc<Limiter>>,
69    minute: Option<Arc<Limiter>>,
70    second: Option<Arc<Limiter>>,
71}
72
73impl EndpointLimiters {
74    fn from_config(cfg: &RateLimitConfig) -> Self {
75        Self {
76            month: cfg.per_month.and_then(|n| {
77                build_custom_period_limiter(n, Duration::from_secs(30 * 24 * 60 * 60))
78            }),
79            day: cfg
80                .per_day
81                .and_then(|n| build_custom_period_limiter(n, Duration::from_secs(24 * 60 * 60))),
82            hour: cfg.per_hour.and_then(|n| build_limiter(n, Quota::per_hour)),
83            minute: cfg
84                .per_minute
85                .and_then(|n| build_limiter(n, Quota::per_minute)),
86            second: cfg
87                .per_second
88                .and_then(|n| build_limiter(n, Quota::per_second)),
89        }
90    }
91
92    fn iter(&self) -> impl Iterator<Item = &Arc<Limiter>> {
93        self.month
94            .iter()
95            .chain(self.day.iter())
96            .chain(self.hour.iter())
97            .chain(self.minute.iter())
98            .chain(self.second.iter())
99    }
100
101    fn is_empty(&self) -> bool {
102        self.second.is_none()
103            && self.minute.is_none()
104            && self.hour.is_none()
105            && self.day.is_none()
106            && self.month.is_none()
107    }
108}

110fn build_limiter<F>(n: u64, quota_fn: F) -> Option<Arc<Limiter>>
111where
112    F: FnOnce(NonZeroU32) -> Quota,
113{
114    let n32 = NonZeroU32::new(u32::try_from(n).ok()?)?;
115    Some(Arc::new(RateLimiter::keyed(quota_fn(n32))))
116}
117
118fn build_custom_period_limiter(n: u64, period: Duration) -> Option<Arc<Limiter>> {
119    let n32 = NonZeroU32::new(u32::try_from(n).ok()?)?;
120    let quota = Quota::with_period(period)?.allow_burst(n32);
121    Some(Arc::new(RateLimiter::keyed(quota)))
122}

132impl RateLimit {
133    /// Global `/api/v0` limits aligned with nginx ingress `limit-rps` and `limit-rpm`; relaxed when `TEST_MODE` is set.
134    pub fn global_api_rate_limit_config(test_mode: bool) -> RateLimitConfig {
135        if test_mode {
136            RateLimitConfig {
137                per_second: Some(10000),
138                per_minute: Some(200000),
139                ..Default::default()
140            }
141        } else {
142            RateLimitConfig {
143                per_second: Some(20),
144                per_minute: Some(1000),
145                per_hour: Some(10000),
146                ..Default::default()
147            }
148        }
149    }
150
151    pub fn new(cfg: RateLimitConfig) -> Self {
152        Self {
153            limiters: Arc::new(EndpointLimiters::from_config(&cfg)),
154            key: RateLimitKey::default(),
155            bypass: Bypass::Off,
156            calls: Arc::new(AtomicU64::new(0)),
157        }
158    }
159
160    /// Shares each quota by `key` instead of by client IP.
161    pub fn keyed_by(self, key: RateLimitKey) -> Self {
162        Self { key, ..self }
163    }
164
165    /// Lets requests carrying the configured `RATELIMIT_PROTECTION_SAFE_API_KEY` in
166    /// [`RATELIMIT_PROTECTION_SAFE_API_KEY_HEADER`] skip this limiter, for trusted servers that
167    /// call on behalf of many users from one IP. Never use it on brute-force limiters.
168    pub fn bypassable(self) -> Self {
169        Self {
170            bypass: Bypass::RuntimeConfigKey,
171            ..self
172        }
173    }
174
175    #[cfg(test)]
176    fn bypassable_with(self, safe_api_key: SecretString) -> Self {
177        Self {
178            bypass: Bypass::Key(safe_api_key),
179            ..self
180        }
181    }
182}

214impl<S> RateLimitInner<S> {
215    fn carries_bypass_key(&self, req: &ServiceRequest) -> bool {
216        let Some(provided) = req
217            .headers()
218            .get(RATELIMIT_PROTECTION_SAFE_API_KEY_HEADER)
219            .and_then(|value| value.to_str().ok())
220        else {
221            return false;
222        };
223        let expected = match &self.bypass {
224            Bypass::Off => return false,
225            Bypass::RuntimeConfigKey => server_runtime_config()
226                .ratelimit_protection_safe_api_key
227                .expose_secret(),
228            #[cfg(test)]
229            Bypass::Key(key) => key.expose_secret(),
230        };
231        !expected.is_empty() && constant_time_eq_str(provided, expected)
232    }
233}
234
235impl<S, B> Service<ServiceRequest> for RateLimitInner<S>
236where
237    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
238    B: MessageBody + 'static,
239{
240    type Response = ServiceResponse<EitherBody<B>>;
241    type Error = Error;
242    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;
243
244    fn poll_ready(&self, cx: &mut Context<'_>) -> Poll<Result<(), Self::Error>> {
245        self.service.poll_ready(cx)
246    }
247
248    fn call(&self, req: ServiceRequest) -> Self::Future {
249        if self.limiters.is_empty() || self.carries_bypass_key(&req) {
250            let fut = self.service.call(req);
251            return Box::pin(async move { fut.await.map(|r| r.map_into_left_body()) });
252        }
253
254        const RETAIN_EVERY: u64 = 1024;
255        const SHRINK_EVERY: u64 = 65_536;
256
257        let n = self.calls.fetch_add(1, Ordering::Relaxed) + 1;
258        if n.is_multiple_of(RETAIN_EVERY) {
259            for limiter in self.limiters.iter() {
260                limiter.retain_recent();
261            }
262            if n.is_multiple_of(SHRINK_EVERY) {
263                for limiter in self.limiters.iter() {
264                    limiter.shrink_to_fit();
265                }
266            }
267        }
268
269        let clock = DefaultClock::default();
270        let key = match self.key {
271            RateLimitKey::User => session_user_id(&req.get_session())
272                .map(|id| format!("user:{id}"))
273                .unwrap_or_else(|| extract_client_ip_key(&req)),
274            RateLimitKey::ClientIp => extract_client_ip_key(&req),
275        };
276
277        let mut retry_after: Option<Duration> = None;
278        for limiter in self.limiters.iter() {
279            if let Err(negative) = limiter.check_key(&key) {
280                let wait = negative.wait_time_from(clock.now());
281                retry_after = Some(retry_after.map_or(wait, |cur| cur.max(wait)));
282            }
283        }
284
285        if let Some(wait) = retry_after {
286            let secs = wait.as_secs().max(1);
287            let resp = HttpResponse::build(StatusCode::TOO_MANY_REQUESTS)
288                .insert_header((header::RETRY_AFTER, secs.to_string()))
289                .content_type("application/json")
290                .body(r#"{"type":"rate_limit","message_key":"rate_limited","message":"Too many requests. Please try again later."}"#.to_string());
291            return Box::pin(async move { Ok(req.into_response(resp).map_into_right_body()) });
292        }
293
294        let fut = self.service.call(req);
295        Box::pin(async move { fut.await.map(|r| r.map_into_left_body()) })
296    }
297}
298
299fn extract_client_ip_key(req: &ServiceRequest) -> String {
300    if let Some(s) = req.connection_info().realip_remote_addr() {
301        let s = s.trim();
302        if !s.is_empty() {
303            return s.to_string();
304        }
305    }
306
307    if let Some(sa) = req.peer_addr() {
308        return sa.ip().to_string();
309    }
310
311    format!("unknown:{}|{}", req.connection_info().host(), req.path())
312}
