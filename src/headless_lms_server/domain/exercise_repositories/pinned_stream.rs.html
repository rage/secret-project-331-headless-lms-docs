1//! The socket stream libgit2 opens every repository-clone connection through.
2//!
3//! Registered as libgit2's standard stream, it carries plain http, https (libgit2's TLS stream
4//! wraps it and still verifies the certificate) and ssh. It resolves the host once, refuses unless
5//! every address is public, and connects only to those addresses, so a DNS answer that changes
6//! between validating the name and connecting to it cannot reach a private address.
7

9    ffi::{CStr, CString, c_char, c_int, c_uint, c_void},
10    io::{self, Read, Write},
11    net::{SocketAddr, TcpStream, ToSocketAddrs},
12    os::fd::AsRawFd,
13    panic::AssertUnwindSafe,
14    sync::OnceLock,
15    time::Duration,
16};

20/// `git_stream` from `include/git2/sys/stream.h`.
21#[repr(C)]
22struct GitStream {
23    version: c_int,
24    /// The `encrypted:1, proxy_support:1` bitfield; both are 0 for a plain socket.
25    flags: c_uint,
26    timeout: c_int,
27    connect_timeout: c_int,
28    connect: unsafe extern "C" fn(*mut GitStream) -> c_int,
29    certificate: unsafe extern "C" fn(*mut *mut c_void, *mut GitStream) -> c_int,
30    set_proxy: unsafe extern "C" fn(*mut GitStream, *const c_void) -> c_int,
31    read: unsafe extern "C" fn(*mut GitStream, *mut c_void, usize) -> isize,
32    write: unsafe extern "C" fn(*mut GitStream, *const c_char, usize, c_int) -> isize,
33    close: unsafe extern "C" fn(*mut GitStream) -> c_int,
34    free: unsafe extern "C" fn(*mut GitStream),
35}

37/// `git_stream_registration` from `include/git2/sys/stream.h`.
38#[repr(C)]
39struct GitStreamRegistration {
40    version: c_int,
41    init: Option<unsafe extern "C" fn(*mut *mut GitStream, *const c_char, *const c_char) -> c_int>,
42    wrap: Option<unsafe extern "C" fn(*mut *mut GitStream, *mut GitStream, *const c_char) -> c_int>,
43}

45/// The libgit2 release whose `git_stream` and `git_socket_stream` layouts the structs here mirror.
46/// Any release may change them, so [`register`] refuses every other one.
47const MIRRORED_LIBGIT2_RELEASE: (u32, u32) = (1, 9);
48
49const GIT_STREAM_VERSION: c_int = 1;
50const GIT_STREAM_STANDARD: c_int = 1;
51
52unsafe extern "C" {
53    fn git_stream_register(kind: c_int, registration: *mut GitStreamRegistration) -> c_int;
54}

56/// libgit2's ssh transport casts the stream to its own `git_socket_stream` and hands `socket` to
57/// libssh2, so the first four fields must match that struct exactly.
58#[repr(C)]
59struct PinnedStream {
60    parent: GitStream,
61    host: *const c_char,
62    port: *const c_char,
63    socket: c_int,
64    host_name: CString,
65    port_name: CString,
66    tcp: Option<TcpStream>,
67}

69/// Makes every connection libgit2 opens in this process go through [`PinnedStream`]. Must run
70/// before any clone; later calls are no-ops that return the first call's result.
71pub(super) fn register() -> anyhow::Result<()> {
72    static REGISTERED: OnceLock<Result<(), String>> = OnceLock::new();
73    REGISTERED
74        .get_or_init(|| {
75            libgit2_sys::init();
76            let version = git2::Version::get().libgit2_version();
77            if (version.0, version.1) != MIRRORED_LIBGIT2_RELEASE {
78                return Err(format!(
79                    "the stream structs mirror libgit2 {}.{}, but {}.{}.{} is linked",
80                    MIRRORED_LIBGIT2_RELEASE.0,
81                    MIRRORED_LIBGIT2_RELEASE.1,
82                    version.0,
83                    version.1,
84                    version.2
85                ));
86            }
87            let mut registration = GitStreamRegistration {
88                version: GIT_STREAM_VERSION,
89                init: Some(init),
90                wrap: None,
91            };
92            // SAFETY: libgit2 is initialized and copies the registration before returning.
93            match unsafe { git_stream_register(GIT_STREAM_STANDARD, &mut registration) } {
94                0 => Ok(()),
95                _ => Err(git2::Error::last_error(-1).message().to_string()),
96            }
97        })
98        .clone()
99        .map_err(|message| anyhow::anyhow!("Could not register the git socket stream: {message}"))
100}

102/// Returns `resolved` if it is non-empty and every address in it is public.
103pub(super) fn public_addresses(
104    host: &str,
105    resolved: Vec<SocketAddr>,
106) -> io::Result<Vec<SocketAddr>> {
107    if resolved.is_empty() || resolved.iter().any(|a| !is_public_address(a.ip())) {
108        return Err(io::Error::new(
109            io::ErrorKind::PermissionDenied,
110            format!("{host} is not a public address"),
111        ));
112    }
113    Ok(resolved)
114}

116/// Resolves `host` exactly once and hands only the checked addresses to `connect`.
117fn connect_to_public_addresses<T>(
118    host: &str,
119    port: u16,
120    resolve: impl FnOnce(&str, u16) -> io::Result<Vec<SocketAddr>>,
121    connect: impl FnOnce(&[SocketAddr]) -> io::Result<T>,
122) -> io::Result<T> {
123    let addresses = public_addresses(host, resolve(host, port)?)?;
124    connect(&addresses)
125}

127/// The resolver clones use; tests substitute their own.
128fn resolve_with_system_resolver(host: &str, port: u16) -> io::Result<Vec<SocketAddr>> {
129    Ok((host, port).to_socket_addrs()?.collect())
130}

132/// Tries the addresses in order and returns the first connection, or the last attempt's error.
133fn connect_to_first_reachable(
134    addresses: &[SocketAddr],
135    connect_timeout: Option<Duration>,
136) -> io::Result<TcpStream> {
137    let mut last_error = io::Error::new(io::ErrorKind::NotFound, "no addresses to connect to");
138    for address in addresses {
139        let attempt = match connect_timeout {
140            Some(timeout) => TcpStream::connect_timeout(address, timeout),
141            None => TcpStream::connect(address),
142        };
143        match attempt {
144            Ok(tcp) => return Ok(tcp),
145            Err(err) => last_error = err,
146        }
147    }
148    Err(last_error)
149}

151/// A libgit2 millisecond setting as a socket timeout; zero or negative means none.
152fn positive_millis(millis: c_int) -> Option<Duration> {
153    u64::try_from(millis)
154        .ok()
155        .filter(|&millis| millis > 0)
156        .map(Duration::from_millis)
157}

159/// Sets the message libgit2 reports for a callback that returns -1.
160fn set_error(message: &str) {
161    let message = CString::new(message.replace('\0', "")).unwrap_or_default();
162    let class = libgit2_sys::GIT_ERROR_NET as c_int;
163    // SAFETY: libgit2 copies the message.
164    unsafe { libgit2_sys::git_error_set_str(class, message.as_ptr()) };
165}

167/// Panics must not unwind into C.
168fn guarded<T>(failure: T, f: impl FnOnce() -> T) -> T {
169    std::panic::catch_unwind(AssertUnwindSafe(f)).unwrap_or_else(|_| {
170        set_error("the git socket stream panicked");
171        failure
172    })
173}

175/// # Safety
176/// `stream` must be a stream created by [`init`] and not yet freed.
177unsafe fn pinned<'a>(stream: *mut GitStream) -> &'a mut PinnedStream {
178    // SAFETY: `parent` is the first field of the repr(C) PinnedStream that `init` allocated.
179    unsafe { &mut *stream.cast::<PinnedStream>() }
180}

182/// libgit2's stream constructor. Only records the host and port; [`connect`] opens the socket.
183unsafe extern "C" fn init(
184    out: *mut *mut GitStream,
185    host: *const c_char,
186    port: *const c_char,
187) -> c_int {
188    guarded(-1, || {
189        // SAFETY: libgit2 passes NUL-terminated host and port strings.
190        let (host_name, port_name) = unsafe {
191            (
192                CStr::from_ptr(host).to_owned(),
193                CStr::from_ptr(port).to_owned(),
194            )
195        };
196        // SAFETY: only reads libgit2's global settings, and libgit2 is initialized by now.
197        let (timeout, connect_timeout) = unsafe {
198            (
199                git2::opts::get_server_timeout_in_milliseconds().unwrap_or(0),
200                git2::opts::get_server_connect_timeout_in_milliseconds().unwrap_or(0),
201            )
202        };
203        let stream = Box::new(PinnedStream {
204            parent: GitStream {
205                version: GIT_STREAM_VERSION,
206                flags: 0,
207                timeout,
208                connect_timeout,
209                connect,
210                certificate,
211                set_proxy,
212                read,
213                write,
214                close,
215                free,
216            },
217            host: host_name.as_ptr(),
218            port: port_name.as_ptr(),
219            socket: -1,
220            host_name,
221            port_name,
222            tcp: None,
223        });
224        // SAFETY: libgit2 passes a valid out pointer and later frees the stream through `free`.
225        unsafe { *out = Box::into_raw(stream).cast() };
226        0
227    })
228}

230/// Connects to the first reachable public address of the host, never to a private one.
231unsafe extern "C" fn connect(stream: *mut GitStream) -> c_int {
232    // SAFETY: libgit2 only calls back with streams `init` created.
233    let stream = unsafe { pinned(stream) };
234    guarded(-1, || {
235        let host = stream.host_name.to_string_lossy().into_owned();
236        let Ok(port) = stream.port_name.to_string_lossy().parse::<u16>() else {
237            set_error(&format!("invalid port for {host}"));
238            return -1;
239        };
240        let connect_timeout = positive_millis(stream.parent.connect_timeout);
241        let connected =
242            connect_to_public_addresses(&host, port, resolve_with_system_resolver, |addresses| {
243                connect_to_first_reachable(addresses, connect_timeout)
244            })
245            .and_then(|tcp| {
246                let timeout = positive_millis(stream.parent.timeout);
247                tcp.set_read_timeout(timeout)?;
248                tcp.set_write_timeout(timeout)?;
249                // libgit2's own socket stream does the same; without it, a read from a peer that
250                // vanished never returns, as no server timeout is configured.
251                socket2::SockRef::from(&tcp).set_tcp_keepalive(
252                    &socket2::TcpKeepalive::new()
253                        .with_time(Duration::from_secs(60))
254                        .with_interval(Duration::from_secs(10))
255                        .with_retries(3),
256                )?;
257                Ok(tcp)
258            });
259        match connected {
260            Ok(tcp) => {
261                stream.socket = tcp.as_raw_fd();
262                stream.tcp = Some(tcp);
263                0
264            }
265            Err(err) => {
266                set_error(&format!("failed to connect to {host}: {err}"));
267                -1
268            }
269        }
270    })
271}

285/// Reads up to `len` bytes into libgit2's buffer, retrying on interruption.
286unsafe extern "C" fn read(stream: *mut GitStream, data: *mut c_void, len: usize) -> isize {
287    // SAFETY: libgit2 only calls back with streams `init` created.
288    let stream = unsafe { pinned(stream) };
289    guarded(-1, || {
290        let Some(tcp) = stream.tcp.as_mut() else {
291            set_error("read from an unconnected socket");
292            return -1;
293        };
294        // SAFETY: libgit2 passes a writable buffer of `len` bytes.
295        let buffer = unsafe { std::slice::from_raw_parts_mut(data.cast::<u8>(), len) };
296        loop {
297            match tcp.read(buffer) {
298                Ok(read) => return read as isize,
299                Err(err) if err.kind() == io::ErrorKind::Interrupted => continue,
300                Err(err) => {
301                    set_error(&format!("error reading from socket: {err}"));
302                    return -1;
303                }
304            }
305        }
306    })
307}

309/// Writes up to `len` bytes from libgit2's buffer, retrying on interruption.
310unsafe extern "C" fn write(
311    stream: *mut GitStream,
312    data: *const c_char,
313    len: usize,
314    _flags: c_int,
315) -> isize {
316    // SAFETY: libgit2 only calls back with streams `init` created.
317    let stream = unsafe { pinned(stream) };
318    guarded(-1, || {
319        let Some(tcp) = stream.tcp.as_mut() else {
320            set_error("write to an unconnected socket");
321            return -1;
322        };
323        // SAFETY: libgit2 passes a readable buffer of `len` bytes.
324        let buffer = unsafe { std::slice::from_raw_parts(data.cast::<u8>(), len) };
325        loop {
326            match tcp.write(buffer) {
327                Ok(written) => return written as isize,
328                Err(err) if err.kind() == io::ErrorKind::Interrupted => continue,
329                Err(err) => {
330                    set_error(&format!("error writing to socket: {err}"));
331                    return -1;
332                }
333            }
334        }
335    })
336}

338/// Closes the socket; the stream itself lives until [`free`].
339unsafe extern "C" fn close(stream: *mut GitStream) -> c_int {
340    // SAFETY: libgit2 only calls back with streams `init` created.
341    let stream = unsafe { pinned(stream) };
342    stream.tcp = None;
343    stream.socket = -1;
344    0
345}
346
347/// Releases the stream [`init`] allocated.
348unsafe extern "C" fn free(stream: *mut GitStream) {
349    // SAFETY: `init` allocated the stream with Box::into_raw and libgit2 frees it exactly once.
350    drop(unsafe { Box::from_raw(stream.cast::<PinnedStream>()) });
351}

371    #[test]
372    fn only_all_public_resolutions_are_accepted() {
373        for refused in [
374            vec!["1.1.1.1", "127.0.0.1"],
375            vec!["10.0.0.1"],
376            vec!["::ffff:127.0.0.1"],
377            vec!["169.254.169.254"],
378            vec![],
379        ] {
380            assert!(
381                public_addresses("example.com", addresses(&refused)).is_err(),
382                "{refused:?}"
383            );
384        }
385        let public = addresses(&["1.1.1.1", "2606:4700:4700::1111"]);
386        assert_eq!(
387            public_addresses("example.com", public.clone()).unwrap(),
388            public
389        );
390    }

392    #[test]
393    fn connects_only_to_the_addresses_it_checked() {
394        let resolutions = RefCell::new(0);
395        let rebinding_resolver = |_: &str, port: u16| {
396            *resolutions.borrow_mut() += 1;
397            let ip = if *resolutions.borrow() == 1 {
398                "1.1.1.1"
399            } else {
400                "127.0.0.1"
401            };
402            Ok(vec![SocketAddr::new(ip.parse().unwrap(), port)])
403        };
404        let mut connected_to = vec![];
405
406        connect_to_public_addresses("rebinding.example", 443, rebinding_resolver, |addresses| {
407            connected_to.extend_from_slice(addresses);
408            Ok(())
409        })
410        .unwrap();
411
412        assert_eq!(*resolutions.borrow(), 1);
413        assert_eq!(connected_to, addresses(&["1.1.1.1"]));
414    }

416    #[test]
417    fn libgit2_clones_cannot_reach_a_loopback_listener() {
418        let listener = TcpListener::bind("127.0.0.1:0").unwrap();
419        listener.set_nonblocking(true).unwrap();
420        let port = listener.local_addr().unwrap().port();
421        let clone_finished = Arc::new(AtomicBool::new(false));
422        let accepted = std::thread::spawn({
423            let clone_finished = Arc::clone(&clone_finished);
424            move || {
425                let mut accepted = 0;
426                while !clone_finished.load(Ordering::SeqCst) {
427                    match listener.accept() {
428                        // Dropping the connection makes an unpinned clone fail fast instead of
429                        // waiting for a response.
430                        Ok(_) => accepted += 1,
431                        Err(err) if err.kind() == io::ErrorKind::WouldBlock => {
432                            std::thread::sleep(Duration::from_millis(10))
433                        }
434                        Err(err) => panic!("{err}"),
435                    }
436                }
437                accepted
438            }
439        });
440        let target = tempfile::tempdir().unwrap();
441
442        register().unwrap();
443        let Err(err) = git2::build::RepoBuilder::new()
444            .clone(&format!("http://localhost:{port}/x.git"), target.path())
445        else {
446            panic!("cloning from a loopback address succeeded");
447        };
448        clone_finished.store(true, Ordering::SeqCst);
449
450        assert!(
451            err.message().contains("not a public address"),
452            "{}",
453            err.message()
454        );
455        assert_eq!(
456            accepted.join().unwrap(),
457            0,
458            "libgit2 connected to the listener"
459        );
460    }
