30/// Accepts a clone URL only over https or ssh (`ssh://` or `git@host:path`) to a host that
31/// currently resolves to public addresses, so a teacher gets a clear error when saving the URL.
32/// This is an early check, not the enforcement: the clone's own connections go through
33/// [`pinned_stream`], which refuses private addresses whatever DNS answers by then.
34pub async fn ensure_clone_url_is_allowed(url: &str) -> anyhow::Result<()> {
35    let (host, port) = clone_url_host_and_port(url)
36        .ok_or_else(|| anyhow::anyhow!("Use an https:// or ssh:// URL, or git@host:path."))?;
37    let addresses: Vec<_> = tokio::net::lookup_host((host.as_str(), port))
38        .await
39        .with_context(|| format!("Could not resolve {host}"))?
40        .collect();
41    pinned_stream::public_addresses(&host, addresses)?;
42    Ok(())
43}

68/// Why an ssh host key was accepted.
69#[derive(Debug, Clone, PartialEq, Eq)]
70enum AcceptedHostKey {
71    /// Pinned, or the key remembered from the first clone.
72    Known,
73    /// The first key seen from this server; it must be remembered once the clone succeeds.
74    FirstSeen(String),
75}
76
77/// Decides whether `seen`, the SHA-256 fingerprint of the host key a server presented, may be
78/// trusted: a pinned host must present a pinned key, any other host the key remembered from its
79/// first clone, or, if none is remembered yet, any key.
80fn check_ssh_host_key(
81    host: &str,
82    port: u16,
83    remembered: Option<&ExerciseRepositorySshHostKey>,
84    seen: Option<&str>,
85) -> Result<AcceptedHostKey, String> {
86    let Some(seen) = seen else {
87        return Err(format!("{host} sent no SHA-256 host key fingerprint"));
88    };
89    let mut pinned = PINNED_SSH_HOST_KEYS
90        .iter()
91        .filter(|(pinned_host, _)| pinned_host.eq_ignore_ascii_case(host))
92        .peekable();
93    if pinned.peek().is_some() {
94        return if pinned.any(|&(_, key)| key == seen) {
95            Ok(AcceptedHostKey::Known)
96        } else {
97            Err(format!(
98                "The ssh host key of {host} (SHA256:{seen}) is not one of its published keys, so the connection may have been intercepted."
99            ))
100        };
101    }
102    match remembered {
103        Some(remembered) if remembered.sha256_fingerprint == seen => Ok(AcceptedHostKey::Known),
104        Some(remembered) => Err(format!(
105            "The ssh host key of {host}:{port} has changed: it is now SHA256:{seen}, but SHA256:{} was seen on the first clone on {}. If the server's key was changed on purpose, ask an administrator to remove the remembered key for {host}:{port}; otherwise the connection may have been intercepted.",
106            remembered.sha256_fingerprint,
107            remembered.created_at.format("%Y-%m-%d")
108        )),
109        None => Ok(AcceptedHostKey::FirstSeen(seen.to_string())),
110    }
111}

113/// Clones `url` into `target`, verifying the server the way [`check_ssh_host_key`] describes and
114/// remembering a first-seen ssh host key once the clone succeeds. TLS certificates are left to
115/// libgit2's own validation.
116async fn clone_repository(
117    conn: &mut PgConnection,
118    url: &str,
119    deploy_key: Option<(Option<&str>, &SecretString)>,
120    target: &Path,
121) -> anyhow::Result<()> {
122    ensure_clone_url_is_allowed(url).await?;
123    pinned_stream::register()?;
124    let (host, port) = clone_url_host_and_port(url)
125        .ok_or_else(|| anyhow::anyhow!("Use an https:// or ssh:// URL, or git@host:path."))?;
126    let remembered = exercise_repository_ssh_host_keys::get(conn, &host, port).await?;
127    let first_seen = Cell::new(None);
128    // libgit2 replaces a rejecting callback's error with a generic one.
129    let rejection = Cell::new(None);
130
131    let mut remote_cbs = RemoteCallbacks::new();
132    // libgit2 uses this one callback for both TLS certificates and ssh host keys.
133    remote_cbs.certificate_check(|cert: &Cert<'_>, _| {
134        let Some(host_key) = cert.as_hostkey() else {
135            return Ok(CertificateCheckStatus::CertificatePassthrough);
136        };
137        let seen = host_key
138            .hash_sha256()
139            .map(|hash| STANDARD_NO_PAD.encode(hash));
140        match check_ssh_host_key(&host, port, remembered.as_ref(), seen.as_deref()) {
141            Ok(AcceptedHostKey::Known) => Ok(CertificateCheckStatus::CertificateOk),
142            Ok(AcceptedHostKey::FirstSeen(key)) => {
143                first_seen.set(Some(key));
144                Ok(CertificateCheckStatus::CertificateOk)
145            }
146            Err(message) => {
147                let error = git2::Error::from_str(&message);
148                rejection.set(Some(message));
149                Err(error)
150            }
151        }
152    });
153    if let Some((public_key, deploy_key)) = deploy_key {
154        remote_cbs.credentials(move |_, username, credential_type| {
155            if credential_type.is_ssh_memory() {
156                Cred::ssh_key_from_memory(
157                    username.unwrap_or("git"),
158                    public_key,
159                    // Exposed only here, where the key is handed to libgit2 in memory.
160                    deploy_key.expose_secret(),
161                    None,
162                )
163            } else {
164                Err(git2::Error::from_str(
165                    "The git server does not support the SSH_MEMORY credential type",
166                ))
167            }
168        });
169    }
170    let mut fetch_opts = FetchOptions::new();
171    // The clone must come from the URL the teacher saved, not from wherever that server redirects.
172    fetch_opts.follow_redirects(RemoteRedirect::None);
173    fetch_opts.remote_callbacks(remote_cbs);
174    if let Err(error) = RepoBuilder::new()
175        .fetch_options(fetch_opts)
176        .clone(url, target)
177    {
178        return Err(rejection
179            .take()
180            .map_or_else(|| error.into(), |message| anyhow::anyhow!(message)));
181    }
182
183    if let Some(seen) = first_seen.take() {
184        let remembered =
185            exercise_repository_ssh_host_keys::remember(conn, &host, port, &seen).await?;
186        if remembered.sha256_fingerprint != seen {
187            anyhow::bail!(
188                "Another clone from {host}:{port} saw a different ssh host key at the same time (SHA256:{} instead of SHA256:{seen}), so neither can be trusted. Ask an administrator to check the server's key.",
189                remembered.sha256_fingerprint
190            );
191        }
192    }
193    Ok(())
194}

196fn clone_url_host_and_port(url: &str) -> Option<(String, u16)> {
197    // libgit2 parses URLs with its own rules, so anything the two parsers may split differently
198    // could make it connect to a host other than the one checked here.
199    if url
200        .chars()
201        .any(|c| c == '\\' || c.is_ascii_control() || c.is_whitespace())
202    {
203        return None;
204    }
205    if let Some((_, rest)) = url.split_once("://") {
206        let authority = rest.split('/').next().unwrap_or_default();
207        if authority.contains(['?', '#']) || authority.matches('@').count() > 1 {
208            return None;
209        }
210    }
211    if url.starts_with("https://") || url.starts_with("ssh://") {
212        let parsed = url::Url::parse(url).ok()?;
213        let host = parsed.host_str()?.trim_matches(['[', ']']);
214        let port = parsed
215            .port()
216            .unwrap_or(if parsed.scheme() == "https" { 443 } else { 22 });
217        return Some((canonical_host(host)?, port));
218    }
219    // scp-like syntax: user@host:path, which git treats as ssh.
220    let (user_and_host, path) = url.split_once(':')?;
221    let (_, host) = user_and_host.split_once('@')?;
222    if host.is_empty() || path.is_empty() || host.contains('/') || path.starts_with("//") {
223        return None;
224    }
225    Some((canonical_host(host)?, 22))
226}

228/// One spelling per host, so that pins and remembered keys cannot be sidestepped by writing the
229/// same host as `GitHub.com` or with the root's trailing dot.
230fn canonical_host(host: &str) -> Option<String> {
231    let host = host.strip_suffix('.').unwrap_or(host);
232    (!host.is_empty()).then(|| host.to_ascii_lowercase())
233}

239/// Processes an exercise repository, creating a repository exercise for each exercise in it.
240/// Each exercise is compressed and uploaded to file storage.
241pub async fn process(
242    conn: &mut PgConnection,
243    repository_id: Uuid,
244    url: &str,
245    public_key: Option<&str>,
246    deploy_key: Option<&SecretString>,
247    file_store: &dyn FileStore,
248    app_conf: &ApplicationConfiguration,
249) -> anyhow::Result<Vec<StoredRepositoryExercise>> {
250    let mut stored_files = vec![];
251    match process_inner(InnerArgs {
252        conn,
253        repository_id,
254        url,
255        public_key,
256        deploy_key,
257        file_store,
258        stored_files: &mut stored_files,
259        app_conf,
260    })
261    .await
262    {
263        Ok(res) => {
264            exercise_repositories::mark_success(conn, repository_id).await?;
265            Ok(res)
266        }
267        Err(err) => {
268            if !stored_files.is_empty() {
269                warn!(
270                    "Failed while creating new exercise repository, cleaning files that were uploaded"
271                );
272                for file in stored_files {
273                    if let Err(err) = file_store.delete(&file).await {
274                        error!("Failed to clean file {}: {err}", file.display());
275                    }
276                }
277            }
278            exercise_repositories::mark_failure(conn, repository_id, &err.to_string()).await?;
279            Err(err)
280        }
281    }
282}

289    deploy_key: Option<&'a SecretString>,

295// implements the logic for process so that we can conveniently handle all errors in process
296async fn process_inner(
297    InnerArgs {
298        conn,
299        repository_id,
300        url,
301        public_key,
302        deploy_key,
303        file_store,
304        stored_files,
305        app_conf,
306    }: InnerArgs<'_>,
307) -> anyhow::Result<Vec<StoredRepositoryExercise>> {
308    // clone repo to temp dir
309    let temp = tempfile::tempdir()?;
310    info!("Cloning {url} to {:?}", temp.path());
311    // Outside the transaction: a first-seen host key must outlive a failed processing, and must not
312    // lock out other clones of the host while processing runs.
313    clone_repository(
314        conn,
315        url,
316        deploy_key.map(|deploy_key| (public_key, deploy_key)),
317        temp.path(),
318    )
319    .await?;
320    info!("Finished cloning {url} to {:?}", temp.path());
321    let mut tx = conn.begin().await?;
322
323    // create exercises in db and store them in file store
324    let found_exercises = find_exercise_directories(temp.path()).await?;
325    let mut repository_exercises = vec![];
326    // (part, name) => exercise
327    let existing_exercises =
328        repository_exercises::get_for_repository(&mut tx, repository_id).await?;
329
330    // we try both the path and the checksum to find existing exercises
331    // these are the only attributes found in the exercise repositories, so if both
332    // the path and checksum change there's no way to detect that it's supposed to be an updated old exercise
333    // rather than a new one
334    // this way we can accommodate both renaming/relocating exercises and changing them, though not at the same time...
335    let existing_exercises_path_map = existing_exercises
336        .iter()
337        .map(|ex| ((&ex.part, &ex.name), ex))
338        .collect::<HashMap<_, _>>();
339    let existing_exercises_checksum_map = existing_exercises
340        .iter()
341        .map(|ex| (ex.checksum.as_slice(), ex))
342        .collect::<HashMap<_, _>>();
343    for fe in &found_exercises {
344        // check if the exercise is new
345        match (
346            existing_exercises_path_map.get(&(&fe.part, &fe.name)),
347            existing_exercises_checksum_map.get(fe.checksum.as_bytes().as_slice()),
348        ) {
349            (Some(_), Some(_)) => {
350                // both the path and checksum are unchanged, no-op
351            }
352            (Some(existing_exercise_by_path), None) => {
353                // found exercise by path but the checksum has changed, exercise has been updated
354                let path = update_exercise(
355                    &mut tx,
356                    repository_id,
357                    existing_exercise_by_path.id,
358                    fe,
359                    file_store,
360                    app_conf,
361                )
362                .await?;
363                stored_files.push(path.clone());
364                let url = file_store.get_direct_download_url(&path).await?;
365                repository_exercises.push(StoredRepositoryExercise { url });
366                repository_exercises::update_checksum(
367                    &mut tx,
368                    existing_exercise_by_path.id,
369                    fe.checksum.as_bytes(),
370                )
371                .await?;
372                // todo: uploaded files get cleaned up on an error, which means that if the refreshing fails
373                // the exercise data will be missing entirely...
374            }
375            (None, Some(existing_exercise_by_checksum)) => {
376                // found exercise by checksum but the path has changed, update path
377                repository_exercises::update_part_and_name(
378                    &mut tx,
379                    existing_exercise_by_checksum.id,
380                    &fe.part,
381                    &fe.name,
382                )
383                .await?;
384            }
385            (None, None) => {
386                // new exercise
387                let new_exercise_id = uuid::Uuid::new_v4();
388                let path = create_and_upload_exercise(
389                    &mut tx,
390                    repository_id,
391                    new_exercise_id,
392                    fe,
393                    file_store,
394                    app_conf,
395                )
396                .await?;
397                stored_files.push(path.clone());
398                let url = file_store.get_direct_download_url(&path).await?;
399                repository_exercises.push(StoredRepositoryExercise { url });
400            }
401        }
402    }
403
404    tx.commit().await?;
405    Ok(repository_exercises)
406}

408/// Updates the given repository using the given url.
409/// Exercises with a known checksum but changed part or name are updated to reflect the new part or name.
410/// Exercises with a known part and name but changed checksum are updated in the file store and the checksum updated.
411/// Errors may leave some exercises updated and others not, since there's no mechanism for rolling back any file store updates.
412/// However, these inconsistencies will be fixed after a successful retry.
413pub async fn update(
414    conn: &mut PgConnection,
415    repository: Uuid,
416    url: &str,
417    file_store: &dyn FileStore,
418    app_conf: &ApplicationConfiguration,
419) -> anyhow::Result<()> {
420    let mut new_stored_files = vec![];
421    match update_inner(
422        conn,
423        repository,
424        url,
425        file_store,
426        &mut new_stored_files,
427        app_conf,
428    )
429    .await
430    {
431        Ok(res) => Ok(res),
432        Err(err) => {
433            if !new_stored_files.is_empty() {
434                debug!(
435                    "Failed while updating exercise repository, cleaning new exercises that were uploaded"
436                );
437                for file in new_stored_files {
438                    if let Err(err) = file_store.delete(&file).await {
439                        error!("Failed to clean file {}: {err}", file.display());
440                    }
441                }
442            }
443            Err(err)
444        }
445    }
446}

448async fn update_inner(
449    conn: &mut PgConnection,
450    repository: Uuid,
451    url: &str,
452    file_store: &dyn FileStore,
453    new_stored_files: &mut Vec<PathBuf>,
454    app_conf: &ApplicationConfiguration,
455) -> anyhow::Result<()> {
456    let temp = tempfile::tempdir()?;
457    clone_repository(conn, url, None, temp.path()).await?;
458    let mut tx = conn.begin().await?;
459
460    let repository_exercises = find_exercise_directories(temp.path()).await?;
461    let current_exercises = repository_exercises::get_for_repository(&mut tx, repository).await?;
462
463    let mut by_name = HashMap::new();
464    let mut by_checksum = HashMap::new();
465    for ex in &current_exercises {
466        by_name.insert((&ex.part, &ex.name), ex);
467        by_checksum.insert(ex.checksum.as_slice(), ex);
468    }
469    for ex in repository_exercises {
470        if let Some(&current) = by_name.get(&(&ex.part, &ex.name)) {
471            // found known exercise by part and name
472            if current.checksum != ex.checksum.as_bytes() {
473                // checksum changed, update files and checksum
474                create_and_upload_exercise(
475                    &mut tx, repository, current.id, &ex, file_store, app_conf,
476                )
477                .await?;
478                repository_exercises::update_checksum(&mut tx, current.id, ex.checksum.as_bytes())
479                    .await?;
480            }
481        } else if let Some(&current) = by_checksum.get(ex.checksum.as_bytes().as_slice()) {
482            // found known exercise by checksum
483            if current.part != ex.part || current.name != ex.name {
484                // part and/or name changed
485                repository_exercises::update_part_and_name(&mut tx, current.id, &ex.part, &ex.name)
486                    .await?;
487            }
488        } else {
489            // unknown part/name and checksum, assume new exercise
490            let path = create_and_upload_exercise(
491                &mut tx,
492                repository,
493                Uuid::new_v4(),
494                &ex,
495                file_store,
496                app_conf,
497            )
498            .await?;
499            new_stored_files.push(path);
500        }
501    }
502
503    tx.commit().await?;
504    Ok(())
505}

507/// Marks the exercises and repository as deleted and removes the associated files from the file store.
508/// Only returns the last error if there are multiple errors when trying to remove the files.
509pub async fn delete(
510    conn: &mut PgConnection,
511    repository_id: Uuid,
512    file_store: &dyn FileStore,
513) -> anyhow::Result<()> {
514    let mut tx = conn.begin().await?;
515
516    let mut latest_error = None;
517    let exercises = repository_exercises::delete_for_repository(&mut tx, repository_id).await?;
518    exercise_repositories::delete(&mut tx, repository_id).await?;
519    for exercise in exercises {
520        let path = file_store::repository_exercise_path(repository_id, exercise);
521        if let Err(err) = file_store.delete(&path).await {
522            error!(
523                "Failed to delete file while deleting repository {}: {err}",
524                path.display()
525            );
526            latest_error = Some(err);
527        }
528    }
529
530    match latest_error {
531        Some(latest_error) => Err(latest_error.into()),
532        _ => {
533            tx.commit().await?;
534            Ok(())
535        }
536    }
537}

539async fn create_and_upload_exercise(
540    conn: &mut PgConnection,
541    repository: Uuid,
542    exercise_id: Uuid,
543    exercise: &FoundExercise,
544    file_store: &dyn FileStore,
545    app_conf: &ApplicationConfiguration,
546) -> anyhow::Result<PathBuf> {
547    // archive and compress
548    let cursor = Cursor::new(vec![]);
549    let mut tar = tar::Builder::new(cursor);
550    // The repository is teacher-controlled; following its symlinks would archive server files.
551    tar.follow_symlinks(false);
552    tar.append_dir_all(".", &exercise.path)?;
553    let mut tar = tar.into_inner()?;
554    // rewind cursor back to the beginning
555    tar.set_position(0);
556    let tar_zstd = zstd::encode_all(tar, 0)?;
557
558    // upload
559    let path = file_store::repository_exercise_path(repository, exercise_id);
560    file_store
561        .upload(&path, tar_zstd, "application/zstd")
562        .await?;
563    let url = file_store.get_download_url(&path, app_conf);
564
565    // create
566    repository_exercises::new(
567        conn,
568        exercise_id,
569        repository,
570        &exercise.part,
571        &exercise.name,
572        exercise.checksum.as_bytes(),
573        &url,
574    )
575    .await?;
576    Ok(path)
577}

622    name: String,
623    checksum: Hash,
624    path: PathBuf,
625}
626
627async fn find_exercise_directories(clone_path: &Path) -> anyhow::Result<Vec<FoundExercise>> {
628    info!("finding exercise directories in {}", clone_path.display());
629
630    let mut exercises = vec![];
631    // exercises in repositories are in subdirs like
632    // part01/01_exercise
633    // part01/02_exercise
634    // part02/01_exercise
635    for entry in WalkDir::new(clone_path)
636        .min_depth(2)
637        .max_depth(2)
638        .into_iter()
639        .filter_entry(|e| {
640            e.file_type().is_dir()
641                && e.file_name() != "private"
642                && !is_hidden_dir(e)
643                && !contains_tmcignore(e)
644                && !is_in_git_dir(e.path())
645        })
646    {
647        let entry = entry?;
648        let checksum = folder_checksum::hash_folder(entry.path()).await?;
649
650        let path = entry.into_path().canonicalize()?;
651        let part = path
652            .parent()
653            .ok_or_else(|| anyhow::anyhow!("Path should be in a subdirectory: {}", path.display()))?
654            .file_name()
655            .ok_or_else(|| {
656                anyhow::anyhow!("The parent file name cannot be missing: {}", path.display())
657            })?
658            .to_str()
659            .context("Invalid directory name in repository")?
660            .to_string();
661        let name = path
662            .file_name()
663            .ok_or_else(|| anyhow::anyhow!("Path should be a directory: {}", path.display()))?
664            .to_str()
665            .context("Invalid directory name in repository")?
666            .to_string();
667        exercises.push(FoundExercise {
668            part,
669            name,
670            checksum,
671            path,
672        });
673    }
674    Ok(exercises)
675}

677// Filter for hidden directories (directories with names starting with '.')
678fn is_hidden_dir(entry: &DirEntry) -> bool {
679    let skip = entry.metadata().map(|e| e.is_dir()).unwrap_or_default()
680        && entry
681            .file_name()
682            .to_str()
683            .map(|s| s.starts_with('.'))
684            .unwrap_or_default();
685    if skip {
686        debug!("is hidden dir: {}", entry.path().display());
687    }
688    skip
689}
690
691// Filter for .git directory
692fn is_in_git_dir(path: &Path) -> bool {
693    let skip = path.parent().map(|p| p.ends_with(".git")).unwrap_or(false);
694    if skip {
695        debug!("is in git dir: {}", path.display());
696    }
697    skip
698}
699
700fn contains_tmcignore(entry: &DirEntry) -> bool {
701    for entry in WalkDir::new(entry.path())
702        .max_depth(1)
703        .into_iter()
704        .filter_map(|e| e.ok())
705    {
706        let is_file = entry.metadata().map(|e| e.is_file()).unwrap_or_default();
707        if is_file && entry.file_name() == ".tmcignore" {
708            debug!("contains .tmcignore: {}", entry.path().display());
709            return true;
710        }
711    }
712    false
713}
