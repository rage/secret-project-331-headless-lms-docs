22/// Accepts a clone URL only over https or ssh (`ssh://` or `git@host:path`) to a host that resolves
23/// to public addresses, so a teacher can't make the server reach local files or the cluster
24/// network.
25pub async fn ensure_clone_url_is_allowed(url: &str) -> anyhow::Result<()> {
26    let (host, port) = clone_url_host_and_port(url)
27        .ok_or_else(|| anyhow::anyhow!("Use an https:// or ssh:// URL, or git@host:path."))?;
28    let addresses: Vec<_> = tokio::net::lookup_host((host.as_str(), port))
29        .await
30        .with_context(|| format!("Could not resolve {host}"))?
31        .collect();
32    if addresses.is_empty()
33        || addresses.iter().any(|address| {
34            !crate::domain::oauth::client_metadata_document::is_public_address(address.ip())
35        })
36    {
37        anyhow::bail!("{host} is not a public address");
38    }
39    Ok(())
40}

42fn clone_url_host_and_port(url: &str) -> Option<(String, u16)> {
43    // libgit2 parses URLs with its own rules, so anything the two parsers may split differently
44    // could make it connect to a host other than the one checked here.
45    if url
46        .chars()
47        .any(|c| c == '\\' || c.is_ascii_control() || c.is_whitespace())
48    {
49        return None;
50    }
51    if let Some((_, rest)) = url.split_once("://") {
52        let authority = rest.split('/').next().unwrap_or_default();
53        if authority.contains(['?', '#']) || authority.matches('@').count() > 1 {
54            return None;
55        }
56    }
57    if url.starts_with("https://") || url.starts_with("ssh://") {
58        let parsed = url::Url::parse(url).ok()?;
59        let host = parsed.host_str()?.trim_matches(['[', ']']).to_string();
60        let port = parsed
61            .port()
62            .unwrap_or(if parsed.scheme() == "https" { 443 } else { 22 });
63        return Some((host, port));
64    }
65    // scp-like syntax: user@host:path, which git treats as ssh.
66    let (user_and_host, path) = url.split_once(':')?;
67    let (_, host) = user_and_host.split_once('@')?;
68    if host.is_empty() || path.is_empty() || host.contains('/') || path.starts_with("//") {
69        return None;
70    }
71    Some((host.to_string(), 22))
72}

78/// Processes an exercise repository, creating a repository exercise for each exercise in it.
79/// Each exercise is compressed and uploaded to file storage.
80pub async fn process(
81    conn: &mut PgConnection,
82    repository_id: Uuid,
83    url: &str,
84    public_key: Option<&str>,
85    deploy_key: Option<&SecretString>,
86    file_store: &dyn FileStore,
87    app_conf: &ApplicationConfiguration,
88) -> anyhow::Result<Vec<StoredRepositoryExercise>> {
89    let mut stored_files = vec![];
90    match process_inner(InnerArgs {
91        conn,
92        repository_id,
93        url,
94        public_key,
95        deploy_key,
96        file_store,
97        stored_files: &mut stored_files,
98        app_conf,
99    })
100    .await
101    {
102        Ok(res) => {
103            exercise_repositories::mark_success(conn, repository_id).await?;
104            Ok(res)
105        }
106        Err(err) => {
107            if !stored_files.is_empty() {
108                warn!(
109                    "Failed while creating new exercise repository, cleaning files that were uploaded"
110                );
111                for file in stored_files {
112                    if let Err(err) = file_store.delete(&file).await {
113                        error!("Failed to clean file {}: {err}", file.display());
114                    }
115                }
116            }
117            exercise_repositories::mark_failure(conn, repository_id, &err.to_string()).await?;
118            Err(err)
119        }
120    }
121}

134// implements the logic for process so that we can conveniently handle all errors in process
135async fn process_inner(
136    InnerArgs {
137        conn,
138        repository_id,
139        url,
140        public_key,
141        deploy_key,
142        file_store,
143        stored_files,
144        app_conf,
145    }: InnerArgs<'_>,
146) -> anyhow::Result<Vec<StoredRepositoryExercise>> {
147    ensure_clone_url_is_allowed(url).await?;
148    let mut tx = conn.begin().await?;
149
150    // clone repo to temp dir
151    let temp = tempfile::tempdir()?;
152    let mut fetch_opts = FetchOptions::new();
153    // A redirect could point the clone at an address ensure_clone_url_is_allowed never saw.
154    fetch_opts.follow_redirects(RemoteRedirect::None);
155    let mut remote_cbs = RemoteCallbacks::new();
156    if let Some(deploy_key) = deploy_key {
157        remote_cbs
158            .certificate_check(|_, _| Ok(CertificateCheckStatus::CertificateOk))
159            .credentials(|_, username, credential_type| {
160                if credential_type.is_ssh_memory() {
161                    Cred::ssh_key_from_memory(
162                        username.unwrap_or("git"),
163                        public_key,
164                        // Exposed only here, where the key is handed to libgit2 in memory.
165                        deploy_key.expose_secret(),
166                        None,
167                    )
168                } else {
169                    Err(git2::Error::from_str(
170                        "The git server does not support the SSH_MEMORY credential type",
171                    ))
172                }
173            });
174    }
175    fetch_opts.remote_callbacks(remote_cbs);
176    info!("Cloning {url} to {:?}", temp.path());
177    RepoBuilder::new()
178        .fetch_options(fetch_opts)
179        .clone(url, temp.path())?;
180    info!("Finished cloning {url} to {:?}", temp.path());
181
182    // create exercises in db and store them in file store
183    let found_exercises = find_exercise_directories(temp.path()).await?;
184    let mut repository_exercises = vec![];
185    // (part, name) => exercise
186    let existing_exercises =
187        repository_exercises::get_for_repository(&mut tx, repository_id).await?;
188
189    // we try both the path and the checksum to find existing exercises
190    // these are the only attributes found in the exercise repositories, so if both
191    // the path and checksum change there's no way to detect that it's supposed to be an updated old exercise
192    // rather than a new one
193    // this way we can accommodate both renaming/relocating exercises and changing them, though not at the same time...
194    let existing_exercises_path_map = existing_exercises
195        .iter()
196        .map(|ex| ((&ex.part, &ex.name), ex))
197        .collect::<HashMap<_, _>>();
198    let existing_exercises_checksum_map = existing_exercises
199        .iter()
200        .map(|ex| (ex.checksum.as_slice(), ex))
201        .collect::<HashMap<_, _>>();
202    for fe in &found_exercises {
203        // check if the exercise is new
204        match (
205            existing_exercises_path_map.get(&(&fe.part, &fe.name)),
206            existing_exercises_checksum_map.get(fe.checksum.as_bytes().as_slice()),
207        ) {
208            (Some(_), Some(_)) => {
209                // both the path and checksum are unchanged, no-op
210            }
211            (Some(existing_exercise_by_path), None) => {
212                // found exercise by path but the checksum has changed, exercise has been updated
213                let path = update_exercise(
214                    &mut tx,
215                    repository_id,
216                    existing_exercise_by_path.id,
217                    fe,
218                    file_store,
219                    app_conf,
220                )
221                .await?;
222                stored_files.push(path.clone());
223                let url = file_store.get_direct_download_url(&path).await?;
224                repository_exercises.push(StoredRepositoryExercise { url });
225                repository_exercises::update_checksum(
226                    &mut tx,
227                    existing_exercise_by_path.id,
228                    fe.checksum.as_bytes(),
229                )
230                .await?;
231                // todo: uploaded files get cleaned up on an error, which means that if the refreshing fails
232                // the exercise data will be missing entirely...
233            }
234            (None, Some(existing_exercise_by_checksum)) => {
235                // found exercise by checksum but the path has changed, update path
236                repository_exercises::update_part_and_name(
237                    &mut tx,
238                    existing_exercise_by_checksum.id,
239                    &fe.part,
240                    &fe.name,
241                )
242                .await?;
243            }
244            (None, None) => {
245                // new exercise
246                let new_exercise_id = uuid::Uuid::new_v4();
247                let path = create_and_upload_exercise(
248                    &mut tx,
249                    repository_id,
250                    new_exercise_id,
251                    fe,
252                    file_store,
253                    app_conf,
254                )
255                .await?;
256                stored_files.push(path.clone());
257                let url = file_store.get_direct_download_url(&path).await?;
258                repository_exercises.push(StoredRepositoryExercise { url });
259            }
260        }
261    }
262
263    tx.commit().await?;
264    Ok(repository_exercises)
265}

267/// Updates the given repository using the given url.
268/// Exercises with a known checksum but changed part or name are updated to reflect the new part or name.
269/// Exercises with a known part and name but changed checksum are updated in the file store and the checksum updated.
270/// Errors may leave some exercises updated and others not, since there's no mechanism for rolling back any file store updates.
271/// However, these inconsistencies will be fixed after a successful retry.
272pub async fn update(
273    conn: &mut PgConnection,
274    repository: Uuid,
275    url: &str,
276    file_store: &dyn FileStore,
277    app_conf: &ApplicationConfiguration,
278) -> anyhow::Result<()> {
279    let mut new_stored_files = vec![];
280    match update_inner(
281        conn,
282        repository,
283        url,
284        file_store,
285        &mut new_stored_files,
286        app_conf,
287    )
288    .await
289    {
290        Ok(res) => Ok(res),
291        Err(err) => {
292            if !new_stored_files.is_empty() {
293                debug!(
294                    "Failed while updating exercise repository, cleaning new exercises that were uploaded"
295                );
296                for file in new_stored_files {
297                    if let Err(err) = file_store.delete(&file).await {
298                        error!("Failed to clean file {}: {err}", file.display());
299                    }
300                }
301            }
302            Err(err)
303        }
304    }
305}

307async fn update_inner(
308    conn: &mut PgConnection,
309    repository: Uuid,
310    url: &str,
311    file_store: &dyn FileStore,
312    new_stored_files: &mut Vec<PathBuf>,
313    app_conf: &ApplicationConfiguration,
314) -> anyhow::Result<()> {
315    ensure_clone_url_is_allowed(url).await?;
316    let mut tx = conn.begin().await?;
317
318    let temp = tempfile::tempdir()?;
319    let mut fetch_opts = FetchOptions::new();
320    fetch_opts.follow_redirects(RemoteRedirect::None);
321    RepoBuilder::new()
322        .fetch_options(fetch_opts)
323        .clone(url, temp.path())?;
324
325    let repository_exercises = find_exercise_directories(temp.path()).await?;
326    let current_exercises = repository_exercises::get_for_repository(&mut tx, repository).await?;
327
328    let mut by_name = HashMap::new();
329    let mut by_checksum = HashMap::new();
330    for ex in &current_exercises {
331        by_name.insert((&ex.part, &ex.name), ex);
332        by_checksum.insert(ex.checksum.as_slice(), ex);
333    }
334    for ex in repository_exercises {
335        if let Some(&current) = by_name.get(&(&ex.part, &ex.name)) {
336            // found known exercise by part and name
337            if current.checksum != ex.checksum.as_bytes() {
338                // checksum changed, update files and checksum
339                create_and_upload_exercise(
340                    &mut tx, repository, current.id, &ex, file_store, app_conf,
341                )
342                .await?;
343                repository_exercises::update_checksum(&mut tx, current.id, ex.checksum.as_bytes())
344                    .await?;
345            }
346        } else if let Some(&current) = by_checksum.get(ex.checksum.as_bytes().as_slice()) {
347            // found known exercise by checksum
348            if current.part != ex.part || current.name != ex.name {
349                // part and/or name changed
350                repository_exercises::update_part_and_name(&mut tx, current.id, &ex.part, &ex.name)
351                    .await?;
352            }
353        } else {
354            // unknown part/name and checksum, assume new exercise
355            let path = create_and_upload_exercise(
356                &mut tx,
357                repository,
358                Uuid::new_v4(),
359                &ex,
360                file_store,
361                app_conf,
362            )
363            .await?;
364            new_stored_files.push(path);
365        }
366    }
367
368    tx.commit().await?;
369    Ok(())
370}

372/// Marks the exercises and repository as deleted and removes the associated files from the file store.
373/// Only returns the last error if there are multiple errors when trying to remove the files.
374pub async fn delete(
375    conn: &mut PgConnection,
376    repository_id: Uuid,
377    file_store: &dyn FileStore,
378) -> anyhow::Result<()> {
379    let mut tx = conn.begin().await?;
380
381    let mut latest_error = None;
382    let exercises = repository_exercises::delete_for_repository(&mut tx, repository_id).await?;
383    exercise_repositories::delete(&mut tx, repository_id).await?;
384    for exercise in exercises {
385        let path = file_store::repository_exercise_path(repository_id, exercise);
386        if let Err(err) = file_store.delete(&path).await {
387            error!(
388                "Failed to delete file while deleting repository {}: {err}",
389                path.display()
390            );
391            latest_error = Some(err);
392        }
393    }
394
395    match latest_error {
396        Some(latest_error) => Err(latest_error.into()),
397        _ => {
398            tx.commit().await?;
399            Ok(())
400        }
401    }
402}

404async fn create_and_upload_exercise(
405    conn: &mut PgConnection,
406    repository: Uuid,
407    exercise_id: Uuid,
408    exercise: &FoundExercise,
409    file_store: &dyn FileStore,
410    app_conf: &ApplicationConfiguration,
411) -> anyhow::Result<PathBuf> {
412    // archive and compress
413    let cursor = Cursor::new(vec![]);
414    let mut tar = tar::Builder::new(cursor);
415    // The repository is teacher-controlled; following its symlinks would archive server files.
416    tar.follow_symlinks(false);
417    tar.append_dir_all(".", &exercise.path)?;
418    let mut tar = tar.into_inner()?;
419    // rewind cursor back to the beginning
420    tar.set_position(0);
421    let tar_zstd = zstd::encode_all(tar, 0)?;
422
423    // upload
424    let path = file_store::repository_exercise_path(repository, exercise_id);
425    file_store
426        .upload(&path, tar_zstd, "application/zstd")
427        .await?;
428    let url = file_store.get_download_url(&path, app_conf);
429
430    // create
431    repository_exercises::new(
432        conn,
433        exercise_id,
434        repository,
435        &exercise.part,
436        &exercise.name,
437        exercise.checksum.as_bytes(),
438        &url,
439    )
440    .await?;
441    Ok(path)
442}

492async fn find_exercise_directories(clone_path: &Path) -> anyhow::Result<Vec<FoundExercise>> {
493    info!("finding exercise directories in {}", clone_path.display());
494
495    let mut exercises = vec![];
496    // exercises in repositories are in subdirs like
497    // part01/01_exercise
498    // part01/02_exercise
499    // part02/01_exercise
500    for entry in WalkDir::new(clone_path)
501        .min_depth(2)
502        .max_depth(2)
503        .into_iter()
504        .filter_entry(|e| {
505            e.file_type().is_dir()
506                && e.file_name() != "private"
507                && !is_hidden_dir(e)
508                && !contains_tmcignore(e)
509                && !is_in_git_dir(e.path())
510        })
511    {
512        let entry = entry?;
513        let checksum = folder_checksum::hash_folder(entry.path()).await?;
514
515        let path = entry.into_path().canonicalize()?;
516        let part = path
517            .parent()
518            .ok_or_else(|| anyhow::anyhow!("Path should be in a subdirectory: {}", path.display()))?
519            .file_name()
520            .ok_or_else(|| {
521                anyhow::anyhow!("The parent file name cannot be missing: {}", path.display())
522            })?
523            .to_str()
524            .context("Invalid directory name in repository")?
525            .to_string();
526        let name = path
527            .file_name()
528            .ok_or_else(|| anyhow::anyhow!("Path should be a directory: {}", path.display()))?
529            .to_str()
530            .context("Invalid directory name in repository")?
531            .to_string();
532        exercises.push(FoundExercise {
533            part,
534            name,
535            checksum,
536            path,
537        });
538    }
539    Ok(exercises)
540}

542// Filter for hidden directories (directories with names starting with '.')
543fn is_hidden_dir(entry: &DirEntry) -> bool {
544    let skip = entry.metadata().map(|e| e.is_dir()).unwrap_or_default()
545        && entry
546            .file_name()
547            .to_str()
548            .map(|s| s.starts_with('.'))
549            .unwrap_or_default();
550    if skip {
551        debug!("is hidden dir: {}", entry.path().display());
552    }
553    skip
554}
555
556// Filter for .git directory
557fn is_in_git_dir(path: &Path) -> bool {
558    let skip = path.parent().map(|p| p.ends_with(".git")).unwrap_or(false);
559    if skip {
560        debug!("is in git dir: {}", path.display());
561    }
562    skip
563}
564
565fn contains_tmcignore(entry: &DirEntry) -> bool {
566    for entry in WalkDir::new(entry.path())
567        .max_depth(1)
568        .into_iter()
569        .filter_map(|e| e.ok())
570    {
571        let is_file = entry.metadata().map(|e| e.is_file()).unwrap_or_default();
572        if is_file && entry.file_name() == ".tmcignore" {
573            debug!("contains .tmcignore: {}", entry.path().display());
574            return true;
575        }
576    }
577    false
578}
