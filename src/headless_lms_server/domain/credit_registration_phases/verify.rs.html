1//! The `verify` phase: asking the study registry what became of a submission.
2//!
3//! The only way out of `submission_uncertain`, and never a way back to `import`. Nothing here ever
4//! fails a row: the attainment may exist, and a failed row is one an admin retries, which for an
5//! uncertain submission would mean sending it twice.
6

39/// The one code that means the submission became an attainment.
40const REGISTERED_CODE: &str = "registered";
41
42/// Both states the poller owns. Withdrawal moves a row out of both, which is what stops the polling
43/// without any query having to know about withdrawal.
44const CLAIMED_STATES: [CreditRegistrationState; 2] = [
45    CreditRegistrationState::AwaitingVerification,
46    CreditRegistrationState::SubmissionUncertain,
47];
48
49pub async fn run(ctx: &PhaseContext<'_>, scope: &PhaseScope) -> anyhow::Result<PhaseRunOutcome> {
50    let mut conn = ctx.pool.acquire().await?;
51    let mut tx = conn.begin().await?;
52    let claimed = claim_due(
53        &mut tx,
54        &CLAIMED_STATES,
55        scope,
56        SuotarEndpoint::VerifyAttainments.max_batch_size() as i64,
57    )
58    .await?;
59    let mut polls = Vec::new();
60    let mut recoveries = Vec::new();
61    for row in claimed {
62        // Counted before the call: the count is part of the request item id, so two polls of one
63        // row are distinguishable in the registry's log.
64        let attempt = increment_verify_attempt_count(&mut tx, row.id).await?;
65        // Pushed out of reach before the request leaves, so a concurrent iteration cannot poll the
66        // same row. The outcome overwrites this.
67        schedule_next_attempt(
68            &mut tx,
69            row.id,
70            next_attempt_at(Utc::now(), verify_backoff_secs(attempt)),
71        )
72        .await?;
73        match row.submitted_attainment_id.clone() {
74            Some(submitted_attainment_id) => polls.push((row, attempt, submitted_attainment_id)),
75            None if row.state == CreditRegistrationState::SubmissionUncertain => {
76                recoveries.push((row, attempt))
77            }
78            None => {
79                warn!(
80                    "Credit registration {} is awaiting verification with no submitted attainment id.",
81                    row.id
82                );
83            }
84        }
85    }
86    tx.commit().await?;
87    // Held only for the claim; `poll` and `recover` re-acquire around their own Suotar calls.
88    drop(conn);
89
90    let mut processed = 0;
91    let mut items_failed = 0;
92    let mut error = None;
93
94    if !polls.is_empty() {
95        let outcome = poll(ctx, &polls).await?;
96        processed += outcome.items_processed;
97        items_failed += outcome.items_failed;
98        error = error.or(outcome.error);
99    }
100    // Recoveries go out on `resolve-enrolments`, whose batch limit is smaller than the one these
101    // rows were claimed at, so an oversized set would be refused whole before anything was sent.
102    for chunk in recoveries.chunks(SuotarEndpoint::ResolveEnrolments.max_batch_size()) {
103        let outcome = recover(ctx, chunk).await?;
104        processed += outcome.items_processed;
105        items_failed += outcome.items_failed;
106        error = error.or(outcome.error);
107    }
108    Ok(PhaseRunOutcome {
109        items_processed: processed,
110        items_failed,
111        error,
112    })
113}

115/// Polls the rows that have something to poll by.
116async fn poll(
117    ctx: &PhaseContext<'_>,
118    polls: &[(CreditRegistration, i32, String)],
119) -> anyhow::Result<PhaseRunOutcome> {
120    let items: Vec<VerifyAttainmentRequestItem> = polls
121        .iter()
122        .map(
123            |(row, attempt, submitted_attainment_id)| VerifyAttainmentRequestItem {
124                request_item_id: verify_request_item_id(row.id, *attempt),
125                submitted_attainment_id: submitted_attainment_id.clone(),
126            },
127        )
128        .collect();
129    let requests = requests_json(&items);
130    let response = ctx
131        .suotar_client
132        .verify_attainments(
133            SuotarCallContext::new(ctx.worker_name(CreditRegistrationPhase::Verify))
134                .for_registrations(polls.iter().map(|(row, _, _)| row.id).collect()),
135            items,
136        )
137        .await;
138    let response = match response {
139        Ok(response) => response,
140        Err(error) => return poll_request_failure(ctx, polls, &requests, &error).await,
141    };
142
143    let mut conn = ctx.pool.acquire().await?;
144    let mut items_failed = 0;
145    for ((row, attempt, _), request) in polls.iter().zip(requests.iter()) {
146        let request_item_id = verify_request_item_id(row.id, *attempt);
147        let response_json = response_item_json(&response.raw_response, &request_item_id);
148        let event = OutcomeEvent {
149            suotar_api_call_id: response.call_id,
150            request: Some(request),
151            response: response_json.as_ref(),
152            ..OutcomeEvent::default()
153        };
154        // The count this poll was made under, not the one the row was claimed with, so the backoff
155        // doubles once per poll.
156        let facts = RowFacts {
157            verify_attempt_count: *attempt,
158            ..row_facts(row)
159        };
160        let item = response.item(&request_item_id);
161        let registered = item.is_some_and(|item| {
162            item.status == SuotarItemStatus::Ok && item.code == REGISTERED_CODE
163        });
164        if registered {
165            if let Some(result) = item.and_then(|item| item.result.as_ref()) {
166                set_sisu_attainment_if_unclaimed(
167                    &mut conn,
168                    row.id,
169                    &result.attainment.id,
170                    Some(&result.attainment.attainment_type),
171                )
172                .await?;
173            }
174            apply_outcome(
175                &mut conn,
176                row,
177                &Outcome {
178                    to_state: CreditRegistrationState::Registered,
179                    error_code: None,
180                    // Confirmed, so whatever an operator was asked to look at is settled.
181                    needs_admin_attention: Some(false),
182                    delay_secs: None,
183                    drop_verified_student_number: false,
184                    increment_submit_retry_count: false,
185                },
186                event,
187                Some(row.state),
188            )
189            .await?;
190            continue;
191        }
192        // Everything else keeps the row where it is: `notRegistered` is a normal polling answer,
193        // and any other answer is one we will not act on blindly.
194        let outcome = item
195            .and_then(|item| map_code(AuditEndpoint::VerifyAttainments, &item.code))
196            .map(|code| verify_error_outcome(row.state, code, &facts))
197            .unwrap_or_else(|| verify_not_registered_outcome(row.state, &facts));
198        apply_outcome(
199            &mut conn,
200            row,
201            &outcome,
202            OutcomeEvent {
203                error_message: item
204                    .and_then(|item| item.error.as_ref())
205                    .map(|error| error.message.as_str()),
206                ..event
207            },
208            Some(row.state),
209        )
210        .await?;
211        if counts_as_failed(&outcome) {
212            items_failed += 1;
213        }
214    }
215    Ok(PhaseRunOutcome {
216        items_processed: i32::try_from(polls.len()).unwrap_or(i32::MAX),
217        items_failed,
218        error: every_item_failed_transiently(&response)
219            .then(|| "Every verify poll came back transiently unavailable.".to_string()),
220    })
221}

223/// A request-level failure of the poll itself (transport, a 5xx, a malformed response, ...).
224///
225/// Deliberately not the shared `request_level_failure`: a failure to ask proves nothing was or was
226/// not created, and moving the row towards `failed_retryable` would let an admin resubmit it. Still
227/// reported as an error so the breaker sees it.
228async fn poll_request_failure(
229    ctx: &PhaseContext<'_>,
230    polls: &[(CreditRegistration, i32, String)],
231    requests: &[serde_json::Value],
232    error: &UtilError,
233) -> anyhow::Result<PhaseRunOutcome> {
234    let mut conn = ctx.pool.acquire().await?;
235    for ((row, attempt, _), request) in polls.iter().zip(requests.iter()) {
236        let facts = RowFacts {
237            verify_attempt_count: *attempt,
238            ..row_facts(row)
239        };
240        apply_outcome(
241            &mut conn,
242            row,
243            &verify_not_registered_outcome(row.state, &facts),
244            OutcomeEvent {
245                message: Some("Could not verify this submission this time."),
246                error_message: Some(error.message()),
247                request: Some(request),
248                ..OutcomeEvent::default()
249            },
250            Some(row.state),
251        )
252        .await?;
253    }
254    Ok(PhaseRunOutcome {
255        items_processed: i32::try_from(polls.len()).unwrap_or(i32::MAX),
256        items_failed: 0,
257        error: Some(error.message().to_string()),
258    })
259}

261/// Looks for the attainment a submission we lost track of would have produced. The row stays
262/// `submission_uncertain` unless it is found: never failed, never re-imported.
263async fn recover(
264    ctx: &PhaseContext<'_>,
265    rows: &[(CreditRegistration, i32)],
266) -> anyhow::Result<PhaseRunOutcome> {
267    let mut conn = ctx.pool.acquire().await?;
268    let contexts = get_submission_contexts(
269        &mut conn,
270        &rows.iter().map(|(row, _)| row.id).collect::<Vec<_>>(),
271    )
272    .await?;
273    let mut items = Vec::new();
274    let mut asked = Vec::new();
275    for (row, attempt) in rows {
276        let Some(context) = contexts.get(&row.id) else {
277            continue;
278        };
279        let (Some(student_number), Some(course_code)) = (
280            row.student_number
281                .clone()
282                .or_else(|| context.student_number.clone()),
283            row.uh_course_code
284                .clone()
285                .or_else(|| context.uh_course_code.clone()),
286        ) else {
287            continue;
288        };
289        items.push(ResolveEnrolmentRequestItem {
290            request_item_id: row.request_item_id.clone(),
291            student_number,
292            course_code,
293        });
294        asked.push((row.clone(), *attempt));
295    }
296    // Held only for `get_submission_contexts` above; the Suotar call below can take up to the
297    // request timeout.
298    drop(conn);
299    if items.is_empty() {
300        return Ok(PhaseRunOutcome::default());
301    }
302
303    let requests = requests_json(&items);
304    let response = ctx
305        .suotar_client
306        .resolve_enrolments(
307            SuotarCallContext::new(ctx.worker_name(CreditRegistrationPhase::Verify))
308                .for_registrations(asked.iter().map(|(row, _)| row.id).collect()),
309            items,
310        )
311        .await;
312    let response = match response {
313        Ok(response) => response,
314        Err(error) => {
315            // Not `request_level_failure`: these rows must stay uncertain whatever the call did.
316            let mut conn = ctx.pool.acquire().await?;
317            for ((row, attempt), request) in asked.iter().zip(requests.iter()) {
318                let facts = RowFacts {
319                    verify_attempt_count: *attempt,
320                    ..row_facts(row)
321                };
322                apply_outcome(
323                    &mut conn,
324                    row,
325                    &uncertain_recheck_outcome(&facts),
326                    OutcomeEvent {
327                        message: Some("Could not look for the attainment this time."),
328                        error_message: Some(error.message()),
329                        request: Some(request),
330                        ..OutcomeEvent::default()
331                    },
332                    Some(row.state),
333                )
334                .await?;
335            }
336            return Ok(PhaseRunOutcome {
337                items_processed: i32::try_from(asked.len()).unwrap_or(i32::MAX),
338                items_failed: i32::try_from(asked.len()).unwrap_or(i32::MAX),
339                error: Some(error.message().to_string()),
340            });
341        }
342    };
343
344    let mut conn = ctx.pool.acquire().await?;
345    for ((row, attempt), request) in asked.iter().zip(requests.iter()) {
346        let response_json = response_item_json(&response.raw_response, &row.request_item_id);
347        let event = OutcomeEvent {
348            suotar_api_call_id: response.call_id,
349            request: Some(request),
350            response: response_json.as_ref(),
351            ..OutcomeEvent::default()
352        };
353        let found = match response.item(&row.request_item_id) {
354            Some(item) if item.status == SuotarItemStatus::Ok => {
355                item.result.as_ref().zip(row.attainment_date).and_then(
356                    |(result, attainment_date)| {
357                        attainment_matching_submission(
358                            &result.existing_attainments,
359                            attainment_date,
360                            row.grade_scale_id.as_deref().unwrap_or_default(),
361                            row.grade_id.as_deref().unwrap_or_default(),
362                        )
363                    },
364                )
365            }
366            _ => None,
367        };
368        match found {
369            Some(attainment) => {
370                set_sisu_attainment_if_unclaimed(
371                    &mut conn,
372                    row.id,
373                    &attainment.id,
374                    Some(&attainment.attainment_type),
375                )
376                .await?;
377                transition(
378                    &mut conn,
379                    row.id,
380                    &Transition {
381                        event_kind: CreditRegistrationEventKind::SuotarResponse,
382                        event_message: Some(
383                            "The attainment this submission would have created is in the study \
384                             registry, so it landed after all."
385                                .to_string(),
386                        ),
387                        needs_admin_attention: Some(false),
388                        suotar_api_call_id: event.suotar_api_call_id,
389                        event_details: Some(
390                            headless_lms_models::credit_registration_events::suotar_exchange_details(
391                                event.request,
392                                event.response,
393                            ),
394                        ),
395                        expected_from_state: Some(row.state),
396                        ..Transition::to(CreditRegistrationState::Duplicate)
397                    },
398                )
399                .await?;
400            }
401            None => {
402                let facts = RowFacts {
403                    verify_attempt_count: *attempt,
404                    ..row_facts(row)
405                };
406                apply_outcome(
407                    &mut conn,
408                    row,
409                    &uncertain_recheck_outcome(&facts),
410                    OutcomeEvent {
411                        message: Some(
412                            "No matching attainment yet, so whether the submission landed is still \
413                             unknown.",
414                        ),
415                        ..event
416                    },
417                    Some(row.state),
418                )
419                .await?;
420            }
421        }
422    }
423    Ok(PhaseRunOutcome {
424        items_processed: i32::try_from(asked.len()).unwrap_or(i32::MAX),
425        // A row still waiting to be resolved is not a failed item; the recheck raises the admin
426        // flag after enough tries instead.
427        items_failed: 0,
428        error: every_item_failed_transiently(&response)
429            .then(|| "Every recovery lookup came back transiently unavailable.".to_string()),
430    })
431}

433#[cfg(test)]
434mod tests {
435    use super::*;
436
437    /// Nothing this phase claims can lead back to a batch.
438    #[test]
439    fn the_poller_owns_exactly_the_two_states_a_submission_can_still_be_in() {
440        assert!(CLAIMED_STATES.contains(&CreditRegistrationState::AwaitingVerification));
441        assert!(CLAIMED_STATES.contains(&CreditRegistrationState::SubmissionUncertain));
442        assert!(!CLAIMED_STATES.contains(&CreditRegistrationState::Submitting));
443        assert!(
444            !CLAIMED_STATES.contains(&CreditRegistrationState::AbandonedByConsentWithdrawal),
445            "withdrawal has to stop the polling"
446        );
447    }
448
449    /// A registry log line has to name one poll, not one row.
450    #[test]
451    fn two_polls_of_one_row_are_addressed_apart() {
452        let id = uuid::Uuid::new_v4();
453        assert_eq!(verify_request_item_id(id, 1), format!("vf-{id}-1"));
454        assert_ne!(verify_request_item_id(id, 1), verify_request_item_id(id, 2));
455    }
