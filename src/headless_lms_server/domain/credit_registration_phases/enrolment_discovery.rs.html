1//! The `enrolment-discovery` phase: who the study registry says is on the course.
2//!
3//! One listing unparks the registrations of people we already have a link for, links the few whose
4//! registry address is an address one of our accounts has proved it controls, and claims an
5//! account-linking mail for everybody else. That middle branch is terminal, never a filter: the
6//! population the linking mail exists to reach is the people whose two addresses differ, and every
7//! fast-track outcome other than a link falls through to the mail.
8

43pub async fn run(ctx: &PhaseContext<'_>, scope: &PhaseScope) -> anyhow::Result<PhaseRunOutcome> {
44    let endpoint = SuotarEndpoint::ListByCourse;
45    let mut conn = ctx.pool.acquire().await?;
46    let mut tx = conn.begin().await?;
47    let claimed =
48        claim_stalest_for_listing(&mut tx, endpoint.max_batch_size() as i64, scope.course_id)
49            .await?;
50    tx.commit().await?;
51    let attempted = i32::try_from(claimed.len()).unwrap_or(i32::MAX);
52
53    let mut items = Vec::new();
54    let mut realisations = Vec::new();
55    let mut items_failed = 0;
56    for realisation in claimed {
57        let Some(course_code) = listable_course_code(&realisation) else {
58            // A configuration problem the config check reports on; a call would only earn
59            // `courseCodeNotFound`.
60            warn!(
61                "Course module {} has a Suotar realisation but no course code, so it cannot be listed.",
62                realisation.course_module_id
63            );
64            items_failed += 1;
65            mark_listing_failed(
66                &mut conn,
67                realisation.id,
68                CreditRegistrationErrorCode::MissingUhCourseCode,
69            )
70            .await?;
71            continue;
72        };
73        items.push(ListByCourseRequestItem {
74            request_item_id: listing_request_item_id(realisation.id),
75            course_code,
76            course_unit_realisation_id: Some(realisation.course_unit_realisation_id.clone()),
77        });
78        realisations.push(realisation);
79    }
80    if items.is_empty() {
81        return Ok(PhaseRunOutcome {
82            items_processed: attempted,
83            items_failed,
84            error: None,
85        });
86    }
87    // Held only for the reads above; the Suotar call can pin it for the whole request timeout.
88    drop(conn);
89
90    let response = ctx
91        .suotar_client
92        .list_enrolments_by_course(
93            SuotarCallContext::new(ctx.worker_name(CreditRegistrationPhase::EnrolmentDiscovery)),
94            items,
95        )
96        .await;
97    let response = match response {
98        // No ledger row to move: the realisations keep their old `last_listed_at` and stay first in
99        // line for the next iteration.
100        Err(error) => return Ok(whole_request_failed(attempted, &error)),
101        Ok(response) => response,
102    };
103
104    let mut conn = ctx.pool.acquire().await?;
105    for realisation in &realisations {
106        let item = response.item(&listing_request_item_id(realisation.id));
107        let listed = match item {
108            Some(item) if item.status == SuotarItemStatus::Ok => Ok(item
109                .result
110                .as_ref()
111                .map(|result| result.people.as_slice())
112                .unwrap_or_default()),
113            Some(item) => {
114                warn!(
115                    "Listing realisation {} failed with {}.",
116                    realisation.course_unit_realisation_id, item.code
117                );
118                Err(map_wire_code(&item.code).unwrap_or(CreditRegistrationErrorCode::Unknown))
119            }
120            None => {
121                warn!(
122                    "The study registry did not answer for realisation {}.",
123                    realisation.course_unit_realisation_id
124                );
125                Err(CreditRegistrationErrorCode::UnexpectedResponse)
126            }
127        };
128        let people = match listed {
129            Ok(people) => people,
130            Err(error) => {
131                items_failed += 1;
132                mark_listing_failed(&mut conn, realisation.id, error).await?;
133                continue;
134            }
135        };
136        let outcome = reconcile(ctx, &mut conn, realisation, people).await?;
137        record_listing_outcome(&mut conn, realisation.id, &outcome).await?;
138    }
139
140    Ok(PhaseRunOutcome {
141        items_processed: attempted,
142        items_failed,
143        error: every_item_failed_transiently(&response).then(|| {
144            "Every realisation of the batch came back transiently unavailable.".to_string()
145        }),
146    })
147}

149/// Applies one realisation's roster and returns the counters the realisation row carries.
150async fn reconcile(
151    ctx: &PhaseContext<'_>,
152    conn: &mut PgConnection,
153    realisation: &RealisationToList,
154    people: &[ListedPerson],
155) -> anyhow::Result<RealisationListingOutcome> {
156    let mut outcome = RealisationListingOutcome {
157        listed_person_count: i32::try_from(people.len()).unwrap_or(i32::MAX),
158        ..RealisationListingOutcome::default()
159    };
160    let person_ids: Vec<String> = people
161        .iter()
162        .map(|person| person.person_id.clone())
163        .collect();
164    let linked = verified_student_numbers::get_by_sisu_person_ids(conn, &person_ids).await?;
165    let linked_person_ids: HashSet<&str> = linked
166        .iter()
167        .map(|row| row.sisu_person_id.as_str())
168        .collect();
169
170    let mut fast_track = FastTrackRun::new(ctx, realisation);
171    let mut discovered = Vec::new();
172    for person in people {
173        if linked_person_ids.contains(person.person_id.as_str()) {
174            outcome.already_linked_count += 1;
175            continue;
176        }
177        let addresses = listed_person_addresses(person);
178        if addresses.is_empty() {
179            // The only genuinely unreachable population, and the reason it has a counter of its own.
180            outcome.no_address_count += 1;
181            continue;
182        }
183        if fast_track.try_link(conn, person, &mut outcome).await? {
184            continue;
185        }
186        discovered.push(DiscoveredPerson {
187            sisu_person_id: person.person_id.clone(),
188            student_number: person.student_number.clone(),
189            first_names: Some(person.first_names.clone()),
190            last_name: Some(person.last_name.clone()),
191            course_id: realisation.course_id,
192            addresses,
193        });
194    }
195    if !discovered.is_empty() {
196        for claimed in claim_linking_mails_batch(conn, &discovered).await? {
197            outcome.mailed_count += claimed.claimed;
198            outcome.suppressed_by_dedup_count += claimed.suppressed_by_dedup;
199            outcome.suppressed_by_rate_cap_count += claimed.suppressed_by_rate_cap;
200        }
201    }
202
203    // The fast way back for a row parked without an enrolment, which would otherwise wait out its
204    // own daily recheck.
205    let linked_user_ids: Vec<_> = linked.iter().map(|row| row.user_id).collect();
206    if !linked_user_ids.is_empty() {
207        recheck_no_usable_enrolment_now(conn, realisation.course_id, &linked_user_ids).await?;
208    }
209    Ok(outcome)
210}

212/// The fast track over one realisation's roster: the config it reads and the template lookup it
213/// caches, so neither is repeated per person.
214struct FastTrackRun<'a> {
215    ctx: &'a PhaseContext<'a>,
216    realisation: &'a RealisationToList,
217    enabled: bool,
218    max_verification_age: chrono::Duration,
219    templates: TemplateCache,
220}
221
222impl<'a> FastTrackRun<'a> {
223    fn new(ctx: &'a PhaseContext<'a>, realisation: &'a RealisationToList) -> Self {
224        let conf = ctx.suotar_conf;
225        Self {
226            ctx,
227            realisation,
228            enabled: conf.fast_track_email_match_enabled,
229            max_verification_age: chrono::Duration::days(
230                conf.fast_track_max_email_verification_age_days.max(0),
231            ),
232            templates: TemplateCache::default(),
233        }
234    }
235
236    /// Whether the person was linked here, in which case no linking mail is owed. `false` for every
237    /// other outcome, including the flag being off, and the caller carries on to the mail.
238    async fn try_link(
239        &mut self,
240        conn: &mut PgConnection,
241        person: &ListedPerson,
242        outcome: &mut RealisationListingOutcome,
243    ) -> anyhow::Result<bool> {
244        if !self.enabled {
245            return Ok(false);
246        }
247        // One transaction, and the candidate query locks the account row: a profile edit landing
248        // between reading the proof and writing the link would leave a link resting on an address
249        // the account no longer holds.
250        let mut tx = conn.begin().await?;
251        // The registry's secondary address is self-entered, so anyone could name someone else's
252        // account address there and be handed their student number.
253        let candidate =
254            find_fast_track_candidate(&mut tx, &person.primary_email, &person.person_id).await?;
255        let decision = decide_fast_track(
256            candidate.as_ref(),
257            RegistryName {
258                first_names: Some(&person.first_names),
259                last_name: Some(&person.last_name),
260            },
261            Utc::now(),
262            self.max_verification_age,
263        );
264        match decision {
265            FastTrackDecision::NoAccountMatch => outcome.fast_track_skipped_no_account_count += 1,
266            FastTrackDecision::UnverifiedAccount => {
267                outcome.fast_track_skipped_unverified_count += 1
268            }
269            FastTrackDecision::StaleVerification => {
270                outcome.fast_track_skipped_stale_verification_count += 1
271            }
272            FastTrackDecision::NameMismatch => outcome.fast_track_skipped_name_mismatch_count += 1,
273            FastTrackDecision::AccountHasStudentNumber => {
274                outcome.fast_track_skipped_account_has_number_count += 1
275            }
276            FastTrackDecision::UnlinkedBefore => {
277                outcome.fast_track_skipped_unlinked_before_count += 1
278            }
279            FastTrackDecision::Link => outcome.fast_tracked_count += 1,
280        }
281        let (FastTrackDecision::Link, Some(candidate)) = (decision, candidate) else {
282            return Ok(false);
283        };
284
285        link_by_email_match(
286            &mut tx,
287            &FastTrackLink {
288                student_number: &person.student_number,
289                sisu_person_id: &person.person_id,
290                first_names: Some(&person.first_names),
291                last_name: Some(&person.last_name),
292                course_id: self.realisation.course_id,
293            },
294            &candidate,
295        )
296        .await?;
297        self.notify(&mut tx, person, &candidate).await?;
298        tx.commit().await?;
299        Ok(true)
300    }
301
302    /// The security notice that makes a wrong link detectable by the one party the link was proved
303    /// against. Off the critical path on purpose: the link is already made and registration proceeds,
304    /// so a missing template is logged and skipped rather than failing the listing.
305    async fn notify(
306        &mut self,
307        conn: &mut PgConnection,
308        person: &ListedPerson,
309        candidate: &FastTrackCandidate,
310    ) -> anyhow::Result<()> {
311        let language = template_language(&self.realisation.course_language_code);
312        let Some(template_id) = self
313            .templates
314            .id_for(
315                conn,
316                EmailTemplateType::CreditRegistrationStudentNumberLinked,
317                &language,
318            )
319            .await?
320        else {
321            warn!(
322                "No credit_registration_student_number_linked email template in {language}, so an automatic link went unannounced."
323            );
324            return Ok(());
325        };
326        insert_email_delivery_with_placeholders(
327            conn,
328            candidate.user_id,
329            template_id,
330            &json!({
331                "NAME": candidate.first_name.clone().unwrap_or_default(),
332                "STUDENT_NUMBER": person.student_number,
333                "LINK": student_number_settings_url(self.ctx.base_url),
334            }),
335        )
336        .await?;
337        Ok(())
338    }
339}

341/// Where the notice's "not you? unlink" link goes.
342fn student_number_settings_url(base_url: &str) -> String {
343    format!(
344        "{}/user-settings/student-number",
345        base_url.trim_end_matches('/')
346    )
347}
348
349fn listable_course_code(realisation: &RealisationToList) -> Option<String> {
350    realisation
351        .uh_course_code
352        .clone()
353        .filter(|code| !code.trim().is_empty())
354}

356fn whole_request_failed(attempted: i32, error: &UtilError) -> PhaseRunOutcome {
357    PhaseRunOutcome {
358        items_processed: attempted,
359        items_failed: attempted,
360        error: Some(scrub_text(error.message())),
361    }
