1//! The `enrolment-discovery` phase: who the study registry says is on the course.
2//!
3//! One listing unparks the registrations of people we already have a link for, and claims an
4//! account-linking mail for the rest. Deliberately no matching of Sisu addresses against accounts:
5//! the population the linking mail exists to reach is the people whose two addresses differ.

33pub async fn run(ctx: &PhaseContext<'_>, scope: &PhaseScope) -> anyhow::Result<PhaseRunOutcome> {
34    let endpoint = SuotarEndpoint::ListByCourse;
35    let mut conn = ctx.pool.acquire().await?;
36    let mut tx = conn.begin().await?;
37    let claimed =
38        claim_stalest_for_listing(&mut tx, endpoint.max_batch_size() as i64, scope.course_id)
39            .await?;
40    tx.commit().await?;
41    let attempted = i32::try_from(claimed.len()).unwrap_or(i32::MAX);
42
43    let mut items = Vec::new();
44    let mut realisations = Vec::new();
45    let mut items_failed = 0;
46    for realisation in claimed {
47        let Some(course_code) = listable_course_code(&realisation) else {
48            // A configuration problem the config check reports on; a call would only earn
49            // `courseCodeNotFound`.
50            warn!(
51                "Course module {} has a Suotar realisation but no course code, so it cannot be listed.",
52                realisation.course_module_id
53            );
54            items_failed += 1;
55            mark_listing_failed(
56                &mut conn,
57                realisation.id,
58                CreditRegistrationErrorCode::MissingUhCourseCode,
59            )
60            .await?;
61            continue;
62        };
63        items.push(ListByCourseRequestItem {
64            request_item_id: listing_request_item_id(realisation.id),
65            course_code,
66            course_unit_realisation_id: Some(realisation.course_unit_realisation_id.clone()),
67        });
68        realisations.push(realisation);
69    }
70    if items.is_empty() {
71        return Ok(PhaseRunOutcome {
72            items_processed: attempted,
73            items_failed,
74            error: None,
75        });
76    }
77    // Held only for the reads above; the Suotar call can pin it for the whole request timeout.
78    drop(conn);
79
80    let response = ctx
81        .suotar_client
82        .list_enrolments_by_course(
83            SuotarCallContext::new(ctx.worker_name(CreditRegistrationPhase::EnrolmentDiscovery)),
84            items,
85        )
86        .await;
87    let response = match response {
88        // No ledger row to move: the realisations keep their old `last_listed_at` and stay first in
89        // line for the next iteration.
90        Err(error) => return Ok(whole_request_failed(attempted, &error)),
91        Ok(response) => response,
92    };
93
94    let mut conn = ctx.pool.acquire().await?;
95    for realisation in &realisations {
96        let item = response.item(&listing_request_item_id(realisation.id));
97        let listed = match item {
98            Some(item) if item.status == SuotarItemStatus::Ok => Ok(item
99                .result
100                .as_ref()
101                .map(|result| result.people.as_slice())
102                .unwrap_or_default()),
103            Some(item) => {
104                warn!(
105                    "Listing realisation {} failed with {}.",
106                    realisation.course_unit_realisation_id, item.code
107                );
108                Err(map_wire_code(&item.code).unwrap_or(CreditRegistrationErrorCode::Unknown))
109            }
110            None => {
111                warn!(
112                    "The study registry did not answer for realisation {}.",
113                    realisation.course_unit_realisation_id
114                );
115                Err(CreditRegistrationErrorCode::UnexpectedResponse)
116            }
117        };
118        let people = match listed {
119            Ok(people) => people,
120            Err(error) => {
121                items_failed += 1;
122                mark_listing_failed(&mut conn, realisation.id, error).await?;
123                continue;
124            }
125        };
126        let outcome = reconcile(&mut conn, realisation, people).await?;
127        record_listing_outcome(&mut conn, realisation.id, &outcome).await?;
128    }
129
130    Ok(PhaseRunOutcome {
131        items_processed: attempted,
132        items_failed,
133        error: every_item_failed_transiently(&response).then(|| {
134            "Every realisation of the batch came back transiently unavailable.".to_string()
135        }),
136    })
137}

139/// Applies one realisation's roster and returns the counters the realisation row carries.
140async fn reconcile(
141    conn: &mut PgConnection,
142    realisation: &RealisationToList,
143    people: &[ListedPerson],
144) -> anyhow::Result<RealisationListingOutcome> {
145    let mut outcome = RealisationListingOutcome {
146        listed_person_count: i32::try_from(people.len()).unwrap_or(i32::MAX),
147        ..RealisationListingOutcome::default()
148    };
149    let person_ids: Vec<String> = people
150        .iter()
151        .map(|person| person.person_id.clone())
152        .collect();
153    let linked = verified_student_numbers::get_by_sisu_person_ids(conn, &person_ids).await?;
154    let linked_person_ids: HashSet<&str> = linked
155        .iter()
156        .map(|row| row.sisu_person_id.as_str())
157        .collect();
158
159    let mut discovered = Vec::new();
160    for person in people {
161        if linked_person_ids.contains(person.person_id.as_str()) {
162            outcome.already_linked_count += 1;
163            continue;
164        }
165        let addresses = listed_person_addresses(person);
166        if addresses.is_empty() {
167            // The only genuinely unreachable population, and the reason it has a counter of its own.
168            outcome.no_address_count += 1;
169            continue;
170        }
171        discovered.push(DiscoveredPerson {
172            sisu_person_id: person.person_id.clone(),
173            student_number: person.student_number.clone(),
174            first_names: Some(person.first_names.clone()),
175            last_name: Some(person.last_name.clone()),
176            course_id: realisation.course_id,
177            addresses,
178        });
179    }
180    if !discovered.is_empty() {
181        for claimed in claim_linking_mails_batch(conn, &discovered).await? {
182            outcome.mailed_count += claimed.claimed;
183            outcome.suppressed_by_dedup_count += claimed.suppressed_by_dedup;
184            outcome.suppressed_by_rate_cap_count += claimed.suppressed_by_rate_cap;
185        }
186    }
187
188    // The fast way back for a row parked without an enrolment, which would otherwise wait out its
189    // own daily recheck.
190    let linked_user_ids: Vec<_> = linked.iter().map(|row| row.user_id).collect();
191    if !linked_user_ids.is_empty() {
192        recheck_no_usable_enrolment_now(conn, realisation.course_id, &linked_user_ids).await?;
193    }
194    Ok(outcome)
195}

197fn listable_course_code(realisation: &RealisationToList) -> Option<String> {
198    realisation
199        .uh_course_code
200        .clone()
201        .filter(|code| !code.trim().is_empty())
202}

204fn whole_request_failed(attempted: i32, error: &UtilError) -> PhaseRunOutcome {
205    PhaseRunOutcome {
206        items_processed: attempted,
207        items_failed: attempted,
208        error: Some(scrub_text(error.message())),
209    }
