1//! The ALTCHA proof of work that the anonymous account forms require: the browser solves a challenge
2//! from [`create_challenge`], and the form handler redeems the solution with [`verify_and_spend`].
3//!
4//! Each challenge is signed with what it is for, so a solution is accepted only by that form.
5
6use std::collections::BTreeMap;
7use std::ops::RangeInclusive;
8
9use altcha::{Challenge, CreateChallengeOptions, Payload, VerifySolutionOptions};
10use base64::Engine;
11use chrono::Duration;
12use headless_lms_models::altcha_spent_challenges;
13use secrecy::ExposeSecret;
14use utoipa::ToSchema;
15
16use crate::config::server_runtime_config;
17use crate::domain::error::{BadRequestReason, bad_request_with_reason};
18use crate::prelude::*;

20const ALGORITHM: &str = "PBKDF2/SHA-256";
21const CHALLENGE_LIFETIME_MINUTES: i64 = 10;
22/// The browser derives a key for every counter up to the secret one, so the work is the counter
23/// times the cost.
24const COUNTER_RANGE: RangeInclusive<u32> = 2_000..=5_000;
25const TEST_MODE_COUNTER_RANGE: RangeInclusive<u32> = 1..=10;
26const EXPIRED_CLEANUP_PROBABILITY: f64 = 0.01;
27
28/// What a challenge may be redeemed for.
29#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
30#[serde(rename_all = "snake_case")]
31pub enum ChallengePurpose {
32    Signup,
33    PasswordReset,
34}
35
36/// What a challenge is issued for.
37pub struct ChallengeContext {
38    pub purpose: ChallengePurpose,
39}
40
41/// A fresh signed challenge for the form widget to solve.
42pub async fn create_challenge(context: ChallengeContext) -> anyhow::Result<Challenge> {
43    // Deterministic mode derives the key here, which is too slow for the async executor.
44    web::block(move || create_challenge_blocking(&context)).await?
45}
46
47fn create_challenge_blocking(context: &ChallengeContext) -> anyhow::Result<Challenge> {
48    let config = server_runtime_config();
49    let counter_range = if config.test_mode {
50        TEST_MODE_COUNTER_RANGE
51    } else {
52        COUNTER_RANGE
53    };
54    let mut data = BTreeMap::new();
55    data.insert(
56        "purpose".to_string(),
57        serde_json::to_value(context.purpose)?,
58    );
59    let secret = config.altcha_hmac_secret.expose_secret().to_string();
60    Ok(altcha::create_challenge(CreateChallengeOptions {
61        algorithm: ALGORITHM.to_string(),
62        cost: config.altcha_cost,
63        counter: Some(rand::rng().random_range(counter_range)),
64        data: Some(data),
65        expires_at: Some((Utc::now() + Duration::minutes(CHALLENGE_LIFETIME_MINUTES)).timestamp()),
66        hmac_signature_secret: Some(secret.clone()),
67        hmac_key_signature_secret: Some(key_signature_secret(&secret)),
68        ..Default::default()
69    })?)
70}
71
72/// Signs the expected key, so a solution is checked with one HMAC instead of a key derivation.
73fn key_signature_secret(secret: &str) -> String {
74    format!("{secret}:key")
75}
76
77/// Accepts `payload`, the widget's base64 solution, once, for `purpose`. Errors with a generic
78/// `BadRequest` when it is malformed, for another purpose, expired, wrongly solved or already
79/// redeemed. Accepts the empty solution the plain-HTTP development host sends in test mode.
80///
81/// Call it before the handler looks at anything else, so a refusal reveals nothing about accounts.
82pub async fn verify_and_spend(
83    conn: &mut PgConnection,
84    payload: &str,
85    purpose: ChallengePurpose,
86) -> Result<(), ControllerError> {
87    if payload.is_empty() && server_runtime_config().test_mode {
88        // Browsers refuse to solve on the plain-HTTP development host; the form sends nothing there.
89        return Ok(());
90    }
91    let submitted = decode(payload).ok_or_else(refusal)?;
92    let signature = submitted.challenge.signature.clone().ok_or_else(refusal)?;
93    let challenge = verified_challenge(submitted).ok_or_else(refusal)?;
94    let signed_purpose = challenge
95        .parameters
96        .data
97        .as_ref()
98        .and_then(|data| data.get("purpose"))
99        .and_then(|value| serde_json::from_value::<ChallengePurpose>(value.clone()).ok());
100    if signed_purpose != Some(purpose) {
101        return Err(refusal());
102    }
103    let expires_at = challenge
104        .parameters
105        .expires_at
106        .and_then(|timestamp| DateTime::from_timestamp(timestamp, 0))
107        .ok_or_else(refusal)?;
108    delete_expired_occasionally(conn).await?;
109    if !altcha_spent_challenges::try_spend(conn, &signature, expires_at).await? {
110        return Err(refusal());
111    }
112    Ok(())
113}
114
115fn decode(payload: &str) -> Option<Payload> {
116    let bytes = base64::engine::general_purpose::STANDARD
117        .decode(payload)
118        .ok()?;
119    serde_json::from_slice(&bytes).ok()
120}
121
122/// Checks the signature, the expiry and the solution. Cheap: the key is checked against its
123/// signature rather than derived again.
124fn verified_challenge(payload: Payload) -> Option<Challenge> {
125    let secret = server_runtime_config().altcha_hmac_secret.expose_secret();
126    let mut options =
127        VerifySolutionOptions::new(&payload.challenge, &payload.solution, secret.to_string());
128    options.hmac_key_signature_secret = Some(key_signature_secret(secret));
129    let result = altcha::verify_solution(options).ok()?;
130    result.verified.then_some(payload.challenge)
131}
132
133fn refusal() -> ControllerError {
134    bad_request_with_reason(
135        BadRequestReason::FormVerificationFailed,
136        "Form verification failed. Please try again.".to_string(),
137    )
138}
139
140async fn delete_expired_occasionally(conn: &mut PgConnection) -> ModelResult<()> {
141    if !rand::rng().random_bool(EXPIRED_CLEANUP_PROBABILITY) {
142        return Ok(());
143    }
144    let deleted = altcha_spent_challenges::delete_expired(conn).await?;
145    info!("Deleted {deleted} expired proof-of-work challenges");
146    Ok(())
147}
