1//! Copying a user's name and email to TMC, which keeps its own copy of both and signs the same
2//! account in by the email. Callers run these last inside their transaction: TMC cannot be rolled
3//! back, so a TMC failure must be the one that rolls the change back.
4
5use headless_lms_utils::services::tmc::{TmcClient, tmc_sync_target};
6
7use crate::prelude::*;
8
9/// Copies a changed name and/or email to TMC in one write, so TMC takes both or neither. A no-op
10/// for an account without a TMC id. Send only an email the user entered on the platform: an
11/// unchanged one may be older than TMC's.
12///
13/// Fails with `BadRequest` when TMC refuses the address, e.g. because another account uses it.
14pub async fn copy_profile_to_tmc(
15    upstream_id: Option<i32>,
16    name: Option<(String, String)>,
17    email: Option<String>,
18    tmc_client: &TmcClient,
19    app_conf: &ApplicationConfiguration,
20) -> Result<(), ControllerError> {
21    let Some(upstream_id) = tmc_sync_target(upstream_id, app_conf) else {
22        return Ok(());
23    };
24    let sends_email = email.is_some();
25    tmc_client
26        .sync_user_profile(upstream_id, name, email)
27        .await
28        .map_err(|error| {
29            if sends_email && error.is_tmc_refused_address() {
30                controller_err!(
31                    BadRequest,
32                    "This email address cannot be used. It may belong to another account."
33                        .to_string(),
34                    error
35                )
36            } else {
37                controller_err!(
38                    InternalServerError,
39                    "Failed to update the account on TMC".to_string(),
40                    error
41                )
42            }
43        })
44}
