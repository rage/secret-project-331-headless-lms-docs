68impl ServerRuntimeConfig {
69    /// Loads runtime configuration from environment variables.
70    pub fn try_from_env() -> anyhow::Result<Self> {
71        let app_conf = ApplicationConfiguration::try_from_env()?;
72        let test_mode = app_conf.test_mode;
73        let file_store_use_google_cloud_storage =
74            ProgramConfig::bool_flag("FILE_STORE_USE_GOOGLE_CLOUD_STORAGE");
75        let google_cloud_storage_bucket_name = if file_store_use_google_cloud_storage {
76            Some(
77                env::var("GOOGLE_CLOUD_STORAGE_BUCKET_NAME")
78                    .context("GOOGLE_CLOUD_STORAGE_BUCKET_NAME must be defined when FILE_STORE_USE_GOOGLE_CLOUD_STORAGE is enabled")?,
79            )
80        } else {
81            None
82        };
83        let ratelimit_protection_safe_api_key = match env::var("RATELIMIT_PROTECTION_SAFE_API_KEY")
84        {
85            Ok(value) => value,
86            Err(_) if cfg!(debug_assertions) || test_mode => "mock-api-key".to_string(),
87            Err(_) => {
88                anyhow::bail!("RATELIMIT_PROTECTION_SAFE_API_KEY must be defined in production")
89            }
90        };
91
92        Ok(Self {
93            database_url: SecretString::new(
94                env::var("DATABASE_URL")
95                    .context("DATABASE_URL must be defined")?
96                    .into(),
97            ),
98            oauth_application_id: env::var("OAUTH_APPLICATION_ID")
99                .context("OAUTH_APPLICATION_ID must be defined")?,
100            oauth_secret: SecretString::new(
101                env::var("OAUTH_SECRET")
102                    .context("OAUTH_SECRET must be defined")?
103                    .into(),
104            ),
105            icu4x_postcard_path: env::var("ICU4X_POSTCARD_PATH")
106                .context("ICU4X_POSTCARD_PATH must be defined")?,
107            redis_url: SecretString::new(
108                env::var("REDIS_URL")
109                    .context("REDIS_URL must be defined")?
110                    .into(),
111            ),
112            mock_suotar_redis_db_index: env::var("MOCK_SUOTAR_REDIS_DB_INDEX")
113                .ok()
114                .and_then(|value| value.trim().parse().ok())
115                .unwrap_or(2),
116            jwt_password: SecretString::new(
117                env::var("JWT_PASSWORD")
118                    .context("JWT_PASSWORD must be defined")?
119                    .into(),
120            ),
121            private_cookie_key: SecretString::new(
122                env::var("PRIVATE_COOKIE_KEY")
123                    .context("PRIVATE_COOKIE_KEY must be defined")?
124                    .into(),
125            ),
126            allow_no_https_for_development: ProgramConfig::bool_flag(
127                "ALLOW_NO_HTTPS_FOR_DEVELOPMENT",
128            ),
129            host: env::var("HOST").unwrap_or_else(|_| "0.0.0.0".to_string()),
130            port: env::var("PORT").unwrap_or_else(|_| "3001".to_string()),
131            file_store: FileStoreRuntimeConfig {
132                use_google_cloud_storage: file_store_use_google_cloud_storage,
133                google_cloud_storage_bucket_name,
134            },
135            tmc_server_secret_for_communicating_to_secret_project: SecretString::new(
136                env::var("TMC_SERVER_SECRET_FOR_COMMUNICATING_TO_SECRET_PROJECT")
137                    .context(
138                        "TMC_SERVER_SECRET_FOR_COMMUNICATING_TO_SECRET_PROJECT must be defined",
139                    )?
140                    .into(),
141            ),
142            ratelimit_protection_safe_api_key: SecretString::new(
143                ratelimit_protection_safe_api_key.into(),
144            ),
145            pod_namespace: env::var("POD_NAMESPACE").unwrap_or_else(|_| "default".to_string()),
146            app_conf,
147            test_mode,
148        })
149    }
150}

152/// Sets global runtime configuration for request-path consumers.
153pub fn set_server_runtime_config(config: ServerRuntimeConfig) -> anyhow::Result<()> {
154    SERVER_RUNTIME_CONFIG.set(config).map_err(|_| {
155        anyhow::anyhow!(
156            "SERVER_RUNTIME_CONFIG was already initialized in set_server_runtime_config"
157        )
158    })
159}

161/// Returns global runtime configuration loaded during startup.
162pub fn server_runtime_config() -> &'static ServerRuntimeConfig {
163    SERVER_RUNTIME_CONFIG
164        .get()
165        .expect("SERVER_RUNTIME_CONFIG has not been initialized; call set_server_runtime_config before request handling")
166}

198                &runtime_config.file_store,
199                &runtime_config.app_conf.base_url,

201            .await,
202            app_conf: runtime_config.app_conf.clone(),
203            redis_url: runtime_config.redis_url.clone(),
204            mock_suotar_redis_db_index: runtime_config.mock_suotar_redis_db_index,

206            tmc_client: TmcClient::new(

214    pub async fn build(self) -> anyhow::Result<ServerConfig> {
215        let json_config = web::JsonConfig::default().limit(2_097_152).error_handler(
216            |err, _req| -> actix_web::Error {
217                info!("Bad request: {}", &err);
218                let body = format!("{{\"title\": \"Bad Request\", \"message\": \"{}\"}}", &err);
219                // create custom error response
220                let response = HttpResponse::with_body(StatusCode::BAD_REQUEST, body.boxed());
221                InternalError::from_response(err, response).into()
222            },
223        );
224        let json_config = Data::new(json_config);
225
226        let payload_config = PayloadConfig::default().limit(2_097_152);
227        let payload_config = Data::new(payload_config);
228
229        let db_pool = PgPoolOptions::new()
230            .max_connections(15)
231            .min_connections(5)
232            .connect(self.database_url.expose_secret())
233            .await?;
234        crate::domain::internal_error_reporting::init_error_reporting(db_pool.clone());
235        let db_pool = Data::new(db_pool);
236
237        let oauth_client: OAuthClient = BasicClient::new(ClientId::new(self.oauth_application_id))
238            .set_client_secret(ClientSecret::new(
239                self.oauth_secret.expose_secret().to_string(),
240            ))
241            .set_auth_uri(AuthUrl::from_url(self.auth_url.clone()))
242            .set_token_uri(TokenUrl::from_url(self.token_url.clone()));
243        let oauth_client = Data::new(oauth_client);
244
245        let icu4x_blob = Icu4xBlob::new(&self.icu4x_postcard_path)?;
246        let icu4x_blob = Data::new(icu4x_blob);
247
248        let app_conf = Data::new(self.app_conf);
249
250        let ip_to_country_mapper = IpToCountryMapper::new(&app_conf)?;
251        let ip_to_country_mapper = Data::new(ip_to_country_mapper);
252
253        let cache = Cache::new(self.redis_url.expose_secret())?;
254        let cache = Data::new(cache);
255
256        // Only the mock's own routes need this, and they exist only under the same flag.
257        let mock_suotar_store = if app_conf.test_suotar {
258            warn!(
259                "MOCK SUOTAR ENABLED - credit registrations are simulated and are NOT recorded in Sisu"
260            );
261            Some(Data::new(MockSuotarStore::new(
262                self.redis_url.expose_secret(),
263                self.mock_suotar_redis_db_index,
264            )?))
265        } else {
266            None
267        };
268
269        let jwt_key = JwtKey::new(&self.jwt_password)?;
270        let jwt_key = Data::new(jwt_key);
271
272        let tmc_client = Data::new(self.tmc_client);
273
274        let sisu_client = Data::new(self.sisu_client);
275
276        // Built here rather than in `from_runtime_config` because auditing every call needs the pool.
277        let suotar_client = Data::new(SuotarClient::new(
278            &app_conf.suotar_configuration,
279            Arc::new(PgSuotarCallAudit::new(db_pool.as_ref().clone())),
280        ));
281
282        let config = ServerConfig {
283            json_config,
284            db_pool,
285            oauth_client,
286            icu4x_blob,
287            ip_to_country_mapper,
288            file_store: self.file_store,
289            app_conf,
290            jwt_key,
291            cache,
292            payload_config,
293            tmc_client,
294            sisu_client,
295            suotar_client,
296            mock_suotar_store,
297        };
298        Ok(config)
299    }

320/// Common configuration that is used by both production and testing.
321pub fn configure(config: &mut ServiceConfig, server_config: ServerConfig) {
322    let ServerConfig {
323        json_config,
324        db_pool,
325        oauth_client,
326        icu4x_blob,
327        ip_to_country_mapper,
328        file_store,
329        app_conf,
330        jwt_key,
331        cache,
332        payload_config,
333        tmc_client,
334        sisu_client,
335        suotar_client,
336        mock_suotar_store,
337    } = server_config;
338    let api_rate_limit_config = RateLimit::global_api_rate_limit_config(app_conf.test_mode);
339    // turns file_store from `dyn FileStore + Send + Sync` to `dyn FileStore` to match controllers
340    // Not using Data::new for file_store to avoid double wrapping it in a arc
341    let file_store = Data::from(file_store as Arc<dyn FileStore>);
342    if let Some(mock_suotar_store) = mock_suotar_store {
343        config.app_data(mock_suotar_store);
344    }
345    config
346        .app_data(payload_config)
347        .app_data(json_config)
348        .app_data(db_pool)
349        .app_data(oauth_client)
350        .app_data(icu4x_blob)
351        .app_data(ip_to_country_mapper)
352        .app_data(file_store)
353        .app_data(app_conf.clone())
354        .app_data(jwt_key)
355        .app_data(cache)
356        .app_data(tmc_client)
357        .app_data(sisu_client)
358        .app_data(suotar_client)
359        .service(
360            web::scope("/api/v0")
361                .wrap(RateLimit::new(api_rate_limit_config))
362                .wrap(RequestSpan)
363                .configure(|c| crate::controllers::configure_controllers(c, app_conf)),
364        );
