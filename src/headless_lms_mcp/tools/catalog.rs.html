14/// What dispatch needs of a tool besides its handler.
15#[derive(Debug, Clone)]
16pub struct ToolDefinition {
17    pub name: &'static str,
18    /// The token needs this scope to see and call the tool.
19    pub scope: McpScope,
20    /// Whether calls without `skill_version` are refused; see [`crate::hints::skill_missing`].
21    pub requires_skill: bool,
22    /// The arguments the tool promotes to `Mcp-Param-*` headers, with the header suffix.
23    pub header_params: Vec<(String, String)>,
24    /// The tool as `tools/list` lists it.
25    pub listing: rmcp::model::Tool,
26}
27
28impl<S> Tool<S> {
29    /// Lets agents without the skill call the tool.
30    fn usable_without_skill(mut self) -> Self {
31        self.definition.requires_skill = false;
32        self
33    }
34
35    /// Asks Claude Code to load the tool up front instead of on demand.
36    fn always_loaded(mut self) -> Self {
37        self.definition
38            .listing
39            .meta
40            .get_or_insert_with(MetaObject::default)
41            .insert("anthropic/alwaysLoad".into(), json!(true));
42        self
43    }
44
45    /// Narrows the shared op enum of the input schema to the ops `target` takes.
46    fn taking_ops(mut self, target: PatchTarget) -> Self {
47        let schema = Arc::make_mut(&mut self.definition.listing.input_schema);
48        if let Some(op) = schema
49            .get_mut("properties")
50            .and_then(|p| p.pointer_mut("/ops/items/properties/op"))
51        {
52            op["enum"] = json!(target.allowed_ops());
53        }
54        self
55    }
56}

74/// How every patch tool stages, appended to their descriptions.
75macro_rules! staging_note {
76    () => {
77        " Every write stages into your open changeset for the course; nothing is live until publish_changeset or the teacher publishes it. Ops apply in order, all or nothing; invariants are checked at the end. dry_run: true stages nothing and returns the diff."
78    };
79}

81fn define<S, H: ToolHandler<S> + 'static>(
82    handler: H,
83    name: &'static str,
84    description: &'static str,
85    annotations: ToolAnnotations,
86    scope: McpScope,
87) -> Tool<S> {
88    let mut input_schema = input_schema::<H::Args>();
89    // Dispatch takes `skill_version` out before the handler parses its arguments.
90    if let Value::Object(properties) = input_schema
91        .entry("properties")
92        .or_insert_with(|| json!({}))
93    {
94        properties.insert(
95            "skill_version".into(),
96            json!({"type": "string", "description": format!("Version from the course-material skill's VERSION file, or \"{SKILL_DECLINED}\" once the teacher has said no to installing the skill. Without either, calls are refused.")}),
97        );
98    }
99    let header_params = header_params(&input_schema);
100    let title = annotations.title.clone().unwrap_or_default();
101    let mut listing =
102        rmcp::model::Tool::new(name, description, Arc::new(input_schema)).with_title(title);
103    if annotations.read_only_hint == Some(true) {
104        listing = listing.with_meta(MetaObject::from(Map::from_iter([(
105            "anthropic/maxResultSizeChars".to_string(),
106            json!(100_000),
107        )])));
108    }
109    Tool {
110        definition: ToolDefinition {
111            name,
112            scope,
113            requires_skill: true,
114            header_params,
115            listing: listing.with_annotations(annotations),
116        },
117        handler: Box::new(handler),
118    }
119}
120
121fn header_params(input_schema: &Map<String, Value>) -> Vec<(String, String)> {
122    input_schema
123        .get("properties")
124        .and_then(Value::as_object)
125        .map(|properties| {
126            properties
127                .iter()
128                .filter_map(|(name, schema)| {
129                    Some((
130                        name.clone(),
131                        schema.get("x-mcp-header")?.as_str()?.to_string(),
132                    ))
133                })
134                .collect()
135        })
136        .unwrap_or_default()
137}

434    const MAX_SCHEMA_BYTES: usize = 4_500;
435
436    fn walk(value: &Value, visit: &mut dyn FnMut(&str, &Value)) {
437        if let Value::Object(map) = value {
438            for (key, child) in map {
439                visit(key, child);
440                walk(child, visit);
441            }
442        } else if let Value::Array(items) = value {
443            items.iter().for_each(|i| walk(i, visit));
444        }
445    }

447    #[test]
448    fn every_input_schema_is_small_and_flat() {
449        for tool in definitions() {
450            let bytes = serde_json::to_vec(&*tool.listing.input_schema)
451                .unwrap()
452                .len();
453            assert!(
454                bytes < MAX_SCHEMA_BYTES,
455                "{} input schema is {bytes} bytes",
456                tool.name
457            );
458            let schema = Value::Object((*tool.listing.input_schema).clone());
459            walk(&schema, &mut |key, value| {
460                assert!(
461                    !matches!(key, "oneOf" | "anyOf" | "allOf" | "$ref" | "$defs"),
462                    "{} uses {key}",
463                    tool.name
464                );
465                if key == "type" {
466                    assert!(value.is_string(), "{} has a type array", tool.name);
467                }
468            });
469            assert_eq!(tool.listing.input_schema["type"], json!("object"));
470        }
471    }

531        const NONDETERMINISTIC_READS: [&str; 1] = ["read_public_course"];
532        for tool in definitions() {
533            let annotations = &serde_json::to_value(&tool.listing).unwrap()["annotations"];
534            let is_read_only = annotations["readOnlyHint"] == json!(true);
535            let is_named_as_read = tool.name.starts_with("get_")
536                || tool.name.starts_with("list_")
537                || tool.name.starts_with("read_")
538                || tool.name == "guide";

540            if is_read_only {
541                assert_eq!(
542                    annotations["idempotentHint"],
543                    json!(!NONDETERMINISTIC_READS.contains(&tool.name)),
544                    "{}",
545                    tool.name
546                );
547            }
548        }
