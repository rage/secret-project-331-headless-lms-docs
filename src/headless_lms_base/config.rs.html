7/// Reads a boolean env var where missing values default to false.
8pub fn bool_env_false_by_default(key: &str) -> bool {
9    match env::var(key) {
10        Ok(value) => {
11            let normalized = value.trim().to_ascii_lowercase();
12            !matches!(
13                normalized.as_str(),
14                "" | "false" | "0" | "no" | "off" | "disabled"
15            )
16        }
17        Err(_) => false,
18    }
19}

21/// Reads an env var, treating a blank value the same as an unset one.
22fn non_empty_env(key: &str) -> Option<String> {
23    match env::var(key) {
24        Ok(value) if !value.trim().is_empty() => Some(value.trim().to_string()),
25        _ => None,
26    }
27}

29/// Reads an integer env var, falling back to `default` when unset, blank or unparseable.
30fn i64_env_or(key: &str, default: i64) -> i64 {
31    non_empty_env(key)
32        .and_then(|value| value.parse().ok())
33        .unwrap_or(default)
34}

36/// Parses a [`Url::join`] base, adding the trailing slash one needs: joining onto a base without it
37/// replaces the base's last path segment instead of extending it.
38///
39/// On the parsed path, never the raw string: a slash appended to the string lands inside any query
40/// or fragment the value carries, and makes trailing whitespace the parser would have trimmed
41/// interior, where it percent-encodes instead.
42fn parse_join_base(url: &str) -> Result<Url, url::ParseError> {
43    let mut url = Url::parse(url)?;
44    if !url.path().ends_with('/') {
45        url.set_path(&format!("{}/", url.path()));
46    }
47    Ok(url)
48}

272impl AzureChatbotConfiguration {
273    /// Attempts to create an AzureChatbotConfiguration from environment variables.
274    ///
275    /// Needs all three of `AZURE_CHATBOT_API_KEY`, `AZURE_CHATBOT_API_ENDPOINT` and
276    /// `AZURE_PROJECT_NAME`. Any of them missing gives `Ok(None)` rather than an error, so a caller
277    /// that requires a chatbot has to reject the `None` itself. Errors only on a set value that
278    /// fails to parse.
279    pub fn try_from_env() -> anyhow::Result<Option<Self>> {
280        let api_key = env::var("AZURE_CHATBOT_API_KEY").ok();
281        let api_endpoint = env::var("AZURE_CHATBOT_API_ENDPOINT").ok();
282        let project_name = env::var("AZURE_PROJECT_NAME").ok();
283
284        if let (Some(api_key), Some(api_endpoint), Some(project_name)) =
285            (api_key, api_endpoint, project_name)
286        {
287            Ok(Some(Self::from_values(
288                api_key,
289                &api_endpoint,
290                project_name,
291            )?))
292        } else {
293            Ok(None)
294        }
295    }
296
297    /// Pure so the join base can be tested without touching process env.
298    fn from_values(
299        api_key: String,
300        api_endpoint: &str,
301        project_name: String,
302    ) -> anyhow::Result<Self> {
303        Ok(Self {
304            api_key: SecretString::new(api_key.into()),
305            api_base: parse_join_base(api_endpoint)
306                .context("Invalid URL in AZURE_CHATBOT_API_ENDPOINT")?,
307            project_name,
308        })
309    }
310

312        Ok(self.api_base.join(&format!(
313            "api/projects/{}/openai/v1/responses",
314            self.project_name

419        ))

484#[derive(Clone)]
485pub struct OAuthServerConfiguration {
486    pub rsa_public_key: String,
487    /// RSA private key (PEM) used to sign OAuth/OIDC tokens. Secret: zeroized on drop,
488    /// redacted from `Debug`; only exposed when handed to the signing key builder.
489    pub rsa_private_key: SecretString,
490    /// Secret key for HMAC-SHA-256 hashing of OAuth tokens (access tokens, refresh tokens, auth codes).
491    pub oauth_token_hmac_key: SecretString,
492    /// Secret key for signing DPoP nonces (HMAC).
493    pub dpop_nonce_key: Arc<SecretBox<String>>,
494}
495
496impl PartialEq for OAuthServerConfiguration {
497    fn eq(&self, other: &Self) -> bool {
498        self.rsa_public_key == other.rsa_public_key
499            && self.rsa_private_key.expose_secret() == other.rsa_private_key.expose_secret()
500            && self.oauth_token_hmac_key.expose_secret()
501                == other.oauth_token_hmac_key.expose_secret()
502            && self.dpop_nonce_key.expose_secret() == other.dpop_nonce_key.expose_secret()
503    }
504}
505
506impl OAuthServerConfiguration {
507    /// Attempts to create an OAuthServerConfiguration.
508    /// Return `Ok(Some(OAuthConfiguration))` if all configurations are set.
509    /// Return `Err` if any is not set.
510    pub fn try_from_env() -> anyhow::Result<Self> {
511        let rsa_public_key =
512            env::var("OAUTH_RSA_PUBLIC_PEM").context("OAUTH_RSA_PUBLIC_KEY must be defined")?;
513        let rsa_private_key = SecretString::new(
514            env::var("OAUTH_RSA_PRIVATE_PEM")
515                .context("OAUTH_RSA_PRIVATE_KEY must be defined")?
516                .into(),
517        );
518        let oauth_token_hmac_key = SecretString::new(
519            env::var("OAUTH_TOKEN_HMAC_KEY")
520                .context("OAUTH_TOKEN_HMAC_KEY must be defined")?
521                .into(),
522        );
523        let dpop_nonce_key = Arc::new(SecretBox::new(Box::new(
524            env::var("OAUTH_DPOP_NONCE_KEY").context("OAUTH_DPOP_NONCE_KEY must be defined")?,
525        )));
526
527        Ok(Self {
528            rsa_public_key,
529            rsa_private_key,
530            oauth_token_hmac_key,
531            dpop_nonce_key,
532        })
533    }
