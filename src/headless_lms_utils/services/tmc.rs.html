164fn format_tmc_errors(errors: &serde_json::Value) -> String {
165    let mut error_messages = Vec::new();
166
167    if let Some(error_array) = errors.as_array() {
168        for error_msg in error_array {
169            match error_msg.as_str() {
170                Some(msg) => error_messages.push(msg.to_string()),
171                None => error_messages.push(error_msg.to_string()),
172            }
173        }
174    } else if let Some(errors_obj) = errors.as_object() {
175        for (field, field_errors) in errors_obj {
176            if let Some(error_array) = field_errors.as_array() {
177                for error_msg in error_array {
178                    if let Some(msg) = error_msg.as_str() {
179                        let field_name = match field.as_str() {
180                            "login" => "username",
181                            _ => field,
182                        };
183                        error_messages.push(format!("{}: {}", field_name, msg));
184                    }
185                }
186            } else if let Some(msg) = field_errors.as_str() {
187                error_messages.push(format!("{}: {}", field, msg));
188            }
189        }
190    }
191
192    if error_messages.is_empty() {
193        errors.to_string()
194    } else {
195        error_messages.join(", ")
196    }
197}

199fn parse_tmc_error_response(error_text: &str, status: Option<reqwest::StatusCode>) -> String {
200    if let Ok(error_json) = serde_json::from_str::<serde_json::Value>(error_text) {
201        // `code` is TMC's stable, machine-readable reason; the human text beside it is neither.
202        let code = error_json.get("code").and_then(|c| c.as_str());
203        let detail = match error_json.get("errors") {
204            Some(errors) => Some(format_tmc_errors(errors)),
205            None => error_json
206                .get("message")
207                .and_then(|m| m.as_str())
208                .map(str::to_string),
209        };
210        match (code, detail) {
211            (Some(code), Some(detail)) => return format!("{code}: {detail}"),
212            (Some(code), None) => return code.to_string(),
213            (None, Some(detail)) => return detail,
214            (None, None) => {}
215        }
216    }
217
218    if let Some(status) = status {
219        format!("Request failed with status {}: {}", status, error_text)
220    } else {
221        format!("Request failed: {}", error_text)
222    }
223}

225/// The error for a non-2xx TMC response, logged without the URL's query string.
226fn status_error(url: &str, status: reqwest::StatusCode, error_text: &str) -> UtilError {
227    if let Ok(parsed) = reqwest::Url::parse(url) {
228        let redacted = format!(
229            "{}{}",
230            parsed.origin().unicode_serialization(),
231            parsed.path()
232        );
233        tracing::warn!("Request to {} failed with status {}", redacted, status);
234    } else {
235        tracing::warn!("Request failed with status {}", status);
236    }
237    tracing::debug!("Response body: {}", error_text);
238
239    let error_message = parse_tmc_error_response(error_text, Some(status));
240
241    // A statusful failure, unlike a send failure, which stays `TmcHttpError`.
242    util_err!(TmcHttpStatusError(status.as_u16()), error_message)
243}

245async fn error_body(res: reqwest::Response) -> String {

251/// Whether a 404 body is the `{"errors": [...]}` TMC's API v8 renders for a missing record, as
252/// opposed to the 404 of a route this TMC deployment does not have.
253fn is_tmc_record_not_found(error_text: &str) -> bool {
254    serde_json::from_str::<serde_json::Value>(error_text)
255        .is_ok_and(|body| body.get("errors").is_some_and(serde_json::Value::is_array))
256}
257
258/// The outcome of a non-2xx answer to an account deletion. Deployments predating the deletion
259/// contract answer a missing user with a 404, but only one carrying TMC's own error body means the
260/// user is gone; any other 404 says nothing about the account.
261fn unsuccessful_account_deletion(
262    url: &str,
263    status: reqwest::StatusCode,
264    error_text: &str,
265) -> UtilResult<TmcAccountDeletion> {
266    if status == reqwest::StatusCode::NOT_FOUND && is_tmc_record_not_found(error_text) {
267        Ok(TmcAccountDeletion::AlreadyDeleted)
268    } else {
269        Err(status_error(url, status, error_text))
270    }
271}

273impl TmcClient {
274    fn check_if_tmc_error_response(response_text: &str) -> Option<UtilError> {
275        if let Ok(error_json) = serde_json::from_str::<serde_json::Value>(response_text)
276            && (error_json.get("errors").is_some()
277                || error_json.get("success") == Some(&serde_json::Value::Bool(false)))
278        {
279            let error_message = parse_tmc_error_response(response_text, None);
280            return Some(UtilError::new(
281                UtilErrorType::TmcErrorResponse,
282                error_message,
283                None,
284            ));
285        }
286        None
287    }

289    async fn deserialize_response_with_tmc_error_check<T: serde::de::DeserializeOwned>(
290        &self,
291        response: reqwest::Response,
292        error_context: &str,
293    ) -> UtilResult<T> {
294        let response_text = response.text().await.map_err(|e| {
295            UtilError::new(
296                UtilErrorType::DeserializationError,
297                format!("Failed to read TMC response body: {}", error_context),
298                Some(e.into()),
299            )
300        })?;
301
302        serde_json::from_str(&response_text).map_err(|e| {
303            if let Some(tmc_error) = Self::check_if_tmc_error_response(&response_text) {
304                tmc_error
305            } else {
306                UtilError::new(
307                    UtilErrorType::DeserializationError,
308                    format!("Failed to parse {}: {}", error_context, e),
309                    Some(e.into()),
310                )
311            }
312        })
313    }

315    pub fn new(
316        admin_access_token: SecretString,
317        ratelimit_api_key: SecretString,
318    ) -> UtilResult<Self> {
319        if admin_access_token.expose_secret().trim().is_empty() {
320            return Err(UtilError::new(
321                UtilErrorType::Other,
322                "TMC_ACCESS_TOKEN cannot be empty".to_string(),
323                None,
324            ));
325        }
326        if ratelimit_api_key.expose_secret().trim().is_empty() {
327            return Err(UtilError::new(
328                UtilErrorType::Other,
329                "RATELIMIT_PROTECTION_SAFE_API_KEY cannot be empty".to_string(),
330                None,
331            ));
332        }
333
334        let client = http_client().map_err(|e| {
335            UtilError::new(
336                UtilErrorType::Other,
337                "Failed to build HTTP client".to_string(),
338                Some(e.into()),
339            )
340        })?;
341
342        Ok(Self {
343            client,
344            admin_access_token,
345            ratelimit_api_key,
346        })
347    }

349    async fn request_with_headers(
350        &self,
351        method: reqwest::Method,
352        url: &str,
353        tmc_request_auth: TMCRequestAuth,
354        body: Option<serde_json::Value>,
355    ) -> UtilResult<reqwest::Response> {
356        self.request_with_headers_and_timeout(method, url, tmc_request_auth, body, None)
357            .await
358    }

360    /// Like [`Self::request_with_headers_and_timeout`], but hands back non-2xx responses as is.
361    async fn send_request(
362        &self,
363        method: reqwest::Method,
364        url: &str,
365        tmc_request_auth: TMCRequestAuth,
366        body: Option<serde_json::Value>,
367        timeout: Option<Duration>,
368    ) -> UtilResult<reqwest::Response> {
369        let mut builder = self
370            .client
371            .request(method, url)
372            .header(
373                "RATELIMIT-PROTECTION-SAFE-API-KEY",
374                self.ratelimit_api_key.expose_secret(),
375            )
376            .header(reqwest::header::CONTENT_TYPE, "application/json")
377            .header(reqwest::header::ACCEPT, "application/json");
378
379        let access_token = match tmc_request_auth {
380            TMCRequestAuth::UseAdminToken => Some(&self.admin_access_token),
381            TMCRequestAuth::UseUserToken(ref token) => Some(token),
382            TMCRequestAuth::NoAuth => None,
383        };
384
385        if let Some(token) = access_token {
386            builder = builder.bearer_auth(token.expose_secret());
387        }
388
389        if let Some(json_body) = body {
390            builder = builder.json(&json_body);
391        }
392
393        if let Some(timeout) = timeout {
394            builder = builder.timeout(timeout);
395        }
396
397        builder.send().await.map_err(|e| {
398            UtilError::new(
399                UtilErrorType::TmcHttpError,
400                "Failed to send HTTP request".to_string(),
401                Some(e.into()),
402            )
403        })
404    }

406    /// `timeout` overrides the shared client's default for this one request.
407    async fn request_with_headers_and_timeout(
408        &self,
409        method: reqwest::Method,
410        url: &str,
411        tmc_request_auth: TMCRequestAuth,
412        body: Option<serde_json::Value>,
413        timeout: Option<Duration>,
414    ) -> UtilResult<reqwest::Response> {
415        let res = self
416            .send_request(method, url, tmc_request_auth, body, timeout)
417            .await?;
418        if res.status().is_success() {
419            Ok(res)
420        } else {
421            let status = res.status();
422            let error_text = error_body(res).await;
423            Err(status_error(url, status, &error_text))
424        }
425    }

427    /// Makes TMC hold the given name and email for user `user_upstream_id` in one write, so TMC
428    /// applies both or neither; `None` leaves that field as it is. The email is written only when
429    /// TMC's differs (ignoring case and surrounding whitespace). TMC signs users in by the email,
430    /// and answers 400 or 422 to one it refuses, such as one another account already uses.
431    pub async fn sync_user_profile(
432        &self,
433        user_upstream_id: i32,
434        name: Option<(String, String)>,
435        email: Option<String>,
436    ) -> UtilResult<()> {
437        let email = match email {
438            Some(email) => {
439                let tmc_email = self
440                    .get_user_from_tmc_mooc_fi_by_tmc_access_token_and_upstream_id(
441                        &user_upstream_id,
442                    )
443                    .await?
444                    .email;
445                (!tmc_email.trim().eq_ignore_ascii_case(email.trim())).then_some(email)
446            }
447            None => None,
448        };
449        let mut payload = serde_json::Map::new();
450        if let Some((first_name, last_name)) = name {
451            payload.insert(
452                "user_field".to_string(),
453                json!({ "first_name": first_name, "last_name": last_name }),
454            );
455        }
456        if let Some(email) = email {
457            payload.insert("user".to_string(), json!({ "email": email }));
458        }
459        if payload.is_empty() {
460            return Ok(());
461        }
462        let url = format!("{}/{}", TMC_API_URL, user_upstream_id);
463        self.request_with_headers(
464            reqwest::Method::PUT,
465            &url,
466            TMCRequestAuth::UseAdminToken,
467            Some(serde_json::Value::Object(payload)),
468        )
469        .await
470        .map(|_| ())
471    }

563    /// Deletes the upstream tmc.mooc.fi account behind `user_upstream_id`.
564    ///
565    /// Every failure is an error carrying TMC's HTTP status, which is what separates a retryable
566    /// outage from a rejection the caller must not retry. A user TMC does not know comes back as
567    /// [`TmcAccountDeletion::AlreadyDeleted`], never as an error.
568    pub async fn delete_user_from_tmc(
569        &self,
570        user_upstream_id: String,
571    ) -> UtilResult<TmcAccountDeletion> {
572        let url = format!("{}/{}", TMC_API_URL, user_upstream_id);
573
574        let res = self
575            .send_request(
576                reqwest::Method::DELETE,
577                &url,
578                TMCRequestAuth::UseAdminToken,
579                None,
580                Some(DELETE_ACCOUNT_TIMEOUT),
581            )
582            .await?;
583        let status = res.status();
584        if !status.is_success() {
585            return unsuccessful_account_deletion(&url, status, &error_body(res).await);
586        }
587
588        let body: TmcDeleteAccountResponse = self
589            .deserialize_response_with_tmc_error_check(res, "delete response from TMC")
590            .await?;
591
592        if !body.success {
593            return Err(util_err!(
594                TmcErrorResponse,
595                "TMC answered the account deletion with success: false".to_string()
596            ));
597        }
598
599        if body.already_deleted.unwrap_or(false) {
600            Ok(TmcAccountDeletion::AlreadyDeleted)
601        } else {
602            Ok(TmcAccountDeletion::Deleted)
603        }
604    }

606    /// Every course the user has progress on in tmc.mooc.fi. A user TMC does not know has none.
607    pub async fn get_user_course_progress(
608        &self,
609        user_upstream_id: i32,
610        timeout: Duration,
611    ) -> UtilResult<Vec<TmcCourseProgress>> {
612        let url = format!("{}/{}/course_progress", TMC_API_URL, user_upstream_id);
613        let res = self
614            .send_request(
615                reqwest::Method::GET,
616                &url,
617                TMCRequestAuth::UseAdminToken,
618                None,
619                Some(timeout),
620            )
621            .await?;
622        let status = res.status();
623        if !status.is_success() {
624            let error_text = error_body(res).await;
625            if status == reqwest::StatusCode::NOT_FOUND && is_tmc_record_not_found(&error_text) {
626                return Ok(Vec::new());
627            }
628            return Err(status_error(&url, status, &error_text));
629        }
630        let body: TmcCourseProgressResponse = self
631            .deserialize_response_with_tmc_error_check(res, "course progress from TMC")
632            .await?;
633        Ok(body.courses)
634    }

636    pub async fn get_user_from_tmc_mooc_fi_by_tmc_access_token(
637        &self,
638        tmc_access_token: &SecretString,
639    ) -> UtilResult<TMCUser> {
640        info!("Getting user details from tmc.mooc.fi");
641
642        let res = self
643            .request_with_headers(
644                reqwest::Method::GET,
645                &format!("{}/current?show_user_fields=1", TMC_API_URL),
646                TMCRequestAuth::UseUserToken(tmc_access_token.clone()),
647                None,
648            )
649            .await?;
650
651        debug!("Received response from TMC, parsing user data");
652        let tmc_user: TMCUser = self
653            .deserialize_response_with_tmc_error_check(res, "current user from TMC by access token")
654            .await?;
655
656        debug!(
657            "Creating or fetching user with TMC id {} and mooc.fi UUID {}",
658            tmc_user.id,
659            tmc_user
660                .courses_mooc_fi_user_id
661                .map(|uuid| uuid.to_string())
662                .unwrap_or_else(|| "None (will generate new UUID)".to_string())
663        );
664        Ok(tmc_user)
665    }

667    /// Doorkeeper's view of `tmc_access_token`. TMC answers 401 for a revoked or expired token and
668    /// for one it did not issue itself, such as a courses.mooc.fi token it accepts elsewhere.
669    pub async fn get_access_token_info(
670        &self,
671        tmc_access_token: &SecretString,
672    ) -> UtilResult<TmcAccessTokenInfo> {
673        let res = self
674            .request_with_headers(
675                reqwest::Method::GET,
676                TMC_TOKEN_INFO_URL,
677                TMCRequestAuth::UseUserToken(tmc_access_token.clone()),
678                None,
679            )
680            .await?;
681        self.deserialize_response_with_tmc_error_check(res, "access token info from TMC")
682            .await
683    }
