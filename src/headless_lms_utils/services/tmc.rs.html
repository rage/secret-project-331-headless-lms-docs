116fn http_client() -> reqwest::Result<Client> {
117    crate::http::base_client_builder()
118        .timeout(Duration::from_secs(15))
119        // Requests carry the admin token or a student's token, which a redirect would forward.

124fn format_tmc_errors(errors: &serde_json::Value) -> String {
125    let mut error_messages = Vec::new();
126
127    if let Some(error_array) = errors.as_array() {
128        for error_msg in error_array {
129            match error_msg.as_str() {
130                Some(msg) => error_messages.push(msg.to_string()),
131                None => error_messages.push(error_msg.to_string()),
132            }
133        }
134    } else if let Some(errors_obj) = errors.as_object() {
135        for (field, field_errors) in errors_obj {
136            if let Some(error_array) = field_errors.as_array() {
137                for error_msg in error_array {
138                    if let Some(msg) = error_msg.as_str() {
139                        let field_name = match field.as_str() {
140                            "login" => "username",
141                            _ => field,
142                        };
143                        error_messages.push(format!("{}: {}", field_name, msg));
144                    }
145                }
146            } else if let Some(msg) = field_errors.as_str() {
147                error_messages.push(format!("{}: {}", field, msg));
148            }
149        }
150    }
151
152    if error_messages.is_empty() {
153        errors.to_string()
154    } else {
155        error_messages.join(", ")
156    }
157}

159fn parse_tmc_error_response(error_text: &str, status: Option<reqwest::StatusCode>) -> String {
160    if let Ok(error_json) = serde_json::from_str::<serde_json::Value>(error_text) {
161        // `code` is TMC's stable, machine-readable reason; the human text beside it is neither.
162        let code = error_json.get("code").and_then(|c| c.as_str());
163        let detail = match error_json.get("errors") {
164            Some(errors) => Some(format_tmc_errors(errors)),
165            None => error_json
166                .get("message")
167                .and_then(|m| m.as_str())
168                .map(str::to_string),
169        };
170        match (code, detail) {
171            (Some(code), Some(detail)) => return format!("{code}: {detail}"),
172            (Some(code), None) => return code.to_string(),
173            (None, Some(detail)) => return detail,
174            (None, None) => {}
175        }
176    }
177
178    if let Some(status) = status {
179        format!("Request failed with status {}: {}", status, error_text)
180    } else {
181        format!("Request failed: {}", error_text)
182    }
183}

185impl TmcClient {
186    fn check_if_tmc_error_response(response_text: &str) -> Option<UtilError> {
187        if let Ok(error_json) = serde_json::from_str::<serde_json::Value>(response_text)
188            && (error_json.get("errors").is_some()
189                || error_json.get("success") == Some(&serde_json::Value::Bool(false)))
190        {
191            let error_message = parse_tmc_error_response(response_text, None);
192            return Some(UtilError::new(
193                UtilErrorType::TmcErrorResponse,
194                error_message,
195                None,
196            ));
197        }
198        None
199    }
200
201    async fn deserialize_response_with_tmc_error_check<T: serde::de::DeserializeOwned>(
202        &self,
203        response: reqwest::Response,
204        error_context: &str,
205    ) -> UtilResult<T> {
206        let response_text = response.text().await.map_err(|e| {
207            UtilError::new(
208                UtilErrorType::DeserializationError,
209                format!("Failed to read TMC response body: {}", error_context),
210                Some(e.into()),
211            )
212        })?;
213
214        serde_json::from_str(&response_text).map_err(|e| {
215            if let Some(tmc_error) = Self::check_if_tmc_error_response(&response_text) {
216                tmc_error
217            } else {
218                UtilError::new(
219                    UtilErrorType::DeserializationError,
220                    format!("Failed to parse {}: {}", error_context, e),
221                    Some(e.into()),
222                )
223            }
224        })
225    }

227    pub fn new(
228        admin_access_token: SecretString,
229        ratelimit_api_key: SecretString,
230    ) -> UtilResult<Self> {
231        if admin_access_token.expose_secret().trim().is_empty() {
232            return Err(UtilError::new(
233                UtilErrorType::Other,
234                "TMC_ACCESS_TOKEN cannot be empty".to_string(),
235                None,
236            ));
237        }
238        if ratelimit_api_key.expose_secret().trim().is_empty() {
239            return Err(UtilError::new(
240                UtilErrorType::Other,
241                "RATELIMIT_PROTECTION_SAFE_API_KEY cannot be empty".to_string(),
242                None,
243            ));
244        }
245
246        let client = http_client().map_err(|e| {
247            UtilError::new(
248                UtilErrorType::Other,
249                "Failed to build HTTP client".to_string(),
250                Some(e.into()),
251            )
252        })?;
253
254        Ok(Self {
255            client,
256            admin_access_token,
257            ratelimit_api_key,
258        })
259    }

261    async fn request_with_headers(
262        &self,
263        method: reqwest::Method,
264        url: &str,
265        tmc_request_auth: TMCRequestAuth,
266        body: Option<serde_json::Value>,
267    ) -> UtilResult<reqwest::Response> {
268        self.request_with_headers_and_timeout(method, url, tmc_request_auth, body, None)
269            .await
270    }
271
272    /// `timeout` overrides the shared client's default for this one request.
273    async fn request_with_headers_and_timeout(
274        &self,
275        method: reqwest::Method,
276        url: &str,
277        tmc_request_auth: TMCRequestAuth,
278        body: Option<serde_json::Value>,
279        timeout: Option<Duration>,
280    ) -> UtilResult<reqwest::Response> {
281        let mut builder = self
282            .client
283            .request(method, url)
284            .header(
285                "RATELIMIT-PROTECTION-SAFE-API-KEY",
286                self.ratelimit_api_key.expose_secret(),
287            )
288            .header(reqwest::header::CONTENT_TYPE, "application/json")
289            .header(reqwest::header::ACCEPT, "application/json");
290
291        let access_token = match tmc_request_auth {
292            TMCRequestAuth::UseAdminToken => Some(&self.admin_access_token),
293            TMCRequestAuth::UseUserToken(ref token) => Some(token),
294            TMCRequestAuth::NoAuth => None,
295        };
296
297        if let Some(token) = access_token {
298            builder = builder.bearer_auth(token.expose_secret());
299        }
300
301        if let Some(json_body) = body {
302            builder = builder.json(&json_body);
303        }
304
305        if let Some(timeout) = timeout {
306            builder = builder.timeout(timeout);
307        }
308
309        let res = builder.send().await.map_err(|e| {
310            UtilError::new(
311                UtilErrorType::TmcHttpError,
312                "Failed to send HTTP request".to_string(),
313                Some(e.into()),
314            )
315        })?;
316
317        if res.status().is_success() {
318            Ok(res)
319        } else {
320            let status = res.status();
321            let error_text = res
322                .text()
323                .await
324                .unwrap_or_else(|e| format!("(Failed to read error body: {e})"));
325
326            if let Ok(parsed) = reqwest::Url::parse(url) {
327                let redacted = format!(
328                    "{}{}",
329                    parsed.origin().unicode_serialization(),
330                    parsed.path()
331                );
332                tracing::warn!("Request to {} failed with status {}", redacted, status);
333            } else {
334                tracing::warn!("Request failed with status {}", status);
335            }
336            tracing::debug!("Response body: {}", error_text);
337
338            let error_message = parse_tmc_error_response(&error_text, Some(status));
339
340            // A statusful failure, unlike the send failure above, which stays `TmcHttpError`.
341            Err(util_err!(
342                TmcHttpStatusError(status.as_u16()),
343                error_message
344            ))
345        }
346    }

348    pub async fn update_user_information(
349        &self,
350        first_name: String,
351        last_name: String,
352        email: Option<String>,
353        user_upstream_id: String,
354    ) -> UtilResult<()> {
355        let mut user_obj = serde_json::Map::new();
356        let mut user_field_obj = serde_json::Map::new();
357
358        if let Some(email) = email {
359            user_obj.insert("email".to_string(), serde_json::Value::String(email));
360        }
361
362        user_field_obj.insert(
363            "first_name".to_string(),
364            serde_json::Value::String(first_name),
365        );
366        user_field_obj.insert(
367            "last_name".to_string(),
368            serde_json::Value::String(last_name),
369        );
370
371        let mut payload = serde_json::Map::new();
372
373        if !user_obj.is_empty() {
374            payload.insert("user".to_string(), serde_json::Value::Object(user_obj));
375        }
376
377        payload.insert(
378            "user_field".to_string(),
379            serde_json::Value::Object(user_field_obj),
380        );
381
382        let payload_value = serde_json::Value::Object(payload);
383
384        let url = format!("{}/{}", TMC_API_URL, user_upstream_id);
385
386        self.request_with_headers(
387            reqwest::Method::PUT,
388            &url,
389            TMCRequestAuth::UseAdminToken,
390            Some(payload_value),
391        )
392        .await
393        .map(|_| ())
394    }

434                &url,

436                Some(payload),

438            .await?;
439
440        let body: TMCUserResponse = self
441            .deserialize_response_with_tmc_error_check(response, "TMC user response")
442            .await?;
443        Ok(body.id)

499    /// Deletes the upstream tmc.mooc.fi account behind `user_upstream_id`.
500    ///
501    /// Every failure is an error carrying TMC's HTTP status, which is what separates a retryable
502    /// outage from a rejection the caller must not retry. An unknown user can arrive either way:
503    /// as [`TmcAccountDeletion::AlreadyDeleted`], or, from deployments predating the deletion
504    /// contract, as a [`UtilErrorType::TmcHttpStatusError`] carrying 404.
505    pub async fn delete_user_from_tmc(
506        &self,
507        user_upstream_id: String,
508    ) -> UtilResult<TmcAccountDeletion> {
509        let url = format!("{}/{}", TMC_API_URL, user_upstream_id);
510
511        let res = self
512            .request_with_headers_and_timeout(
513                reqwest::Method::DELETE,
514                &url,
515                TMCRequestAuth::UseAdminToken,
516                None,
517                Some(DELETE_ACCOUNT_TIMEOUT),
518            )
519            .await?;
520
521        let body: TmcDeleteAccountResponse = self
522            .deserialize_response_with_tmc_error_check(res, "delete response from TMC")
523            .await?;
524
525        if !body.success {
526            return Err(util_err!(
527                TmcErrorResponse,
528                "TMC answered the account deletion with success: false".to_string()
529            ));
530        }
531
532        if body.already_deleted.unwrap_or(false) {
533            Ok(TmcAccountDeletion::AlreadyDeleted)
534        } else {
535            Ok(TmcAccountDeletion::Deleted)
536        }
537    }

539    pub async fn get_user_from_tmc_mooc_fi_by_tmc_access_token(
540        &self,
541        tmc_access_token: &SecretString,
542    ) -> UtilResult<TMCUser> {
543        info!("Getting user details from tmc.mooc.fi");
544
545        let res = self
546            .request_with_headers(
547                reqwest::Method::GET,
548                &format!("{}/current?show_user_fields=1", TMC_API_URL),
549                TMCRequestAuth::UseUserToken(tmc_access_token.clone()),
550                None,
551            )
552            .await?;
553
554        debug!("Received response from TMC, parsing user data");
555        let tmc_user: TMCUser = self
556            .deserialize_response_with_tmc_error_check(res, "current user from TMC by access token")
557            .await?;
558
559        debug!(
560            "Creating or fetching user with TMC id {} and mooc.fi UUID {}",
561            tmc_user.id,
562            tmc_user
563                .courses_mooc_fi_user_id
564                .map(|uuid| uuid.to_string())
565                .unwrap_or_else(|| "None (will generate new UUID)".to_string())
566        );
567        Ok(tmc_user)
568    }
