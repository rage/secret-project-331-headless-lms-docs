71impl SuotarEndpoint {
72    /// Relative to the configured base url, which ends in `/`.
73    pub fn path(self) -> &'static str {
74        match self {
75            Self::ResolvePersons => "persons/resolve-by-student-numbers",
76            Self::ResolveEnrolments => "enrolments/resolve",
77            Self::ImportAttainments => "attainments/import",
78            Self::VerifyAttainments => "attainments/verify",
79            Self::ListByCourse => "enrolments/list-by-course",
80            Self::ValidateCourseCodes => "course-codes/validate",
81        }
82    }
83
84    /// Suotar's own per-endpoint limits; a larger batch is refused whole.
85    pub fn max_batch_size(self) -> usize {
86        match self {
87            Self::ResolvePersons
88            | Self::ResolveEnrolments
89            | Self::VerifyAttainments
90            | Self::ValidateCourseCodes => 1000,
91            Self::ImportAttainments => 100,
92            Self::ListByCourse => 50,
93        }
94    }
95
96    /// How long one worker call may take before it is abandoned. On `import` an abandoned call
97    /// leaves the whole batch uncertain, so it is sized above Suotar's worst case for a full batch;
98    /// the read-only endpoints sit below theirs, as a timeout there costs only a retry.
99    pub const fn request_timeout(self) -> Duration {
100        let minutes = match self {
101            Self::ImportAttainments => 60,
102            Self::VerifyAttainments => 25,
103            Self::ResolveEnrolments | Self::ListByCourse => 20,
104            Self::ResolvePersons | Self::ValidateCourseCodes => 10,
105        };
106        Duration::from_secs(minutes * 60)
107    }
108
109    /// An item this endpoint never answered is uncertain, not retryable: re-sending it can put a
110    /// second attainment on a real transcript.
111    pub fn creates_attainments(self) -> bool {
112        matches!(self, Self::ImportAttainments)
113    }
114}

451/// Importer dates arrive as `YYYY-MM-DD` or as an instant (Sisu's UTC midnight), and an instant
452/// means its Helsinki date, the zone Suotar compares days in. Anything else reads as absent.
453fn lenient_date<'de, D: Deserializer<'de>>(deserializer: D) -> Result<Option<NaiveDate>, D::Error> {
454    let value = Option::<serde_json::Value>::deserialize(deserializer)?;
455    Ok(value
456        .as_ref()
457        .and_then(serde_json::Value::as_str)
458        .and_then(|text| {
459            NaiveDate::parse_from_str(text, "%Y-%m-%d")
460                .ok()
461                .or_else(|| {
462                    DateTime::parse_from_rfc3339(text)
463                        .ok()
464                        .map(|instant| helsinki_date(instant.with_timezone(&Utc)))
465                })
466        }))
467}

469/// A value that does not read as `T` reads as absent.
470fn lenient<'de, D: Deserializer<'de>, T: DeserializeOwned>(
471    deserializer: D,
472) -> Result<Option<T>, D::Error> {
473    let value = Option::<serde_json::Value>::deserialize(deserializer)?;
474    Ok(value.and_then(|value| serde_json::from_value(value).ok()))
475}

477/// A Sisu id that should be a string but, for at least `gradeId`, has arrived as a bare JSON number.
478/// A strict `String` field would fail to parse and drop the whole record in [`readable_elements`],
479/// silencing whatever check depends on it. Coerces either shape into a `String`; anything else reads
480/// as absent.
481fn lenient_id<'de, D: Deserializer<'de>>(deserializer: D) -> Result<Option<String>, D::Error> {
482    let value = Option::<serde_json::Value>::deserialize(deserializer)?;
483    Ok(value.and_then(|value| match value {
484        serde_json::Value::String(text) => Some(text),
485        serde_json::Value::Number(number) => Some(number.to_string()),
486        _ => None,
487    }))
488}

490/// An RFC 3339 instant, or Sisu's zoneless local date-time read as UTC, which is close enough to
491/// order enrolments by. Anything else reads as absent.
492fn lenient_instant<'de, D: Deserializer<'de>>(
493    deserializer: D,
494) -> Result<Option<DateTime<Utc>>, D::Error> {
495    let value = Option::<serde_json::Value>::deserialize(deserializer)?;
496    Ok(value
497        .as_ref()
498        .and_then(serde_json::Value::as_str)
499        .and_then(|text| {
500            DateTime::parse_from_rfc3339(text)
501                .map(|instant| instant.with_timezone(&Utc))
502                .ok()
503                .or_else(|| {
504                    chrono::NaiveDateTime::parse_from_str(text, "%Y-%m-%dT%H:%M:%S%.f")
505                        .ok()
506                        .map(|local| local.and_utc())
507                })
508        }))
509}

511/// Keeps the elements that parse and logs how many did not.
512fn readable_elements<'de, D, T>(deserializer: D) -> Result<Vec<T>, D::Error>
513where
514    D: Deserializer<'de>,
515    T: DeserializeOwned,
516{
517    let values = Vec::<serde_json::Value>::deserialize(deserializer)?;
518    let total = values.len();
519    let readable: Vec<T> = values
520        .into_iter()
521        .filter_map(|value| serde_json::from_value(value).ok())
522        .collect();
523    if readable.len() < total {
524        warn!(
525            unreadable = total - readable.len(),
526            total, "Suotar answered with list elements that could not be read; skipping them"
527        );
528    }
529    Ok(readable)
530}

545#[derive(Debug, Clone, PartialEq, Serialize)]
546#[serde(rename_all = "camelCase")]
547pub struct SuotarResponseItem<R> {
548    pub request_item_id: String,
549    pub status: SuotarItemStatus,
550    /// A string, not an enum: Suotar may add codes, and a strict enum would take the pipeline down
551    /// the day it does.
552    pub code: String,
553    /// Also present on the error items that carry one: `sisuTimeout`, `duplicateRequestItem`,
554    /// `submissionPending`, `enrolmentNotFound` and `enrolmentNotAccepted`. An error item's result
555    /// that cannot be read reads as `None`; an ok item's makes the whole item unreadable.
556    #[serde(skip_serializing_if = "Option::is_none")]
557    pub result: Option<R>,
558    #[serde(skip_serializing_if = "Option::is_none")]
559    pub error: Option<SuotarItemError>,
560}

562impl<'de, R: DeserializeOwned> Deserialize<'de> for SuotarResponseItem<R> {
563    fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
564        #[derive(Deserialize)]
565        #[serde(rename_all = "camelCase")]
566        struct Wire {
567            request_item_id: String,
568            status: SuotarItemStatus,
569            code: String,
570            #[serde(default)]
571            result: Option<serde_json::Value>,
572            #[serde(default)]
573            error: Option<SuotarItemError>,
574        }
575
576        let wire = Wire::deserialize(deserializer)?;
577        let result = match wire.result.map(serde_json::from_value::<R>) {
578            None => None,
579            Some(Ok(result)) => Some(result),
580            // The code is the answer on an error item, and the result only adds to it.
581            Some(Err(_)) if wire.status == SuotarItemStatus::Error => {
582                warn!(
583                    code = %wire.code,
584                    "Suotar answered with a result that could not be read; ignoring the result"
585                );
586                None
587            }
588            // Not the serde message: it quotes the offending value, which may be personal data.
589            Some(Err(_)) => {
590                return Err(serde::de::Error::custom(format!(
591                    "the result of a `{}` item could not be read",
592                    wire.code
593                )));
594            }
595        };
596        Ok(Self {
597            request_item_id: wire.request_item_id,
598            status: wire.status,
599            code: wire.code,
600            result,
601            error: wire.error,
602        })
603    }
604}

626/// How a call to Suotar failed at the request level. Per-item failures are not errors: they come
627/// back inside a successful batch response.
628#[derive(Debug, Clone, Copy, PartialEq, Eq)]
629pub enum SuotarErrorVariant {
630    /// Our credentials. Loud, and never attributed to the rows in the batch.
631    Unauthorized,
632    /// Our request. Loud, and never attributed to the rows in the batch.
633    MalformedRequest,
634    /// Another 4xx carrying the documented `{ error: { code, message } }` body.
635    RequestLevelError,
636    /// Suotar's 503 `serviceTemporarilyUnavailable`: an importer lookup failed before anything was
637    /// written or sent.
638    ServiceTemporarilyUnavailable,
639    ServerError,
640    /// The connection itself failed, so the request provably never arrived.
641    TransportNotDelivered,
642    /// The request left and the answer did not arrive. A timeout is this, not the above.
643    TransportUnknown,
644    /// Suotar answered, and the answer was not a batch response.
645    Deserialization,
646}
647
648/// A Suotar call that got no batch response. [`SuotarError::variant`] and [`SuotarError::was_sent`]
649/// are what a caller decides by; the message and the source are for the logs and the audit trail.
650#[derive(Debug)]
651pub struct SuotarError {
652    pub variant: SuotarErrorVariant,
653    /// `false` for a request refused before it left, which says nothing about Suotar.
654    pub was_sent: bool,
655    error: UtilError,
656}
657
658impl SuotarError {
659    #[track_caller]
660    fn new(variant: SuotarErrorVariant, message: impl Into<String>) -> Self {
661        Self {
662            variant,
663            was_sent: true,
664            error: util_err!(SuotarClientError, message.into()),
665        }
666    }
667
668    #[track_caller]
669    fn caused_by(
670        variant: SuotarErrorVariant,
671        message: impl Into<String>,
672        source: impl Into<anyhow::Error>,
673    ) -> Self {
674        Self {
675            variant,
676            was_sent: true,
677            error: util_err!(SuotarClientError, message.into(), source.into()),
678        }
679    }
680
681    fn unsent(self) -> Self {
682        Self {
683            was_sent: false,
684            ..self
685        }
686    }
687
688    /// The error's text without the variant prefix `Display` adds.
689    pub fn message(&self) -> &str {
690        self.error.message()
691    }
692}

697    }

823    /// Sends one batch to `E`'s endpoint. An empty batch is not sent.
824    pub async fn post<E: BatchEndpoint>(
825        &self,
826        context: SuotarCallContext,
827        items: Vec<E::Item>,
828    ) -> Result<SuotarBatchResponse<E::Result>, SuotarError> {
829        let endpoint = E::ENDPOINT;
830        if items.is_empty() {
831            return Ok(empty_batch_response(endpoint));
832        }
833        // Serialized once: the audited body and the wire body must be byte-for-byte the same.
834        let not_encoded = |error| {
835            SuotarError::caused_by(
836                SuotarErrorVariant::TransportNotDelivered,
837                format!("Could not encode a {} request", endpoint.path()),
838                error,
839            )
840            .unsent()
841        };
842        let request_body = serde_json::to_value(&items).map_err(not_encoded)?;
843        let encoded = serde_json::to_vec(&request_body).map_err(not_encoded)?;
844        // Before the pre-flight checks, so a refused batch still leaves an audit row to diagnose.
845        let call_id = self
846            .audit
847            .started(SuotarCallStarted {
848                endpoint,
849                request_item_count: items.len(),
850                worker_name: context.worker_name,
851                credit_registration_ids: context.credit_registration_ids,
852                request_item_ids: items
853                    .iter()
854                    .map(|item| item.request_item_id().to_string())
855                    .collect(),
856                started_at: Utc::now(),
857                request_body,
858            })
859            .await;
860        if call_id.is_none() {
861            error!(
862                endpoint = endpoint.path(),
863                "Could not write a suotar_api_calls row; sending the call unaudited"
864            );
865        }
866
867        let sent_ids = match check_batch(endpoint, &items) {
868            Ok(sent_ids) => sent_ids,
869            Err(error) => return self.refused(call_id, error).await,
870        };
871        if encoded.len() > MAX_REQUEST_BODY_BYTES {
872            return self
873                .refused(
874                    call_id,
875                    SuotarError::new(SuotarErrorVariant::MalformedRequest, format!(
876                            "A {} request of {} items encodes to {} bytes, over the {MAX_REQUEST_BODY_BYTES} byte limit.",
877                            endpoint.path(),
878                            sent_ids.len(),
879                            encoded.len()
880                        )),
881                )
882                .await;
883        }
884
885        let url = match self.api_base_url.join(endpoint.path()) {
886            Ok(url) => url,
887            Err(error) => {
888                return self
889                    .refused(
890                        call_id,
891                        SuotarError::caused_by(
892                            SuotarErrorVariant::TransportNotDelivered,
893                            format!("Could not build the Suotar {} url", endpoint.path()),
894                            error,
895                        ),
896                    )
897                    .await;
898            }
899        };
900        if !self.allow_http && url.scheme() != "https" {
901            return self
902                .refused(
903                    call_id,
904                    SuotarError::new(
905                        SuotarErrorVariant::TransportNotDelivered,
906                        format!(
907                            "Refusing to send a {} request over plain http",
908                            endpoint.path()
909                        ),
910                    ),
911                )
912                .await;
913        }
914        let clock = Instant::now();
915        #[cfg(any(test, feature = "test-support"))]
916        let http_client = if self.allow_http {
917            &*SUOTAR_PLAIN_HTTP_CLIENT
918        } else {
919            &*SUOTAR_HTTP_CLIENT
920        };
921        // `allow_http` comes only from `TEST_MODE` here, where the shared client allows http too.
922        #[cfg(not(any(test, feature = "test-support")))]
923        let http_client = &*SUOTAR_HTTP_CLIENT;
924        let mut request = http_client
925            .post(url)
926            .timeout(
927                context
928                    .request_timeout
929                    .unwrap_or_else(|| endpoint.request_timeout()),
930            )
931            .header(AUTHORIZATION, self.authorization.expose_secret())
932            .header(CONTENT_TYPE, "application/json");
933        if let Some(call_id) = call_id {
934            request = request.header(CORRELATION_ID_HEADER, call_id.to_string());
935        }
936
937        let (mut outcome, finished) = self
938            .exchange(endpoint, request, encoded, sent_ids, clock)
939            .await;
940        if let Ok(response) = &mut outcome {
941            response.call_id = call_id;
942        }
943        if let Some(call_id) = call_id {
944            self.audit.finished(call_id, finished).await;
945        }
946        outcome
947    }

949    /// Finishes the `suotar_api_calls` row of a call that never left, tagged so the health rules can
950    /// tell our own refusal from Suotar being unreachable.
951    async fn refused<R>(
952        &self,
953        call_id: Option<Uuid>,
954        error: SuotarError,
955    ) -> Result<SuotarBatchResponse<R>, SuotarError> {
956        if let Some(call_id) = call_id {
957            self.audit
958                .finished(
959                    call_id,
960                    SuotarCallFinished {
961                        request_level_error_code: Some(REFUSED_BEFORE_SENDING_CODE.to_string()),
962                        error_message: Some(error.message().to_string()),
963                        ..SuotarCallFinished::default()
964                    },
965                )
966                .await;
967        }
968        Err(error.unsent())
969    }

971    /// Returns the audit record alongside the result: only this function knows the status, the
972    /// duration and the request-level code, and the row needs all three.
973    async fn exchange<R: DeserializeOwned>(
974        &self,
975        endpoint: SuotarEndpoint,
976        request: reqwest::RequestBuilder,
977        body: Vec<u8>,
978        sent_ids: Vec<String>,
979        clock: Instant,
980    ) -> Exchanged<R> {
981        let response = match request.body(body).send().await {
982            Ok(response) => response,
983            Err(error) => {
984                return failed(
985                    SuotarError::caused_by(
986                        transport_variant(&error),
987                        format!("Request to Suotar {} failed", endpoint.path()),
988                        error,
989                    ),
990                    None,
991                    clock.elapsed(),
992                    None,
993                    None,
994                );
995            }
996        };
997
998        let http_status = response.status().as_u16();
999        let text = match response.text().await {
1000            Ok(text) => text,
1001            Err(error) => {
1002                return failed(
1003                    SuotarError::caused_by(
1004                        transport_variant(&error),
1005                        format!(
1006                            "Reading the Suotar {} response body failed",
1007                            endpoint.path()
1008                        ),
1009                        error,
1010                    ),
1011                    Some(http_status),
1012                    clock.elapsed(),
1013                    None,
1014                    None,
1015                );
1016            }
1017        };
1018        let duration = clock.elapsed();
1019
1020        if !(200..300).contains(&http_status) {
1021            let detail = serde_json::from_str::<RequestLevelErrorBody>(&text)
1022                .ok()
1023                .map(|parsed| parsed.error);
1024            let code = detail
1025                .as_ref()
1026                .and_then(RequestLevelErrorDetail::code)
1027                .map(str::to_string);
1028            let error = request_level_error(endpoint, http_status, detail.as_ref());
1029            return failed(
1030                error,
1031                Some(http_status),
1032                duration,
1033                code,
1034                Some(Arc::new(body_for_audit(&text))),
1035            );
1036        }
1037
1038        let raw_response: Arc<serde_json::Value> = match serde_json::from_str(&text) {
1039            Ok(value) => Arc::new(value),
1040            Err(error) => {
1041                return failed(
1042                    SuotarError::caused_by(
1043                        SuotarErrorVariant::Deserialization,
1044                        format!(
1045                            "Suotar {} answered {http_status} with a body that is not JSON",
1046                            endpoint.path()
1047                        ),
1048                        error,
1049                    ),
1050                    Some(http_status),
1051                    duration,
1052                    None,
1053                    Some(Arc::new(body_for_audit(&text))),
1054                );
1055            }
1056        };
1057        let Some(array) = raw_response.as_array() else {
1058            return failed(
1059                SuotarError::new(
1060                    SuotarErrorVariant::Deserialization,
1061                    format!(
1062                        "Suotar {} answered {http_status} with a body that is not a batch response",
1063                        endpoint.path()
1064                    ),
1065                ),
1066                Some(http_status),
1067                duration,
1068                None,
1069                Some(raw_response),
1070            );
1071        };
1072        // Item by item, so one malformed entry costs only its own row: parsing the array as a whole
1073        // would park every other row of the batch as unanswered too. `reconcile` then reports the
1074        // dropped ids as missing, which is what an item we cannot read amounts to.
1075        let items: Vec<SuotarResponseItem<R>> = array
1076            .iter()
1077            .filter_map(|item| match SuotarResponseItem::<R>::deserialize(item) {
1078                Ok(parsed) => Some(parsed),
1079                // Not the serde message: it quotes the offending value, which may be personal data.
1080                Err(_) => {
1081                    error!(
1082                        endpoint = endpoint.path(),
1083                        "Suotar answered with an item that could not be read; treating it as unanswered"
1084                    );
1085                    None
1086                }
1087            })
1088            .collect();
1089
1090        let response = reconcile(endpoint, sent_ids, items, duration, raw_response);
1091        let finished = SuotarCallFinished {
1092            http_status: Some(http_status),
1093            duration,
1094            succeeded: true,
1095            ok_item_count: response
1096                .items
1097                .iter()
1098                .filter(|item| item.status == SuotarItemStatus::Ok)
1099                .count(),
1100            error_item_count: response
1101                .items
1102                .iter()
1103                .filter(|item| item.status == SuotarItemStatus::Error)
1104                .count(),
1105            request_level_error_code: None,
1106            error_message: None,
1107            response_body: Some(Arc::clone(&response.raw_response)),
1108        };
1109        (Ok(response), finished)
1110    }
1111}

1143/// Refuses our own bugs before a request goes out; both would come back as a request-level error
1144/// rejecting the whole batch.
1145fn check_batch<T: SuotarRequestItem>(
1146    endpoint: SuotarEndpoint,
1147    items: &[T],
1148) -> Result<Vec<String>, SuotarError> {
1149    if items.len() > endpoint.max_batch_size() {
1150        return Err(SuotarError::new(
1151            SuotarErrorVariant::MalformedRequest,
1152            format!(
1153                "A {} request carries {} items, over the batch size of {}.",
1154                endpoint.path(),
1155                items.len(),
1156                endpoint.max_batch_size()
1157            ),
1158        ));
1159    }
1160    let mut seen = HashSet::with_capacity(items.len());
1161    for item in items {
1162        if !seen.insert(item.request_item_id()) {
1163            return Err(SuotarError::new(
1164                SuotarErrorVariant::MalformedRequest,
1165                format!(
1166                    "A {} request repeats requestItemId `{}`.",
1167                    endpoint.path(),
1168                    item.request_item_id()
1169                ),
1170            ));
1171        }
1172    }
1173    Ok(items
1174        .iter()
1175        .map(|item| item.request_item_id().to_string())
1176        .collect())
1177}

1190/// The requestItemIds a response does not pair with what was sent.
1191#[derive(Debug, PartialEq)]
1192struct UnpairedItemIds {
1193    /// Sent, but answered by nothing. Unknown outcome on [`SuotarEndpoint::creates_attainments`].
1194    missing: Vec<String>,
1195    /// Answered, but never sent. Logged and otherwise ignored.
1196    unexpected: Vec<String>,
1197}
1198
1199fn unpaired_item_ids<R>(sent_ids: &[String], items: &[SuotarResponseItem<R>]) -> UnpairedItemIds {
1200    let sent: HashSet<&str> = sent_ids.iter().map(String::as_str).collect();
1201    let answered: HashSet<&str> = items
1202        .iter()
1203        .map(|item| item.request_item_id.as_str())
1204        .collect();
1205    UnpairedItemIds {
1206        missing: sent_ids
1207            .iter()
1208            .filter(|id| !answered.contains(id.as_str()))
1209            .cloned()
1210            .collect(),
1211        unexpected: items
1212            .iter()
1213            .filter(|item| !sent.contains(item.request_item_id.as_str()))
1214            .map(|item| item.request_item_id.clone())
1215            .collect(),
1216    }
1217}

1219/// Pairs the response against what was sent by `requestItemId`; order is not consulted.
1220fn reconcile<R>(
1221    endpoint: SuotarEndpoint,
1222    sent_ids: Vec<String>,
1223    items: Vec<SuotarResponseItem<R>>,
1224    duration: Duration,
1225    raw_response: Arc<serde_json::Value>,
1226) -> SuotarBatchResponse<R> {
1227    let unpaired = unpaired_item_ids(&sent_ids, &items);
1228    if !unpaired.unexpected.is_empty() {
1229        warn!(
1230            endpoint = endpoint.path(),
1231            unexpected = unpaired.unexpected.len(),
1232            "Suotar answered with requestItemIds that were not sent; ignoring them"
1233        );
1234    }
1235    if !unpaired.missing.is_empty() && endpoint.creates_attainments() {
1236        error!(
1237            endpoint = endpoint.path(),
1238            missing = unpaired.missing.len(),
1239            sent = sent_ids.len(),
1240            "Suotar left items unanswered; their attainments may or may not exist and must not be re-sent"
1241        );
1242    }

1253#[derive(Debug, Deserialize)]
1254struct RequestLevelErrorBody {
1255    error: RequestLevelErrorDetail,
1256}
1257
1258/// The envelope's `{code, message}`, or the bare string Suotar's fall-through route answers with.
1259#[derive(Debug, Deserialize)]
1260#[serde(untagged)]
1261enum RequestLevelErrorDetail {
1262    Coded { code: String, message: String },
1263    Bare(String),
1264}
1265
1266impl RequestLevelErrorDetail {
1267    fn code(&self) -> Option<&str> {
1268        match self {
1269            Self::Coded { code, .. } => Some(code),
1270            Self::Bare(_) => None,
1271        }
1272    }
1273}
1274
1275fn request_level_error(
1276    endpoint: SuotarEndpoint,
1277    http_status: u16,
1278    detail: Option<&RequestLevelErrorDetail>,
1279) -> SuotarError {
1280    let path = endpoint.path();
1281    // A path the moocfi router does not serve falls through to a route that refuses our key; the
1282    // key is fine and the base url is wrong.
1283    if let Some(RequestLevelErrorDetail::Bare(message)) = detail
1284        && http_status == 401
1285    {
1286        return SuotarError::new(
1287            SuotarErrorVariant::RequestLevelError,
1288            format!(
1289                "Suotar answered {path} with 401 `{message}`, which means the moocfi API does not serve that path. Check SUOTAR_API_BASE_URL."
1290            ),
1291        );
1292    }
1293    let variant = match (http_status, detail.and_then(RequestLevelErrorDetail::code)) {
1294        (401 | 403, _) | (_, Some("unauthorized")) => SuotarErrorVariant::Unauthorized,
1295        (413, _) | (_, Some("malformedRequest" | "requestTooLarge")) => {
1296            SuotarErrorVariant::MalformedRequest
1297        }
1298        (503, Some("serviceTemporarilyUnavailable")) => {
1299            SuotarErrorVariant::ServiceTemporarilyUnavailable
1300        }
1301        (500..=599, _) => SuotarErrorVariant::ServerError,
1302        _ => SuotarErrorVariant::RequestLevelError,
1303    };
1304    let detail = match detail {
1305        Some(RequestLevelErrorDetail::Coded { code, message }) => format!("`{code}`: {message}"),
1306        Some(RequestLevelErrorDetail::Bare(message)) => format!("`{message}`"),
1307        None => "no documented error body".to_string(),
1308    };
1309    SuotarError::new(
1310        variant,
1311        format!("Suotar {path} rejected the whole request with {http_status}, {detail}"),
1312    )
1313}

1315/// `is_connect` is the one case where the request provably never reached Suotar; everything else, a
1316/// timeout above all, may have been processed.
1317fn transport_variant(error: &reqwest::Error) -> SuotarErrorVariant {
1318    if error.is_connect() || error.is_builder() {
1319        SuotarErrorVariant::TransportNotDelivered
1320    } else {
1321        SuotarErrorVariant::TransportUnknown
1322    }
1323}
