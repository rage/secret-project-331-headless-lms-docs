1//! Removing personal data from free text kept or logged about an external service's answers.
2
3use std::sync::LazyLock;
4
5use regex::{Captures, Regex};
6
7/// Replaces a redacted value; the key is kept so the payload shape survives.
8pub const REDACTED: &str = "[redacted]";
9
10static EMAIL_RE: LazyLock<Regex> = LazyLock::new(|| {
11    Regex::new(r"[A-Za-z0-9._%+\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}").expect("hardcoded regex")
12});
13
14/// Student-number-shaped digit runs. The id shapes come first because `regex` has no lookaround:
15/// leftmost-first alternation is the only way to exclude digits inside an id. Word boundaries keep
16/// the digit branch off longer numbers such as millisecond timestamps.
17///
18/// Known cost of the `prefixed` branch: a student number hyphen-joined to a word,
19/// `person-012345678`, survives. A bare digit run, the shape Suotar's messages use, still goes.
20static STUDENT_NUMBER_RE: LazyLock<Regex> = LazyLock::new(|| {
21    Regex::new(
22        r"(?P<uuid>\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\b)|(?P<prefixed>\b[A-Za-z][A-Za-z0-9-]*-[0-9]{6,12}\b)|(?P<digits>\b[0-9]{6,12}\b)",
23    )
24    .expect("hardcoded regex")
25});
26
27/// Removes email addresses and student-number-shaped digit runs from `text`. Best effort, not an
28/// exhaustive PII filter: a name in free text is kept.
29pub fn scrub_text(text: &str) -> String {
30    let without_emails = EMAIL_RE.replace_all(text, REDACTED);
31    STUDENT_NUMBER_RE
32        .replace_all(&without_emails, |captures: &Captures| {
33            // The id branches exist to beat the digit branch to the match; put them back unchanged.
34            if captures.name("digits").is_some() {
35                REDACTED.to_string()
36            } else {
37                captures[0].to_string()
38            }
39        })
40        .into_owned()
41}
