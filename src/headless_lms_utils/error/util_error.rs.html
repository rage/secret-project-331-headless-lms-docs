45/// How a call to Suotar failed at the request level. Per-item failures are not errors: they come
46/// back inside a successful batch response.
47#[derive(Debug, Clone, Copy, PartialEq, Eq)]
48pub enum SuotarErrorVariant {
49    /// Our credentials. Loud, and never attributed to the rows in the batch.
50    Unauthorized,
51    /// Our request. Loud, and never attributed to the rows in the batch.
52    MalformedRequest,
53    /// Another 4xx carrying the documented `{ error: { code, message } }` body.
54    RequestLevelError,
55    ServerError,
56    /// The connection itself failed, so the request provably never arrived.
57    TransportNotDelivered,
58    /// The request left and the answer did not arrive. A timeout is this, not the above.
59    TransportUnknown,
60    /// Suotar answered, and the answer was not a batch response.
61    Deserialization,
62}
63
64impl SuotarErrorVariant {
65    /// Whether Suotar may have acted on the request. An import that may have landed must be
66    /// verified rather than re-sent, or a transcript gets a second attainment.
67    ///
68    /// A 4xx (`Unauthorized`, `MalformedRequest`, `RequestLevelError`) never reached Suotar's
69    /// business logic, so it is as resendable as a connection that never opened
70    /// (`TransportNotDelivered`); a genuine 5xx (`ServerError`), a response that never arrived
71    /// (`TransportUnknown`), or one that arrived malformed (`Deserialization`) all leave the
72    /// outcome unknown.
73    pub fn outcome_may_have_landed(self) -> bool {
74        !matches!(
75            self,
76            Self::Unauthorized
77                | Self::MalformedRequest
78                | Self::RequestLevelError
79                | Self::TransportNotDelivered
80        )
81    }
82}

84/**
85Error type used by all models. Used as the error type in [UtilError], which is used by all the controllers in the application.
86
87All the information in the error is meant to be seen by the user. The type of error is determined by the [UtilErrorType] enum, which is stored inside this struct.
88
89## Examples
90
91### Usage without source error
92
93```no_run
94# use headless_lms_utils::prelude::*;
95# fn random_function() -> UtilResult<()> {
96#    let erroneous_condition = 1 == 1;
97if erroneous_condition {
98    return Err(UtilError::new(
99        UtilErrorType::Other,
100        "File not found".to_string(),
101        None,
102    ));
103}
104# Ok(())
105# }
106```
107
108### Usage with a source error
109
110Used when calling a function that returns an error that cannot be automatically converted to an UtilError. (See `impl From<X>` implementations on this struct.)
111
112```no_run
113# use headless_lms_utils::prelude::*;
114# fn some_function_returning_an_error() -> UtilResult<()> {
115#    return Err(UtilError::new(
116#        UtilErrorType::Other,
117#        "File not found".to_string(),
118#        None,
119#    ));
120# }
121#
122# fn random_function() -> UtilResult<()> {
123#    let erroneous_condition = 1 == 1;
124some_function_returning_an_error().map_err(|original_error| {
125    UtilError::new(
126        UtilErrorType::Other,
127        "Library x failed to do y".to_string(),
128        Some(original_error.into()),
129    )
130})?;
131# Ok(())
132# }
133```
134*/
135pub struct UtilError {
136    error_type: <UtilError as BackendError>::ErrorType,
137    message: String,
138    /// Original error that caused this error.
139    source: Option<anyhow::Error>,
140    /// A trace of tokio tracing spans, generated automatically when the error is generated.
141    span_trace: Box<SpanTrace>,
142    /// Stack trace, generated automatically when the error is created.
143    backtrace: Box<Backtrace>,
144    /// Source location where the error was raised.
145    location: Option<&'static Location<'static>>,
146}
147
148impl std::error::Error for UtilError {
149    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
150        self.source
151            .as_deref()
152            .map(|e| e as &(dyn std::error::Error + 'static))
153    }
154
155    fn cause(&self) -> Option<&dyn std::error::Error> {
156        self.source()
157    }
158}

208    }

296/// Helper function for `.map_err()` chains to wrap any error as UtilError.
297///
298/// This function creates a closure that converts any error into a `UtilError`
299/// with the specified error type and message, including the original error as the source.
300///
301/// # Examples
302///
303/// ```ignore
304/// // Instead of:
305/// .map_err(|e| UtilError::new(UtilErrorType::Other, e.to_string(), Some(e.into())))?
306///
307/// // You can write:
308/// .map_err(as_util_error(UtilErrorType::Other, "Failed to process".to_string()))?
309/// ```
310pub fn as_util_error<E>(
311    error_type: UtilErrorType,
312    message: impl Into<String>,
313) -> impl FnOnce(E) -> UtilError
314where
315    E: Into<anyhow::Error>,
316{
317    let msg = message.into();
318    move |e| UtilError::new(error_type, msg, Some(e.into()))
319}
320
321/// Helper function for `.ok_or_else()` to create UtilError on None.
322///
323/// This function creates a closure that generates a `UtilError` with the
324/// specified error type and message when called.
325///
326/// # Examples
327///
328/// ```ignore
329/// // Instead of:
330/// .ok_or_else(|| UtilError::new(UtilErrorType::Other, "Item not found".to_string(), None))
331///
332/// // You can write:
333/// .ok_or_else(missing_util_error(UtilErrorType::Other, "Item not found".to_string()))
334/// ```
335pub fn missing_util_error(
336    error_type: UtilErrorType,
337    message: impl Into<String>,
338) -> impl FnOnce() -> UtilError {
339    let msg = message.into();
340    move || UtilError::new(error_type, msg, None)
341}
