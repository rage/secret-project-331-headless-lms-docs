12static ALL_TAG_REGEX: Lazy<Regex> =
13    Lazy::new(|| Regex::new(r"<.+?>").expect("invalid all_tags regex"));
14static LINK_REGEX: Lazy<Regex> = Lazy::new(|| {
15    Regex::new(r#"(?s)<a\s[^>]*?href="([^"]*)"[^>]*>(.*?)</a>"#).expect("invalid link regex")
16});
17static WHITESPACE_REGEX: Lazy<Regex> =
18    Lazy::new(|| Regex::new(r"\s+").expect("invalid whitespace regex"));
19static PLACEHOLDER_REGEX: Lazy<Regex> =
20    Lazy::new(|| Regex::new(r"\{\{(\w+)\}\}").expect("invalid placeholder regex"));
21// Gutenberg's link inputs turn `{{LINK}}` into `https://{{LINK}}`. The CMS strips this on save, but
22// templates saved before it did still have it.
23static PREPENDED_PROTOCOL_BEFORE_PLACEHOLDER_REGEX: Lazy<Regex> = Lazy::new(|| {
24    Regex::new(r"^https?://(\{\{\w+\}\})$").expect("invalid prepended_protocol regex")
25});
26static PREPENDED_PROTOCOL_IN_HREF_REGEX: Lazy<Regex> = Lazy::new(|| {
27    Regex::new(r#"(<a\s[^>]*?href=")https?://(\{\{\w+\}\}")"#)
28        .expect("invalid prepended_protocol_in_href regex")
29});
30static LAYOUT_PLACEHOLDER_REGEX: Lazy<Regex> = Lazy::new(|| {
31    Regex::new(r"\{\{(CONTENT|SUBJECT|PREHEADER|LANGUAGE)\}\}")
32        .expect("invalid layout_placeholder regex")
33});
34static IMG_TAG_REGEX: Lazy<Regex> =
35    Lazy::new(|| Regex::new(r"(?is)<img\b[^>]*>").expect("invalid img_tag regex"));
36static SRC_ATTRIBUTE_REGEX: Lazy<Regex> = Lazy::new(|| {
37    Regex::new(r#"(?is)\ssrc\s*=\s*["']?([^"'\s>]*)"#).expect("invalid src_attribute regex")
38});
39static HREFLESS_LINK_REGEX: Lazy<Regex> =
40    Lazy::new(|| Regex::new(r"(?s)<a>(.*?)</a>").expect("invalid hrefless_link regex"));

144/// One block of an email body. Rich-text attributes (see `EmailBlockName::is_rich_text_attribute`)
145/// are HTML, sanitized when rendered; every other attribute is plain.
146#[derive(Debug, Deserialize, Serialize, PartialEq, Clone)]
147#[serde(rename_all = "camelCase")]
148pub struct EmailGutenbergBlock {
149    pub name: EmailBlockName,
150    #[serde(default)]
151    pub attributes: Map<String, Value>,
152    #[serde(default)]
153    pub inner_blocks: Vec<EmailGutenbergBlock>,
154}
155
156impl EmailBlockName {
157    /// Table sections count as rich text because their cells' `content` is.
158    fn is_rich_text_attribute(self, key: &str) -> bool {
159        match self {
160            Self::Paragraph | Self::Heading | Self::ListItem | Self::Code => key == "content",
161            Self::Image => key == "caption",
162            Self::Table => matches!(key, "caption" | "head" | "body" | "foot"),
163            Self::Button => key == "text",
164            Self::Quote => matches!(key, "value" | "citation"),
165            Self::Callout => key == "title",
166            Self::List | Self::Buttons | Self::Separator | Self::Spacer | Self::OneTimeCode => {
167                false
168            }
169        }
170    }
171}

173struct TableCell<'a> {
174    tag: &'static str,
175    content: &'a str,
176    colspan: Option<u32>,
177    rowspan: Option<u32>,
178}

180impl EmailGutenbergBlock {
181    fn str_attribute(&self, key: &str) -> &str {
182        self.attributes
183            .get(key)
184            .and_then(Value::as_str)
185            .unwrap_or_default()
186    }
187
188    fn rich_text(&self, key: &str) -> String {
189        sanitize_rich_text(self.str_attribute(key), &RICH_TEXT_SANITIZER)
190    }
191
192    fn rich_text_as_text(&self, key: &str) -> String {
193        html_to_text(&self.rich_text(key))
194    }
195
196    /// Whether the author picked the block style `name`, which Gutenberg records in `className`.
197    fn has_style(&self, name: &str) -> bool {
198        self.str_attribute("className")
199            .split_whitespace()
200            .any(|class| class.strip_prefix("is-style-") == Some(name))
201    }
202
203    /// A length in px; Gutenberg stores it as a string like "100px", old content as a number.
204    fn px_attribute(&self, key: &str) -> Option<f64> {
205        let px = match self.attributes.get(key) {
206            Some(Value::Number(number)) => number.as_f64(),
207            Some(Value::String(text)) => text.trim().trim_end_matches("px").trim().parse().ok(),
208            _ => None,
209        };
210        px.filter(|px| px.is_finite())
211    }
212
213    fn spacer_height_px(&self) -> u32 {
214        self.px_attribute("height")
215            .map(|height| height.clamp(0.0, MAX_SPACER_HEIGHT_PX).round() as u32)
216            .unwrap_or(DEFAULT_SPACER_HEIGHT_PX)
217    }
218
219    /// The width an image is shown at: the author's, but never wider than the content column.
220    /// Without one the image fills the column.
221    fn image_width_px(&self) -> u32 {
222        self.px_attribute("width")
223            .filter(|width| *width >= 1.0)
224            .unwrap_or(MAX_IMAGE_WIDTH_PX)
225            .min(MAX_IMAGE_WIDTH_PX)
226            .round() as u32
227    }
228
229    fn image_alt(&self) -> &str {
230        match self.str_attribute("alt").trim() {
231            UNCHANGED_ALT_TEXT => "",
232            alt => alt,
233        }
234    }
235
236    fn table_rows(&self, section: &str) -> Vec<Vec<TableCell<'_>>> {
237        let Some(Value::Array(rows)) = self.attributes.get(section) else {
238            return vec![];
239        };
240        let span = |cell: &Value, key: &str| {
241            let span = match cell.get(key)? {
242                Value::Number(number) => number.as_u64(),
243                Value::String(text) => text.trim().parse().ok(),
244                _ => None,
245            }?;
246            u32::try_from(span).ok().filter(|span| *span > 1)
247        };
248        rows.iter()
249            .map(|row| {
250                row.get("cells")
251                    .and_then(Value::as_array)
252                    .map(|cells| {
253                        cells
254                            .iter()
255                            .map(|cell| TableCell {
256                                tag: match cell.get("tag").and_then(Value::as_str) {
257                                    Some("th") => "th",
258                                    _ => "td",
259                                },
260                                content: cell
261                                    .get("content")
262                                    .and_then(Value::as_str)
263                                    .unwrap_or_default(),
264                                colspan: span(cell, "colspan"),
265                                rowspan: span(cell, "rowspan"),
266                            })
267                            .collect()
268                    })
269                    .unwrap_or_default()
270            })
271            .collect()
272    }
273
274    /// Its items' numbers if ordered, honouring `start` and `reversed` like a browser does.
275    fn list_item_numbers(&self) -> impl Iterator<Item = i64> + '_ {
276        let count = self.inner_blocks.len() as i64;
277        let is_reversed = self.is_reversed();
278        let start = self
279            .attributes
280            .get("start")
281            .and_then(Value::as_i64)
282            .unwrap_or(if is_reversed { count } else { 1 });
283        (0..count).map(move |index| {
284            if is_reversed {
285                start.saturating_sub(index)
286            } else {
287                start.saturating_add(index)
288            }
289        })
290    }
291
292    fn is_ordered(&self) -> bool {
293        self.attributes
294            .get("ordered")
295            .and_then(Value::as_bool)
296            .unwrap_or(false)
297    }
298
299    fn is_reversed(&self) -> bool {
300        self.attributes
301            .get("reversed")
302            .and_then(Value::as_bool)
303            .unwrap_or(false)
304    }
305}

307fn rich_text_sanitizer(extra_tags: &[&'static str]) -> ammonia::Builder<'static> {
308    let mut sanitizer = ammonia::Builder::empty();
309    sanitizer
310        .tags(RICH_TEXT_TAGS.iter().chain(extra_tags).copied().collect())
311        .generic_attributes(HashSet::new())
312        .tag_attributes(HashMap::from([("a", HashSet::from(["href"]))]))
313        .url_schemes(HashSet::from(LINK_SCHEMES))
314        .url_relative(ammonia::UrlRelative::Deny)
315        .link_rel(None);
316    sanitizer
317}
318
319fn sanitize_rich_text(html: &str, sanitizer: &ammonia::Builder) -> String {
320    let sanitized = sanitizer.clean(html).to_string();
321    // A link whose target was rejected loses only its href; unwrap it so it does not look like one.
322    HREFLESS_LINK_REGEX
323        .replace_all(&sanitized, "$1")
324        .into_owned()
325}

327/// `url` trimmed, if it is absolute with one of `schemes`.
328fn url_with_scheme<'a>(url: &'a str, schemes: &[&str]) -> Option<&'a str> {
329    let url = url.trim();
330    let (scheme, _) = url.split_once(':')?;
331    schemes
332        .iter()
333        .any(|allowed| scheme.eq_ignore_ascii_case(allowed))
334        .then_some(url)
335}

337/// Replaces `{{KEY}}` in every attribute, nested blocks included. Values are HTML-escaped in rich
338/// text and inserted as is elsewhere; a protocol Gutenberg prepended to a link whose whole target is
339/// a placeholder is dropped. Returns the keys found that `replacements` has no value for;
340/// those are left in place.
341pub fn fill_placeholders(
342    blocks: &mut [EmailGutenbergBlock],
343    replacements: &HashMap<String, String>,
344) -> BTreeSet<String> {
345    let mut unfilled = BTreeSet::new();
346    fill_placeholders_in_blocks(blocks, replacements, &mut unfilled);
347    unfilled
348}
349
350fn fill_placeholders_in_blocks(
351    blocks: &mut [EmailGutenbergBlock],
352    replacements: &HashMap<String, String>,
353    unfilled: &mut BTreeSet<String>,
354) {
355    for block in blocks {
356        for (key, value) in block.attributes.iter_mut() {
357            let is_html = block.name.is_rich_text_attribute(key);
358            fill_placeholders_in_value(value, replacements, is_html, unfilled);
359        }
360        fill_placeholders_in_blocks(&mut block.inner_blocks, replacements, unfilled);
361    }
362}
363
364fn fill_placeholders_in_value(
365    value: &mut Value,
366    replacements: &HashMap<String, String>,
367    is_html: bool,
368    unfilled: &mut BTreeSet<String>,
369) {
370    match value {
371        Value::String(text) => {
372            if let Cow::Owned(filled) =
373                fill_placeholders_in_text(text, replacements, is_html, unfilled)
374            {
375                *text = filled;
376            }
377        }
378        Value::Array(values) => values
379            .iter_mut()
380            .for_each(|v| fill_placeholders_in_value(v, replacements, is_html, unfilled)),
381        Value::Object(fields) => fields
382            .values_mut()
383            .for_each(|v| fill_placeholders_in_value(v, replacements, is_html, unfilled)),
384        _ => {}
385    }
386}
387
388fn fill_placeholders_in_text<'a>(
389    text: &'a str,
390    replacements: &HashMap<String, String>,
391    is_html: bool,
392    unfilled: &mut BTreeSet<String>,
393) -> Cow<'a, str> {
394    let unprefixed = if is_html {
395        PREPENDED_PROTOCOL_IN_HREF_REGEX.replace_all(text, "$1$2")
396    } else {
397        PREPENDED_PROTOCOL_BEFORE_PLACEHOLDER_REGEX.replace(text, "$1")
398    };
399    match unprefixed {
400        Cow::Borrowed(text) => replace_placeholders(text, replacements, is_html, unfilled),
401        Cow::Owned(text) => {
402            Cow::Owned(replace_placeholders(&text, replacements, is_html, unfilled).into_owned())
403        }
404    }
405}
406
407fn replace_placeholders<'a>(
408    text: &'a str,
409    replacements: &HashMap<String, String>,
410    is_html: bool,
411    unfilled: &mut BTreeSet<String>,
412) -> Cow<'a, str> {
413    PLACEHOLDER_REGEX.replace_all(text, |caps: &Captures| match replacements.get(&caps[1]) {
414        Some(replacement) if is_html => escape_html(replacement),
415        Some(replacement) => replacement.clone(),
416        None => {
417            unfilled.insert(caps[1].to_string());
418            caps[0].to_string()
419        }
420    })
421}

423/// Renders a body into the HTML a shell's `{{CONTENT}}` is replaced with.
424pub fn process_content_to_html(blocks: &[EmailGutenbergBlock], theme: EmailTheme) -> String {
425    blocks
426        .iter()
427        .map(|block| block_to_html(block, theme))
428        .collect()
429}
430
431fn block_to_html(block: &EmailGutenbergBlock, theme: EmailTheme) -> String {
432    match block.name {
433        // Inline, not `h1 + p` in the shell: Gmail and Outlook ignore sibling selectors.
434        EmailBlockName::Paragraph if block.has_style("lead") => format!(
435            r#"<p class="email-lead" style="font-size: 18px; line-height: 28px;">{}</p>"#,
436            block.rich_text("content")
437        ),
438        EmailBlockName::Paragraph => format!("<p>{}</p>", block.rich_text("content")),
439        EmailBlockName::Heading => {
440            let level = block
441                .attributes
442                .get("level")
443                .and_then(Value::as_u64)
444                .filter(|level| (1..=6).contains(level))
445                .unwrap_or(2);
446            format!("<h{level}>{}</h{level}>", block.rich_text("content"))
447        }
448        EmailBlockName::Image => {
449            // The width attribute is for Outlook desktop, which ignores CSS widths on images.
450            let img = match url_with_scheme(block.str_attribute("url"), &IMAGE_SCHEMES) {
451                Some(src) => {
452                    let width = block.image_width_px();
453                    format!(
454                        r#"<img src="{}" alt="{}" width="{width}" style="width: 100%; max-width: {width}px; height: auto;">"#,
455                        escape_html(src),
456                        escape_html(block.image_alt())
457                    )
458                }
459                None => String::new(),
460            };
461            let img = match url_with_scheme(block.str_attribute("href"), &LINK_SCHEMES) {
462                Some(href) if !img.is_empty() => {
463                    format!(r#"<a href="{}">{img}</a>"#, escape_html(href))
464                }
465                _ => img,
466            };
467            match block.rich_text("caption").as_str() {
468                "" => img,
469                caption => format!(
470                    r#"<div class="email-image">{img}<p class="email-image-caption">{caption}</p></div>"#
471                ),
472            }
473        }
474        EmailBlockName::List => {
475            let items = process_content_to_html(&block.inner_blocks, theme);
476            if !block.is_ordered() {
477                return format!("<ul>{items}</ul>");
478            }
479            let start = match block.attributes.get("start").and_then(Value::as_i64) {
480                Some(start) => format!(r#" start="{start}""#),
481                None => String::new(),
482            };
483            let reversed = if block.is_reversed() { " reversed" } else { "" };
484            format!("<ol{start}{reversed}>{items}</ol>")
485        }
486        EmailBlockName::ListItem => format!(
487            "<li>{}{}</li>",
488            block.rich_text("content"),
489            process_content_to_html(&block.inner_blocks, theme)
490        ),
491        EmailBlockName::Table => {
492            let section = |name: &str, tag: &str| {
493                let rows = block.table_rows(name);
494                if rows.is_empty() {
495                    return String::new();
496                }
497                let rows: String = rows
498                    .iter()
499                    .map(|cells| {
500                        let cells: String = cells
501                            .iter()
502                            .map(|cell| {
503                                let mut spans = String::new();
504                                if let Some(colspan) = cell.colspan {
505                                    spans.push_str(&format!(r#" colspan="{colspan}""#));
506                                }
507                                if let Some(rowspan) = cell.rowspan {
508                                    spans.push_str(&format!(r#" rowspan="{rowspan}""#));
509                                }
510                                format!(
511                                    "<{tag}{spans}>{}</{tag}>",
512                                    sanitize_rich_text(cell.content, &RICH_TEXT_SANITIZER),
513                                    tag = cell.tag
514                                )
515                            })
516                            .collect();
517                        format!("<tr>{cells}</tr>")
518                    })
519                    .collect();
520                format!("<{tag}>{rows}</{tag}>")
521            };
522            let caption = match block.rich_text("caption").as_str() {
523                "" => String::new(),
524                caption => format!("<caption>{caption}</caption>"),
525            };
526            format!(
527                "<table>{caption}{}{}{}</table>",
528                section("head", "thead"),
529                section("body", "tbody"),
530                section("foot", "tfoot")
531            )
532        }
533        EmailBlockName::Buttons => format!(
534            r#"<div class="email-buttons">{}</div>"#,
535            process_content_to_html(&block.inner_blocks, theme)
536        ),
537        // Outlook desktop drops padding and background on <a>, so the cell repeats them in bgcolor
538        // and `mso-padding-alt`, which only Outlook reads. The arrow is text because Gmail and
539        // Outlook drop CSS `::after`.
540        EmailBlockName::Button => {
541            let (open_tag, close_tag) =
542                match url_with_scheme(block.str_attribute("url"), &LINK_SCHEMES) {
543                    Some(url) => (format!(r#"a href="{}""#, escape_html(url)), "a"),
544                    None => ("span".to_string(), "span"),
545                };
546            format!(
547                r#"<table role="presentation" class="email-button" cellpadding="0" cellspacing="0" border="0"><tr><td class="email-button-cell" bgcolor="{background}" style="background-color: {background}; border-radius: 6px; mso-padding-alt: 14px 26px;"><{open_tag} class="email-button-link" style="display: inline-block; padding: 14px 26px; font-size: 16px; line-height: 20px; font-weight: 600; color: {text}; text-decoration: none;">{}{}</{close_tag}></td></tr></table>"#,
548                block.rich_text("text"),
549                if block.has_style("arrow") {
550                    "&nbsp;&rarr;"
551                } else {
552                    ""
553                },
554                background = escape_html(theme.button_background_color),
555                text = escape_html(theme.button_text_color),
556            )
557        }
558        // A cell, not <blockquote>: Outlook desktop ignores padding on a blockquote, so its text would
559        // touch the bar.
560        EmailBlockName::Quote => {
561            let citation = match block.rich_text("citation").as_str() {
562                "" => String::new(),
563                citation => format!(
564                    r#"<cite style="display: block; margin: 0; font-size: 14px; font-style: normal;">{citation}</cite>"#
565                ),
566            };
567            let body = if block.inner_blocks.is_empty() {
568                sanitize_rich_text(block.str_attribute("value"), &QUOTE_VALUE_SANITIZER)
569            } else {
570                process_content_to_html(&block.inner_blocks, theme)
571            };
572            format!(
573                r#"<table role="presentation" class="email-quote" width="100%" cellpadding="0" cellspacing="0" border="0" style="width: 100%; margin: 0 0 18px;"><tr><td style="padding: 0 0 0 16px; border: 0; border-left: 4px solid {QUOTE_BORDER_COLOR}; font-size: 16px; line-height: 26px; color: {QUOTE_TEXT_COLOR};">{body}{citation}</td></tr></table>"#
574            )
575        }
576        // Tables, not <hr> or a sized <div>: Outlook desktop ignores their borders, heights and margins.
577        EmailBlockName::Separator => r##"<table role="presentation" class="email-separator" width="100%" cellpadding="0" cellspacing="0" border="0"><tr><td height="1" style="height: 1px; font-size: 1px; line-height: 1px; mso-line-height-rule: exactly; border-top: 1px solid #DDDEE0;">&nbsp;</td></tr></table>"##.to_string(),
578        EmailBlockName::Spacer => {
579            let height = block.spacer_height_px();
580            format!(
581                r#"<table role="presentation" class="email-spacer" width="100%" cellpadding="0" cellspacing="0" border="0"><tr><td height="{height}" style="height: {height}px; font-size: {height}px; line-height: {height}px; mso-line-height-rule: exactly; padding: 0; border: 0;">&nbsp;</td></tr></table>"#
582            )
583        }
584        EmailBlockName::Code => format!(
585            "<pre class=\"email-code\"><code>{}</code></pre>",
586            block.rich_text("content").replace("<br>", "\n")
587        ),
588        EmailBlockName::OneTimeCode => {
589            let code = block.str_attribute("code").trim();
590            if code.is_empty() {
591                return String::new();
592            }
593            // The extra left padding offsets the letter-spacing trailing the last character.
594            format!(
595                r#"<table role="presentation" class="email-one-time-code" width="100%" cellpadding="0" cellspacing="0" border="0" bgcolor="{ONE_TIME_CODE_BACKGROUND_COLOR}" style="width: 100%; margin: 0 0 18px; background-color: {ONE_TIME_CODE_BACKGROUND_COLOR}; border: 1px solid {ONE_TIME_CODE_BORDER_COLOR}; border-radius: 6px; border-collapse: separate;"><tr><td align="center" style="padding: 16px 16px 16px 20px; border: 0; text-align: center; font-family: {ONE_TIME_CODE_FONT_FAMILY}; font-size: 22px; line-height: 30px; mso-line-height-rule: exactly; font-weight: 600; letter-spacing: 4px; color: {ONE_TIME_CODE_TEXT_COLOR}; overflow-wrap: anywhere; word-break: break-word; user-select: all;">{}</td></tr></table>"#,
596                escape_html(code)
597            )
598        }
599        EmailBlockName::Callout => {
600            let icon_cell = match callout_icon_file(block.str_attribute("icon")) {
601                Some(file) => format!(
602                    r#"<td class="email-callout-icon" width="24" valign="top" style="width: 24px; padding: 20px 0 20px 20px; border: 0;"><img src="{EMAIL_ASSET_BASE_URL}/{file}" width="24" height="24" alt="" style="display: block; width: 24px; height: 24px; margin: 1px 0 0; border: 0; border-radius: 0;"></td>"#
603                ),
604                None => String::new(),
605            };
606            let body_padding = if icon_cell.is_empty() {
607                "20px"
608            } else {
609                "20px 20px 20px 14px"
610            };
611            let title = match block.rich_text("title").as_str() {
612                "" => String::new(),
613                title => format!(
614                    r#"<p class="email-callout-title" style="margin: 0 0 6px;"><strong>{title}</strong></p>"#
615                ),
616            };
617            format!(
618                r#"<table role="presentation" class="email-callout" width="100%" cellpadding="0" cellspacing="0" border="0" bgcolor="{CALLOUT_BACKGROUND_COLOR}" style="width: 100%; background-color: {CALLOUT_BACKGROUND_COLOR}; border: 1px solid {CALLOUT_BORDER_COLOR}; border-radius: 8px; border-collapse: separate;"><tr>{icon_cell}<td class="email-callout-body" valign="top" style="padding: {body_padding}; border: 0;">{title}{}</td></tr></table>"#,
619                process_content_to_html(&block.inner_blocks, theme)
620            )
621        }
622    }
623}

636/// Renders a body into the plain-text part of a message. Links keep their targets in the text.
637pub fn process_content_to_plaintext(blocks: &[EmailGutenbergBlock]) -> String {
638    blocks
639        .iter()
640        .map(block_to_plaintext)
641        .filter(|text| !text.is_empty())
642        .collect::<Vec<_>>()
643        .join("\n\n")
644}
645
646fn block_to_plaintext(block: &EmailGutenbergBlock) -> String {
647    match block.name {
648        EmailBlockName::Paragraph | EmailBlockName::Heading => block.rich_text_as_text("content"),
649        EmailBlockName::Image => {
650            let alt = block.image_alt().replace('"', "");
651            let image =
652                url_with_scheme(block.str_attribute("url"), &IMAGE_SCHEMES).map(|url| {
653                    match (
654                        url_with_scheme(block.str_attribute("href"), &LINK_SCHEMES),
655                        alt.as_str(),
656                    ) {
657                        (Some(href), "") => href.to_string(),
658                        (Some(href), alt) => format!("{alt}: {href}"),
659                        (None, "") => format!("<{url}>"),
660                        (None, alt) => format!("\"{alt}\", <{url}>"),
661                    }
662                });
663            let caption = Some(block.rich_text_as_text("caption")).filter(|text| !text.is_empty());
664            image
665                .into_iter()
666                .chain(caption)
667                .collect::<Vec<_>>()
668                .join("\n")
669        }
670        EmailBlockName::List | EmailBlockName::ListItem => list_to_plaintext(block, 0),
671        EmailBlockName::Table => {
672            let mut lines = vec![];
673            let caption = block.rich_text_as_text("caption");
674            if !caption.is_empty() {
675                lines.push(caption);
676            }
677            for section in ["head", "body", "foot"] {
678                for cells in block.table_rows(section) {
679                    let cells: Vec<String> = cells
680                        .iter()
681                        .map(|cell| {
682                            html_to_text(&sanitize_rich_text(cell.content, &RICH_TEXT_SANITIZER))
683                        })
684                        .collect();
685                    lines.push(cells.join(" | "));
686                }
687            }
688            lines.join("\n")
689        }
690        EmailBlockName::Buttons => block
691            .inner_blocks
692            .iter()
693            .map(block_to_plaintext)
694            .collect::<Vec<_>>()
695            .join("\n"),
696        EmailBlockName::Button => {
697            let text = block.rich_text_as_text("text");
698            match url_with_scheme(block.str_attribute("url"), &LINK_SCHEMES) {
699                Some(url) => format!("{text}: {url}"),
700                None => text,
701            }
702        }
703        EmailBlockName::Quote => {
704            let body = if block.inner_blocks.is_empty() {
705                let value =
706                    sanitize_rich_text(block.str_attribute("value"), &QUOTE_VALUE_SANITIZER);
707                html_to_text(&value.replace("</p>", "\n"))
708            } else {
709                process_content_to_plaintext(&block.inner_blocks)
710            };
711            let mut lines: Vec<String> = body
712                .lines()
713                .map(|line| format!("> {line}").trim_end().to_string())
714                .collect();
715            let citation = block.rich_text_as_text("citation");
716            if !citation.is_empty() {
717                lines.push(format!("— {citation}"));
718            }
719            lines.join("\n")
720        }
721        EmailBlockName::Separator => "---".to_string(),
722        EmailBlockName::Spacer => String::new(),
723        EmailBlockName::Code => html_to_preformatted_text(&block.rich_text("content"))
724            .lines()
725            .map(|line| format!("    {line}").trim_end().to_string())
726            .collect::<Vec<_>>()
727            .join("\n"),
728        EmailBlockName::OneTimeCode => block.str_attribute("code").trim().to_string(),
729        EmailBlockName::Callout => [
730            block.rich_text_as_text("title"),
731            process_content_to_plaintext(&block.inner_blocks),
732        ]
733        .into_iter()
734        .filter(|text| !text.is_empty())
735        .collect::<Vec<_>>()
736        .join("\n"),
737    }
738}
739
740fn list_to_plaintext(list: &EmailGutenbergBlock, depth: usize) -> String {
741    let indent = "  ".repeat(depth);
742    let mut numbers = list.list_item_numbers();
743    let is_ordered = list.is_ordered();
744    let mut lines = vec![];
745    for item in &list.inner_blocks {
746        let number = numbers.next();
747        let marker = match number {
748            Some(number) if is_ordered => format!("{number}."),
749            _ => "*".to_string(),
750        };
751        lines.push(format!(
752            "{indent}{marker} {}",
753            item.rich_text_as_text("content")
754        ));
755        for nested in &item.inner_blocks {
756            lines.push(list_to_plaintext(nested, depth + 1));
757        }
758    }
759    lines.join("\n")
760}

762/// The inbox preview: the first paragraph as one line of text, or empty when there is none.
763pub fn preheader(blocks: &[EmailGutenbergBlock]) -> String {
764    blocks
765        .iter()
766        .find(|block| block.name == EmailBlockName::Paragraph)
767        .map(|block| {
768            let text = block.rich_text_as_text("content");
769            WHITESPACE_REGEX.replace_all(&text, " ").into_owned()
770        })
771        .unwrap_or_default()
772}

774/// Rich text to plain text. A link keeps its target, since the plain-text part has no other way to
775/// show it.
776fn html_to_text(html: &str) -> String {
777    html_to_preformatted_text(html).trim().to_string()
778}
779
780/// Like [`html_to_text`], but keeps the first line's indentation.
781fn html_to_preformatted_text(html: &str) -> String {
782    let with_links = LINK_REGEX.replace_all(html, |caps: &Captures| {
783        let href = &caps[1];
784        let text = ALL_TAG_REGEX.replace_all(&caps[2], "");
785        if text.trim() == href || text.trim().is_empty() {
786            href.to_string()
787        } else {
788            format!("{text} ({href})")
789        }
790    });
791    let with_breaks = with_links.replace("<br>", "\n");
792    let text = ALL_TAG_REGEX.replace_all(&with_breaks, "");
793    decode_html_entities(&text)
794        .trim_start_matches(['\r', '\n'])
795        .trim_end()
796        .to_string()
797}
798
799fn decode_html_entities(text: &str) -> String {
800    text.replace("&nbsp;", " ")
801        .replace("&lt;", "<")
802        .replace("&gt;", ">")
803        .replace("&quot;", "\"")
804        .replace("&#39;", "'")
805        .replace("&amp;", "&")
806}

808/// What a shell's placeholders are filled with.
809pub struct EmailLayoutFields<'a> {
810    /// Rendered body HTML, inserted as is.
811    pub content_html: &'a str,
812    pub subject: &'a str,
813    pub preheader: &'a str,
814    /// BCP 47 tag for `lang`; empty when unknown.
815    pub language: &'a str,
816}
817
818/// Wraps a rendered body in an email shell: `{{CONTENT}}` gets the body as is, `{{SUBJECT}}`,
819/// `{{PREHEADER}}` and `{{LANGUAGE}}` their escaped values. One pass, so placeholders inside the body
820/// are never expanded.
821pub fn wrap_in_layout(layout_html: &str, fields: &EmailLayoutFields) -> String {
822    LAYOUT_PLACEHOLDER_REGEX
823        .replace_all(layout_html, |caps: &Captures| match &caps[1] {
824            "CONTENT" => fields.content_html.to_string(),
825            "SUBJECT" => escape_html(fields.subject),
826            "PREHEADER" => escape_html(fields.preheader),
827            _ => escape_html(fields.language),
828        })
829        .into_owned()
830}

832/// What a message is rendered from: a body and subject with `{{KEY}}` placeholders, and the layout
833/// it is wrapped in.
834pub struct EmailRenderInput<'a> {
835    pub layout_html: &'a str,
836    pub theme: EmailTheme<'a>,
837    pub subject: &'a str,
838    pub body: Vec<EmailGutenbergBlock>,
839    /// BCP 47 tag for `lang`; empty when unknown.
840    pub language: &'a str,
841    pub replacements: &'a HashMap<String, String>,
842}
843
844/// A message ready to hand to the mail relay.
845#[derive(Debug, Clone, PartialEq, Eq)]
846pub struct RenderedEmail {
847    pub subject: String,
848    pub html: String,
849    pub plain_text: String,
850    /// Placeholders in the subject or body that `replacements` had no value for, left as typed.
851    pub unfilled_placeholders: BTreeSet<String>,
852}
853
854/// Renders a whole message. The one path both sending and previewing go through, so a preview shows
855/// what is sent.
856pub fn render_email(input: EmailRenderInput) -> RenderedEmail {
857    let EmailRenderInput {
858        layout_html,
859        theme,
860        subject,
861        mut body,
862        language,
863        replacements,
864    } = input;
865    let mut unfilled_placeholders = fill_placeholders(&mut body, replacements);
866    // Not escaped: the subject header is not HTML, and `wrap_in_layout` escapes it for the shell.
867    let subject =
868        fill_placeholders_in_text(subject, replacements, false, &mut unfilled_placeholders)
869            .into_owned();
870    let html = wrap_in_layout(
871        layout_html,
872        &EmailLayoutFields {
873            content_html: &process_content_to_html(&body, theme),
874            subject: &subject,
875            preheader: &preheader(&body),
876            language,
877        },
878    );
879    RenderedEmail {
880        html: inline_css(html),
881        plain_text: process_content_to_plaintext(&body),
882        subject,
883        unfilled_placeholders,
884    }
885}
886
887/// Copies the `<style>` rules onto the elements they match, for clients that drop `<style>`.
888fn inline_css(html: String) -> String {
889    match CSS_INLINER.inline(&html) {
890        Ok(inlined) => inlined,
891        Err(err) => {
892            tracing::warn!("Could not inline the CSS of an email, sending it as is: {err}");
893            html
894        }
895    }
896}

898/// Problems that make a shell render badly: a missing `{{SUBJECT}}`, `{{PREHEADER}}` or
899/// `{{LANGUAGE}}`, or an image not loaded over https, which clients block or warn about. Empty when
900/// none is found; a heuristic, not validation.
901pub fn layout_warnings(layout_html: &str) -> Vec<String> {
902    let mut warnings: Vec<String> = ["{{SUBJECT}}", "{{PREHEADER}}", "{{LANGUAGE}}"]
903        .into_iter()
904        .filter(|placeholder| !layout_html.contains(placeholder))
905        .map(|placeholder| format!("missing {placeholder}"))
906        .collect();
907    for img in IMG_TAG_REGEX.find_iter(layout_html) {
908        let src = SRC_ATTRIBUTE_REGEX
909            .captures(img.as_str())
910            .map(|caps| caps[1].to_string())
911            .unwrap_or_default();
912        if !src.to_ascii_lowercase().starts_with("https://") {
913            warnings.push(format!("image source is not https: {src:?}"));
914        }
915    }
916    warnings
917}
918
919fn escape_html(text: &str) -> String {
920    let mut escaped = String::with_capacity(text.len());
921    for c in text.chars() {
922        match c {
923            '&' => escaped.push_str("&amp;"),
924            '<' => escaped.push_str("&lt;"),
925            '>' => escaped.push_str("&gt;"),
926            '"' => escaped.push_str("&quot;"),
927            '\'' => escaped.push_str("&#39;"),
928            _ => escaped.push(c),
929        }
930    }
931    escaped
932}
