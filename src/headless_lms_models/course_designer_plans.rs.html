231pub fn validate_schedule_input(stages: &[CourseDesignerScheduleStageInput]) -> ModelResult<()> {
232    let expected_order = fixed_stage_order();
233    if stages.len() != expected_order.len() {
234        return Err(ModelError::new(
235            ModelErrorType::InvalidRequest,
236            "Schedule must contain exactly 5 stages.".to_string(),
237            None,
238        ));
239    }
240
241    for (idx, stage) in stages.iter().enumerate() {
242        if stage.stage != expected_order[idx] {
243            return Err(ModelError::new(
244                ModelErrorType::InvalidRequest,
245                "Schedule stages must be in the fixed order: analysis, design, development, implementation, evaluation."
246                    .to_string(),
247                None,
248            ));
249        }
250        if stage.planned_starts_on > stage.planned_ends_on {
251            return Err(ModelError::new(
252                ModelErrorType::InvalidRequest,
253                format!("Stage {:?} starts after it ends.", stage.stage),
254                None,
255            ));
256        }
257        if idx > 0 {
258            let prev = &stages[idx - 1];
259            if !no_gap_between(prev.planned_ends_on, stage.planned_starts_on) {
260                return Err(ModelError::new(
261                    ModelErrorType::InvalidRequest,
262                    "Schedule must have no gaps or overlaps between consecutive stages."
263                        .to_string(),
264                    None,
265                ));
266            }
267        }
268    }
269
270    Ok(())
271}

273fn first_day_of_month(date: NaiveDate) -> ModelResult<NaiveDate> {
274    NaiveDate::from_ymd_opt(date.year(), date.month(), 1).ok_or_else(|| {
275        ModelError::new(
276            ModelErrorType::InvalidRequest,
277            "Invalid date while generating schedule suggestion.".to_string(),
278            None,
279        )
280    })

283fn last_day_of_month(year: i32, month: u32) -> ModelResult<u32> {
284    for day in (28..=31).rev() {
285        if NaiveDate::from_ymd_opt(year, month, day).is_some() {
286            return Ok(day);
287        }
288    }
289
290    Err(ModelError::new(
291        ModelErrorType::InvalidRequest,
292        "Invalid month while generating schedule suggestion.".to_string(),
293        None,
294    ))
295}

297pub(crate) fn add_months_clamped(date: NaiveDate, months: u32) -> ModelResult<NaiveDate> {
298    let total_months = date.year() * 12 + date.month0() as i32 + months as i32;
299    let target_year = total_months.div_euclid(12);
300    let target_month0 = total_months.rem_euclid(12) as u32;
301    let target_month = target_month0 + 1;
302    let target_day = date
303        .day()
304        .min(last_day_of_month(target_year, target_month)?);
305
306    NaiveDate::from_ymd_opt(target_year, target_month, target_day).ok_or_else(|| {
307        ModelError::new(
308            ModelErrorType::InvalidRequest,
309            "Failed to generate schedule suggestion date.".to_string(),
310            None,
311        )
312    })
313}

315fn suggestion_months(size: CourseDesignerCourseSize) -> [u32; 5] {
316    match size {
317        CourseDesignerCourseSize::Small => [1, 1, 2, 1, 1],
318        CourseDesignerCourseSize::Medium => [1, 2, 3, 2, 1],
319        CourseDesignerCourseSize::Large => [2, 2, 4, 3, 1],
320    }
321}
322
323pub fn build_schedule_suggestion(
324    size: CourseDesignerCourseSize,
325    starts_on: NaiveDate,
326) -> ModelResult<Vec<CourseDesignerScheduleStageInput>> {
327    let mut current_start = first_day_of_month(starts_on)?;
328    let stage_order = fixed_stage_order();
329    let month_durations = suggestion_months(size);
330    let mut out = Vec::with_capacity(stage_order.len());
331
332    for (stage, months) in stage_order.into_iter().zip(month_durations) {
333        let next_stage_start = add_months_clamped(current_start, months)?;
334        let planned_ends_on = next_stage_start - Duration::days(1);
335        out.push(CourseDesignerScheduleStageInput {
336            stage,
337            planned_starts_on: current_start,
338            planned_ends_on,
339        });
340        current_start = planned_ends_on + Duration::days(1);
341    }
342
343    Ok(out)
344}

421/// A plan with its members, stages, tasks and course status, if `user_id` is a member.
422pub async fn get_plan_details_for_user(
423    conn: &mut PgConnection,
424    plan_id: Uuid,
425    user_id: Uuid,
426) -> ModelResult<CourseDesignerPlanDetails> {
427    let plan = course_designer_plan_members::get_plan_for_user(conn, plan_id, user_id).await?;
428    let members = course_designer_plan_members::get_plan_members(conn, plan_id).await?;
429    let stages = course_designer_plan_members::get_plan_stages(conn, plan_id).await?;
430    let tasks = course_designer_plan_members::get_plan_tasks(conn, plan_id).await?;
431    let course_status = course_designer_course_status::get_for_plan(conn, plan_id).await?;
432    let mut tasks_by_stage: std::collections::HashMap<Uuid, Vec<CourseDesignerPlanStageTask>> =
433        tasks
434            .into_iter()
435            .fold(std::collections::HashMap::new(), |mut acc, mut t| {
436                t.apply_course_status(course_status.as_ref());
437                acc.entry(t.course_designer_plan_stage_id)
438                    .or_default()
439                    .push(t);
440                acc
441            });
442    let stages_with_tasks: Vec<CourseDesignerPlanStageWithTasks> = stages
443        .into_iter()
444        .map(|stage| {
445            let stage_id = stage.id;
446            let stage_tasks = tasks_by_stage.remove(&stage_id).unwrap_or_default();
447            CourseDesignerPlanStageWithTasks {
448                stage,
449                tasks: stage_tasks,
450            }
451        })
452        .collect();
453    Ok(CourseDesignerPlanDetails {
454        plan,
455        members,
456        stages: stages_with_tasks,
457        course_status,
458    })
459}

529/// Saves a stage's workspace form. The workspace schema must belong to `stage`.
530///
531/// `base_revision` is the `workspace_revision` the edit started from. When another save has
532/// happened since, refuses with [`ModelErrorType::StaleVersion`] so members never overwrite each
533/// other's work unseen.
534pub async fn update_stage_workspace_for_user(
535    conn: &mut PgConnection,
536    plan_id: Uuid,
537    user_id: Uuid,
538    stage: CourseDesignerStage,
539    workspace: CourseDesignerStageWorkspace,
540    base_revision: i32,
541) -> ModelResult<CourseDesignerPlanDetails> {
542    if workspace.stage() != stage {
543        return Err(ModelError::new(
544            ModelErrorType::InvalidRequest,
545            format!("This workspace schema does not belong to the {stage:?} stage."),
546            None,
547        ));
548    }
549    let mut tx = conn.begin().await?;
550    let current_revision =
551        lock_stage_workspace_revision_for_user(&mut tx, plan_id, user_id, stage).await?;
552    if current_revision != base_revision {
553        return Err(model_err!(
554            StaleVersion,
555            "Someone else saved this form after you opened it.".to_string()
556        ));
557    }
558    update_locked_stage_workspace(&mut tx, plan_id, user_id, &workspace).await?;
559    tx.commit().await?;
560    get_plan_details_for_user(conn, plan_id, user_id).await
561}

563/// Saves the workspace of the stage its schema belongs to and bumps the stage's
564/// `workspace_revision`, for edits that read the workspace with [`lock_stage_workspace_for_user`]
565/// earlier in `tx` and so need no revision check of their own.
566pub async fn update_locked_stage_workspace(
567    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
568    plan_id: Uuid,
569    user_id: Uuid,
570    workspace: &CourseDesignerStageWorkspace,
571) -> ModelResult<()> {
572    let stage = workspace.stage();
573    let workspace_json = course_designer_stage_workspace::workspace_to_json(workspace)?;
574    sqlx::query!(
575        r#"
576UPDATE course_designer_plan_stages
577SET workspace_data = $3,
578  workspace_revision = workspace_revision + 1
579WHERE course_designer_plan_id = $1
580  AND stage = $2
581  AND deleted_at IS NULL
582"#,
583        plan_id,
584        stage as CourseDesignerStage,
585        workspace_json
586    )
587    .execute(&mut **tx)
588    .await?;
589    // Autosave fires on every edit; one event per member and stage an hour keeps the timeline readable.
590    sqlx::query!(
591        r#"
592INSERT INTO course_designer_plan_events (
593  course_designer_plan_id,
594  actor_user_id,
595  event_type,
596  stage
597)
598SELECT $1, $2, 'workspace_saved', $3
599WHERE NOT EXISTS (
600  SELECT 1
601  FROM course_designer_plan_events
602  WHERE course_designer_plan_id = $1
603    AND actor_user_id = $2
604    AND event_type = 'workspace_saved'
605    AND stage = $3
606    AND occurred_at > NOW() - INTERVAL '1 hour'
607    AND deleted_at IS NULL
608)
609"#,
610        plan_id,
611        user_id,
612        stage as CourseDesignerStage
613    )
614    .execute(&mut **tx)
615    .await?;
616    Ok(())
617}

619/// Saves the plan's "what makes this MOOC great" text; blank text clears it.
620pub async fn set_great_mooc_vision_for_user(
621    conn: &mut PgConnection,
622    plan_id: Uuid,
623    user_id: Uuid,
624    great_mooc_vision: Option<String>,
625) -> ModelResult<CourseDesignerPlan> {
626    let great_mooc_vision = great_mooc_vision.filter(|v| !v.trim().is_empty());
627    if great_mooc_vision
628        .as_ref()
629        .is_some_and(|v| v.chars().count() > MAX_GREAT_MOOC_VISION_CHARS)
630    {
631        return Err(ModelError::new(
632            ModelErrorType::InvalidRequest,
633            format!("The vision can be at most {MAX_GREAT_MOOC_VISION_CHARS} characters long."),
634            None,
635        ));
636    }
637    let mut tx = conn.begin().await?;
638    course_designer_plan_members::lock_editable_plan_for_user(&mut tx, plan_id, user_id).await?;
639    let plan = sqlx::query_as!(
640        CourseDesignerPlan,
641        r#"
642UPDATE course_designer_plans p
643SET great_mooc_vision = $3
644FROM course_designer_plan_members m
645WHERE p.id = $1
646  AND p.deleted_at IS NULL
647  AND m.course_designer_plan_id = p.id
648  AND m.user_id = $2
649  AND m.deleted_at IS NULL
650RETURNING
651  p.id,
652  p.created_at,
653  p.updated_at,
654  p.deleted_at,
655  p.created_by_user_id,
656  p.name,
657  p.status,
658  p.active_stage,
659  p.last_weekly_stage_email_sent_at,
660  p.great_mooc_vision,
661  p.created_course_id
662"#,
663        plan_id,
664        user_id,
665        great_mooc_vision
666    )
667    .fetch_optional(&mut *tx)
668    .await?
669    .ok_or_else(|| {
670        ModelError::new(
671            ModelErrorType::PreconditionFailed,
672            "Plan not found or user is not a plan member.".to_string(),
673            None,
674        )
675    })?;
676    tx.commit().await?;
677    Ok(plan)
678}

680/// Archives a plan, or restores an archived one, keeping its stages and active stage as they were.
681///
682/// Restoring derives the status from the schedule: in progress with an active stage, completed
683/// when every stage is completed, scheduling when stages exist, draft otherwise.
684pub async fn set_archived_for_user(
685    conn: &mut PgConnection,
686    plan_id: Uuid,
687    user_id: Uuid,
688    archived: bool,
689) -> ModelResult<CourseDesignerPlan> {
690    let mut tx = conn.begin().await?;
691    let plan = course_designer_plan_members::lock_plan_for_user(&mut tx, plan_id, user_id).await?;
692    let is_archived = plan.status == CourseDesignerPlanStatus::Archived;
693    if is_archived == archived {
694        return Err(ModelError::new(
695            ModelErrorType::PreconditionFailed,
696            if archived {
697                "The plan is already archived."
698            } else {
699                "The plan is not archived."
700            }
701            .to_string(),
702            None,
703        ));
704    }
705    let new_status = if archived {
706        CourseDesignerPlanStatus::Archived
707    } else if plan.active_stage.is_some() {
708        CourseDesignerPlanStatus::InProgress
709    } else {
710        let counts = sqlx::query!(
711            r#"
712SELECT
713  COUNT(*)::BIGINT AS "stage_count!",
714  COUNT(*) FILTER (WHERE status = 'completed')::BIGINT AS "completed_stage_count!"
715FROM course_designer_plan_stages
716WHERE course_designer_plan_id = $1
717  AND deleted_at IS NULL
718"#,
719            plan_id
720        )
721        .fetch_one(&mut *tx)
722        .await?;
723        if counts.stage_count == 0 {
724            CourseDesignerPlanStatus::Draft
725        } else if counts.completed_stage_count == counts.stage_count {
726            CourseDesignerPlanStatus::Completed
727        } else {
728            CourseDesignerPlanStatus::Scheduling
729        }
730    };
731    let updated = sqlx::query_as!(
732        CourseDesignerPlan,
733        r#"
734UPDATE course_designer_plans
735SET status = $2
736WHERE id = $1
737  AND deleted_at IS NULL
738RETURNING *
739"#,
740        plan_id,
741        new_status as CourseDesignerPlanStatus
742    )
743    .fetch_one(&mut *tx)
744    .await?;
745    insert_plan_event(
746        &mut tx,
747        plan_id,
748        Some(user_id),
749        if archived {
750            "plan_archived"
751        } else {
752            "plan_unarchived"
753        },
754        None,
755        json!({ "status": new_status }),
756    )
757    .await?;
758    tx.commit().await?;
759    Ok(updated)
760}

808/// Every non-deleted plan with its members and stage progress, newest activity first. Not scoped
809/// to a user: the caller must have checked global admin access.
810pub async fn list_plans_for_support(
811    conn: &mut PgConnection,
812) -> ModelResult<Vec<CourseDesignerSupportPlan>> {
813    let plans = sqlx::query_as!(
814        CourseDesignerPlan,
815        r#"
816SELECT *
817FROM course_designer_plans
818WHERE deleted_at IS NULL
819ORDER BY updated_at DESC, id DESC
820"#
821    )
822    .fetch_all(&mut *conn)
823    .await?;
824    let plan_ids: Vec<Uuid> = plans.iter().map(|p| p.id).collect();
825    let member_rows = sqlx::query!(
826        r#"
827SELECT
828  m.course_designer_plan_id,
829  m.id,
830  m.user_id,
831  ud.first_name,
832  ud.last_name,
833  ud.email,
834  m.created_at,
835  m.receives_weekly_digest
836FROM course_designer_plan_members m
837JOIN user_details ud ON ud.user_id = m.user_id
838WHERE m.course_designer_plan_id = ANY($1)
839  AND m.deleted_at IS NULL
840ORDER BY m.created_at ASC, m.id ASC
841"#,
842        &plan_ids
843    )
844    .fetch_all(&mut *conn)
845    .await?;
846    let stage_rows = sqlx::query!(
847        r#"
848SELECT
849  s.course_designer_plan_id,
850  s.stage AS "stage: CourseDesignerStage",
851  s.status AS "status: CourseDesignerPlanStageStatus",
852  s.planned_starts_on,
853  s.planned_ends_on,
854  COUNT(t.id)::BIGINT AS "task_count!",
855  COUNT(t.id) FILTER (
856    WHERE t.is_completed
857      AND t.automatic_check IS NULL
858  )::BIGINT AS "completed_task_count!",
859  COALESCE(
860    ARRAY_AGG(t.automatic_check) FILTER (WHERE t.automatic_check IS NOT NULL),
861    '{}'
862  ) AS "automatic_checks!"
863FROM course_designer_plan_stages s
864LEFT JOIN course_designer_plan_stage_tasks t
865  ON t.course_designer_plan_stage_id = s.id
866  AND t.deleted_at IS NULL
867WHERE s.course_designer_plan_id = ANY($1)
868  AND s.deleted_at IS NULL
869GROUP BY s.id
870ORDER BY
871  CASE s.stage
872    WHEN 'analysis' THEN 1
873    WHEN 'design' THEN 2
874    WHEN 'development' THEN 3
875    WHEN 'implementation' THEN 4
876    WHEN 'evaluation' THEN 5
877  END
878"#,
879        &plan_ids
880    )
881    .fetch_all(&mut *conn)
882    .await?;
883    let plan_ids_with_course: Vec<Uuid> = plans
884        .iter()
885        .filter(|p| p.created_course_id.is_some())
886        .map(|p| p.id)
887        .collect();
888    let course_statuses =
889        course_designer_course_status::get_for_plans(conn, &plan_ids_with_course).await?;
890
891    let mut support_plans: Vec<CourseDesignerSupportPlan> = plans
892        .into_iter()
893        .map(|plan| CourseDesignerSupportPlan {
894            plan,
895            members: vec![],
896            stages: vec![],
897        })
898        .collect();
899    let index_by_plan: std::collections::HashMap<Uuid, usize> = support_plans
900        .iter()
901        .enumerate()
902        .map(|(index, p)| (p.plan.id, index))
903        .collect();
904    for row in member_rows {
905        if let Some(&index) = index_by_plan.get(&row.course_designer_plan_id) {
906            support_plans[index].members.push(PlanMemberWithDetails {
907                id: row.id,
908                user_id: row.user_id,
909                first_name: row.first_name,
910                last_name: row.last_name,
911                email: row.email,
912                created_at: row.created_at,
913                receives_weekly_digest: row.receives_weekly_digest,
914            });
915        }
916    }
917    for row in stage_rows {
918        if let Some(&index) = index_by_plan.get(&row.course_designer_plan_id) {
919            let course_status = course_statuses.get(&row.course_designer_plan_id);
920            let passed_check_count = course_designer_course_status::count_passed_checks(
921                course_status,
922                &row.automatic_checks,
923            );
924            support_plans[index]
925                .stages
926                .push(CourseDesignerSupportStage {
927                    stage: row.stage,
928                    status: row.status,
929                    planned_starts_on: row.planned_starts_on,
930                    planned_ends_on: row.planned_ends_on,
931                    task_count: row.task_count,
932                    completed_task_count: row.completed_task_count + passed_check_count,
933                });
934        }
935    }
936    Ok(support_plans)
937}
