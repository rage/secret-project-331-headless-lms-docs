52pub async fn insert(
53    conn: &mut PgConnection,
54    pkey_policy: PKeyPolicy<Uuid>,
55    page_id: Uuid,
56    title: &str,
57    content: &PageHistoryContent,
58    history_change_reason: HistoryChangeReason,
59    author: HistoryAuthor,
60    restored_from_id: Option<Uuid>,
61) -> ModelResult<Uuid> {
62    let res = sqlx::query!(
63        "
64INSERT INTO page_history (
65    id,
66    page_id,
67    title,
68    content,
69    history_change_reason,
70    author_user_id,
71    restored_from_id,
72    agent_call_id
73  )
74VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
75RETURNING id
76        ",
77        pkey_policy.into_uuid(),
78        page_id,
79        title,
80        serde_json::to_value(content)?,
81        history_change_reason as HistoryChangeReason,
82        author.user_id,
83        restored_from_id,
84        author.agent_call_id,
85    )
86    .fetch_one(&mut *conn)
87    .await?;
88    // A restore has to be able to bring back the private specs in this snapshot, so the files they
89    // name must stay out of the abandoned-upload reaper's reach for as long as the snapshot exists.
90    // Only the private ones: a restore re-derives the other two from them.
91    let spec_files: Vec<Uuid> = content
92        .exercise_tasks
93        .iter()
94        .flat_map(|task| task.private_spec_files.iter().copied())
95        .collect();
96    crate::page_history_spec_files::insert_many(conn, res.id, &spec_files).await?;
97    Ok(res.id)
98}

106pub async fn get_history_data(conn: &mut PgConnection, id: Uuid) -> ModelResult<PageHistoryData> {
107    let record = sqlx::query!(
108        "
109SELECT page_history.content,
110  page_history.title,
111  pages.exam_id
112FROM page_history
113  JOIN pages ON pages.id = page_history.page_id
114WHERE page_history.id = $1
115  AND pages.deleted_at IS NULL
116  AND page_history.deleted_at IS NULL
117        ",
118        id,
119    )
120    .fetch_one(conn)
121    .await?;
122    Ok(PageHistoryData {
123        content: serde_json::from_value(record.content)?,
124        title: record.title,
125        exam_id: record.exam_id,
126    })
127}

156pub async fn history(
157    conn: &mut PgConnection,
158    page_id: Uuid,
159    pagination: Pagination,
160) -> ModelResult<Vec<PageHistory>> {
161    let res = sqlx::query_as!(
162        PageHistory,
163        r#"
164SELECT h.id AS "id!",
165  h.created_at AS "created_at!",
166  h.updated_at AS "updated_at!",
167  h.deleted_at,
168  h.title AS "title!",
169  h.content AS "content!",
170  h.history_change_reason AS "history_change_reason!",
171  h.restored_from_id,
172  h.author_user_id AS "author_user_id!",
173  h.page_id AS "page_id!",
174  c.client_name AS "agent_client_name?"
175FROM page_history h
176  LEFT JOIN agent_changeset_calls c ON c.id = h.agent_call_id
177WHERE h.page_id = $1
178  AND h.deleted_at IS NULL
179ORDER BY h.created_at DESC,
180  h.id
181LIMIT $2 OFFSET $3
182"#,
183        page_id,
184        pagination.limit(),
185        pagination.offset()
186    )
187    .fetch_all(conn)
188    .await?;
189    Ok(res)
190}

207/// Latest non-deleted `page_history` row id per page for pages in the given courses.
208pub async fn get_latest_page_history_ids_by_course_ids(
209    conn: &mut PgConnection,
210    course_ids: &[Uuid],
211) -> ModelResult<HashMap<Uuid, Uuid>> {
212    let rows = sqlx::query!(
213        r#"
214SELECT DISTINCT ON (ph.page_id)
215  ph.id,
216  ph.page_id
217FROM page_history ph
218  INNER JOIN pages p ON p.id = ph.page_id
219WHERE p.course_id = ANY($1)
220  AND ph.deleted_at IS NULL
221ORDER BY ph.page_id,
222  ph.created_at DESC,
223  ph.id DESC
224"#,
225        course_ids
226    )
227    .fetch_all(conn)
228    .await?;
229
230    Ok(rows.into_iter().map(|row| (row.page_id, row.id)).collect())
