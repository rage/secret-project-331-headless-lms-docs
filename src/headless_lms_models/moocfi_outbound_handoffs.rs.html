1//! Single-use handoffs that sign a student in on mooc.fi when they leave courses for it.

7/// Where on mooc.fi a handoff sends the student; mooc.fi maps it to its own path.
8#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, Type, ToSchema)]
9#[sqlx(type_name = "moocfi_handoff_destination", rename_all = "snake_case")]
10#[serde(rename_all = "snake_case")]
11pub enum MoocfiHandoffDestination {
12    Certificate,
13}
14
15/// What mooc.fi learns from redeeming a handoff.
16#[derive(Debug, Clone, PartialEq)]
17pub struct RedeemedMoocfiHandoff {
18    /// The student's TMC user id.
19    pub upstream_id: i32,
20    pub moocfi_course_id: Uuid,
21    pub destination: MoocfiHandoffDestination,
22}
23
24/// Why a handoff could not be redeemed.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26pub enum RedeemRefusal {
27    /// Minted from another IP than the one redeeming it. The handoff is left unspent.
28    IpMismatch,
29    /// Redeemed already, or expired.
30    Spent,
31    /// No such handoff, pruned long ago, or for a completion or target deleted since.
32    Unknown,
33}
34
35/// Stores a handoff for `external_completion_id`, redeemable for `lifetime` from the IP
36/// `client_ip_hash` is of, and prunes handoffs expired over a day ago.
37pub async fn insert(
38    conn: &mut PgConnection,
39    token_digest: &[u8],
40    external_completion_id: Uuid,
41    destination: MoocfiHandoffDestination,
42    client_ip_hash: &str,
43    lifetime: chrono::Duration,
44) -> ModelResult<()> {
45    let mut tx = conn.begin().await?;
46    sqlx::query!(
47        r#"
48DELETE FROM moocfi_outbound_handoffs
49WHERE expires_at < now() - INTERVAL '1 day'
50        "#,
51    )
52    .execute(&mut *tx)
53    .await?;
54    sqlx::query!(
55        r#"
56INSERT INTO moocfi_outbound_handoffs (
57    token_digest,
58    external_completion_id,
59    destination,
60    client_ip_hash,
61    expires_at
62  )
63VALUES ($1, $2, $3, $4, now() + $5 * INTERVAL '1 millisecond')
64        "#,
65        token_digest,
66        external_completion_id,
67        destination as MoocfiHandoffDestination,
68        client_ip_hash,
69        lifetime.num_milliseconds() as f64,
70    )
71    .execute(&mut *tx)
72    .await?;
73    tx.commit().await?;
74    Ok(())
75}

77/// The courses account of the student a live handoff with `token_digest` is for, spent or not.
78pub async fn get_student_id(
79    conn: &mut PgConnection,
80    token_digest: &[u8],
81) -> ModelResult<Option<Uuid>> {
82    let res = sqlx::query_scalar!(
83        r#"
84SELECT ec.user_id
85FROM moocfi_outbound_handoffs h
86  JOIN external_completions ec ON ec.id = h.external_completion_id
87  JOIN external_credit_targets t ON t.id = ec.external_credit_target_id
88WHERE h.token_digest = $1
89  AND h.deleted_at IS NULL
90  AND ec.deleted_at IS NULL
91  AND t.deleted_at IS NULL
92        "#,
93        token_digest,
94    )
95    .fetch_optional(conn)
96    .await?;
97    Ok(res)
98}

100/// Consumes the handoff with `token_digest` if it was minted from the IP `client_ip_hash` is of.
101/// Atomic: of two concurrent redeems only one wins, and a redeem from another IP never spends it.
102pub async fn redeem(
103    conn: &mut PgConnection,
104    token_digest: &[u8],
105    client_ip_hash: &str,
106) -> ModelResult<Result<RedeemedMoocfiHandoff, RedeemRefusal>> {
107    let redeemed = sqlx::query_as!(
108        RedeemedMoocfiHandoff,
109        r#"
110WITH redeemed AS (
111  UPDATE moocfi_outbound_handoffs h
112  SET redeemed_at = now()
113  WHERE h.token_digest = $1
114    AND h.client_ip_hash = $2
115    AND h.redeemed_at IS NULL
116    AND h.expires_at > now()
117    AND h.deleted_at IS NULL
118    AND EXISTS (
119      SELECT 1
120      FROM external_completions ec
121        JOIN external_credit_targets t ON t.id = ec.external_credit_target_id
122      WHERE ec.id = h.external_completion_id
123        AND ec.deleted_at IS NULL
124        AND t.deleted_at IS NULL
125    )
126  RETURNING h.external_completion_id,
127    h.destination
128)
129SELECT ec.moocfi_user_upstream_id AS upstream_id,
130  t.moocfi_course_id,
131  r.destination AS "destination: MoocfiHandoffDestination"
132FROM redeemed r
133  JOIN external_completions ec ON ec.id = r.external_completion_id
134  JOIN external_credit_targets t ON t.id = ec.external_credit_target_id
135        "#,
136        token_digest,
137        client_ip_hash,
138    )
139    .fetch_optional(&mut *conn)
140    .await?;
141    if let Some(redeemed) = redeemed {
142        return Ok(Ok(redeemed));
143    }
144    let is_bound = sqlx::query_scalar!(
145        r#"
146SELECT h.client_ip_hash = $2 AS "is_bound!"
147FROM moocfi_outbound_handoffs h
148  JOIN external_completions ec ON ec.id = h.external_completion_id
149  JOIN external_credit_targets t ON t.id = ec.external_credit_target_id
150WHERE h.token_digest = $1
151  AND h.deleted_at IS NULL
152  AND ec.deleted_at IS NULL
153  AND t.deleted_at IS NULL
154        "#,
155        token_digest,
156        client_ip_hash,
157    )
158    .fetch_optional(conn)
159    .await?;
160    Ok(Err(match is_bound {
161        Some(false) => RedeemRefusal::IpMismatch,
162        Some(true) => RedeemRefusal::Spent,
163        None => RedeemRefusal::Unknown,
164    }))
165}
