25/// The products whose access tokens are worth refreshing: those configured on an enabled, unpaused
26/// module, least recently attempted first. One product can back several modules, so each appears
27/// once.
28///
29/// Ordered by the last attempt rather than the last success, so a product whose refresh keeps
30/// failing rotates to the back instead of holding the head of the queue and starving everything
31/// behind it.
32pub async fn get_stalest_product_ids_for_enabled_modules(
33    conn: &mut PgConnection,
34    limit: i64,
35    course_id: Option<Uuid>,
36) -> ModelResult<Vec<String>> {
37    let res = sqlx::query_scalar!(
38        r#"
39SELECT c.open_university_product_id AS "open_university_product_id!"
40FROM course_module_suotar_configurations c
41  JOIN course_modules cm ON cm.id = c.course_module_id
42  LEFT JOIN open_university_product_access_tokens t ON t.open_university_product_id = c.open_university_product_id
43  AND t.deleted_at IS NULL
44WHERE c.open_university_product_id IS NOT NULL
45  AND c.paused_at IS NULL
46  AND c.deleted_at IS NULL
47  AND cm.enable_credit_registration_via_suotar
48  AND cm.deleted_at IS NULL
49  AND ($2::uuid IS NULL OR cm.course_id = $2)
50GROUP BY c.open_university_product_id,
51  t.last_refreshed_at,
52  t.last_refresh_failed_at
53ORDER BY GREATEST(t.last_refreshed_at, t.last_refresh_failed_at) ASC NULLS FIRST,
54  c.open_university_product_id
55LIMIT $1
56        "#,
57        limit,
58        course_id,
59    )
60    .fetch_all(conn)
61    .await?;
62    Ok(res)
63}

65/// Whether the module already has a live configuration row. Lets a caller tell "nothing to store"
66/// apart from "the teacher cleared what was stored", which look the same in an edit payload.
67pub async fn exists(conn: &mut PgConnection, course_module_id: Uuid) -> ModelResult<bool> {
68    let res = sqlx::query_scalar!(
69        r#"
70SELECT EXISTS (
71    SELECT 1
72    FROM course_module_suotar_configurations
73    WHERE course_module_id = $1
74      AND deleted_at IS NULL
75  ) AS "exists!"
76        "#,
77        course_module_id,
78    )
79    .fetch_one(conn)
80    .await?;
81    Ok(res)
82}

84/// Writes the module's Suotar configuration, creating the row if the module has none. The pause and
85/// config-check columns are left alone; their writers are separate.
86///
87/// Resurrects a soft-deleted row rather than inserting beside it: `ON CONFLICT` can only infer
88/// against `uq_course_module_suotar_configurations`, which is keyed on `course_module_id` alone.
89pub async fn upsert(
90    conn: &mut PgConnection,
91    course_module_id: Uuid,
92    open_university_product_id: Option<&str>,
93    grade_scale_id: Option<&str>,
94) -> ModelResult<CourseModuleSuotarConfiguration> {
95    let res = sqlx::query_as!(
96        CourseModuleSuotarConfiguration,
97        r#"
98INSERT INTO course_module_suotar_configurations (
99    course_module_id,
100    open_university_product_id,
101    grade_scale_id
102  )
103VALUES ($1, $2, $3) ON CONFLICT (course_module_id) DO
104UPDATE
105SET open_university_product_id = $2,
106  grade_scale_id = $3,
107  deleted_at = NULL
108RETURNING *
109        "#,
110        course_module_id,
111        open_university_product_id,
112        grade_scale_id,
113    )
114    .fetch_one(conn)
115    .await?;
116    Ok(res)
117}

119/// Who paused a module's credit registration, and why. One value rather than three arguments
120/// because `course_module_suotar_configurations_pause_pair` rejects a timestamp without an actor.
121#[derive(Debug, Clone)]
122pub struct SuotarPause<'a> {
123    pub paused_at: DateTime<Utc>,
124    pub paused_by_user_id: Uuid,
125    pub reason: Option<&'a str>,
126}

128/// Pauses or resumes the module. Every phase's claim query skips a paused module, so pausing freezes
129/// its ledger rows where they stand instead of cancelling them. `None` resumes.
130pub async fn set_paused(
131    conn: &mut PgConnection,
132    course_module_id: Uuid,
133    pause: Option<SuotarPause<'_>>,
134) -> ModelResult<()> {
135    let pause = pause.as_ref();
136    sqlx::query!(
137        r#"
138UPDATE course_module_suotar_configurations
139SET paused_at = $2,
140  paused_by_user_id = $3,
141  pause_reason = $4
142WHERE course_module_id = $1
143  AND deleted_at IS NULL
144        "#,
145        course_module_id,
146        pause.map(|pause| pause.paused_at),
147        pause.map(|pause| pause.paused_by_user_id),
148        pause.and_then(|pause| pause.reason),
149    )
150    .execute(conn)
151    .await?;
152    Ok(())
153}
