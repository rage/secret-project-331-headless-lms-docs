8/// Backdates `state_entered_at` for a registration, so a test can simulate a row that has been
9/// sitting in its state long enough for a backoff or timeout to fire.
10///
11/// Exists only for test setup: [`transition`](super::transition::transition) owns this stamp, and calling this
12/// from a live path would desynchronize it from the state it is supposed to describe.
13pub async fn set_state_entered_at_for_testing(
14    conn: &mut PgConnection,
15    id: Uuid,
16    state_entered_at: DateTime<Utc>,
17) -> ModelResult<()> {
18    sqlx::query!(
19        "
20UPDATE credit_registrations
21SET state_entered_at = $2
22WHERE id = $1
23        ",
24        id,
25        state_entered_at,
26    )
27    .execute(conn)
28    .await?;
29    Ok(())
30}

56/// Excuses every one of a user's rows (or, with `course_id`, just that course's) from every
57/// unscoped claim until `held_until`; a scoped claim ignores every hold regardless. Keyed on
58/// identity rather than a row id so a spec can hold before materialize creates the row it means to
59/// protect, closing the window a row-id hold could only ever narrow: the live background worker
60/// ticks every 10s regardless of any single test, so a hold applied after the row exists still
61/// races the worker's own next tick.
62///
63/// Exists only for test setup: nothing in the product ever needs to hide a user's rows from the
64/// worker that owns them.
65pub async fn set_test_exclusive_hold_for_testing(
66    conn: &mut PgConnection,
67    user_id: Uuid,
68    course_id: Option<Uuid>,
69    held_until: DateTime<Utc>,
70) -> ModelResult<()> {
71    sqlx::query!(
72        "
73INSERT INTO credit_registration_test_exclusive_holds (user_id, course_id, held_until)
74VALUES ($1, $2, $3)
75        ",
76        user_id,
77        course_id,
78        held_until,
79    )
80    .execute(conn)
81    .await?;
82    Ok(())
83}

85/// Backdates the row's last enrolment check and last check request past the limit on asking, so a
86/// spec can press a recheck button without waiting it out. With `clear_restarts`, also forgets the
87/// day's restarts. Exists only for test setup.
88pub async fn expire_enrolment_recheck_allowance_for_testing(
89    conn: &mut PgConnection,
90    id: Uuid,
91    clear_restarts: bool,
92) -> ModelResult<()> {
93    use crate::library::credit_registration::enrolment_check_schedule::CHECK_REQUEST_MIN_INTERVAL;
94    sqlx::query!(
95        "
96UPDATE credit_registrations
97SET enrolment_checked_at = enrolment_checked_at - $2::interval,
98  enrolment_check_requested_at = enrolment_check_requested_at - $2::interval,
99  enrolment_check_restart_count = CASE
100    WHEN $3 THEN 0
101    ELSE enrolment_check_restart_count
102  END
103WHERE id = $1
104        ",
105        id,
106        CHECK_REQUEST_MIN_INTERVAL as TimeDelta,
107        clear_restarts,
108    )
109    .execute(conn)
110    .await?;
111    Ok(())
112}

114/// Makes every row in `scope` that waits for an enrolment check due now, as if its next rung had
115/// come, and returns how many it moved. Exists only for test setup: specs cannot wait out a day for
116/// a row's first check.
117pub async fn make_enrolment_checks_due_for_testing(
118    conn: &mut PgConnection,
119    scope: &RegistrationScope,
120) -> ModelResult<u64> {
121    let res = sqlx::query!(
122        r#"
123UPDATE credit_registrations
124SET next_attempt_at = now()
125WHERE state = 'no_usable_enrolment'
126  AND next_attempt_at > now()
127  AND deleted_at IS NULL
128  AND ($1::uuid IS NULL OR course_id = $1)
129  AND ($2::uuid IS NULL OR user_id = $2)
130  AND (
131    cardinality($3::uuid []) = 0
132    OR id = ANY($3::uuid [])
133  )
134        "#,
135        scope.course_id,
136        scope.user_id,
137        &scope.credit_registration_ids,
138    )
139    .execute(conn)
140    .await?;
141    Ok(res.rows_affected())
142}
