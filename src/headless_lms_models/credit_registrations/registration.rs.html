141/// Whether a row is waiting for an enrolment: parked without a usable one, check schedule started
142/// or not, or on its first check or a retry on its way there. Only such a row is moved by a visit
143/// or a check request, and kept waiting through a lookup that fails in transit.
144pub fn is_waiting_for_enrolment(
145    state: CreditRegistrationState,
146    enrolment_check_anchor_at: Option<DateTime<Utc>>,
147    no_usable_enrolment_since: Option<DateTime<Utc>>,
148) -> bool {
149    state.keeps_enrolment_check_schedule()
150        && (enrolment_check_anchor_at.is_some() || no_usable_enrolment_since.is_some())
151}

153/// A completion a ledger row can register, named by its own id and kind: what the per-completion
154/// side tables (enrolment check signals, enrolment routes) are keyed by. Its [`id`](Self::id) is the
155/// ledger's `subject_id`.
156#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
157pub enum SubjectId {
158    CourseModuleCompletion(Uuid),
159    ExternalCompletion(Uuid),
160}
161
162impl SubjectId {
163    /// The completion's id in its own table, which is also the ledger's `subject_id`.
164    pub fn id(self) -> Uuid {
165        match self {
166            Self::CourseModuleCompletion(id) | Self::ExternalCompletion(id) => id,
167        }
168    }
169
170    /// The id for a `course_module_completion_id` column, `None` for a mooc.fi completion.
171    pub fn course_module_completion_id(self) -> Option<Uuid> {
172        match self {
173            Self::CourseModuleCompletion(id) => Some(id),
174            Self::ExternalCompletion(_) => None,
175        }
176    }
177
178    /// The id for an `external_completion_id` column, `None` for a course module completion.
179    pub fn external_completion_id(self) -> Option<Uuid> {
180        match self {
181            Self::CourseModuleCompletion(_) => None,
182            Self::ExternalCompletion(id) => Some(id),
183        }
184    }
185
186    /// The completion `registration` registers.
187    pub fn of(registration: &CreditRegistration) -> Self {
188        match registration.external_completion_id {
189            Some(id) => Self::ExternalCompletion(id),
190            None => Self::CourseModuleCompletion(registration.subject_id),
191        }
192    }
193}

195/// The completion a new ledger row registers, with the ids its kind of row carries.
196#[derive(Debug, Clone, PartialEq)]
197pub enum NewRegistrationSubject {
198    CourseModule {
199        course_module_completion_id: Uuid,
200        course_id: Uuid,
201        course_module_id: Uuid,
202        course_instance_id: Uuid,
203    },
204    Moocfi {
205        external_completion_id: Uuid,
206        external_credit_target_id: Uuid,
207    },
208}
209
210impl NewRegistrationSubject {
211    /// The subject a ledger row's id columns name, as read back from the row; errors on a mix
212    /// `credit_registrations_one_subject` refuses.
213    pub fn from_columns(
214        course_module_completion_id: Option<Uuid>,
215        course_id: Option<Uuid>,
216        course_module_id: Option<Uuid>,
217        course_instance_id: Option<Uuid>,
218        external_completion_id: Option<Uuid>,
219        external_credit_target_id: Option<Uuid>,
220    ) -> ModelResult<Self> {
221        match (
222            course_module_completion_id,
223            course_id,
224            course_module_id,
225            course_instance_id,
226            external_completion_id,
227            external_credit_target_id,
228        ) {
229            (
230                Some(course_module_completion_id),
231                Some(course_id),
232                Some(course_module_id),
233                Some(course_instance_id),
234                None,
235                None,
236            ) => Ok(Self::CourseModule {
237                course_module_completion_id,
238                course_id,
239                course_module_id,
240                course_instance_id,
241            }),
242            (
243                None,
244                None,
245                None,
246                None,
247                Some(external_completion_id),
248                Some(external_credit_target_id),
249            ) => Ok(Self::Moocfi {
250                external_completion_id,
251                external_credit_target_id,
252            }),
253            _ => Err(model_err!(
254                Generic,
255                "A ledger row names neither exactly a course module completion nor a mooc.fi one."
256            )),
257        }
258    }
259}

268/// Creates a ledger row at `pending` with a `created` event.
269pub async fn insert(
270    conn: &mut PgConnection,
271    pkey_policy: PKeyPolicy<Uuid>,
272    new: &NewCreditRegistration,
273    event_message: Option<&str>,
274) -> ModelResult<Uuid> {
275    let id = pkey_policy.into_uuid();
276    let (
277        course_module_completion_id,
278        course_id,
279        course_module_id,
280        course_instance_id,
281        external_completion_id,
282        external_credit_target_id,
283    ) = match new.subject {
284        NewRegistrationSubject::CourseModule {
285            course_module_completion_id,
286            course_id,
287            course_module_id,
288            course_instance_id,
289        } => (
290            Some(course_module_completion_id),
291            Some(course_id),
292            Some(course_module_id),
293            Some(course_instance_id),
294            None,
295            None,
296        ),
297        NewRegistrationSubject::Moocfi {
298            external_completion_id,
299            external_credit_target_id,
300        } => (
301            None,
302            None,
303            None,
304            None,
305            Some(external_completion_id),
306            Some(external_credit_target_id),
307        ),
308    };
309    let mut tx = conn.begin().await?;
310    sqlx::query!(
311        r#"
312INSERT INTO credit_registrations (
313    id,
314    course_module_completion_id,
315    user_id,
316    course_id,
317    course_module_id,
318    course_instance_id,
319    external_completion_id,
320    external_credit_target_id,
321    attempt_number
322  )
323VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
324        "#,
325        id,
326        course_module_completion_id,
327        new.user_id,
328        course_id,
329        course_module_id,
330        course_instance_id,
331        external_completion_id,
332        external_credit_target_id,
333        new.attempt_number,
334    )
335    .execute(&mut *tx)
336    .await?;
337
338    crate::credit_registration_events::insert(
339        &mut tx,
340        &NewCreditRegistrationEvent {
341            message: event_message.map(str::to_string),
342            ..NewCreditRegistrationEvent::new(id, CreditRegistrationEventKind::Created)
343        },
344    )
345    .await?;
346
347    tx.commit().await?;
348    Ok(id)
349}

367/// The named rows, locked until the caller's transaction ends. Must be called inside one.
368///
369/// For a caller that judges each row and then transitions it: holding the lock is what keeps the
370/// judgement true, so [`transition`](super::transition::transition)'s `expected_from_state` cannot fail halfway
371/// and roll the whole batch back. Locks in id order, which every batch caller shares, so two of
372/// them cannot deadlock.
373pub async fn get_by_ids_for_update(
374    conn: &mut PgConnection,
375    ids: &[Uuid],
376) -> ModelResult<Vec<CreditRegistration>> {
377    let res = sqlx::query_as!(
378        CreditRegistration,
379        r#"
380SELECT *
381FROM credit_registrations
382WHERE id = ANY($1::uuid [])
383  AND deleted_at IS NULL
384ORDER BY id FOR UPDATE
385        "#,
386        ids
387    )
388    .fetch_all(conn)
389    .await?;
390    Ok(res)
391}

433/// Whether this account has any attempt, live or replaced, on this course.
434///
435/// For course-scoped handlers that take a user id from a request body: without it, holding one
436/// course lets a teacher ask questions about accounts that have nothing to do with it.
437pub async fn exists_for_user_and_course(
438    conn: &mut PgConnection,
439    user_id: Uuid,
440    course_id: Uuid,
441) -> ModelResult<bool> {
442    let exists = sqlx::query_scalar!(
443        r#"
444SELECT EXISTS (
445    SELECT 1
446    FROM credit_registrations
447    WHERE user_id = $1
448      AND course_id = $2
449      AND deleted_at IS NULL
450  ) AS "exists!"
451        "#,
452        user_id,
453        course_id,
454    )
455    .fetch_one(conn)
456    .await?;
457    Ok(exists)
