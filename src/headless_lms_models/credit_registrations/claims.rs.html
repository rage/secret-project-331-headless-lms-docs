1//! Claiming due rows for a worker phase.
2//!
3//! Every claim takes up to `limit` due rows in its states, the longest due first. The row locks
4//! live until the caller's transaction ends, so callers must pass a transaction. Rows on a paused
5//! course module, or on one whose credit registration has been switched off, are never claimed:
6//! enforced here so no phase can forget it. Both freeze a row where it stands rather than
7//! cancelling it, so switching the module back on resumes the rows that were already in flight.
8//!
9//! An unscoped claim (the live background worker) also skips a row whose user (and, if the hold
10//! names one, course) has a live row in `credit_registration_test_exclusive_holds`. A scoped claim
11//! always ignores holds, so a spec driving its own rows through explicit ticks is unaffected
12//! either way.
13

21/// Which rows a phase iteration may touch. Empty means every row, which is what production runs; a
22/// narrowed scope lets a test drive the pipeline for its own course on a shared database.
23#[derive(Debug, Clone, Default, PartialEq)]
24pub struct RegistrationScope {
25    pub course_id: Option<Uuid>,
26    pub user_id: Option<Uuid>,
27    /// The precision escape hatch, for a caller that already knows its ledger rows.
28    pub credit_registration_ids: Vec<Uuid>,
29}
30
31impl RegistrationScope {
32    pub fn is_unscoped(&self) -> bool {
33        self.course_id.is_none()
34            && self.user_id.is_none()
35            && self.credit_registration_ids.is_empty()
36    }
37
38    pub fn for_course(course_id: Uuid) -> Self {
39        Self {
40            course_id: Some(course_id),
41            ..Self::default()
42        }
43    }
44}

46/// The states resolve-enrolments looks up from: rows on their way to a first resolve, and rows
47/// parked in `no_usable_enrolment` whose next enrolment check is due, which are checked where they
48/// stand.
49const LOOKUP_STATES: [CreditRegistrationState; 2] = [
50    CreditRegistrationState::ReadyToSubmit,
51    CreditRegistrationState::NoUsableEnrolment,
52];
53
54/// Claims, for the person lookup that precedes resolve-enrolments, the rows
55/// [`claim_due_for_resolve`] would take, the later [`EnrolmentCheckGroup`](crate::library::credit_registration::enrolment_check_schedule::EnrolmentCheckGroup) first.
56pub async fn claim_due_for_person_lookup(
57    conn: &mut PgConnection,
58    scope: &RegistrationScope,
59    limit: i64,
60) -> ModelResult<Vec<CreditRegistration>> {
61    claim(conn, &LOOKUP_STATES, scope, limit, ClaimKind::PersonLookup).await
62}
63
64/// Claims, for resolve-enrolments, `ready_to_submit` rows and parked rows due an enrolment check,
65/// the later [`EnrolmentCheckGroup`](crate::library::credit_registration::enrolment_check_schedule::EnrolmentCheckGroup) first, minus any whose student already has another live row
66/// for the module somewhere between resolving and a known outcome. First pulls slow checks due soon
67/// into the batch; see
68/// [`crate::library::credit_registration::enrolment_checks::pull_forward_batched_checks`].
69///
70/// Only one completion per student and module goes past resolve at a time, so each is weighed
71/// against the outcome of the one before it rather than racing it to the registry; see
72/// [`lock_live_successes_for_same_module`](super::lock_live_successes_for_same_module). Of two such
73/// rows claimed together only the first comes back; the other stays claimable where it is. A parked
74/// row sent for a check must be marked with [`claim_enrolment_checks`] before the claim's
75/// transaction commits.
76pub async fn claim_due_for_resolve(
77    conn: &mut PgConnection,
78    scope: &RegistrationScope,
79    limit: i64,
80) -> ModelResult<Vec<CreditRegistration>> {
81    crate::library::credit_registration::enrolment_checks::pull_forward_batched_checks(conn, scope)
82        .await?;
83    let claimed = claim(conn, &LOOKUP_STATES, scope, limit, ClaimKind::Resolve).await?;
84    Ok(first_per(
85        claimed,
86        |row| (row.user_id, row.course_module_id),
87        "another attempt for the same student and module",
88    ))
89}

91/// Keeps parked rows out of every claim while a lookup for them is out, as `resolving_enrolment`
92/// does for a row on its first resolve. The answer's [`transition`](super::transition::transition) ends the
93/// claim; one a worker died holding expires after
94/// [`RESOLVING_RECOVERY_GRACE`](crate::library::credit_registration::backoff::RESOLVING_RECOVERY_GRACE).
95/// A no-op for a row in any other state.
96pub async fn claim_enrolment_checks(conn: &mut PgConnection, ids: &[Uuid]) -> ModelResult<()> {
97    use crate::library::credit_registration::backoff::RESOLVING_RECOVERY_GRACE;
98    if ids.is_empty() {
99        return Ok(());
100    }
101    sqlx::query!(
102        r#"
103UPDATE credit_registrations
104SET enrolment_check_claimed_until = now() + $2::interval
105WHERE id = ANY($1)
106  AND state = 'no_usable_enrolment'
107        "#,
108        ids,
109        RESOLVING_RECOVERY_GRACE as TimeDelta,
110    )
111    .execute(conn)
112    .await?;
113    Ok(())
114}

116/// Claims, for import, `checking_enrolment` rows, minus any whose student and course code already
117/// have a submission in flight, which Suotar's hour-old copy of Sisu would not stop from
118/// registering twice, and any whose person and module slot in
119/// `uq_credit_registrations_person_module` is taken. Of two rows for the same student and course
120/// code claimed together only the first comes back; the other stays claimable where it is.
121pub async fn claim_due_for_import(
122    conn: &mut PgConnection,
123    scope: &RegistrationScope,
124    limit: i64,
125) -> ModelResult<Vec<CreditRegistration>> {
126    let claimed = claim(
127        conn,
128        &[CreditRegistrationState::CheckingEnrolment],
129        scope,
130        limit,
131        ClaimKind::Import,
132    )
133    .await?;
134    Ok(first_per(
135        claimed,
136        |row| {
137            (
138                row.student_number
139                    .as_ref()
140                    .map(|number| number.expose_secret().to_string()),
141                row.uh_course_code.clone(),
142            )
143        },
144        "another attempt for the same student and course code",
145    ))
146}

148/// The first of the claimed rows per `key`. The claim query holds back a row whose twin is already
149/// in flight, but cannot see a twin claimed alongside it; the rows dropped here keep their lock
150/// until the claim's transaction ends, and are claimable where they are after it.
151fn first_per<K: Eq + std::hash::Hash>(
152    claimed: Vec<CreditRegistration>,
153    key: impl Fn(&CreditRegistration) -> K,
154    twin: &str,
155) -> Vec<CreditRegistration> {
156    let mut seen = std::collections::HashSet::new();
157    claimed
158        .into_iter()
159        .filter(|row| {
160            let is_first = seen.insert(key(row));
161            if !is_first {
162                debug!(
163                    credit_registration_id = %row.id,
164                    "Leaving row claimable: {twin} is already in this claim"
165                );
166            }
167            is_first
168        })
169        .collect()
170}

172/// The states verify polls from. Withdrawal moves a row out of both, which is what stops the
173/// polling without any query having to know about withdrawal.
174const VERIFY_STATES: [CreditRegistrationState; 2] = [
175    CreditRegistrationState::AwaitingVerification,
176    CreditRegistrationState::SubmissionUncertain,
177];
178
179/// Which of verify's flows a claim is for.
180#[derive(Debug, Clone, Copy, PartialEq, Eq)]
181pub enum VerifyFlow {
182    /// Rows with a submitted attainment to poll by, and `awaiting_verification` rows stuck without
183    /// one.
184    Poll,
185    /// `submission_uncertain` rows with no submitted attainment id, looked up through
186    /// resolve-enrolments instead.
187    UncertainRecovery,
188}
189
190/// Claims rows for one of verify's flows. Each flow is claimed on its own, so that rows one flow
191/// has no allowance to send cannot fill the other's claim.
192pub async fn claim_due_for_verify(
193    conn: &mut PgConnection,
194    flow: VerifyFlow,
195    scope: &RegistrationScope,
196    limit: i64,
197) -> ModelResult<Vec<CreditRegistration>> {
198    claim(conn, &VERIFY_STATES, scope, limit, ClaimKind::Verify(flow)).await
199}

201/// Which caller a claim is for, which decides the rows that hold a row back and the order.
202#[derive(Debug, Clone, Copy, PartialEq, Eq)]
203enum ClaimKind {
204    /// See [`claim_due_for_verify`].
205    Verify(VerifyFlow),
206    /// See [`claim_due_for_person_lookup`].
207    PersonLookup,
208    /// See [`claim_due_for_resolve`].
209    Resolve,
210    /// See [`claim_due_for_import`].
211    Import,
212}
213
214/// Shares its eligibility filters with
215/// [`count_due_enrolment_checks`](super::count_due_enrolment_checks) and
216/// [`pull_forward_batched_checks`](crate::library::credit_registration::enrolment_checks::pull_forward_batched_checks);
217/// change all three together.
218async fn claim(
219    conn: &mut PgConnection,
220    states: &[CreditRegistrationState],
221    scope: &RegistrationScope,
222    limit: i64,
223    kind: ClaimKind,
224) -> ModelResult<Vec<CreditRegistration>> {
225    let ignores_test_holds = !scope.is_unscoped();
226    let excludes_twins_in_flight = kind == ClaimKind::Import;
227    let waits_behind_other_attempts = kind == ClaimKind::Resolve;
228    let orders_by_check_group = matches!(kind, ClaimKind::PersonLookup | ClaimKind::Resolve);
229    // `Some(true)` claims only uncertain rows without a submitted attainment, `Some(false)` every
230    // other verify row.
231    let only_uncertain_without_attainment_id = match kind {
232        ClaimKind::Verify(flow) => Some(flow == VerifyFlow::UncertainRecovery),
233        _ => None,
234    };
235    let res = sqlx::query_as!(
236        CreditRegistration,
237        r#"
238WITH due AS (
239  SELECT cr.id
240  FROM credit_registrations cr
241    JOIN credit_registration_active_course_modules acm ON acm.course_module_id = cr.course_module_id
242    JOIN course_module_completions cmc ON cmc.id = cr.course_module_completion_id
243  WHERE cr.deleted_at IS NULL
244    AND cr.superseded_by_id IS NULL
245    -- A completion opted out by hand is the pull path's again; only a row already sent carries on.
246    AND (
247      cmc.register_credits_via_suotar
248      OR cr.submitted_at IS NOT NULL
249    )
250    AND cr.state = ANY($1::credit_registration_state [])
251    AND cr.next_attempt_at <= now()
252    AND (
253      cr.enrolment_check_claimed_until IS NULL
254      OR cr.enrolment_check_claimed_until <= now()
255    )
256    AND ($3::uuid IS NULL OR cr.course_id = $3)
257    AND ($4::uuid IS NULL OR cr.user_id = $4)
258    AND (
259      cardinality($5::uuid []) = 0
260      OR cr.id = ANY($5::uuid [])
261    )
262    AND (
263      $12::boolean IS NULL
264      OR (
265        cr.state = 'submission_uncertain'
266        AND cr.submitted_attainment_id IS NULL
267      ) = $12
268    )
269    AND (
270      $6::boolean
271      OR NOT EXISTS (
272        SELECT 1
273        FROM credit_registration_test_exclusive_holds h
274        WHERE h.user_id = cr.user_id
275          AND (
276            h.course_id IS NULL
277            OR h.course_id = cr.course_id
278          )
279          AND h.held_until > now()
280      )
281    )
282    AND (
283      NOT $7::boolean
284      OR (
285        NOT EXISTS (
286          SELECT 1
287          FROM credit_registrations twin
288          WHERE twin.student_number = cr.student_number
289            AND twin.uh_course_code = cr.uh_course_code
290            AND twin.id <> cr.id
291            AND twin.deleted_at IS NULL
292            AND twin.state = ANY($10::credit_registration_state [])
293        )
294        -- Mirrors uq_credit_registrations_person_module, which moving to submitting would violate.
295        AND NOT EXISTS (
296          SELECT 1
297          FROM credit_registrations holder
298          WHERE holder.sisu_person_id = cr.sisu_person_id
299            AND holder.course_module_id = cr.course_module_id
300            AND holder.id <> cr.id
301            AND holder.deleted_at IS NULL
302            AND holder.superseded_by_id IS NULL
303            AND holder.pending_superseded_by_id IS NULL
304            AND (
305              holder.state = ANY($10::credit_registration_state [])
306              OR holder.state = ANY($11::credit_registration_state [])
307            )
308        )
309      )
310    )
311    AND (
312      NOT $8::boolean
313      OR NOT EXISTS (
314        SELECT 1
315        FROM credit_registrations ahead
316        WHERE ahead.user_id = cr.user_id
317          AND ahead.course_module_id = cr.course_module_id
318          AND ahead.id <> cr.id
319          AND ahead.deleted_at IS NULL
320          AND ahead.superseded_by_id IS NULL
321          -- failed_retryable because its backoff may resume it at checking_enrolment or later.
322          AND (
323            ahead.state IN (
324              'resolving_enrolment',
325              'checking_enrolment',
326              'failed_retryable'
327            )
328            OR ahead.state = ANY($10::credit_registration_state [])
329            OR ahead.enrolment_check_claimed_until > now()
330          )
331      )
332    )
333  ORDER BY CASE
334      WHEN $9 THEN cr.enrolment_check_group
335    END DESC NULLS LAST,
336    cr.next_attempt_at
337  FOR UPDATE OF cr SKIP LOCKED
338  LIMIT $2
339)
340UPDATE credit_registrations cr
341SET last_attempt_at = now()
342FROM due
343WHERE cr.id = due.id
344RETURNING cr.*
345        "#,
346        states as &[CreditRegistrationState],
347        limit,
348        scope.course_id,
349        scope.user_id,
350        &scope.credit_registration_ids,
351        ignores_test_holds,
352        excludes_twins_in_flight,
353        waits_behind_other_attempts,
354        orders_by_check_group,
355        &CreditRegistrationState::IN_FLIGHT_STATES as &[CreditRegistrationState],
356        &CreditRegistrationState::SUCCESS_STATES as &[CreditRegistrationState],
357        only_uncertain_without_attainment_id,
358    )
359    .fetch_all(conn)
360    .await?;
361    Ok(res)
