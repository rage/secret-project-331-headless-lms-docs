87impl AdminCreditRegistration {
88    /// What decides whether a human may move this row; see [`ResubmissionFacts`].
89    pub fn resubmission_facts(&self) -> ResubmissionFacts {
90        ResubmissionFacts {
91            state: self.state,
92            is_superseded: self.superseded_by_id.is_some(),
93            error_code: self.error_code,
94            resubmit_not_before: self.resubmit_not_before,
95            submitted_at: self.submitted_at,
96        }
97    }
98
99    /// See [`is_waiting_for_enrolment`].
100    pub fn is_waiting_for_enrolment(&self) -> bool {
101        is_waiting_for_enrolment(
102            self.state,
103            self.enrolment_check_anchor_at,
104            self.no_usable_enrolment_since,
105        )
106    }
107
108    /// What a `pending` row is waiting on, as the row's precondition flags say.
109    pub fn preconditions(&self) -> PendingPreconditions {
110        PendingPreconditions {
111            completion_eligible: self.completion_eligible,
112            has_verified_student_number: self.has_verified_student_number,
113            course_code_allowed: self.course_code_allowed,
114        }
115    }
116
117    /// Where the row stands on the admin timeline.
118    pub fn position(&self) -> TimelinePosition {
119        TimelinePosition::of(
120            self.state,
121            self.preconditions(),
122            self.selected_enrolment_id.is_some(),
123            Engagement::of(self.pressed_at, self.last_visited_at),
124        )
125    }
126
127    /// When the row entered its current phase. A row held for its course code moved there from
128    /// waiting for a student number when the number was linked, which changed no state.
129    pub fn phase_started_at(&self) -> DateTime<Utc> {
130        match self.position().step {
131            TimelineStep::HeldForCourseCode => self
132                .verified_student_number_at
133                .map_or(self.phase_started_at, |linked_at| {
134                    linked_at.max(self.phase_started_at)
135                }),
136            _ => self.phase_started_at,
137        }
138    }
139
140    /// What this row is waiting on, or `None` where it is not waiting at all: outside `pending` the
141    /// preconditions say nothing about why the row is where it is.
142    pub fn pending_reason(&self) -> Option<CreditRegistrationPendingReason> {
143        (self.state == CreditRegistrationState::Pending)
144            .then(|| self.preconditions().reason())
145            .flatten()
146    }
147}

149/// How the explorer orders a page. Descending only: an ops table is read newest-worst first.
150#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
151pub enum AdminCreditRegistrationSort {
152    #[default]
153    LastActivity,
154    Created,
155    TimeInState,
156    Attempts,
157}
158
159impl AdminCreditRegistrationSort {
160    /// Bound into the query's `ORDER BY` as a `text` parameter.
161    fn as_str(self) -> &'static str {
162        match self {
163            Self::LastActivity => "last_activity",
164            Self::Created => "created",
165            Self::TimeInState => "time_in_state",
166            Self::Attempts => "attempts",
167        }
168    }
169}

171/// The narrowings the admin explorer applies, all of them in SQL.
172#[derive(Debug, Clone, Default)]
173pub struct AdminCreditRegistrationFilters<'a> {
174    pub states: Option<&'a [CreditRegistrationState]>,
175    pub error_codes: Option<&'a [CreditRegistrationErrorCode]>,
176    pub course_id: Option<Uuid>,
177    pub course_module_id: Option<Uuid>,
178    pub user_id: Option<Uuid>,
179    pub student_number: Option<&'a str>,
180    /// Rows at one of these timeline steps. Empty means no narrowing. A row waiting for a student
181    /// number matches only if it completed since `account_linking_since`, as the linking page and
182    /// [`count_by_step_and_engagement`] count them.
183    pub steps: &'a [TimelineStep],
184    /// Rows with one of these engagements, at steps that wait on the student. Empty means no
185    /// narrowing.
186    pub engagements: &'a [Engagement],
187    /// Leaves out not-started rows at steps that wait on the student.
188    pub hide_not_started: bool,
189    pub account_linking_since: Option<DateTime<Utc>>,
190    /// Listed first, in the sort order among themselves.
191    pub first_ids: Option<&'a [Uuid]>,
192    pub submitted_after: Option<DateTime<Utc>>,
193    pub submitted_before: Option<DateTime<Utc>>,
194    /// Matched against the student's name and email, either student number, the attainment ids and
195    /// the stored error text. Searching that text is not rendering it.
196    pub search: Option<&'a str>,
197    /// A uuid typed into the search box: a registration, a user or a completion id. Ambiguous by
198    /// design, for a human's paste. A caller that already knows which single field it means should
199    /// use `id` or `course_module_completion_id` instead, not this plus a Rust-side filter.
200    pub search_id: Option<Uuid>,
201    /// Exactly one registration.
202    pub id: Option<Uuid>,
203    /// Every attempt against one completion.
204    pub course_module_completion_id: Option<Uuid>,
205    /// An exact set of rows, for a caller that already knows which ones it wants.
206    pub credit_registration_ids: Option<&'a [Uuid]>,
207    /// Off by default, or a course that regrades shows two rows per student.
208    pub include_superseded: bool,
209}

211/// The query behind [`get_admin_facing`]. `total_count` is computed before the limit, so every row
212/// of a page carries the count of the whole filtered ledger.
213async fn admin_facing_page(
214    conn: &mut PgConnection,
215    filters: &AdminCreditRegistrationFilters<'_>,
216    sort: AdminCreditRegistrationSort,
217    limit: i64,
218    offset: i64,
219) -> ModelResult<Vec<AdminCreditRegistration>> {
220    let search_pattern = filters.search.map(search_pattern_of);
221    let step_match = StepMatch::all();
222    let res = sqlx::query_as!(
223        AdminCreditRegistration,
224        r#"
225SELECT cr.id,
226  cr.created_at,
227  cr.user_id,
228  ud.first_name AS "first_name?",
229  ud.last_name AS "last_name?",
230  ud.email AS "email?",
231  cr.course_id,
232  c.name AS course_name,
233  cr.course_module_id,
234  cm.name AS course_module_name,
235  cr.course_instance_id,
236  cr.course_module_completion_id,
237  cmc.completion_date,
238  cr.state,
239  cr.state_entered_at,
240  cr.state_changed_at,
241  cr.phase_started_at,
242  cr.error_code AS "error_code?",
243  cr.needs_admin_attention,
244  cr.next_attempt_at,
245  cr.last_attempt_at,
246  cr.submitted_at,
247  cr.registered_at,
248  cr.terminal_at,
249  cr.student_number,
250  cr.sisu_person_id,
251  cr.uh_course_code,
252  cr.selected_enrolment_id,
253  cr.grade_scale_id,
254  cr.grade_id,
255  cr.credits,
256  cr.submitted_attainment_id,
257  cr.sisu_attainment_id,
258  cr.submit_retry_count,
259  cr.verify_attempt_count,
260  cr.attempt_number,
261  cr.superseded_by_id,
262  vsn.student_number AS "verified_student_number?",
263  vsn.verified_at AS "verified_student_number_at?",
264  vsn.verified_via AS "verified_student_number_via?",
265  cr.resubmit_not_before,
266  cr.partially_registered_at,
267  cr.not_registered_reimport_count,
268  cr.no_usable_enrolment_since,
269  cr.enrolment_checked_at,
270  cr.enrolment_check_anchor_at,
271  cr.enrolment_check_due_at,
272  cr.enrolment_checks_stopped_at,
273  p.completion_eligible AS "completion_eligible!",
274  p.has_verified_student_number AS "has_verified_student_number!",
275  p.course_code_allowed AS "course_code_allowed!",
276  route.enrolment_confirmed_at AS "pressed_at?",
277  route.route AS "enrolment_route?: CreditRegistrationEnrolmentRoute",
278  sig.last_visited_at AS "last_visited_at?",
279  COUNT(*) OVER () AS "total_count!"
280FROM credit_registrations cr
281  JOIN courses c ON c.id = cr.course_id
282  JOIN course_modules cm ON cm.id = cr.course_module_id
283  JOIN course_module_completions cmc ON cmc.id = cr.course_module_completion_id
284  JOIN credit_registration_preconditions p ON p.credit_registration_id = cr.id
285  LEFT JOIN user_details ud ON ud.user_id = cr.user_id
286  LEFT JOIN verified_student_numbers vsn ON vsn.user_id = cr.user_id
287  AND vsn.deleted_at IS NULL
288  LEFT JOIN credit_registration_enrolment_routes route ON route.course_module_completion_id = cr.course_module_completion_id
289  AND route.deleted_at IS NULL
290  LEFT JOIN credit_registration_enrolment_check_signals sig ON sig.course_module_completion_id = cr.course_module_completion_id
291  AND sig.deleted_at IS NULL
292  -- The step and engagement are computed exactly as in count_by_step_and_engagement, so a list
293  -- filtered to a step and engagement holds as many rows as that function counts.
294  LEFT JOIN LATERAL (
295    SELECT m.step
296    FROM UNNEST(
297        $19::credit_registration_state [],
298        $20::boolean [],
299        $21::boolean [],
300        $22::boolean [],
301        $23::boolean [],
302        $24::credit_registration_timeline_step []
303      ) AS m(
304        state,
305        completion_eligible,
306        has_verified_student_number,
307        course_code_allowed,
308        enrolment_resolved,
309        step
310      )
311    WHERE m.state = cr.state
312      AND m.completion_eligible = p.completion_eligible
313      AND m.has_verified_student_number = p.has_verified_student_number
314      AND m.course_code_allowed = p.course_code_allowed
315      AND m.enrolment_resolved = (cr.selected_enrolment_id IS NOT NULL)
316  ) ts ON TRUE
317  CROSS JOIN LATERAL (
318    SELECT CASE
319        WHEN ts.step = ANY($25::credit_registration_timeline_step []) THEN CASE
320          WHEN route.enrolment_confirmed_at IS NOT NULL THEN 'pressed'::credit_registration_engagement
321          WHEN sig.last_visited_at IS NOT NULL THEN 'visited'::credit_registration_engagement
322          ELSE 'not_started'::credit_registration_engagement
323        END
324      END AS engagement,
325      ts.step = 'waiting_for_student_number'
326      AND cmc.completion_date < $29::timestamptz AS is_before_linking
327  ) eng
328WHERE cr.deleted_at IS NULL
329  AND ($1::bool OR cr.superseded_by_id IS NULL)
330  AND (
331    $2::credit_registration_state [] IS NULL
332    OR cr.state = ANY($2)
333  )
334  AND (
335    $3::credit_registration_error_code [] IS NULL
336    OR cr.error_code = ANY($3)
337  )
338  AND ($4::uuid IS NULL OR cr.course_id = $4)
339  AND ($5::uuid IS NULL OR cr.course_module_id = $5)
340  AND ($6::uuid IS NULL OR cr.user_id = $6)
341  AND (
342    $7::text IS NULL
343    OR cr.student_number = $7
344    OR vsn.student_number = $7
345  )
346  AND (
347    CARDINALITY($8::credit_registration_timeline_step []) = 0
348    OR (
349      ts.step = ANY($8)
350      AND eng.is_before_linking IS NOT TRUE
351    )
352  )
353  AND (
354    CARDINALITY($26::credit_registration_engagement []) = 0
355    OR eng.engagement = ANY($26)
356  )
357  AND (
358    NOT $27::bool
359    OR eng.engagement IS DISTINCT FROM 'not_started'
360  )
361  AND ($9::timestamptz IS NULL OR cr.submitted_at >= $9)
362  AND ($10::timestamptz IS NULL OR cr.submitted_at <= $10)
363  AND (
364    $11::text IS NULL
365    OR ud.name_search_helper LIKE '%' || $11 || '%' ESCAPE '\'
366    OR ud.email_search_helper LIKE '%' || $11 || '%' ESCAPE '\'
367    OR LOWER(cr.student_number) LIKE '%' || $11 || '%' ESCAPE '\'
368    OR LOWER(vsn.student_number) LIKE '%' || $11 || '%' ESCAPE '\'
369    OR LOWER(cr.submitted_attainment_id) LIKE '%' || $11 || '%' ESCAPE '\'
370    OR LOWER(cr.sisu_attainment_id) LIKE '%' || $11 || '%' ESCAPE '\'
371    OR LOWER(cr.error_message) LIKE '%' || $11 || '%' ESCAPE '\'
372  )
373  AND (
374    $12::uuid IS NULL
375    OR cr.id = $12
376    OR cr.user_id = $12
377    OR cr.course_module_completion_id = $12
378  )
379  AND (
380    $13::uuid [] IS NULL
381    OR cr.id = ANY($13)
382  )
383  AND ($17::uuid IS NULL OR cr.id = $17)
384  AND (
385    $18::uuid IS NULL
386    OR cr.course_module_completion_id = $18
387  )
388ORDER BY COALESCE(cr.id = ANY($28::uuid []), FALSE) DESC,
389  CASE
390    WHEN $14::text = 'attempts' THEN cr.submit_retry_count + cr.verify_attempt_count
391  END DESC NULLS LAST,
392  CASE $14::text
393    WHEN 'created' THEN cr.created_at
394    WHEN 'time_in_state' THEN cr.state_changed_at
395    ELSE COALESCE(cr.last_attempt_at, cr.state_entered_at)
396  END DESC,
397  cr.id
398LIMIT $15 OFFSET $16
399        "#,
400        filters.include_superseded,
401        filters.states as Option<&[CreditRegistrationState]>,
402        filters.error_codes as Option<&[CreditRegistrationErrorCode]>,
403        filters.course_id,
404        filters.course_module_id,
405        filters.user_id,
406        filters.student_number,
407        filters.steps as &[TimelineStep],
408        filters.submitted_after,
409        filters.submitted_before,
410        search_pattern.as_deref(),
411        filters.search_id,
412        filters.credit_registration_ids as Option<&[Uuid]>,
413        sort.as_str(),
414        limit,
415        offset,
416        filters.id,
417        filters.course_module_completion_id,
418        &step_match.states as &[CreditRegistrationState],
419        &step_match.completion_eligible,
420        &step_match.has_verified_student_number,
421        &step_match.course_code_allowed,
422        &step_match.enrolment_resolved,
423        &step_match.steps as &[TimelineStep],
424        &TimelineStep::ENGAGEMENT_STEPS as &[TimelineStep],
425        filters.engagements as &[Engagement],
426        filters.hide_not_started,
427        filters.first_ids as Option<&[Uuid]>,
428        filters.account_linking_since,
429    )
430    .fetch_all(conn)
431    .await?;
432    Ok(res)
433}

446/// Live rows of one module at one step, with one engagement where the step waits on the student.
447#[derive(Debug, Clone, PartialEq)]
448pub struct StepCount {
449    pub course_module_id: Uuid,
450    pub step: TimelineStep,
451    pub engagement: Option<Engagement>,
452    pub count: i64,
453}

455/// Live rows per module, step and engagement: the overview's "Where registrations stand", the
456/// Courses tab's per-module table and the daily snapshot.
457///
458/// Counts exactly what [`get_admin_facing`] lists filtered to the same step, engagement and module,
459/// superseded rows left out: a row waiting for a student number is counted only if it completed
460/// since `account_linking_since`.
461pub async fn count_by_step_and_engagement(
462    conn: &mut PgConnection,
463    account_linking_since: Option<DateTime<Utc>>,
464) -> ModelResult<Vec<StepCount>> {
465    let step_match = StepMatch::all();
466    let res = sqlx::query_as!(
467        StepCount,
468        r#"
469SELECT cr.course_module_id,
470  ts.step AS "step!: TimelineStep",
471  eng.engagement AS "engagement?: Engagement",
472  COUNT(*) AS "count!"
473FROM credit_registrations cr
474  JOIN course_module_completions cmc ON cmc.id = cr.course_module_completion_id
475  JOIN credit_registration_preconditions p ON p.credit_registration_id = cr.id
476  LEFT JOIN credit_registration_enrolment_routes route ON route.course_module_completion_id = cr.course_module_completion_id
477  AND route.deleted_at IS NULL
478  LEFT JOIN credit_registration_enrolment_check_signals sig ON sig.course_module_completion_id = cr.course_module_completion_id
479  AND sig.deleted_at IS NULL
480  -- Computed exactly as in admin_facing_page; see there.
481  JOIN LATERAL (
482    SELECT m.step
483    FROM UNNEST(
484        $1::credit_registration_state [],
485        $2::boolean [],
486        $3::boolean [],
487        $4::boolean [],
488        $5::boolean [],
489        $6::credit_registration_timeline_step []
490      ) AS m(
491        state,
492        completion_eligible,
493        has_verified_student_number,
494        course_code_allowed,
495        enrolment_resolved,
496        step
497      )
498    WHERE m.state = cr.state
499      AND m.completion_eligible = p.completion_eligible
500      AND m.has_verified_student_number = p.has_verified_student_number
501      AND m.course_code_allowed = p.course_code_allowed
502      AND m.enrolment_resolved = (cr.selected_enrolment_id IS NOT NULL)
503  ) ts ON TRUE
504  CROSS JOIN LATERAL (
505    SELECT CASE
506        WHEN ts.step = ANY($7::credit_registration_timeline_step []) THEN CASE
507          WHEN route.enrolment_confirmed_at IS NOT NULL THEN 'pressed'::credit_registration_engagement
508          WHEN sig.last_visited_at IS NOT NULL THEN 'visited'::credit_registration_engagement
509          ELSE 'not_started'::credit_registration_engagement
510        END
511      END AS engagement,
512      ts.step = 'waiting_for_student_number'
513      AND cmc.completion_date < $8::timestamptz AS is_before_linking
514  ) eng
515WHERE cr.deleted_at IS NULL
516  AND cr.superseded_by_id IS NULL
517  AND eng.is_before_linking IS NOT TRUE
518GROUP BY cr.course_module_id,
519  ts.step,
520  eng.engagement
521        "#,
522        &step_match.states as &[CreditRegistrationState],
523        &step_match.completion_eligible,
524        &step_match.has_verified_student_number,
525        &step_match.course_code_allowed,
526        &step_match.enrolment_resolved,
527        &step_match.steps as &[TimelineStep],
528        &TimelineStep::ENGAGEMENT_STEPS as &[TimelineStep],
529        account_linking_since,
530    )
531    .fetch_all(conn)
532    .await?;
533    Ok(res)
534}

536/// [`count_by_step_and_engagement`]'s rows summed over modules, in timeline order.
537pub fn sum_step_counts_over_modules<'a>(
538    rows: impl IntoIterator<Item = &'a StepCount>,
539) -> BTreeMap<(TimelineStep, Option<Engagement>), i64> {
540    let mut totals = BTreeMap::new();
541    for row in rows {
542        *totals.entry((row.step, row.engagement)).or_insert(0) += row.count;
543    }
544    totals
545}

547/// Live rows in each of the given states, newest activity first within each state, for the
548/// Reconciliation lists. `limit_per_state` caps every state independently, via `ROW_NUMBER`, so one
549/// state with many rows cannot crowd another out of a shared `LIMIT`.
550pub async fn get_live_by_states(
551    conn: &mut PgConnection,
552    states: &[CreditRegistrationState],
553    limit_per_state: i64,
554) -> ModelResult<Vec<CreditRegistration>> {
555    let res = sqlx::query_as!(
556        CreditRegistration,
557        r#"
558SELECT cr.*
559FROM credit_registrations cr
560  JOIN (
561    SELECT id,
562      ROW_NUMBER() OVER (
563        PARTITION BY state
564        ORDER BY state_entered_at DESC
565      ) AS rn
566    FROM credit_registrations
567    WHERE state = ANY($1)
568      AND superseded_by_id IS NULL
569      AND deleted_at IS NULL
570  ) ranked ON ranked.id = cr.id
571WHERE ranked.rn <= $2
572ORDER BY cr.state_entered_at DESC
573        "#,
574        states as &[CreditRegistrationState],
575        limit_per_state,
576    )
577    .fetch_all(conn)
578    .await?;
579    Ok(res)
580}

582/// What the student did and when, around one registration's completion, for the admin timeline.
583#[derive(Debug, Clone, PartialEq)]
584pub struct AdminRegistrationJourney {
585    /// When they first picked any language version of the course.
586    pub course_started_at: Option<DateTime<Utc>>,
587    /// When the completion's page signals were first recorded: the first visit, unless a check
588    /// request came first.
589    pub first_visited_at: Option<DateTime<Utc>>,
590    pub last_visited_at: Option<DateTime<Utc>>,
591    pub last_check_requested_at: Option<DateTime<Utc>>,
592    pub check_request_source: Option<EnrolmentCheckSource>,
593    pub enrolment_route: Option<CreditRegistrationEnrolmentRoute>,
594    pub pressed_at: Option<DateTime<Utc>>,
595    /// Sisu's own enrolment time, from the latest enrolment check that found it.
596    pub checked_enrolled_at: Option<DateTime<Utc>>,
597    /// Sisu's own enrolment time as the selected enrolment's lookup answer gave it, unparsed.
598    pub selected_enrolment_date_time: Option<String>,
599}
600
601/// The journey around `credit_registration_id`'s completion. `None` for no such live row.
602pub async fn get_registration_journey(
603    conn: &mut PgConnection,
604    credit_registration_id: Uuid,
605) -> ModelResult<Option<AdminRegistrationJourney>> {
606    let res = sqlx::query_as!(
607        AdminRegistrationJourney,
608        r#"
609SELECT ucs.created_at AS "course_started_at?",
610  CASE
611    WHEN sig.last_visited_at IS NOT NULL THEN LEAST(sig.created_at, sig.last_visited_at)
612  END AS "first_visited_at?",
613  sig.last_visited_at AS "last_visited_at?",
614  sig.last_check_requested_at AS "last_check_requested_at?",
615  sig.check_request_source AS "check_request_source?: EnrolmentCheckSource",
616  route.route AS "enrolment_route?: CreditRegistrationEnrolmentRoute",
617  route.enrolment_confirmed_at AS "pressed_at?",
618  (
619    SELECT o.enrolled_at
620    FROM credit_registration_enrolment_check_outcomes o
621      JOIN credit_registrations attempt ON attempt.id = o.credit_registration_id
622    WHERE attempt.course_module_completion_id = cr.course_module_completion_id
623      AND o.is_enrolment_found
624      AND o.enrolled_at IS NOT NULL
625    ORDER BY o.checked_at DESC
626    LIMIT 1
627  ) AS "checked_enrolled_at?",
628  (
629    SELECT found.value #>> '{}'
630    FROM credit_registration_events e
631      CROSS JOIN LATERAL jsonb_path_query_first(
632        e.details,
633        'lax $.** ? (@.id == $id).enrolmentDateTime',
634        jsonb_build_object('id', cr.selected_enrolment_id)
635      ) AS found(value)
636    WHERE e.credit_registration_id = cr.id
637      AND e.deleted_at IS NULL
638      AND cr.selected_enrolment_id IS NOT NULL
639      AND found.value IS NOT NULL
640    ORDER BY e.created_at DESC
641    LIMIT 1
642  ) AS "selected_enrolment_date_time?"
643FROM credit_registrations cr
644  JOIN courses c ON c.id = cr.course_id
645  LEFT JOIN user_course_settings ucs ON ucs.user_id = cr.user_id
646  AND ucs.course_language_group_id = c.course_language_group_id
647  AND ucs.deleted_at IS NULL
648  LEFT JOIN credit_registration_enrolment_check_signals sig ON sig.course_module_completion_id = cr.course_module_completion_id
649  AND sig.deleted_at IS NULL
650  LEFT JOIN credit_registration_enrolment_routes route ON route.course_module_completion_id = cr.course_module_completion_id
651  AND route.deleted_at IS NULL
652WHERE cr.id = $1
653  AND cr.deleted_at IS NULL
654        "#,
655        credit_registration_id,
656    )
657    .fetch_optional(conn)
658    .await?;
659    Ok(res)
660}
