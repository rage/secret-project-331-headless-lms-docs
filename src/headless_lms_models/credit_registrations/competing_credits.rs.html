1//! The other credits of a student's course code that a new attempt is weighed against before it is
2//! sent, whichever module or mooc.fi target they came from: the study registry holds one attainment
3//! per person and course unit.
4
5use super::state::CreditRegistrationState;
6use crate::library::credit_registration::grade_mapping::{GradeSource, MappedGrade, map_grade};
7use crate::prelude::*;
8
9/// Another live attempt, of any completion of the same student and course code, that the study
10/// registry already holds.
11#[derive(Debug, Clone)]
12pub struct LiveSuccessForCourseCode {
13    pub id: Uuid,
14    pub credit: RecordedCredit,
15}
16
17/// A credit for the student and course code that our own records say the study registry holds.
18#[derive(Debug, Clone)]
19pub struct RecordedCredit {
20    /// `None` for a credit the pull path registered, which freezes nothing.
21    pub grade_scale_id: Option<String>,
22    pub grade_id: Option<String>,
23    pub completion_passed: bool,
24    pub completion_grade: Option<i32>,
25}
26
27impl RecordedCredit {
28    /// The grade the registry holds: the frozen one, or for a credit with nothing frozen (settled
29    /// before a freeze, or registered by the pull path), its completion's, which the registry held
30    /// at least as well.
31    pub fn held_grade(&self) -> Option<MappedGrade> {
32        MappedGrade::from_columns(self.grade_scale_id.as_deref(), self.grade_id.as_deref()).or_else(
33            || {
34                map_grade(GradeSource {
35                    passed: self.completion_passed,
36                    grade: self.completion_grade,
37                    enrolment_grade_scale_id: None,
38                })
39                .ok()
40            },
41        )
42    }
43}

45/// Every credit for the row's student and course code that our records say the registry holds:
46/// another of our live attempts in a success state, or a registrar's pull-path registration. The
47/// row's code is its unit's current one, which is what a payload frozen now would carry; credits of
48/// the same unit count whatever code they were frozen with.
49///
50/// Not [`lock_live_successes_for_same_course_code`], which locks, and leaves out the pull path
51/// because nothing there can be marked for replacement.
52pub async fn get_recorded_credits_for_same_course_code(
53    conn: &mut PgConnection,
54    id: Uuid,
55) -> ModelResult<Vec<RecordedCredit>> {
56    let res = sqlx::query_as!(
57        RecordedCredit,
58        r#"
59WITH this AS (
60  SELECT cr.id,
61    cr.user_id,
62    cr.credit_unit_id,
63    UPPER(TRIM(s.uh_course_code)) AS course_code
64  FROM credit_registrations cr
65    JOIN credit_registration_subjects s ON s.subject_id = cr.subject_id
66  WHERE cr.id = $1
67)
68SELECT other.grade_scale_id::text AS "grade_scale_id?",
69  other.grade_id::text AS "grade_id?",
70  os.passed IS TRUE AS "completion_passed!",
71  os.grade AS "completion_grade?"
72FROM this
73  JOIN credit_registrations other ON other.user_id = this.user_id
74  AND (
75    UPPER(TRIM(other.uh_course_code)) = this.course_code
76    OR other.credit_unit_id = this.credit_unit_id
77  )
78  AND other.id <> this.id
79  JOIN credit_registration_subjects os ON os.subject_id = other.subject_id
80WHERE other.deleted_at IS NULL
81  AND other.superseded_by_id IS NULL
82  AND other.state = ANY($2::credit_registration_state [])
83UNION ALL
84SELECT NULL::text,
85  NULL::text,
86  cmc.passed,
87  cmc.grade
88FROM this
89  JOIN course_module_completion_registered_to_study_registries r ON r.user_id = this.user_id
90  JOIN course_modules cm ON cm.id = r.course_module_id
91  AND (
92    UPPER(TRIM(cm.uh_course_code)) = this.course_code
93    OR cm.id = this.credit_unit_id
94  )
95  JOIN course_module_completions cmc ON cmc.id = r.course_module_completion_id
96  -- A null registrar is our own mirror of a row the first half already reads.
97WHERE r.study_registry_registrar_id IS NOT NULL
98  AND r.deleted_at IS NULL
99        "#,
100        id,
101        &CreditRegistrationState::SUCCESS_STATES as &[CreditRegistrationState],
102    )
103    .fetch_all(conn)
104    .await?;
105    Ok(res)
106}

108/// Readies a row that resolve-enrolments settles as `duplicate` without sending it, in the caller's
109/// transaction and before the transition.
110///
111/// Drops the row's claim on `uq_credit_registrations_person_course_code`, left from an earlier
112/// freeze, since the slot belongs to the credit the registry holds, and the frozen credits, which
113/// nothing registered. Keeps `weighed_grade`, the grade we would have sent, as the grade a later
114/// regrade of the completion has to beat.
115pub async fn prepare_unsent_duplicate(
116    conn: &mut PgConnection,
117    id: Uuid,
118    weighed_grade: Option<&MappedGrade>,
119) -> ModelResult<()> {
120    sqlx::query!(
121        r#"
122UPDATE credit_registrations
123SET sisu_person_id = NULL,
124  grade_scale_id = $2,
125  grade_id = $3,
126  credits = NULL
127WHERE id = $1
128  AND deleted_at IS NULL
129        "#,
130        id,
131        weighed_grade.map(|grade| grade.grade_scale_id.as_str()),
132        weighed_grade.map(|grade| grade.grade_id.as_str()),
133    )
134    .execute(conn)
135    .await?;
136    Ok(())
137}

139/// The student's other live attempts for the row's course code that the registry already holds,
140/// locked until the caller's transaction ends. The row's code is its unit's current one, as in
141/// [`get_recorded_credits_for_same_course_code`].
142///
143/// What another attempt has to beat to be sent, and what it replaces when it does, through
144/// [`mark_pending_superseded`](super::mark_pending_superseded). Includes rows already waiting on an
145/// earlier replacement, which a better one takes over.
146pub async fn lock_live_successes_for_same_course_code(
147    conn: &mut PgConnection,
148    id: Uuid,
149) -> ModelResult<Vec<LiveSuccessForCourseCode>> {
150    let rows = sqlx::query!(
151        r#"
152WITH this AS (
153  SELECT cr.id,
154    cr.user_id,
155    cr.credit_unit_id,
156    UPPER(TRIM(s.uh_course_code)) AS course_code
157  FROM credit_registrations cr
158    JOIN credit_registration_subjects s ON s.subject_id = cr.subject_id
159  WHERE cr.id = $1
160)
161SELECT other.id,
162  other.grade_scale_id,
163  other.grade_id,
164  os.passed IS TRUE AS "completion_passed!",
165  os.grade AS "completion_grade?"
166FROM this
167  JOIN credit_registrations other ON other.user_id = this.user_id
168  AND (
169    UPPER(TRIM(other.uh_course_code)) = this.course_code
170    OR other.credit_unit_id = this.credit_unit_id
171  )
172  AND other.id <> this.id
173  JOIN credit_registration_subjects os ON os.subject_id = other.subject_id
174WHERE other.deleted_at IS NULL
175  AND other.superseded_by_id IS NULL
176  AND other.state = ANY($2::credit_registration_state [])
177ORDER BY other.id FOR
178UPDATE OF other
179        "#,
180        id,
181        &CreditRegistrationState::SUCCESS_STATES as &[CreditRegistrationState],
182    )
183    .fetch_all(conn)
184    .await?;
185    Ok(rows
186        .into_iter()
187        .map(|row| LiveSuccessForCourseCode {
188            id: row.id,
189            credit: RecordedCredit {
190                grade_scale_id: row.grade_scale_id,
191                grade_id: row.grade_id,
192                completion_passed: row.completion_passed,
193                completion_grade: row.completion_grade,
194            },
195        })
196        .collect())
197}
