23pub async fn insert_course_audiences(
24    conn: &mut PgConnection,
25    course_id: Uuid,
26    audiences: Vec<String>,
27    embeddings: Vec<Vec<f32>>,
28) -> ModelResult<Vec<CourseAudience>> {
29    let embed_vecs: Vec<Vector> = embeddings.into_iter().map(Vector::from).collect();
30    let res = sqlx::query_as!(
31        CourseAudience,
32        r#"
33INSERT INTO course_audiences (
34    course_id,
35    audience,
36    embedding
37  )
38SELECT $1,
39       t.audience,
40       t.embedding
41FROM UNNEST(
42    $2::text[],
43    $3::vector[]
44) AS t(audience, embedding)
45RETURNING *
46    "#,
47        course_id,
48        &audiences,
49        &embed_vecs as _
50    )
51    .fetch_all(conn)
52    .await?;
53    Ok(res)
54}

95pub async fn get_course_ids_by_audience_vectors(
96    conn: &mut PgConnection,
97    audience_vecs: Vec<Vec<f32>>,
98    audience_keywords: Vec<String>,
99) -> ModelResult<Vec<Uuid>> {
100    let vectors: Vec<Vector> = audience_vecs.into_iter().map(Vector::from).collect();
101    let res = sqlx::query_scalar!(
102        r#"
103SELECT course_id
104FROM (
105    SELECT
106        a.course_id,
107        MIN(a.embedding <#> v.embedding) AS distance
108    FROM course_audiences a
109    CROSS JOIN unnest($1::vector[]) AS v(embedding)
110    WHERE deleted_at IS NULL
111    GROUP BY a.course_id
112    ORDER BY distance ASC
113    LIMIT 5
114) t
115UNION ALL
116SELECT DISTINCT a.course_id
117FROM course_audiences a
118CROSS JOIN unnest($2::text[]) AS k(keyword)
119WHERE deleted_at IS NULL
120AND a.audience % k.keyword
121        "#,
122        &vectors as _,
123        &audience_keywords
124    )
125    .fetch_all(conn)
126    .await?;
127    Ok(res.into_iter().flatten().collect())
128}
