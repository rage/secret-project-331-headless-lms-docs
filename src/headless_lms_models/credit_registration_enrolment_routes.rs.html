1//! What a student said about enrolling, on the new registration page: the route, the questions
2//! that steer students who would struggle to enrol towards a certificate, and the certificate
3//! choice.
4//!
5//! Advisory throughout: the pipeline never reads these rows. They decide which enrolment
6//! instructions the page shows and how far its step list has come, and the study registry is what
7//! actually settles whether an enrolment exists.
8

14/// Which university relationship a student picked, which decides only where they are told to enrol.
15#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, Type, ToSchema)]
16#[sqlx(
17    type_name = "credit_registration_enrolment_route",
18    rename_all = "snake_case"
19)]
20#[serde(rename_all = "snake_case")]
21pub enum CreditRegistrationEnrolmentRoute {
22    UniversityOfHelsinki,
23    OpenUniversity,
24}
25
26/// Whether a student said they need a certificate or credits in the study registry.
27#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, Type, ToSchema)]
28#[sqlx(type_name = "credit_registration_need", rename_all = "snake_case")]
29#[serde(rename_all = "snake_case")]
30pub enum CreditRegistrationNeed {
31    Certificate,
32    Credits,
33}

35/// How a student without a Finnish personal identity code said they can identify themselves for

42#[serde(rename_all = "snake_case")]

48    NoSuomiFi,

51/// The student's answers to the enrolment questions, written together and replaced as a whole.
52///
53/// The detour answers are kept whichever way the answers above them point, so a student who
54/// changes an earlier answer and changes it back finds their place.
55#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
56pub struct EnrolmentAnswers {
57    pub route: CreditRegistrationEnrolmentRoute,
58    pub has_finnish_personal_identity_code: Option<bool>,
59    pub need: Option<CreditRegistrationNeed>,
60    pub identification: Option<CreditRegistrationIdentification>,
61    /// What they need when asked again after saying they cannot identify themselves.
62    pub need_after_reconsider: Option<CreditRegistrationNeed>,
63}
64
65#[derive(Debug, Serialize, Deserialize, PartialEq, Clone, ToSchema)]
66pub struct EnrolmentRouteAnswer {
67    /// The completion answered about, whichever kind it is.
68    pub subject_id: Uuid,
69    pub route: CreditRegistrationEnrolmentRoute,
70    pub has_finnish_personal_identity_code: Option<bool>,
71    pub need: Option<CreditRegistrationNeed>,
72    pub identification: Option<CreditRegistrationIdentification>,
73    pub need_after_reconsider: Option<CreditRegistrationNeed>,
74    /// When the student pressed Done. `None` until they do, and cleared again if they take it back.
75    pub enrolment_confirmed_at: Option<DateTime<Utc>>,
76    /// When the student took the certificate instead of registering credits. `None` until they
77    /// do, and cleared again if they take it back.
78    pub certificate_chosen_at: Option<DateTime<Utc>>,
79}

81pub async fn get_by_subject(
82    conn: &mut PgConnection,
83    subject: SubjectId,
84) -> ModelResult<Option<EnrolmentRouteAnswer>> {
85    let res = sqlx::query_as!(
86        EnrolmentRouteAnswer,
87        r#"
88SELECT COALESCE(course_module_completion_id, external_completion_id) AS "subject_id!",
89  route AS "route: _",
90  has_finnish_personal_identity_code,
91  need AS "need: _",
92  identification AS "identification: _",
93  need_after_reconsider AS "need_after_reconsider: _",
94  enrolment_confirmed_at,
95  certificate_chosen_at
96FROM credit_registration_enrolment_routes
97WHERE COALESCE(course_module_completion_id, external_completion_id) = $1
98  AND deleted_at IS NULL
99        "#,
100        subject.id(),
101    )
102    .fetch_optional(conn)
103    .await?;
104    Ok(res)
105}

107/// Records the student's answers, replacing any earlier ones.
108///
109/// Changing the route clears the Done that went with it: the confirmation was about enrolling the
110/// way the old route described, and it says nothing about the new one.
111pub async fn set_answers(
112    conn: &mut PgConnection,
113    subject: SubjectId,
114    user_id: Uuid,
115    answers: &EnrolmentAnswers,
116) -> ModelResult<EnrolmentRouteAnswer> {
117    let res = sqlx::query_as!(
118        EnrolmentRouteAnswer,
119        r#"
120INSERT INTO credit_registration_enrolment_routes (
121    course_module_completion_id,
122    external_completion_id,
123    user_id,
124    route,
125    has_finnish_personal_identity_code,
126    need,
127    identification,
128    need_after_reconsider
129  )
130VALUES ($1, $8, $2, $3, $4, $5, $6, $7)
131ON CONFLICT (
132    (COALESCE(course_module_completion_id, external_completion_id)),
133    deleted_at
134  ) DO
135UPDATE
136SET route = EXCLUDED.route,
137  has_finnish_personal_identity_code = EXCLUDED.has_finnish_personal_identity_code,
138  need = EXCLUDED.need,
139  identification = EXCLUDED.identification,
140  need_after_reconsider = EXCLUDED.need_after_reconsider,
141  enrolment_confirmed_at = CASE
142    WHEN credit_registration_enrolment_routes.route = EXCLUDED.route THEN credit_registration_enrolment_routes.enrolment_confirmed_at
143    ELSE NULL
144  END
145RETURNING COALESCE(course_module_completion_id, external_completion_id) AS "subject_id!",
146  route AS "route: _",
147  has_finnish_personal_identity_code,
148  need AS "need: _",
149  identification AS "identification: _",
150  need_after_reconsider AS "need_after_reconsider: _",
151  enrolment_confirmed_at,
152  certificate_chosen_at
153        "#,
154        subject.course_module_completion_id(),
155        user_id,
156        answers.route as CreditRegistrationEnrolmentRoute,
157        answers.has_finnish_personal_identity_code,
158        answers.need as Option<CreditRegistrationNeed>,
159        answers.identification as Option<CreditRegistrationIdentification>,
160        answers.need_after_reconsider as Option<CreditRegistrationNeed>,
161        subject.external_completion_id(),
162    )
163    .fetch_one(conn)
164    .await?;
165    Ok(res)
166}

168/// Sets or clears the Done the student pressed to say they had enrolled.
169///
170/// Errors with not-found when the student has not answered yet.
171pub async fn set_enrolment_confirmed(
172    conn: &mut PgConnection,
173    subject: SubjectId,
174    confirmed: bool,
175) -> ModelResult<EnrolmentRouteAnswer> {
176    let res = sqlx::query_as!(
177        EnrolmentRouteAnswer,
178        r#"
179UPDATE credit_registration_enrolment_routes
180SET enrolment_confirmed_at = CASE
181    WHEN $2 THEN COALESCE(enrolment_confirmed_at, now())
182    ELSE NULL
183  END
184WHERE COALESCE(course_module_completion_id, external_completion_id) = $1
185  AND deleted_at IS NULL
186RETURNING COALESCE(course_module_completion_id, external_completion_id) AS "subject_id!",
187  route AS "route: _",
188  has_finnish_personal_identity_code,
189  need AS "need: _",
190  identification AS "identification: _",
191  need_after_reconsider AS "need_after_reconsider: _",
192  enrolment_confirmed_at,
193  certificate_chosen_at
194        "#,
195        subject.id(),
196        confirmed,
197    )
198    .fetch_one(conn)
199    .await?;
200    Ok(res)
201}

238/// Erases every answer the user gave. Called from account deletion, which is why this is a hard
239/// delete: the answers are personal circumstances the student told us.
240pub async fn delete_all_by_user_id(conn: &mut PgConnection, user_id: Uuid) -> ModelResult<()> {
241    sqlx::query!(
242        "
243DELETE FROM credit_registration_enrolment_routes
244WHERE user_id = $1
245        ",
246        user_id,
247    )
248    .execute(conn)
249    .await?;
250    Ok(())
