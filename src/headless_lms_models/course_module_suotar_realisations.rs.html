41pub async fn upsert(
42    conn: &mut PgConnection,
43    course_module_id: Uuid,
44    course_unit_realisation_id: &str,
45    label: Option<&str>,
46    active: bool,
47) -> ModelResult<Uuid> {
48    let res = sqlx::query!(
49        r#"
50INSERT INTO course_module_suotar_realisations (
51    course_module_id,
52    course_unit_realisation_id,
53    label,
54    active
55  )
56VALUES ($1, $2, $3, $4) ON CONFLICT (
57    course_module_id,
58    course_unit_realisation_id,
59    deleted_at
60  ) DO
61UPDATE
62SET label = $3,
63  active = $4
64RETURNING id
65        "#,
66        course_module_id,
67        course_unit_realisation_id,
68        label,
69        active,
70    )
71    .fetch_one(conn)
72    .await?;
73    Ok(res.id)
74}

76/// Makes the module's live realisations exactly `wanted`, soft-deleting the rest rather than setting
77/// `active = false`, which means "configured but not polled now".
78pub async fn replace_for_course_module(
79    conn: &mut PgConnection,
80    course_module_id: Uuid,
81    wanted: &[CourseModuleSuotarRealisationEdit],
82) -> ModelResult<()> {
83    let existing = get_by_course_module_id(conn, course_module_id).await?;
84    let kept: Vec<&str> = wanted
85        .iter()
86        .map(|row| row.course_unit_realisation_id.trim())
87        .filter(|id| !id.is_empty())
88        .collect();
89    for row in existing {
90        if !kept.contains(&row.course_unit_realisation_id.as_str()) {
91            soft_delete(conn, row.id).await?;
92        }
93    }
94    for row in wanted {
95        let realisation_id = row.course_unit_realisation_id.trim();
96        if realisation_id.is_empty() {
97            continue;
98        }
99        let label = row
100            .label
101            .as_deref()
102            .map(str::trim)
103            .filter(|l| !l.is_empty());
104        upsert(conn, course_module_id, realisation_id, label, row.active).await?;
105    }
106    Ok(())
107}

109/// Shared by [`get_by_course_module_id`] and [`get_by_course_id`], which differ only in which of
110/// these is `Some`.
111async fn realisations_for(
112    conn: &mut PgConnection,
113    course_module_id: Option<Uuid>,
114    course_id: Option<Uuid>,
115) -> ModelResult<Vec<CourseModuleSuotarRealisation>> {
116    let res = sqlx::query_as!(
117        CourseModuleSuotarRealisation,
118        r#"
119SELECT cmsr.id,
120  cmsr.created_at,
121  cmsr.updated_at,
122  cmsr.deleted_at,
123  cmsr.course_module_id,
124  cmsr.course_unit_realisation_id,
125  cmsr.label,
126  cmsr.active,
127  cmsr.last_listed_at,
128  cmsr.last_listed_person_count,
129  cmsr.last_already_linked_count,
130  cmsr.last_mailed_count,
131  cmsr.last_suppressed_by_dedup_count,
132  cmsr.last_suppressed_by_rate_cap_count,
133  cmsr.last_no_address_count,
134  cmsr.last_listing_attempted_at,
135  cmsr.last_listing_error AS "last_listing_error?",
136  cmsr.consecutive_listing_failures
137FROM course_module_suotar_realisations cmsr
138  JOIN course_modules cm ON cm.id = cmsr.course_module_id
139WHERE ($1::uuid IS NULL OR cmsr.course_module_id = $1)
140  AND ($2::uuid IS NULL OR cm.course_id = $2)
141  AND cm.deleted_at IS NULL
142  AND cmsr.deleted_at IS NULL
143ORDER BY cm.order_number,
144  cmsr.created_at
145        "#,
146        course_module_id,
147        course_id,
148    )
149    .fetch_all(conn)
150    .await?;
151    Ok(res)
152}

185/// Claims the realisations one discovery iteration takes, stalest attempt first.
186///
187/// Locks the rows `FOR UPDATE SKIP LOCKED` and stamps `last_listing_attempted_at` in the same
188/// transaction the caller commits before dropping the connection: the stamp is what keeps a second
189/// concurrent run from reselecting the same realisations once the lock itself is released ahead of
190/// the (potentially slow) Suotar call.
191///
192/// Ordering only: due-ness comes from the phase's own interval, never from the timestamps. Attempts
193/// order the queue rather than successes, so a realisation that keeps failing cannot starve the rest.
194pub async fn claim_stalest_for_listing(
195    conn: &mut PgConnection,
196    limit: i64,
197    course_id: Option<Uuid>,
198) -> ModelResult<Vec<RealisationToList>> {
199    let res = sqlx::query_as!(
200        RealisationToList,
201        r#"
202SELECT cmsr.id AS "id!",
203  cmsr.course_module_id AS "course_module_id!",
204  cm.course_id AS "course_id!",
205  cmsr.course_unit_realisation_id AS "course_unit_realisation_id!",
206  cm.uh_course_code AS "uh_course_code?"
207FROM course_module_suotar_realisations cmsr
208  JOIN course_modules cm ON cm.id = cmsr.course_module_id
209  LEFT JOIN course_module_suotar_configurations c ON c.course_module_id = cm.id
210  AND c.deleted_at IS NULL
211WHERE cmsr.active
212  AND cmsr.deleted_at IS NULL
213  AND cm.enable_credit_registration_via_suotar
214  AND c.paused_at IS NULL
215  AND cm.deleted_at IS NULL
216  AND ($2::uuid IS NULL OR cm.course_id = $2)
217ORDER BY COALESCE(cmsr.last_listing_attempted_at, cmsr.last_listed_at) ASC NULLS FIRST,
218  cmsr.id
219LIMIT $1
220FOR UPDATE OF cmsr SKIP LOCKED
221        "#,
222        limit,
223        course_id,
224    )
225    .fetch_all(&mut *conn)
226    .await?;
227    let ids: Vec<Uuid> = res.iter().map(|row| row.id).collect();
228    if !ids.is_empty() {
229        sqlx::query!(
230            r#"
231UPDATE course_module_suotar_realisations
232SET last_listing_attempted_at = now()
233WHERE id = ANY($1)
234            "#,
235            &ids,
236        )
237        .execute(conn)
238        .await?;
239    }
240    Ok(res)
241}

301pub async fn get_active_discovery_reports(
302    conn: &mut PgConnection,
303) -> ModelResult<Vec<RealisationDiscoveryReport>> {
304    let res = sqlx::query_as!(
305        RealisationDiscoveryReport,
306        r#"
307SELECT cmsr.id,
308  cm.course_id,
309  c.name AS course_name,
310  cmsr.course_module_id,
311  cm.name AS course_module_name,
312  cmsr.course_unit_realisation_id,
313  cmsr.label,
314  cm.uh_course_code,
315  cmsr.last_listed_at,
316  cmsr.last_listed_person_count,
317  cmsr.last_already_linked_count,
318  cmsr.last_mailed_count,
319  cmsr.last_suppressed_by_dedup_count,
320  cmsr.last_suppressed_by_rate_cap_count,
321  cmsr.last_no_address_count,
322  cmsr.last_listing_attempted_at,
323  cmsr.last_listing_error AS "last_listing_error?: CreditRegistrationErrorCode",
324  cmsr.consecutive_listing_failures
325FROM course_module_suotar_realisations cmsr
326  JOIN course_modules cm ON cm.id = cmsr.course_module_id
327  JOIN courses c ON c.id = cm.course_id
328WHERE cmsr.active
329  AND cmsr.deleted_at IS NULL
330  AND cm.deleted_at IS NULL
331  AND c.deleted_at IS NULL
332ORDER BY c.name,
333  cmsr.course_unit_realisation_id
334        "#,
335    )
336    .fetch_all(conn)
337    .await?;
338    Ok(res)
339}

341/// Records a listing attempt whose roster never arrived. `last_listed_at` and the counters keep
342/// describing the last roster that did, because zeroing them would make a failed listing read as an
343/// empty course.
344pub async fn mark_listing_failed(
345    conn: &mut PgConnection,
346    id: Uuid,
347    error: CreditRegistrationErrorCode,
348) -> ModelResult<()> {
349    sqlx::query!(
350        r#"
351UPDATE course_module_suotar_realisations
352SET last_listing_attempted_at = now(),
353  last_listing_error = $2,
354  consecutive_listing_failures = consecutive_listing_failures + 1
355WHERE id = $1
356  AND deleted_at IS NULL
357        "#,
358        id,
359        error as CreditRegistrationErrorCode,
360    )
361    .execute(conn)
362    .await?;
363    Ok(())
