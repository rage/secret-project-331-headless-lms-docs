1//! When enrolment discovery fetches each course code's enrolment list.
2//!
3//! A code is due weekly, and sooner while somebody on its modules waits for a student number: each
4//! such person is on the ladders of the signals they have given that they are enrolling, and the
5//! code is due at the earliest rung no fetch that could claim linking mail has served. Every due
6//! time snaps up to the code's own grid, see [`fetch_slot`], so codes spread evenly over the hour.
7//! Both are derived whenever they are needed; the table keeps only what cannot be: when fetches
8//! ran, the failure backoff and how many fetches of the code went out today.
9

22/// How far apart the points of a code's fetch grid are. Divides an hour.
23pub const FETCH_GRID_STEP: TimeDelta = TimeDelta::minutes(20);
24/// A code fetched this recently is not fetched again, whatever is due.
25pub const MIN_REFETCH_GAP: TimeDelta = TimeDelta::minutes(15);
26/// The shorter gap the first fetch after an "I have enrolled" press or an admin's request needs,
27/// since somebody is waiting on it.
28pub const PRESS_MIN_REFETCH_GAP: TimeDelta = TimeDelta::minutes(5);
29/// How often every code is fetched, whether or not anybody on it is waiting.
30pub const BASELINE_INTERVAL: TimeDelta = TimeDelta::weeks(1);
31/// How long after its anchor a waiting person still makes the code due: every ladder ends here.
32pub const WAITING_HORIZON: TimeDelta = TimeDelta::days(90);
33/// How many fetches of one code may go out in a UTC day. Past it, only the weekly fetch and an
34/// admin's request make the code due: a third of the grid points, so one busy code cannot crowd
35/// out the rest.
36pub const MAX_FETCHES_PER_DAY: i32 = 24;
37/// How long a code that failed on its own is left alone, by how many times in a row it has.
38const ALONE_FAILURE_BACKOFF_SECS: [i64; 3] = [HOUR_SECS, 4 * HOUR_SECS, DAY_SECS];
39
40const WAITING_HORIZON_SECS: i64 = WAITING_HORIZON.num_seconds();
41const VISITED_LADDER: [i64; 33] = ladder(
42    &[1, 2, 3, 5, 8, 12, 24],
43    HOUR_SECS,
44    &[
45        (DAY_SECS, 14 * DAY_SECS),
46        (3 * DAY_SECS, 30 * DAY_SECS),
47        (WEEK_SECS, WAITING_HORIZON_SECS),
48    ],
49);
50const PRESSED_LADDER: [i64; 35] = ladder(
51    &[
52        0, 20, 40, 60, 80, 100, 120, 150, 180, 240, 360, 540, 720, 1440,
53    ],
54    MINUTE_SECS,
55    &[
56        (DAY_SECS, 7 * DAY_SECS),
57        (3 * DAY_SECS, 30 * DAY_SECS),
58        (WEEK_SECS, WAITING_HORIZON_SECS),
59    ],
60);

62/// The first point of `course_code`'s fetch grid at or after `at`: [`FETCH_GRID_STEP`] apart, at an
63/// offset into the step fixed per code. The only place that decides when within the hour a code
64/// goes out.
65pub fn fetch_slot(course_code: &str, at: DateTime<Utc>) -> DateTime<Utc> {
66    // FNV-1a: the standard library's hashers do not promise a stable output, and a code should
67    // keep its grid across processes and releases.
68    let hash = course_code
69        .bytes()
70        .fold(0xcbf2_9ce4_8422_2325_u64, |hash, byte| {
71            (hash ^ u64::from(byte)).wrapping_mul(0x0100_0000_01b3)
72        });
73    let step = FETCH_GRID_STEP.num_seconds();
74    let offset = i64::try_from(hash % step.unsigned_abs()).unwrap_or(0);
75    let secs = at.timestamp() + i64::from(at.timestamp_subsec_nanos() > 0);
76    let point = (secs - offset + step - 1).div_euclid(step) * step + offset;
77    DateTime::from_timestamp(point, 0).unwrap_or(at)
78}

95/// One code's schedule, with the waiting people its next fetch is derived from.
96#[derive(Debug, Clone, PartialEq)]
97pub struct RosterSchedule {
98    pub course_code: String,
99    pub last_fetched_at: Option<DateTime<Utc>>,
100    pub last_fetch_started_at: Option<DateTime<Utc>>,
101    pub last_fetch_duration_ms: Option<i32>,
102    pub last_listed_person_count: Option<i32>,
103    pub is_fetched_alone: bool,
104    pub consecutive_failures: i32,
105    pub retry_not_before: Option<DateTime<Utc>>,
106    pub last_error: Option<CreditRegistrationErrorCode>,
107    /// The active modules on the code, which share its enrolment list.
108    pub module_count: i64,
109    /// Registrations on the code's modules waiting for a student number within
110    /// [`WAITING_HORIZON`] of an anchor. Always zero with account linking off.
111    pub waiting_count: i64,
112    /// The earliest unserved rung of a waiting person's ladder.
113    pub next_rung_at: Option<DateTime<Utc>>,
114    /// The latest unserved "I have enrolled" press a waiting person's ladder is anchored on.
115    pub unserved_press_at: Option<DateTime<Utc>>,
116    /// Fetches of the code sent today (UTC), against [`MAX_FETCHES_PER_DAY`].
117    pub fetches_today: i32,
118    /// An admin's fetch request; served once a fetch sent after it arrives.
119    pub fetch_requested_at: Option<DateTime<Utc>>,
120    /// The last enrolment list that fed account linking.
121    pub linking_counters: Option<AccountLinkingCodeCounters>,
122}
123
124impl RosterSchedule {
125    /// When the code is next due, ignoring the failure backoff: at once if it has never been
126    /// fetched.
127    pub fn next_fetch_at(&self, now: DateTime<Utc>) -> DateTime<Utc> {
128        let Some(last_fetched_at) = self.last_fetched_at else {
129            return now;
130        };
131        let mut reasons = vec![(last_fetched_at + BASELINE_INTERVAL, MIN_REFETCH_GAP)];
132        reasons.extend(
133            self.unserved_fetch_request_at()
134                .map(|requested_at| (requested_at, PRESS_MIN_REFETCH_GAP)),
135        );
136        if self.fetches_today < MAX_FETCHES_PER_DAY {
137            reasons.extend(self.next_rung_at.map(|rung_at| (rung_at, MIN_REFETCH_GAP)));
138            reasons.extend(
139                self.unserved_press_at
140                    .map(|pressed_at| (pressed_at, PRESS_MIN_REFETCH_GAP)),
141            );
142        }
143        reasons
144            .into_iter()
145            .map(|(ready_at, gap)| self.slot_after(ready_at, gap))
146            .min()
147            .unwrap_or(now)
148    }
149
150    /// The first grid point at or after `ready_at` that is at least `gap` after the last fetch.
151    fn slot_after(&self, ready_at: DateTime<Utc>, gap: TimeDelta) -> DateTime<Utc> {
152        let ready_at = self
153            .last_fetched_at
154            .map_or(ready_at, |fetched_at| ready_at.max(fetched_at + gap));
155        fetch_slot(&self.course_code, ready_at)
156    }
157
158    /// An admin's fetch request no fetch sent after it has served yet.
159    pub fn unserved_fetch_request_at(&self) -> Option<DateTime<Utc>> {
160        self.fetch_requested_at.filter(|requested_at| {
161            self.last_fetch_started_at
162                .is_none_or(|started_at| *requested_at > started_at)
163        })
164    }
165
166    /// Whether the code is to be fetched now, failure backoff included.
167    pub fn is_due(&self, now: DateTime<Utc>) -> bool {
168        self.retry_not_before.is_none_or(|retry| retry <= now) && self.next_fetch_at(now) <= now
169    }
170}

172/// Creates the schedule rows of every listable code, and the per-module configuration rows the
173/// listing status is written to. `course_id` narrows it to one course.
174pub async fn ensure_rows(conn: &mut PgConnection, course_id: Option<Uuid>) -> ModelResult<()> {
175    sqlx::query!(
176        r#"
177INSERT INTO course_module_suotar_configurations (course_module_id)
178SELECT acm.course_module_id
179FROM credit_registration_active_course_modules acm
180  JOIN course_modules cm ON cm.id = acm.course_module_id AND cm.deleted_at IS NULL
181WHERE TRIM(COALESCE(cm.uh_course_code, '')) <> ''
182  AND ($1::uuid IS NULL OR acm.course_id = $1) ON CONFLICT (course_module_id) DO NOTHING
183        "#,
184        course_id,
185    )
186    .execute(&mut *conn)
187    .await?;
188    sqlx::query!(
189        r#"
190INSERT INTO credit_registration_roster_schedules (course_code)
191SELECT DISTINCT TRIM(cm.uh_course_code)
192FROM credit_registration_active_course_modules acm
193  JOIN course_modules cm ON cm.id = acm.course_module_id AND cm.deleted_at IS NULL
194WHERE TRIM(COALESCE(cm.uh_course_code, '')) <> ''
195  AND ($1::uuid IS NULL OR acm.course_id = $1) ON CONFLICT (course_code) DO NOTHING
196        "#,
197        course_id,
198    )
199    .execute(conn)
200    .await?;
201    Ok(())
202}

204/// Which codes [`get_schedules`] reads.
205#[derive(Debug, Clone, Copy, PartialEq, Eq)]
206pub enum ScheduleSelection<'a> {
207    Every,
208    /// Only the codes not fetched in the last [`PRESS_MIN_REFETCH_GAP`] and out of backoff, a superset
209    /// [`RosterSchedule::is_due`] narrows down: the waiting people are then read for those alone.
210    DueCandidates,
211    /// Only this code, which must be passed trimmed.
212    Code(&'a str),
213}

215/// Every listable code that has a schedule row; see [`ensure_rows`]. `course_id` narrows the codes
216/// and the modules their waiting people come from to one course, for a scoped tick. Only
217/// completions on or after `account_linking_since` can wait, and with it `None` nobody does.
218pub async fn get_schedules(
219    conn: &mut PgConnection,
220    course_id: Option<Uuid>,
221    selection: ScheduleSelection<'_>,
222    account_linking_since: Option<DateTime<Utc>>,
223) -> ModelResult<Vec<RosterSchedule>> {
224    let only_code = match selection {
225        ScheduleSelection::Code(course_code) => Some(course_code),
226        ScheduleSelection::Every | ScheduleSelection::DueCandidates => None,
227    };
228    let rows = sqlx::query!(
229        r#"
230WITH modules AS (
231  SELECT TRIM(cm.uh_course_code) AS course_code
232  FROM credit_registration_active_course_modules acm
233    JOIN course_modules cm ON cm.id = acm.course_module_id AND cm.deleted_at IS NULL
234  WHERE TRIM(COALESCE(cm.uh_course_code, '')) <> ''
235    AND ($1::uuid IS NULL OR acm.course_id = $1)
236    AND (
237      $4::text IS NULL
238      OR TRIM(cm.uh_course_code) = $4
239    )
240)
241SELECT s.course_code,
242  s.last_fetched_at,
243  s.last_fetch_started_at,
244  s.last_mailing_fetch_started_at,
245  s.last_fetch_duration_ms,
246  s.last_listed_person_count,
247  s.is_fetched_alone,
248  s.consecutive_failures,
249  s.retry_not_before,
250  s.last_error,
251  s.fetch_requested_at,
252  s.linking_listed_count,
253  s.linking_already_linked_count,
254  s.linking_mailed_count,
255  s.linking_suppressed_by_dedup_count,
256  s.linking_suppressed_by_rate_cap_count,
257  s.linking_no_address_count,
258  CASE
259    WHEN s.fetch_day = (now() AT TIME ZONE 'UTC')::date THEN s.fetch_day_count
260    ELSE 0
261  END AS "fetches_today!",
262  (
263    SELECT COUNT(*)
264    FROM modules m
265    WHERE m.course_code = s.course_code
266  ) AS "module_count!"
267FROM credit_registration_roster_schedules s
268WHERE s.course_code IN (
269    SELECT course_code
270    FROM modules
271  )
272  AND (
273    NOT $2::boolean
274    OR (
275      (
276        s.retry_not_before IS NULL
277        OR s.retry_not_before <= now()
278      )
279      AND (
280        s.last_fetched_at IS NULL
281        OR s.last_fetched_at < $3
282      )
283    )
284  )
285ORDER BY s.course_code
286        "#,
287        course_id,
288        selection == ScheduleSelection::DueCandidates,
289        Utc::now() - PRESS_MIN_REFETCH_GAP,
290        only_code,
291    )
292    .fetch_all(&mut *conn)
293    .await?;
294    let codes: Vec<String> = rows.iter().map(|row| row.course_code.clone()).collect();
295    // Every rung comes after the epoch, so a code never fetched has served none.
296    let served_until: Vec<DateTime<Utc>> = rows
297        .iter()
298        .map(|row| {
299            row.last_mailing_fetch_started_at
300                .unwrap_or(DateTime::UNIX_EPOCH)
301        })
302        .collect();
303    let mut waiting = get_waiting(
304        conn,
305        course_id,
306        &codes,
307        &served_until,
308        account_linking_since,
309    )
310    .await?;
311    Ok(rows
312        .into_iter()
313        .map(|row| {
314            let waiting = waiting.remove(&row.course_code).unwrap_or_default();
315            let linking_counters =
316                row.linking_listed_count
317                    .map(|listed_person_count| AccountLinkingCodeCounters {
318                        listed_person_count,
319                        already_linked_count: row.linking_already_linked_count.unwrap_or(0),
320                        mailed_count: row.linking_mailed_count.unwrap_or(0),
321                        suppressed_by_dedup_count: row
322                            .linking_suppressed_by_dedup_count
323                            .unwrap_or(0),
324                        suppressed_by_rate_cap_count: row
325                            .linking_suppressed_by_rate_cap_count
326                            .unwrap_or(0),
327                        no_address_count: row.linking_no_address_count.unwrap_or(0),
328                    });
329            RosterSchedule {
330                course_code: row.course_code,
331                last_fetched_at: row.last_fetched_at,
332                last_fetch_started_at: row.last_fetch_started_at,
333                last_fetch_duration_ms: row.last_fetch_duration_ms,
334                last_listed_person_count: row.last_listed_person_count,
335                is_fetched_alone: row.is_fetched_alone,
336                consecutive_failures: row.consecutive_failures,
337                retry_not_before: row.retry_not_before,
338                last_error: row.last_error,
339                module_count: row.module_count,
340                waiting_count: waiting.waiting_count,
341                next_rung_at: waiting.next_rung_at,
342                unserved_press_at: waiting.unserved_press_at,
343                fetches_today: row.fetches_today,
344                fetch_requested_at: row.fetch_requested_at,
345                linking_counters,
346            }
347        })
348        .collect())
349}

353struct CodeWaiting {
354    waiting_count: i64,
355    next_rung_at: Option<DateTime<Utc>>,
356    unserved_press_at: Option<DateTime<Utc>>,
357}

359/// The people waiting on each code, their rungs and presses counted after the code's
360/// `served_until`.
361async fn get_waiting(
362    conn: &mut PgConnection,
363    course_id: Option<Uuid>,
364    course_codes: &[String],
365    served_until: &[DateTime<Utc>],
366    account_linking_since: Option<DateTime<Utc>>,
367) -> ModelResult<HashMap<String, CodeWaiting>> {
368    if account_linking_since.is_none() || course_codes.is_empty() {
369        return Ok(HashMap::new());
370    }
371    let rows = sqlx::query!(
372        r#"
373WITH codes AS (
374  SELECT *
375  FROM UNNEST($2::text [], $3::timestamptz []) AS code(course_code, served_until)
376),
377waiting AS (
378  SELECT codes.course_code,
379    codes.served_until,
380    cr.id AS credit_registration_id,
381    ladder.anchor_at,
382    ladder.offsets,
383    ladder.is_press
384  FROM codes
385    JOIN course_modules cm ON TRIM(cm.uh_course_code) = codes.course_code
386    AND cm.deleted_at IS NULL
387    JOIN credit_registration_active_course_modules acm ON acm.course_module_id = cm.id
388    JOIN credit_registrations cr ON cr.course_module_id = cm.id
389    JOIN course_module_completions cmc ON cmc.id = cr.course_module_completion_id
390    JOIN credit_registration_preconditions p ON p.credit_registration_id = cr.id
391    LEFT JOIN credit_registration_enrolment_check_signals sig ON sig.course_module_completion_id = cr.course_module_completion_id
392    AND sig.deleted_at IS NULL
393    CROSS JOIN LATERAL (
394      VALUES (cmc.completion_date, $4::bigint [], FALSE),
395        (sig.visit_ladder_anchor_at, $5::bigint [], FALSE),
396        (sig.check_request_ladder_anchor_at, $6::bigint [], TRUE)
397    ) AS ladder(anchor_at, offsets, is_press)
398  WHERE ($1::uuid IS NULL OR acm.course_id = $1)
399    AND cr.state = 'pending'
400    AND cr.superseded_by_id IS NULL
401    AND cr.deleted_at IS NULL
402    AND p.completion_eligible
403    AND NOT p.has_verified_student_number
404    AND cmc.completion_date >= $7::timestamptz
405    AND ladder.anchor_at > now() - ($8::bigint * INTERVAL '1 second')
406)
407SELECT w.course_code AS "course_code!",
408  COUNT(DISTINCT w.credit_registration_id) AS "waiting_count!",
409  MIN(rung.due_at) FILTER (
410    WHERE rung.due_at > w.served_until
411  ) AS next_rung_at,
412  MAX(w.anchor_at) FILTER (
413    WHERE w.is_press
414      AND w.anchor_at > w.served_until
415  ) AS unserved_press_at
416FROM waiting w
417  CROSS JOIN LATERAL (
418    SELECT w.anchor_at + (o.secs * INTERVAL '1 second') AS due_at
419    FROM unnest(w.offsets) AS o(secs)
420  ) rung
421GROUP BY w.course_code
422        "#,
423        course_id,
424        course_codes,
425        served_until,
426        &COMPLETED_LADDER[..],
427        &VISITED_LADDER[..],
428        &PRESSED_LADDER[..],
429        account_linking_since,
430        WAITING_HORIZON.num_seconds(),
431    )
432    .fetch_all(conn)
433    .await?;
434    Ok(rows
435        .into_iter()
436        .map(|row| {
437            (
438                row.course_code,
439                CodeWaiting {
440                    waiting_count: row.waiting_count,
441                    next_rung_at: row.next_rung_at,
442                    unserved_press_at: row.unserved_press_at,
443                },
444            )
445        })
446        .collect())
447}

449/// The active modules on each of `course_codes`, which share its roster. `course_id` narrows them
450/// to one course, as it does for [`get_schedules`].
451pub async fn get_modules_by_code(
452    conn: &mut PgConnection,
453    course_id: Option<Uuid>,
454    course_codes: &[String],
455) -> ModelResult<HashMap<String, Vec<ModuleToList>>> {
456    let modules = sqlx::query_as!(
457        ModuleToList,
458        r#"
459SELECT acm.course_module_id AS "course_module_id!",
460  acm.course_id AS "course_id!",
461  TRIM(cm.uh_course_code) AS "uh_course_code!",
462  co.language_code AS "course_language_code!"
463FROM credit_registration_active_course_modules acm
464  JOIN course_modules cm ON cm.id = acm.course_module_id AND cm.deleted_at IS NULL
465  JOIN courses co ON co.id = acm.course_id AND co.deleted_at IS NULL
466WHERE TRIM(cm.uh_course_code) = ANY($2::text [])
467  AND ($1::uuid IS NULL OR acm.course_id = $1)
468ORDER BY cm.id
469        "#,
470        course_id,
471        course_codes,
472    )
473    .fetch_all(conn)
474    .await?;
475    let mut modules_by_code: HashMap<String, Vec<ModuleToList>> = HashMap::new();
476    for module in modules {
477        modules_by_code
478            .entry(module.uh_course_code.clone())
479            .or_default()
480            .push(module);
481    }
482    Ok(modules_by_code)
483}

485/// Stamps the codes a listing request is about to go out for and counts the fetch against
486/// [`MAX_FETCHES_PER_DAY`]. Returns the stamp to hand to [`mark_fetched`].
487pub async fn mark_attempted(
488    conn: &mut PgConnection,
489    course_codes: &[String],
490) -> ModelResult<DateTime<Utc>> {
491    let started_at = sqlx::query_scalar!(
492        r#"
493WITH stamped AS (
494  UPDATE credit_registration_roster_schedules
495  SET last_attempted_at = now(),
496    fetch_day_count = CASE
497      WHEN fetch_day = (now() AT TIME ZONE 'UTC')::date THEN fetch_day_count + 1
498      ELSE 1
499    END,
500    fetch_day = (now() AT TIME ZONE 'UTC')::date
501  WHERE course_code = ANY($1::text [])
502)
503SELECT now() AS "started_at!"
504        "#,
505        course_codes,
506    )
507    .fetch_one(conn)
508    .await?;
509    Ok(started_at)
510}

512/// Records an enrolment list that arrived, or Suotar saying it holds no realisation of the code,
513/// and clears the failure streak. `started_at` is what [`mark_attempted`] returned for the request,
514/// and `could_claim_mail` whether the list was free to claim linking mail; only such a fetch serves
515/// the waiting people's rungs and presses. Returns when the code was fetched before this.
516pub async fn mark_fetched(
517    conn: &mut PgConnection,
518    course_code: &str,
519    started_at: DateTime<Utc>,
520    could_claim_mail: bool,
521    listed_person_count: i32,
522    duration_ms: i32,
523) -> ModelResult<Option<DateTime<Utc>>> {
524    let previous = sqlx::query_scalar!(
525        r#"
526WITH previous AS (
527  SELECT id,
528    last_fetched_at
529  FROM credit_registration_roster_schedules
530  WHERE course_code = $1
531  FOR UPDATE
532)
533UPDATE credit_registration_roster_schedules s
534SET last_fetched_at = now(),
535  last_fetch_started_at = $4,
536  last_mailing_fetch_started_at = CASE
537    WHEN $5 THEN $4
538    ELSE s.last_mailing_fetch_started_at
539  END,
540  last_fetch_duration_ms = $3,
541  last_listed_person_count = $2,
542  is_fetched_alone = FALSE,
543  consecutive_failures = 0,
544  retry_not_before = NULL,
545  last_error = NULL
546FROM previous
547WHERE s.id = previous.id
548RETURNING previous.last_fetched_at
549        "#,
550        course_code,
551        listed_person_count,
552        duration_ms,
553        started_at,
554        could_claim_mail,
555    )
556    .fetch_optional(conn)
557    .await?;
558    Ok(previous.flatten())
559}

591/// Makes the code due at its next grid point, past [`MAX_FETCHES_PER_DAY`]. Returns the schedule
592/// row's id, or `None` for a code with no row.
593pub async fn request_fetch(
594    conn: &mut PgConnection,
595    course_code: &str,
596) -> ModelResult<Option<Uuid>> {
597    let id = sqlx::query_scalar!(
598        r#"
599UPDATE credit_registration_roster_schedules
600SET fetch_requested_at = now()
601WHERE course_code = $1
602RETURNING id
603        "#,
604        course_code,
605    )
606    .fetch_optional(conn)
607    .await?;
608    Ok(id)
609}

611/// Records a request batching several codes that failed as a whole. Each is listed on its own from
612/// now on, until it succeeds, so one bad code stops failing the others.
613pub async fn mark_batch_failed(
614    conn: &mut PgConnection,
615    course_codes: &[String],
616    error: CreditRegistrationErrorCode,
617) -> ModelResult<()> {
618    sqlx::query!(
619        r#"
620UPDATE credit_registration_roster_schedules
621SET is_fetched_alone = TRUE,
622  last_error = $2
623WHERE course_code = ANY($1::text [])
624        "#,
625        course_codes,
626        error as CreditRegistrationErrorCode,
627    )
628    .execute(conn)
629    .await?;
630    Ok(())
631}

633/// Records a listing of the code on its own that failed, and backs the code off: an hour, four
634/// hours, then daily.
635pub async fn mark_alone_failed(
636    conn: &mut PgConnection,
637    course_code: &str,
638    error: CreditRegistrationErrorCode,
639) -> ModelResult<()> {
640    sqlx::query!(
641        r#"
642UPDATE credit_registration_roster_schedules
643SET is_fetched_alone = TRUE,
644  consecutive_failures = consecutive_failures + 1,
645  last_error = $2,
646  retry_not_before = now() + (
647    ($3::bigint [])[LEAST(consecutive_failures + 1, CARDINALITY($3::bigint []))] * INTERVAL '1 second'
648  )
649WHERE course_code = $1
650        "#,
651        course_code,
652        error as CreditRegistrationErrorCode,
653        &ALONE_FAILURE_BACKOFF_SECS[..],
654    )
655    .execute(conn)
656    .await?;
657    Ok(())
658}

667}

669/// The codes whose latest listings failed, most consecutive failures first.
670pub async fn get_failing_codes(conn: &mut PgConnection) -> ModelResult<Vec<FailingRosterCode>> {
671    let res = sqlx::query_as!(

673        r#"

675  consecutive_failures,
676  last_error,
677  last_attempted_at

680ORDER BY consecutive_failures DESC,
681  course_code
682        "#,
683    )
684    .fetch_all(conn)
685    .await?;
686    Ok(res)

702            r#"
703UPDATE credit_registration_roster_schedules

705  retry_not_before = NULL
706WHERE course_code IN (
707    SELECT TRIM(cm.uh_course_code)
708    FROM credit_registration_active_course_modules acm
709      JOIN course_modules cm ON cm.id = acm.course_module_id AND cm.deleted_at IS NULL
710    WHERE acm.course_id = $1
711  )
712        "#,
713            course_id,
714            (BASELINE_INTERVAL + FETCH_GRID_STEP).num_seconds(),
715        )
716        .execute(conn)
717        .await?;
718        Ok(res.rows_affected())
