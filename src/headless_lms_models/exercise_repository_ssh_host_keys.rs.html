1//! ssh host keys remembered on the first clone from a git server (trust on first use).
2
3use crate::{error::missing_model_error, prelude::*};
4
5/// A git server's remembered ssh host key.
6#[derive(Debug, Clone, PartialEq, Eq)]
7pub struct ExerciseRepositorySshHostKey {
8    pub id: Uuid,
9    /// When the key was first seen.
10    pub created_at: DateTime<Utc>,
11    pub updated_at: DateTime<Utc>,
12    pub deleted_at: Option<DateTime<Utc>>,
13    /// Lowercased.
14    pub host: String,
15    pub port: i32,
16    pub sha256_fingerprint: String,
17}
18
19/// The key remembered for `host` (compared case-insensitively) and `port`, if any.
20pub async fn get(
21    conn: &mut PgConnection,
22    host: &str,
23    port: u16,
24) -> ModelResult<Option<ExerciseRepositorySshHostKey>> {
25    let res = sqlx::query_as!(
26        ExerciseRepositorySshHostKey,
27        "
28SELECT *
29FROM exercise_repository_ssh_host_keys
30WHERE host = lower($1)
31  AND port = $2
32  AND deleted_at IS NULL
33",
34        host,
35        i32::from(port)
36    )
37    .fetch_optional(conn)
38    .await?;
39    Ok(res)
40}
41
42/// Remembers `sha256_fingerprint` for the server unless a key is already remembered for it, and
43/// returns whichever key is remembered afterwards. A caller that sees a different key back lost a
44/// race with a clone that saw another key, and must not trust its own.
45pub async fn remember(
46    conn: &mut PgConnection,
47    host: &str,
48    port: u16,
49    sha256_fingerprint: &str,
50) -> ModelResult<ExerciseRepositorySshHostKey> {
51    sqlx::query!(
52        "
53INSERT INTO exercise_repository_ssh_host_keys (host, port, sha256_fingerprint)
54VALUES (lower($1), $2, $3)
55ON CONFLICT (host, port, deleted_at) DO NOTHING
56",
57        host,
58        i32::from(port),
59        sha256_fingerprint
60    )
61    .execute(&mut *conn)
62    .await?;
63    get(conn, host, port).await?.ok_or_else(missing_model_error(
64        ModelErrorType::RecordNotFound,
65        "The remembered ssh host key disappeared",
66    ))
67}

69#[cfg(test)]
70mod test {
71    use super::*;
72    use crate::test_helper::*;
73
74    #[tokio::test]
75    async fn the_first_remembered_key_wins_and_hosts_are_case_insensitive() {
76        insert_data!(:tx);
77        let host = format!("Git-{}.Example", Uuid::new_v4());
78
79        assert_eq!(get(tx.as_mut(), &host, 22).await.unwrap(), None);
80        let first = remember(tx.as_mut(), &host, 22, "first").await.unwrap();
81        assert_eq!(first.sha256_fingerprint, "first");
82
83        let second = remember(tx.as_mut(), &host.to_lowercase(), 22, "second")
84            .await
85            .unwrap();
86        assert_eq!(second.sha256_fingerprint, "first");
87        assert_eq!(get(tx.as_mut(), &host, 2222).await.unwrap(), None);
88        tx.rollback().await;
89    }
