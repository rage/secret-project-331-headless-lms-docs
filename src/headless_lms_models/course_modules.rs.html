119impl From<CourseModulesSchema> for CourseModule {
120    fn from(schema: CourseModulesSchema) -> Self {
121        let completion_policy = if schema.automatic_completion {
122            CompletionPolicy::Automatic(AutomaticCompletionRequirements {
123                course_module_id: schema.id,
124                number_of_exercises_attempted_treshold: schema
125                    .automatic_completion_number_of_exercises_attempted_treshold,
126                number_of_points_treshold: schema.automatic_completion_number_of_points_treshold,
127                requires_exam: schema.automatic_completion_requires_exam,
128            })
129        } else {
130            CompletionPolicy::Manual
131        };
132        Self {
133            id: schema.id,
134            created_at: schema.created_at,
135            updated_at: schema.updated_at,
136            deleted_at: schema.deleted_at,
137            name: schema.name,
138            course_id: schema.course_id,
139            order_number: schema.order_number,
140            copied_from: schema.copied_from,
141            uh_course_code: schema.uh_course_code,
142            completion_policy,
143            completion_registration_link_override: schema.completion_registration_link_override,
144            ects_credits: schema.ects_credits,
145            enable_registering_completion_to_uh_open_university: schema
146                .enable_registering_completion_to_uh_open_university,
147            certification_enabled: schema.certification_enabled,
148        }
149    }
150}

290pub async fn delete(conn: &mut PgConnection, id: Uuid) -> ModelResult<()> {
291    let associated_chapters = chapters::get_for_module(conn, id).await?;
292    if !associated_chapters.is_empty() {
293        return Err(ModelError::new(
294            ModelErrorType::InvalidRequest,
295            format!(
296                "Cannot remove module {id} because it has {} chapters associated with it",
297                associated_chapters.len()
298            ),
299            None,
300        ));
301    }
302    sqlx::query!(
303        "
304UPDATE course_modules
305SET deleted_at = now()
306WHERE id = $1
307AND deleted_at IS NULL
308",
309        id
310    )
311    .execute(conn)
312    .await?;
313    Ok(())
314}

498/// Gets course module where the given exercise belongs to. This will result in an error in the case
499/// of an exam exercise.
500pub async fn get_by_exercise_id(
501    conn: &mut PgConnection,
502    exercise_id: Uuid,
503) -> ModelResult<CourseModule> {
504    let res = sqlx::query_as!(
505        CourseModulesSchema,
506        r#"
507SELECT
508    course_modules.id AS "id!",
509    course_modules.created_at AS "created_at!",
510    course_modules.updated_at AS "updated_at!",
511    course_modules.deleted_at,
512    course_modules.name,
513    course_modules.course_id AS "course_id!",
514    course_modules.order_number AS "order_number!",
515    course_modules.copied_from,
516    course_modules.uh_course_code,
517    course_modules.automatic_completion AS "automatic_completion!",
518    course_modules.automatic_completion_number_of_exercises_attempted_treshold,
519    course_modules.automatic_completion_number_of_points_treshold,
520    course_modules.automatic_completion_requires_exam AS "automatic_completion_requires_exam!",
521    course_modules.completion_registration_link_override,
522    course_modules.ects_credits,
523    course_modules.enable_registering_completion_to_uh_open_university AS "enable_registering_completion_to_uh_open_university!",
524    course_modules.certification_enabled AS "certification_enabled!"
525FROM exercises
526  LEFT JOIN chapters ON (exercises.chapter_id = chapters.id)
527  LEFT JOIN course_modules ON (chapters.course_module_id = course_modules.id)
528WHERE exercises.id = $1
529AND chapters.deleted_at IS NULL
530AND course_modules.deleted_at IS NULL
531        "#,
532        exercise_id,
533    )
534    .fetch_one(conn)
535    .await?;
536    Ok(res.into())
537}

777pub struct AutomaticCompletionRequirements {
778    /// Course module associated with these requirements.
779    pub course_module_id: Uuid,
780    pub number_of_exercises_attempted_treshold: Option<i32>,
781    pub number_of_points_treshold: Option<i32>,
782    pub requires_exam: bool,
783}
784
785impl AutomaticCompletionRequirements {
786    /// Shorthand for checking whether the given exercise related values pass their respective
787    /// tresholds.
788    pub fn passes_exercise_tresholds(
789        &self,
790        exercises_attempted: i32,
791        exercise_points: i32,
792    ) -> bool {
793        self.passes_number_of_exercises_attempted_treshold(exercises_attempted)
794            && self.passes_number_of_exercise_points_treshold(exercise_points)
795    }
796
797    /// Whether the given number is higher than the exercises attempted treshold. Always returns
798    /// true if there is no treshold.
799    pub fn passes_number_of_exercises_attempted_treshold(&self, exercises_attempted: i32) -> bool {
800        self.number_of_exercises_attempted_treshold
801            .map(|x| x <= exercises_attempted)
802            .unwrap_or(true)
803    }
804
805    /// Whether the given number is higher than the exercise points treshold. Always returns true
806    /// if there is no treshold.
807    pub fn passes_number_of_exercise_points_treshold(&self, exercise_points: i32) -> bool {
808        self.number_of_points_treshold
809            .map(|x| x <= exercise_points)
810            .unwrap_or(true)
811    }
812}

814#[derive(Debug, PartialEq, Clone, Serialize, Deserialize, ToSchema)]
815#[serde(tag = "policy", rename_all = "kebab-case")]
816pub enum CompletionPolicy {
817    Automatic(AutomaticCompletionRequirements),
818    Manual,
819}
820
821impl CompletionPolicy {
822    /// Returns associated data for `Automatic` variant, if matches.
823    pub fn automatic(&self) -> Option<&AutomaticCompletionRequirements> {
824        match self {
825            CompletionPolicy::Automatic(requirements) => Some(requirements),
826            CompletionPolicy::Manual => None,
827        }
828    }
829
830    fn to_database_fields(&self) -> (bool, Option<i32>, Option<i32>, bool) {
831        match self {
832            CompletionPolicy::Automatic(requirements) => (
833                true,
834                requirements.number_of_exercises_attempted_treshold,
835                requirements.number_of_points_treshold,
836                requirements.requires_exam,
837            ),
838            CompletionPolicy::Manual => (false, None, None, false),
839        }
840    }
841}

1072pub async fn update_modules(
1073    conn: &mut PgConnection,
1074    course_id: Uuid,
1075    updates: ModuleUpdates,
1076) -> ModelResult<()> {
1077    let mut tx = conn.begin().await?;
1078
1079    // scramble order of modified and deleted modules
1080    for module_id in updates
1081        .modified_modules
1082        .iter()
1083        // do not scramble the default module, it should always be first
1084        .filter(|m| m.order_number != 0)
1085        .map(|m| m.id)
1086        .chain(updates.deleted_modules.iter().copied())
1087    {
1088        update_with_order_number(&mut tx, module_id, None, rand::random()).await?;
1089    }
1090    let mut modified_and_new_modules = updates.modified_modules;
1091    for new in updates.new_modules {
1092        // destructure so new fields cause a compilation error here
1093        let NewModule {
1094            name,
1095            order_number,
1096            chapters,
1097            uh_course_code,
1098            ects_credits,
1099            completion_policy,
1100            completion_registration_link_override,
1101            enable_registering_completion_to_uh_open_university,
1102        } = new;
1103        // insert with a random order number to avoid conflicts
1104        let new_course_module = NewCourseModule::new(course_id, Some(name.clone()), rand::random())
1105            .set_completion_policy(completion_policy.clone())
1106            .set_completion_registration_link_override(completion_registration_link_override)
1107            .set_ects_credits(ects_credits)
1108            .set_uh_course_code(uh_course_code)
1109            .set_enable_registering_completion_to_uh_open_university(
1110                enable_registering_completion_to_uh_open_university,
1111            );
1112        let module = insert(&mut tx, PKeyPolicy::Generate, &new_course_module).await?;
1113        for chapter in chapters {
1114            chapters::set_module(&mut tx, chapter, module.id).await?;
1115        }
1116        //modify the order number with the rest
1117        modified_and_new_modules.push(ModifiedModule {
1118            id: module.id,
1119            name: None,
1120            order_number,
1121            uh_course_code: module.uh_course_code,
1122            ects_credits,
1123            completion_policy,
1124            completion_registration_link_override: module.completion_registration_link_override,
1125            enable_registering_completion_to_uh_open_university: module
1126                .enable_registering_completion_to_uh_open_university,
1127        })
1128    }
1129    // update modified and new modules
1130    for module in modified_and_new_modules {
1131        // destructure so new fields cause a compilation error here
1132        let ModifiedModule {
1133            id,
1134            name,
1135            order_number,
1136            uh_course_code,
1137            ects_credits,
1138            completion_policy,
1139            completion_registration_link_override,
1140            enable_registering_completion_to_uh_open_university,
1141        } = module;
1142        update(
1143            &mut tx,
1144            id,
1145            &NewCourseModule::new(course_id, name.clone(), order_number)
1146                .set_completion_policy(completion_policy)
1147                .set_completion_registration_link_override(completion_registration_link_override)
1148                .set_ects_credits(ects_credits)
1149                .set_uh_course_code(uh_course_code)
1150                .set_enable_registering_completion_to_uh_open_university(
1151                    enable_registering_completion_to_uh_open_university,
1152                ),
1153        )
1154        .await?;
1155    }
1156    for (chapter, module) in updates.moved_chapters {
1157        chapters::set_module(&mut tx, chapter, module).await?;
1158    }
1159    for deleted in updates.deleted_modules {
1160        delete(&mut tx, deleted).await?;
1161    }
1162
1163    tx.commit().await?;
1164    Ok(())
1165}
