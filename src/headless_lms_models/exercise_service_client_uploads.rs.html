1//! Files uploaded through the exercise-services client API, bound to the exercise and user
2//! they were uploaded for.
3//!
4//! The binding is what lets submit reject a file uploaded for a different exercise; ownership
5//! alone would let any exercise's submission name any of the user's uploads. It lives in its own
6//! table rather than as a nullable column on `file_uploads` because `file_uploads` is shared with
7//! CMS media, organization images, certificates and iframe-uploaded answer files, and the reaper's
8//! safety depends on the distinction being structural.
9
10use crate::prelude::*;
11use chrono::Duration;
12
13/// A client upload as seen by submit validation. `deleted` distinguishes a reaped upload
14/// (answerable with `upload_expired`) from one that was never recorded.
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub struct ClientUpload {
17    pub file_upload_id: Uuid,
18    pub deleted: bool,
19}
20
21/// Binds freshly uploaded files to the exercise and user they were uploaded for.
22pub async fn insert_many(
23    conn: &mut PgConnection,
24    exercise_id: Uuid,
25    user_id: Uuid,
26    file_upload_ids: &[Uuid],
27) -> ModelResult<()> {
28    sqlx::query!(
29        "
30INSERT INTO exercise_service_client_uploads (file_upload_id, exercise_id, user_id)
31SELECT file_upload_id,
32  $2,
33  $3
34FROM UNNEST($1::uuid []) AS t(file_upload_id)
35",
36        file_upload_ids,
37        exercise_id,
38        user_id
39    )
40    .execute(conn)
41    .await?;
42    Ok(())
43}

45/// The requested uploads that belong to this exercise and user, soft-deleted ones included.
46///
47/// Rows bound to another exercise or another user are deliberately not returned: to the caller
48/// they must be indistinguishable from ids that were never uploaded, so a foreign id leaks
49/// nothing beyond "not yours".
50pub async fn get_for_exercise_and_user(
51    conn: &mut PgConnection,
52    exercise_id: Uuid,
53    user_id: Uuid,
54    file_upload_ids: &[Uuid],
55) -> ModelResult<Vec<ClientUpload>> {
56    let res = sqlx::query!(
57        "
58SELECT file_upload_id,
59  deleted_at
60FROM exercise_service_client_uploads
61WHERE file_upload_id = ANY($1)
62  AND exercise_id = $2
63  AND user_id = $3
64",
65        file_upload_ids,
66        exercise_id,
67        user_id
68    )
69    .fetch_all(conn)
70    .await?;
71    Ok(res
72        .into_iter()
73        .map(|row| ClientUpload {
74            file_upload_id: row.file_upload_id,
75            deleted: row.deleted_at.is_some(),
76        })
77        .collect())
78}

80/// Like [`get_for_exercise_and_user`], but takes a row lock on the bindings so the reaper cannot
81/// retire them between this check and the caller's `exercise_task_submission_files` insert.
82///
83/// Must be called inside the transaction that later records the association, or the lock is
84/// released before it protects anything. Without it, a reap that lands between validation and the
85/// association commits a submission whose files are already gone: `download_submission` then
86/// reports zero files forever and the student never sees `upload_expired`.
87pub async fn lock_for_exercise_and_user(
88    tx: &mut sqlx::Transaction<'_, sqlx::Postgres>,
89    exercise_id: Uuid,
90    user_id: Uuid,
91    file_upload_ids: &[Uuid],
92) -> ModelResult<Vec<ClientUpload>> {
93    let res = sqlx::query!(
94        "
95SELECT file_upload_id,
96  deleted_at
97FROM exercise_service_client_uploads
98WHERE file_upload_id = ANY($1)
99  AND exercise_id = $2
100  AND user_id = $3
101FOR UPDATE
102",
103        file_upload_ids,
104        exercise_id,
105        user_id
106    )
107    .fetch_all(&mut **tx)
108    .await?;
109    Ok(res
110        .into_iter()
111        .map(|row| ClientUpload {
112            file_upload_id: row.file_upload_id,
113            deleted: row.deleted_at.is_some(),
114        })
115        .collect())
116}

118/// Oldest upload time the reaper still spares. Kept short because an upload is only ever meant
119/// to live for the few seconds between a client's upload call and its submit call.
120pub fn retention_cutoff(now: DateTime<Utc>) -> DateTime<Utc> {
121    now - Duration::hours(1)
122}

133/// Client uploads created before `cutoff` that no submission was ever made from.
134///
135/// `FROM exercise_service_client_uploads` is the safety property of this whole feature, not an
136/// optimisation: `file_uploads` also holds CMS media, organization images, certificates and
137/// iframe-uploaded answer files, and iframe uploads are referenced only from inside opaque
138/// answer blobs the host cannot parse, so the host cannot tell whether one is still needed.
139/// Widening this query to `file_uploads` would silently destroy student answers and course
140/// media. Never do it.
141///
142/// Progress is tracked by `file_uploads.deleted_at`, not by the binding's: the binding is retired
143/// first and the object removed afterwards, so a row whose object delete failed still has a live
144/// `file_uploads` row and comes back on the next run. Filtering on `u.deleted_at IS NULL` instead
145/// would make every transient object-store error orphan its object permanently.
146pub async fn get_reapable(
147    conn: &mut PgConnection,
148    cutoff: DateTime<Utc>,
149) -> ModelResult<Vec<ReapableUpload>> {
150    let res = sqlx::query_as!(
151        ReapableUpload,
152        "
153SELECT u.id,
154  u.file_upload_id,
155  f.path
156FROM exercise_service_client_uploads AS u
157  JOIN file_uploads AS f ON f.id = u.file_upload_id
158WHERE f.deleted_at IS NULL
159  AND u.created_at < $1
160  AND NOT EXISTS (
161    SELECT 1
162    FROM exercise_task_submission_files AS s
163    WHERE s.file_upload_id = u.file_upload_id
164      AND s.deleted_at IS NULL
165  )
166ORDER BY u.created_at
167",
168        cutoff
169    )
170    .fetch_all(conn)
171    .await?;
172    Ok(res)
173}

175/// Retires an upload, keeping the row so a submit naming it answers `upload_expired` instead of
176/// the misleading `unknown_upload`. Idempotent, so a run retrying a failed object delete can call
177/// it again.
178///
179/// Re-checks that no submission has come to reference the upload since `get_reapable` listed it,
180/// and reports `false` if one has. That re-check plus the row lock
181/// [`lock_for_exercise_and_user`] takes is what stops a reap from landing in the middle of a
182/// submit and destroying its files.
183///
184/// The lock is taken in a statement of its own, and that ordering is the whole point. Under READ
185/// COMMITTED a statement's snapshot is fixed when the statement starts, and Postgres refreshes only
186/// the row an `UPDATE` locks — never the rows its subqueries read. So a single locking `UPDATE`
187/// blocks on the submit's lock, unblocks, and then evaluates `NOT EXISTS` against a snapshot from
188/// *before* the submit committed: it finds no submission and reaps the files of a submission that
189/// already answered 200. Locking first makes the `UPDATE` a later statement, so its snapshot
190/// includes that commit.
191pub async fn mark_reaped(conn: &mut PgConnection, id: Uuid) -> ModelResult<bool> {
192    let mut tx = conn.begin().await?;
193    let locked = sqlx::query_scalar!(
194        "
195SELECT id
196FROM exercise_service_client_uploads
197WHERE id = $1
198FOR UPDATE
199",
200        id
201    )
202    .fetch_optional(&mut *tx)
203    .await?;
204    if locked.is_none() {
205        tx.rollback().await?;
206        return Ok(false);
207    }
208    let retired = sqlx::query_scalar!(
209        "
210UPDATE exercise_service_client_uploads AS u
211SET deleted_at = COALESCE(u.deleted_at, now())
212WHERE u.id = $1
213  AND NOT EXISTS (
214    SELECT 1
215    FROM exercise_task_submission_files AS s
216    WHERE s.file_upload_id = u.file_upload_id
217      AND s.deleted_at IS NULL
218  )
219RETURNING u.id
220",
221        id
222    )
223    .fetch_optional(&mut *tx)
224    .await?;
225    tx.commit().await?;
226    Ok(retired.is_some())
227}

560    /// The submit side of the reap race. A reaper that listed this upload as orphaned before the
561    /// submission referenced it must not go through with the reap, or the submission commits with
562    /// its files already gone.
563    ///
564    /// This covers the interleaving logically, in one transaction. The row lock that makes the two
565    /// orderings the *only* possibilities needs two connections, and so committed fixtures, which
566    /// this harness cannot produce; that half lives in the server crate, as
567    /// `programs::exercise_service_client_upload_reaper`'s
568    /// `a_concurrent_reaper_blocks_on_the_submit_lock_and_then_declines_to_reap`.
569    #[tokio::test]
570    async fn a_reap_cannot_retire_an_upload_a_submission_just_referenced() {
571        insert_data!(:tx, user:user_id, :org, course:course_id, instance:_instance, course_module:_cm, chapter:_chapter, page:_page, exercise:exercise_id, slide:slide_id, task:task_id);
572        let file_id = insert_file(tx.as_mut(), "raced").await;
573        insert_many(tx.as_mut(), exercise_id, user_id, &[file_id])
574            .await
575            .unwrap();
576        backdate(tx.as_mut(), file_id, Duration::hours(2)).await;
577
578        // The reaper lists it while the submit is still validating.
579        let listed = get_reapable(tx.as_mut(), retention_cutoff(Utc::now()))
580            .await
581            .unwrap();
582        assert_eq!(listed.len(), 1);
583
584        // The submit wins the race and records the association.
585        let task_submission = insert_task_submission_referencing(
586            tx.as_mut(),
587            course_id,
588            user_id,
589            exercise_id,
590            slide_id,
591            task_id,
592            &[file_id],
593        )
594        .await;
595        assert!(task_submission.is_some());
596
597        assert!(
598            !mark_reaped(tx.as_mut(), listed[0].id).await.unwrap(),
599            "the reaper must abandon an upload that became referenced after it was listed"
600        );
601        assert_eq!(
602            get_for_exercise_and_user(tx.as_mut(), exercise_id, user_id, &[file_id])
603                .await
604                .unwrap(),
605            vec![ClientUpload {
606                file_upload_id: file_id,
607                deleted: false
608            }],
609            "the upload must stay usable, so download_submission can still serve it"
610        );

645    /// already retired; that must succeed, not read as "someone referenced it".
646    #[tokio::test]
