5/// A continuation still waiting: unexpired and unused.
6#[derive(Debug, Clone, PartialEq)]
7pub struct PendingContinuation {
8    pub user_id: Uuid,
9    pub moocfi_course_id: Uuid,
10    pub ui_language: Option<String>,
11    pub client_ip_hash: String,
12}

14/// What a continuation is created from.
15#[derive(Debug, Clone)]
16pub struct NewContinuation<'a> {
17    pub user_id: Uuid,
18    pub moocfi_course_id: Uuid,
19    pub ui_language: Option<&'a str>,
20    pub client_ip_hash: &'a str,
21}
22
23/// Stores a continuation usable for `lifetime`, and prunes those expired over a day ago.
24pub async fn insert(
25    conn: &mut PgConnection,
26    token_digest: &[u8],
27    continuation: &NewContinuation<'_>,
28    lifetime: chrono::Duration,
29) -> ModelResult<()> {
30    let mut tx = conn.begin().await?;
31    sqlx::query!(
32        r#"
33DELETE FROM moocfi_handoff_continuations
34WHERE expires_at < now() - INTERVAL '1 day'
35        "#,
36    )
37    .execute(&mut *tx)
38    .await?;
39    sqlx::query!(
40        r#"
41INSERT INTO moocfi_handoff_continuations (
42    token_digest,
43    user_id,
44    moocfi_course_id,
45    ui_language,
46    client_ip_hash,
47    expires_at
48  )
49VALUES ($1, $2, $3, $4, $5, now() + $6 * INTERVAL '1 millisecond')
50        "#,
51        token_digest,
52        continuation.user_id,
53        continuation.moocfi_course_id,
54        continuation.ui_language,
55        continuation.client_ip_hash,
56        lifetime.num_milliseconds() as f64,
57    )
58    .execute(&mut *tx)
59    .await?;
60    tx.commit().await?;
61    Ok(())
62}

64/// The waiting continuation for `moocfi_course_id` with `token_digest`, if any.
65pub async fn get_pending(
66    conn: &mut PgConnection,
67    token_digest: &[u8],
68    moocfi_course_id: Uuid,
69) -> ModelResult<Option<PendingContinuation>> {
70    let res = sqlx::query_as!(
71        PendingContinuation,
72        r#"
73SELECT user_id,
74  moocfi_course_id,
75  ui_language,
76  client_ip_hash
77FROM moocfi_handoff_continuations
78WHERE token_digest = $1
79  AND moocfi_course_id = $2
80  AND used_at IS NULL
81  AND expires_at > now()
82  AND deleted_at IS NULL
83        "#,
84        token_digest,
85        moocfi_course_id,
86    )
87    .fetch_optional(conn)
88    .await?;
89    Ok(res)
90}

92/// Spends the waiting continuation, returning it. Atomic, so of two concurrent continues only one
93/// gets it.
94pub async fn take(
95    conn: &mut PgConnection,
96    token_digest: &[u8],
97    moocfi_course_id: Uuid,
98) -> ModelResult<Option<PendingContinuation>> {
99    let res = sqlx::query_as!(
100        PendingContinuation,
101        r#"
102UPDATE moocfi_handoff_continuations
103SET used_at = now()
104WHERE token_digest = $1
105  AND moocfi_course_id = $2
106  AND used_at IS NULL
107  AND expires_at > now()
108  AND deleted_at IS NULL
109RETURNING user_id,
110  moocfi_course_id,
111  ui_language,
112  client_ip_hash
113        "#,
114        token_digest,
115        moocfi_course_id,
116    )
117    .fetch_optional(conn)
118    .await?;
119    Ok(res)
120}
