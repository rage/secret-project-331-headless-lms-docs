171/// The slide of each exercise that the course site shows the user: the one already selected for
172/// them in `course_id`, otherwise a random one, as for anonymous visitors. Read-only: unlike
173/// [`crate::exercises::get_or_select_exercise_slide`], a random pick isn't stored, so it can differ
174/// between calls. Exercises without slides are left out.
175pub async fn get_shown_slide_ids_for_exercises(
176    conn: &mut PgConnection,
177    user_id: Option<Uuid>,
178    course_id: Uuid,
179    exercise_ids: &[Uuid],
180) -> ModelResult<HashMap<Uuid, Uuid>> {
181    let rows = sqlx::query!(
182        "
183SELECT DISTINCT ON (s.exercise_id) s.exercise_id,
184  s.id
185FROM exercise_slides s
186  LEFT JOIN user_exercise_states ues ON ues.exercise_id = s.exercise_id
187  AND ues.user_id = $1
188  AND ues.course_id = $2
189  AND ues.deleted_at IS NULL
190WHERE s.exercise_id = ANY($3)
191  AND s.deleted_at IS NULL
192ORDER BY s.exercise_id,
193  s.id = ues.selected_exercise_slide_id DESC NULLS LAST,
194  random()
195        ",
196        user_id,
197        course_id,
198        exercise_ids,
199    )
200    .fetch_all(conn)
201    .await?;
202    Ok(rows
203        .into_iter()
204        .map(|row| (row.exercise_id, row.id))
205        .collect())
206}
