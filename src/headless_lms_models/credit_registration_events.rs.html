14/// Replaces a redacted value; the key is kept so the payload shape survives.
15pub const REDACTED: &str = "[redacted]";
16
17/// Keys whose values identify a person or authenticate a request. Matched case-insensitively at any
18/// depth.
19const REDACTED_KEYS: &[&str] = &[
20    "studentnumber",
21    "firstnames",
22    "lastname",
23    "fullname",
24    "primaryemail",
25    "secondaryemail",
26    "email",
27    "emailedto",
28    "accesstoken",
29    "personid",
30    "sisupersonid",
31];
32
33/// Keys whose values the value scan must leave alone: they carry ids shaped like student numbers —
34/// `cr-{uuid}` request item ids, Sisu ids such as `hy-CUR-135176012` — that the scan would mangle.
35///
36/// Container keys do not belong here: an exemption stops at the objects below it.
37const NEVER_SCANNED_KEYS: &[&str] = &[
38    "requestitemid",
39    "code",
40    "coursecode",
41    "gradescaleid",
42    "gradeid",
43    "credits",
44    "attainmentdate",
45    "attainmentlanguage",
46    "submittedattainmentid",
47    "attainmentid",
48    "sisuattainmentid",
49    "courseunitrealisationid",
50    "openuniversityproductid",
51];
52
53static EMAIL_RE: LazyLock<Regex> = LazyLock::new(|| {
54    Regex::new(r"[A-Za-z0-9._%+\-]+@[A-Za-z0-9.\-]+\.[A-Za-z]{2,}").expect("hardcoded regex")
55});
56
57/// Student-number-shaped digit runs. The id shapes come first because the `regex` crate has no
58/// lookaround: leftmost-first alternation is the only way to say "digits not part of an id". The
59/// word boundaries keep the digit branch off longer numbers such as millisecond timestamps.
60///
61/// Known cost of the `prefixed` branch: a student number hyphen-joined to a word,
62/// `person-012345678`, survives. A bare digit run, the shape Suotar's messages use, still goes.
63static STUDENT_NUMBER_RE: LazyLock<Regex> = LazyLock::new(|| {
64    Regex::new(
65        r"(?P<uuid>\b[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}\b)|(?P<prefixed>\b[A-Za-z][A-Za-z0-9-]*-[0-9]{6,12}\b)|(?P<digits>\b[0-9]{6,12}\b)",
66    )
67    .expect("hardcoded regex")
68});
69
70/// Best-effort removal of personal data from a Suotar request or response body. Not a guarantee and
71/// not an exhaustive PII filter.
72///
73/// Two mechanisms, because either alone leaks: key matching removes the structured person fields,
74/// and the free-text scan is the backstop for the input Suotar's error messages quote back. A key's
75/// treatment covers its own scalar value and the scalars of an array under it, but objects are
76/// always classified again key by key, so an exemption never spreads over a subtree.
77///
78/// The scan only removes email addresses and student-number-shaped digit runs. A name in free text
79/// is kept: no pattern separates it from error prose, and the study registry holds it anyway.
80pub fn scrub_suotar_body(value: &Value) -> Value {
81    scrub(value)
82}
83
84enum KeyPolicy {
85    /// Replaced by [`REDACTED`], subtree and all.
86    FullyRedact,
87    /// Passed through verbatim, except for objects, which are classified by their own keys.
88    NeverScan,
89    /// The default.
90    ScanFreeText,
91}
92
93fn key_policy(key: &str) -> KeyPolicy {
94    let normalized = normalize_key(key);
95    if REDACTED_KEYS.contains(&normalized.as_str()) {
96        KeyPolicy::FullyRedact
97    } else if NEVER_SCANNED_KEYS.contains(&normalized.as_str()) {
98        KeyPolicy::NeverScan
99    } else {
100        KeyPolicy::ScanFreeText
101    }
102}
103
104fn scrub(value: &Value) -> Value {
105    match value {
106        Value::Object(map) => Value::Object(
107            map.iter()
108                .map(|(key, child)| {
109                    let scrubbed = match key_policy(key) {
110                        KeyPolicy::FullyRedact => json!(REDACTED),
111                        KeyPolicy::NeverScan => keep_scalars(child),
112                        KeyPolicy::ScanFreeText => scrub(child),
113                    };
114                    (key.clone(), scrubbed)
115                })
116                .collect(),
117        ),
118        Value::Array(items) => Value::Array(items.iter().map(scrub).collect()),
119        Value::String(text) => Value::String(scrub_free_text(text)),
120        other => other.clone(),
121    }
122}
123
124/// Keeps bare ids and lists of ids, but hands objects back to [`scrub`] so an exemption cannot
125/// smuggle a nested error message past the value scan.
126fn keep_scalars(value: &Value) -> Value {
127    match value {
128        Value::Object(_) => scrub(value),
129        Value::Array(items) => Value::Array(items.iter().map(keep_scalars).collect()),
130        other => other.clone(),
131    }
132}
133
134fn normalize_key(key: &str) -> String {
135    key.chars()
136        .filter(|c| c.is_ascii_alphanumeric())
137        .flat_map(|c| c.to_lowercase())
138        .collect()
139}
140
141fn scrub_free_text(text: &str) -> String {
142    let without_emails = EMAIL_RE.replace_all(text, REDACTED);
143    STUDENT_NUMBER_RE
144        .replace_all(&without_emails, |captures: &Captures| {
145            // The id branches exist to beat the digit branch to the match; put them back unchanged.
146            if captures.name("digits").is_some() {
147                REDACTED.to_string()
148            } else {
149                captures[0].to_string()
150            }
151        })
152        .into_owned()
153}
154
155/// Scrubs a bare error message with the same rules as a JSON body.
156pub fn scrub_text(text: &str) -> String {
157    scrub_free_text(text)
158}

160/// Both sides of a Suotar exchange, scrubbed on construction so there is no way to build an
161/// unscrubbed `details`. The request is kept because the ledger row no longer reflects it after a
162/// retry.
163pub fn suotar_exchange_details(request: Option<&Value>, response: Option<&Value>) -> Value {
164    let mut details = serde_json::Map::new();
165    if let Some(request) = request {
166        details.insert("request".to_string(), scrub_suotar_body(request));
167    }
168    if let Some(response) = response {
169        details.insert("response".to_string(), scrub_suotar_body(response));
170    }
171    Value::Object(details)
172}

237/// Callers that also change `state` must go through `credit_registrations::transition` instead,
238/// which writes both in one transaction.
239pub async fn insert(
240    conn: &mut PgConnection,
241    new: &NewCreditRegistrationEvent,
242) -> ModelResult<Uuid> {
243    let res = sqlx::query!(
244        r#"
245INSERT INTO credit_registration_events (
246    credit_registration_id,
247    kind,
248    from_state,
249    to_state,
250    error_code,
251    message,
252    suotar_api_call_id,
253    actor_user_id,
254    details
255  )
256VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
257RETURNING id
258        "#,
259        new.credit_registration_id,
260        new.kind as CreditRegistrationEventKind,
261        new.from_state as Option<CreditRegistrationState>,
262        new.to_state as Option<CreditRegistrationState>,
263        new.error_code as Option<CreditRegistrationErrorCode>,
264        new.message,
265        new.suotar_api_call_id,
266        new.actor_user_id,
267        new.details,
268    )
269    .fetch_one(conn)
270    .await?;
271    Ok(res.id)
272}

274/// Inserts the same event on many rows in one round trip.
275pub async fn insert_many(
276    conn: &mut PgConnection,
277    credit_registration_ids: &[Uuid],
278    kind: CreditRegistrationEventKind,
279    actor_user_id: Option<Uuid>,
280    message: Option<&str>,
281) -> ModelResult<()> {
282    if credit_registration_ids.is_empty() {
283        return Ok(());
284    }
285    sqlx::query!(
286        r#"
287INSERT INTO credit_registration_events (credit_registration_id, kind, actor_user_id, message)
288SELECT id, $2, $3, $4
289FROM UNNEST($1::uuid []) AS id
290        "#,
291        credit_registration_ids,
292        kind as CreditRegistrationEventKind,
293        actor_user_id,
294        message,
295    )
296    .execute(conn)
297    .await?;
298    Ok(())
299}

301/// The per-item timeline, newest first.
302pub async fn get_by_registration_id(
303    conn: &mut PgConnection,
304    credit_registration_id: Uuid,

306    let res = sqlx::query_as!(
307        CreditRegistrationEvent,
308        r#"
309SELECT *
310FROM credit_registration_events
311WHERE credit_registration_id = $1
312  AND deleted_at IS NULL
313ORDER BY created_at DESC
314        "#,
315        credit_registration_id
316    )
317    .fetch_all(conn)
318    .await?;
319    Ok(res)
320}

498    #[test]
499    fn value_scan_keeps_request_item_ids_quoted_in_free_text_whole() {
500        // The last UUID group is 12 digits, so a naive digit-run scan would eat it.
501        let body = json!({ "message": "item cr-2a4b0d6e-0000-4000-8000-000000000001 rejected" });
502        assert_eq!(scrub_suotar_body(&body), body);

506        assert_eq!(scrub_suotar_body(&numeric), numeric);
