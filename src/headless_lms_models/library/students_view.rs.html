33/// Escapes the `LIKE`/`ILIKE` metacharacters `\`, `%` and `_` so a search string is matched
34/// literally (used together with `ESCAPE '\'` in the query).
35pub fn escape_like_pattern(input: &str) -> String {
36    input
37        .replace('\\', "\\\\")
38        .replace('%', "\\%")
39        .replace('_', "\\_")
40}

42/// Grade filter values accepted by [`get_course_students_page`], beyond a literal numeric grade
43/// string (the sis-0-5 scale, `"0"`..`"5"`).

46pub const GRADE_FILTER_FAILED: &str = "failed";

48/// Returns a filtered, sorted, paginated page of the course's enrolled users (identity only).
49///
50/// `sort_column` (`last_name` | `first_name` | `email` | `total_points`) and `sort_direction` are
51/// narrowed to fixed literals and bound as parameters, never interpolated from raw input. `search`
52/// matches name/email substrings via the trigram `name_search_helper` / `email_search_helper`
53/// columns, plus an exact user-id match when it parses as a UUID. `course_instance_id` narrows to a
54/// single instance.
55///
56/// `module_id` + `grade` together narrow to students whose *latest* completion of that module matches:
57/// a numeric grade string (sis-0-5 scale), [`GRADE_FILTER_PASSED`]/[`GRADE_FILTER_FAILED`] (the
58/// sis-hyv-hyl scale, i.e. `grade IS NULL`), or [`GRADE_FILTER_NOT_COMPLETED`] (no completion row at
59/// all). A numerically graded module's completions never match `passed`/`failed` -- those only ever
60/// apply to modules that use the pass/fail scale, mirroring how `CompletionsTab` renders the grade
61/// column (a numeric grade takes precedence over passed/failed). `grade` is ignored unless `module_id`
62/// is also set.
63///
64/// `registration_stages` narrows to students holding at least one live credit registration at one of
65/// those stages, within `course_instance_id` and `module_id` where either is given. Empty means no
66/// narrowing.
67#[allow(clippy::too_many_arguments)]
68pub async fn get_course_students_page(
69    conn: &mut PgConnection,
70    course_id: Uuid,
71    pagination: Pagination,
72    search: Option<&str>,
73    sort_column: Option<&str>,
74    sort_direction: Option<&str>,
75    course_instance_id: Option<Uuid>,
76    module_id: Option<Uuid>,
77    grade: Option<&str>,
78    registration_stages: &[StudentFacingCreditRegistrationStatus],
79) -> ModelResult<StudentsListPage> {
80    let stages = StageMatch::of(registration_stages);
81    // Empty/blank search behaves like no search.
82    let search = search.map(str::trim).filter(|s| !s.is_empty());
83    let user_id_exact = search.and_then(|s| Uuid::parse_str(s).ok());
84    // The helper columns are lowercased generated columns, so lowercase the term and escape the LIKE
85    // metacharacters (matched literally via `ESCAPE '\'`). The GiST trigram indexes serve LIKE.
86    let search_pattern = search.map(|s| escape_like_pattern(&s.to_lowercase()));
87    // A `grade` without a `module_id` has nothing to scope it to, so it is dropped rather than
88    // matched against every module.
89    let grade_filter = module_id.and(grade);
90
91    // Both the sort column and the direction are narrowed to a fixed literal here, so the query can
92    // bind them and stay one offline-checked shape instead of being built by string formatting.
93    let sort_column = match sort_column {
94        Some("first_name") => "first_name",
95        Some("email") => "email",
96        Some("total_points") => "total_points",
97        _ => "last_name",
98    };
99    let sort_direction = match sort_direction {
100        Some("desc") | Some("DESC") => "desc",
101        _ => "asc",
102    };
103
104    let total_count = sqlx::query_scalar!(
105        r#"
106SELECT COUNT(*) AS "count!"
107FROM (
108  SELECT u.id
109  FROM course_instance_enrollments cie
110    JOIN users u ON u.id = cie.user_id
111    LEFT JOIN user_details ud ON ud.user_id = u.id
112    LEFT JOIN LATERAL (
113      SELECT cmc.grade, cmc.passed
114      FROM course_module_completions cmc
115      WHERE cmc.user_id = u.id
116        AND cmc.course_id = $1
117        AND cmc.course_module_id = $5
118        AND cmc.deleted_at IS NULL
119      ORDER BY cmc.completion_date DESC
120      LIMIT 1
121    ) gm ON $5::uuid IS NOT NULL
122  WHERE cie.course_id = $1
123    AND cie.deleted_at IS NULL
124    AND u.deleted_at IS NULL
125    AND ($2::uuid IS NULL OR cie.course_instance_id = $2)
126    AND (
127      $3::text IS NULL
128      OR ud.name_search_helper LIKE '%' || $3 || '%' ESCAPE '\'
129      OR ud.email_search_helper LIKE '%' || $3 || '%' ESCAPE '\'
130      OR ($4::uuid IS NOT NULL AND u.id = $4)
131    )
132    AND (
133      $6::text IS NULL
134      OR ($6 = 'not_completed' AND gm.grade IS NULL AND gm.passed IS NULL)
135      OR ($6 = 'passed' AND gm.grade IS NULL AND gm.passed = true)
136      OR ($6 = 'failed' AND gm.grade IS NULL AND gm.passed = false)
137      OR ($6 ~ '^[0-9]+$' AND gm.grade = $6::int)
138    )
139    AND (
140      CARDINALITY($7::credit_registration_state []) = 0
141      OR EXISTS (
142        SELECT 1
143        FROM credit_registrations cr
144          JOIN credit_registration_preconditions p ON p.credit_registration_id = cr.id
145          JOIN UNNEST(
146              $7::credit_registration_state [],
147              $8::boolean [],
148              $9::boolean [],
149              $10::boolean []
150            ) AS stage(
151              state,
152              completion_eligible,
153              has_verified_student_number,
154              enrolment_resolved
155            )
156            ON stage.state = cr.state
157            AND stage.completion_eligible = p.completion_eligible
158            AND stage.has_verified_student_number = p.has_verified_student_number
159            AND stage.enrolment_resolved = (cr.selected_enrolment_id IS NOT NULL)
160        WHERE cr.user_id = u.id
161          AND cr.course_id = $1
162          AND cr.superseded_by_id IS NULL
163          AND cr.deleted_at IS NULL
164          AND ($2::uuid IS NULL OR cr.course_instance_id = $2)
165          AND ($5::uuid IS NULL OR cr.course_module_id = $5)
166      )
167    )
168  GROUP BY u.id
169) t
170        "#,
171        course_id,
172        course_instance_id,
173        search_pattern.as_deref(),
174        user_id_exact,
175        module_id,
176        grade_filter,
177        &stages.states as &[CreditRegistrationState],
178        &stages.completion_eligible as &[bool],
179        &stages.has_verified_student_number as &[bool],
180        &stages.enrolment_resolved as &[bool],
181    )
182    .fetch_one(&mut *conn)
183    .await?;
184
185    // Each sort key appears twice, once per direction, and yields NULL in every row that the bound
186    // column and direction do not select -- an all-NULL key orders nothing, which is what lets one
187    // fixed ORDER BY stand in for the eight column/direction combinations. `u.id` breaks ties so
188    // paging over equal sort keys (duplicate/NULL names, duplicate emails) never skips or repeats a
189    // student.
190    let data = sqlx::query_as!(
191        CourseStudentListRow,
192        r#"
193SELECT
194  u.id AS "user_id!",
195  ud.first_name AS "first_name?",
196  ud.last_name AS "last_name?",
197  ud.email AS "email?",
198  COALESCE(
199    array_agg(DISTINCT ci.name) FILTER (WHERE ci.name IS NOT NULL),
200    ARRAY[]::text[]
201  ) AS "course_instances!: Vec<String>",
202  COALESCE(bool_or(ci.id IS NOT NULL), false) AS "has_active_instance!"
203FROM course_instance_enrollments cie
204  JOIN users u ON u.id = cie.user_id
205  LEFT JOIN user_details ud ON ud.user_id = u.id
206  LEFT JOIN course_instances ci
207    ON ci.id = cie.course_instance_id
208   AND ci.deleted_at IS NULL
209  LEFT JOIN LATERAL (
210    SELECT cmc.grade, cmc.passed
211    FROM course_module_completions cmc
212    WHERE cmc.user_id = u.id
213      AND cmc.course_id = $1
214      AND cmc.course_module_id = $7
215      AND cmc.deleted_at IS NULL
216    ORDER BY cmc.completion_date DESC
217    LIMIT 1
218  ) gm ON $7::uuid IS NOT NULL
219  LEFT JOIN (
220    SELECT ues.user_id, COALESCE(SUM(ues.score_given), 0)::double precision AS total_points
221    FROM user_exercise_states ues
222      JOIN exercises ex ON ex.id = ues.exercise_id
223    WHERE ues.course_id = $1
224      AND ues.deleted_at IS NULL
225      AND ex.deleted_at IS NULL
226    GROUP BY ues.user_id
227  ) points ON points.user_id = u.id
228WHERE cie.course_id = $1
229  AND cie.deleted_at IS NULL
230  AND u.deleted_at IS NULL
231  AND ($4::uuid IS NULL OR cie.course_instance_id = $4)
232  AND (
233    $2::text IS NULL
234    OR ud.name_search_helper LIKE '%' || $2 || '%' ESCAPE '\'
235    OR ud.email_search_helper LIKE '%' || $2 || '%' ESCAPE '\'
236    OR ($3::uuid IS NOT NULL AND u.id = $3)
237  )
238  AND (
239    $8::text IS NULL
240    OR ($8 = 'not_completed' AND gm.grade IS NULL AND gm.passed IS NULL)
241    OR ($8 = 'passed' AND gm.grade IS NULL AND gm.passed = true)
242    OR ($8 = 'failed' AND gm.grade IS NULL AND gm.passed = false)
243    OR ($8 ~ '^[0-9]+$' AND gm.grade = $8::int)
244  )
245  AND (
246    CARDINALITY($11::credit_registration_state []) = 0
247    OR EXISTS (
248      SELECT 1
249      FROM credit_registrations cr
250        JOIN credit_registration_preconditions p ON p.credit_registration_id = cr.id
251        JOIN UNNEST(
252            $11::credit_registration_state [],
253            $12::boolean [],
254            $13::boolean [],
255            $14::boolean []
256          ) AS stage(
257            state,
258            completion_eligible,
259            has_verified_student_number,
260            enrolment_resolved
261          )
262          ON stage.state = cr.state
263          AND stage.completion_eligible = p.completion_eligible
264          AND stage.has_verified_student_number = p.has_verified_student_number
265          AND stage.enrolment_resolved = (cr.selected_enrolment_id IS NOT NULL)
266      WHERE cr.user_id = u.id
267        AND cr.course_id = $1
268        AND cr.superseded_by_id IS NULL
269        AND cr.deleted_at IS NULL
270        AND ($4::uuid IS NULL OR cr.course_instance_id = $4)
271        AND ($7::uuid IS NULL OR cr.course_module_id = $7)
272    )
273  )
274GROUP BY u.id, ud.first_name, ud.last_name, ud.email
275ORDER BY
276  CASE
277    WHEN $9 = 'total_points' AND $10 = 'asc' THEN COALESCE(MAX(points.total_points), 0)
278  END ASC NULLS LAST,
279  CASE
280    WHEN $9 = 'total_points' AND $10 = 'desc' THEN COALESCE(MAX(points.total_points), 0)
281  END DESC NULLS LAST,
282  CASE
283    WHEN $10 <> 'asc' THEN NULL
284    WHEN $9 = 'first_name' THEN LOWER(TRIM(ud.first_name))
285    WHEN $9 = 'email' THEN LOWER(ud.email)
286    WHEN $9 = 'last_name' THEN LOWER(TRIM(ud.last_name))
287  END ASC NULLS LAST,
288  CASE
289    WHEN $10 <> 'desc' THEN NULL
290    WHEN $9 = 'first_name' THEN LOWER(TRIM(ud.first_name))
291    WHEN $9 = 'email' THEN LOWER(ud.email)
292    WHEN $9 = 'last_name' THEN LOWER(TRIM(ud.last_name))
293  END DESC NULLS LAST,
294  CASE
295    WHEN $9 = 'first_name' THEN LOWER(TRIM(ud.last_name))
296    WHEN $9 = 'last_name' THEN LOWER(TRIM(ud.first_name))
297  END ASC NULLS LAST,
298  u.id ASC
299LIMIT $5 OFFSET $6
300        "#,
301        course_id,
302        search_pattern.as_deref(),
303        user_id_exact,
304        course_instance_id,
305        pagination.limit(),
306        pagination.offset(),
307        module_id,
308        grade_filter,
309        sort_column,
310        sort_direction,
311        &stages.states as &[CreditRegistrationState],
312        &stages.completion_eligible as &[bool],
313        &stages.has_verified_student_number as &[bool],
314        &stages.enrolment_resolved as &[bool],
315    )
316    .fetch_all(&mut *conn)
317    .await?;
318
319    Ok(StudentsListPage {
320        data,
321        total_pages: pagination.total_pages(total_count as u32),
322    })
323}

337/// Returns student × module completion rows for the given users, keyed by `user_id`.
338pub async fn get_completions_grid_for_users(
339    conn: &mut PgConnection,
340    course_id: Uuid,
341    user_ids: &[Uuid],
342) -> ModelResult<Vec<CompletionGridRow>> {
343    let rows = sqlx::query_as!(
344        CompletionGridRow,
345        r#"
346WITH modules AS (
347  SELECT id AS module_id, name AS module_name, order_number
348  FROM course_modules
349  WHERE course_id = $1
350    AND deleted_at IS NULL
351),
352targets AS (
353  SELECT DISTINCT user_id
354  FROM course_instance_enrollments
355  WHERE course_id = $1
356    AND deleted_at IS NULL
357    AND user_id = ANY($2::uuid[])
358),
359latest_cmc AS (
360  SELECT DISTINCT ON (cmc.user_id, cmc.course_module_id)
361    cmc.id,
362    cmc.user_id,
363    cmc.course_module_id,
364    cmc.grade,
365    cmc.passed,
366    cmc.completion_date,
367    cmc.needs_to_be_reviewed
368  FROM course_module_completions cmc
369  WHERE cmc.course_id = $1
370    AND cmc.deleted_at IS NULL
371    AND cmc.user_id = ANY($2::uuid[])
372  ORDER BY cmc.user_id, cmc.course_module_id, cmc.completion_date DESC
373),
374cmcr AS (
375  SELECT course_module_completion_id
376  FROM course_module_completion_registered_to_study_registries
377  WHERE course_id = $1
378    AND deleted_at IS NULL
379)
380SELECT
381  e.user_id AS "user_id!",
382  m.module_id AS "module_id!",
383  m.module_name AS "module?",
384  r.grade AS "grade?",
385  r.passed AS "passed?",
386  (r.id IS NOT NULL AND r.id IN (SELECT course_module_completion_id FROM cmcr)) AS "registered!",
387  COALESCE(r.needs_to_be_reviewed, false) AS "needs_to_be_reviewed!"
388FROM modules m
389CROSS JOIN targets e
390LEFT JOIN latest_cmc r
391  ON r.user_id = e.user_id
392 AND r.course_module_id = m.module_id
393ORDER BY m.order_number, e.user_id
394        "#,
395        course_id,
396        user_ids
397    )
398    .fetch_all(&mut *conn)
399    .await?;
400
401    Ok(rows)
402}

414/// Returns the latest course certificate (if any) for each of the given users, keyed by `user_id`.
415pub async fn get_certificates_grid_for_users(
416    conn: &mut PgConnection,
417    course_id: Uuid,
418    user_ids: &[Uuid],
419) -> ModelResult<Vec<CertificateGridRow>> {
420    let rows = sqlx::query_as!(
421        CertificateGridRow,
422        r#"
423WITH targets AS (
424  SELECT DISTINCT user_id
425  FROM course_instance_enrollments
426  WHERE course_id = $1
427    AND deleted_at IS NULL
428    AND user_id = ANY($2::uuid[])
429),
430user_certs AS (
431  -- one latest certificate per user for this course
432  SELECT DISTINCT ON (gc.user_id)
433    gc.user_id,
434    gc.id,
435    gc.created_at AS latest_issued_at,
436    gc.verification_id,
437    gc.name_on_certificate
438  FROM generated_certificates gc
439  JOIN certificate_configuration_to_requirements cctr
440    ON gc.certificate_configuration_id = cctr.certificate_configuration_id
441   AND cctr.deleted_at IS NULL
442  JOIN course_modules cm
443    ON cm.id = cctr.course_module_id
444   AND cm.deleted_at IS NULL
445  WHERE cm.course_id = $1
446    AND gc.deleted_at IS NULL
447    AND gc.user_id = ANY($2::uuid[])
448  ORDER BY gc.user_id, gc.created_at DESC
449)
450SELECT
451  e.user_id AS "user_id!",
452  uc.latest_issued_at AS "date_issued?",
453  uc.verification_id AS "verification_id?",
454  uc.id AS "certificate_id?",
455  uc.name_on_certificate AS "name_on_certificate?"
456FROM targets e
457LEFT JOIN user_certs uc ON uc.user_id = e.user_id
458        "#,
459        course_id,
460        user_ids
461    )
462    .fetch_all(&mut *conn)
463    .await?;
464
465    Ok(rows)
466}

478/// Per-user progress detail for the Progress tab, scoped to the requested `user_ids`.
479#[derive(Clone, PartialEq, Deserialize, Serialize, ToSchema)]

481pub struct CourseStudentsProgressUsers {
482    pub user_chapter_progress: Vec<UserChapterProgress>,
483    pub user_chapter_locking_statuses: Vec<UserChapterLockingStatus>,
484}
485
486/// Returns the course-level chapter structure shared by every page of the Progress tab.
487pub async fn get_progress_structure(
488    conn: &mut PgConnection,
489    course_id: Uuid,
490) -> ModelResult<CourseStudentsProgressStructure> {
491    let course = crate::courses::get_course(conn, course_id).await?;
492    let chapters = crate::chapters::get_course_chapters(conn, course_id).await?;
493    let chapter_availability = chapters::fetch_chapter_availability(conn, course_id).await?;
494
495    Ok(CourseStudentsProgressStructure {
496        chapter_locking_enabled: course.chapter_locking_enabled,
497        chapters,
498        chapter_availability,
499    })
500}

502/// Returns per-user chapter progress and locking statuses for the given `user_ids`.
503pub async fn get_progress_for_users(
504    conn: &mut PgConnection,
505    course_id: Uuid,
506    user_ids: &[Uuid],
507) -> ModelResult<CourseStudentsProgressUsers> {
508    let course = crate::courses::get_course(conn, course_id).await?;
509    let user_chapter_progress =
510        chapters::fetch_user_chapter_progress(conn, course_id, Some(user_ids)).await?;
511    let user_chapter_locking_statuses =
512        crate::user_chapter_locking_statuses::get_for_users_and_course(conn, user_ids, &course)
513            .await?;
514
515    Ok(CourseStudentsProgressUsers {
516        user_chapter_progress,
517        user_chapter_locking_statuses,
518    })
