139/// Reads the plan a course is about to be created from, for drafting its texts before
140/// [`create_course_from_plan`].
141///
142/// Fails like [`create_course_from_plan`] when no course may be created from the plan, so the
143/// caller does not draft for nothing.
144pub async fn get_course_from_plan_source(
145    conn: &mut PgConnection,
146    plan_id: Uuid,
147    user_id: Uuid,
148    options: &CourseFromPlanOptions,
149) -> ModelResult<CourseFromPlanSource> {
150    let mut tx = conn.begin().await?;
151    let plan =
152        crate::course_designer_plan_members::lock_editable_plan_for_user(&mut tx, plan_id, user_id)
153            .await?;
154    ensure_course_can_be_created(&mut tx, &plan).await?;
155    let (analysis, design) = get_analysis_and_design(&mut tx, plan_id).await?;
156    if design.planned_content_formats.chatbots {
157        default_chatbot_model(&mut tx).await?;
158    }
159    tx.commit().await?;
160
161    let course_name = options
162        .name
163        .as_deref()
164        .and_then(non_empty_trimmed)
165        .or_else(|| analysis.course_title.as_deref().and_then(non_empty_trimmed))
166        .or_else(|| plan.name.as_deref().and_then(non_empty_trimmed))
167        .unwrap_or("Untitled course")
168        .chars()
169        .take(MAX_COURSE_NAME_CHARS)
170        .collect::<String>();
171    let language_code = options
172        .language_code
173        .as_deref()
174        .and_then(non_empty_trimmed)
175        .map(|code| {
176            if is_ietf_language_code_like(code) {
177                Ok(code.to_string())
178            } else {
179                Err(model_err!(
180                    InvalidRequest,
181                    "Malformed language code.".to_string()
182                ))
183            }
184        })
185        .transpose()?
186        .unwrap_or_else(|| language_code_from_analysis(analysis.language.as_deref()));
187    Ok(CourseFromPlanSource {
188        plan_id,
189        course_name,
190        language_code,
191        great_mooc_vision: plan.great_mooc_vision,
192        analysis,
193        design,
194    })
195}

197/// A plan has at most one live course, and gets one only once Design is done.
198async fn ensure_course_can_be_created(
199    conn: &mut PgConnection,
200    plan: &CourseDesignerPlan,
201) -> ModelResult<()> {
202    let row = sqlx::query!(
203        r#"
204SELECT (c.id IS NOT NULL) AS "has_live_course!",
205  (
206    SELECT s.status
207    FROM course_designer_plan_stages s
208    WHERE s.course_designer_plan_id = p.id
209      AND s.stage = 'design'
210      AND s.deleted_at IS NULL
211  ) AS design_status
212FROM course_designer_plans p
213  LEFT JOIN courses c ON c.id = p.created_course_id
214  AND c.deleted_at IS NULL
215WHERE p.id = $1
216  AND p.deleted_at IS NULL
217"#,
218        plan.id
219    )
220    .fetch_one(conn)
221    .await?;
222    if row.has_live_course {
223        return Err(model_err!(
224            PreconditionFailed,
225            "A course has already been created from this plan.".to_string()
226        ));
227    }
228    if row.design_status != Some(CourseDesignerPlanStageStatus::Completed) {
229        return Err(model_err!(
230            PreconditionFailed,
231            "The course can be created once the Design phase is done.".to_string()
232        ));
233    }
234    Ok(())
235}

252/// Creates a draft course from a plan read with [`get_course_from_plan_source`] and the texts an
253/// AI drafted from it: one chapter per planned chapter, each with an Introduction page, one page
254/// per topic and a Conclusion page, plus the front page, Contributors and How to study pages,
255/// glossary, chapter colours and, when planned, a disabled chatbot. The creator becomes a teacher
256/// on the course and the plan remembers the course.
257///
258/// The caller must have checked that `user_id` may create courses in the organization. Fails if
259/// the user is not a plan member, Design is not done or the plan already has a course.
260#[allow(clippy::too_many_arguments)]
261pub async fn create_course_from_plan(
262    conn: &mut PgConnection,
263    app_config: &ApplicationConfiguration,
264    user_id: Uuid,
265    options: &CourseFromPlanOptions,
266    source: &CourseFromPlanSource,
267    drafts: &CourseFromPlanDrafts,
268    spec_fetcher: impl SpecFetcher,
269    fetch_service_info: impl Fn(Url) -> BoxFuture<'static, ModelResult<ExerciseServiceInfoApi>> + Copy,
270) -> ModelResult<Course> {
271    let plan_id = source.plan_id;
272    let mut tx = conn.begin().await?;
273    let plan =
274        crate::course_designer_plan_members::lock_editable_plan_for_user(&mut tx, plan_id, user_id)
275            .await?;
276    ensure_course_can_be_created(&mut tx, &plan).await?;
277    let (analysis, design) = (&source.analysis, &source.design);
278    let copy = CourseCopy::for_language(&source.language_code);
279
280    let teacher = crate::user_details::get_user_details_by_user_id(&mut tx, user_id).await?;
281    let teacher_name = crate::user_details::full_name_or_email(
282        teacher.first_name.as_deref(),
283        teacher.last_name.as_deref(),
284        &teacher.email,
285    );
286    let slug = unique_course_slug(&mut tx, &source.course_name).await?;
287
288    let new_course = NewCourse {
289        name: source.course_name.clone(),
290        slug,
291        organization_id: options.organization_id,
292        language_code: source.language_code.clone(),
293        teacher_in_charge_name: teacher_name,
294        teacher_in_charge_email: teacher.email,
295        description: drafts.course.description.trim().to_string(),
296        is_draft: true,
297        is_test_mode: false,
298        is_unlisted: false,
299        copy_user_permissions: false,
300        is_joinable_by_code_only: false,
301        join_code: None,
302        ask_marketing_consent: false,
303        flagged_answers_threshold: None,
304        can_add_chatbot: source.has_planned_chatbot(),
305    };
306    let ects_credits = analysis
307        .credits
308        .filter(|credits| *credits > 0.0)
309        .map(|credits| credits as f32);
310    let (course, _front_page, _instance, default_module) =
311        content_management::create_new_course_with_content(
312            &mut tx,
313            app_config,
314            PKeyPolicy::Generate,
315            new_course,
316            user_id,
317            NewCourseContent {
318                front_page_content: Some(course_content::front_page_content(
319                    copy,
320                    &source.course_name,
321                    &source.learning_goals(),
322                    &drafts.course,
323                )),
324                default_module_ects_credits: ects_credits,
325            },
326            &spec_fetcher,
327            fetch_service_info,
328        )
329        .await?;
330    roles::insert(
331        &mut tx,
332        user_id,
333        UserRole::Teacher,
334        RoleDomain::Course(course.id),
335    )
336    .await?;
337    if options.should_give_members_course_roles {
338        let members =
339            crate::course_designer_plan_members::get_plan_members(&mut tx, plan_id).await?;
340        for member in members.iter().filter(|member| member.user_id != user_id) {
341            roles::insert(
342                &mut tx,
343                member.user_id,
344                UserRole::Teacher,
345                RoleDomain::Course(course.id),
346            )
347            .await?;
348        }
349    }
350
351    let chapter_colors = course_content::chapter_colors(&design.style_guide.colour_palette);
352    if drafts.chapters.len() != design.chapters.len() {
353        return Err(model_err!(
354            InvalidRequest,
355            "The drafts must have one chapter draft per planned chapter.".to_string()
356        ));
357    }
358    for (index, planned_chapter) in design.chapters.iter().enumerate() {
359        let chapter_number = i32::try_from(index + 1)
360            .map_err(|_| model_err!(InvalidRequest, "Too many chapters.".to_string()))?;
361        let chapter_name = non_empty_trimmed(&planned_chapter.title)
362            .map(str::to_string)
363            .unwrap_or_else(|| format!("Chapter {chapter_number}"));
364        let color = (!chapter_colors.is_empty())
365            .then(|| chapter_colors[index % chapter_colors.len()].clone());
366        let objectives = course_content::learning_objectives(planned_chapter);
367        let (chapter, _) = content_management::create_new_chapter_with_content(
368            &mut tx,
369            PKeyPolicy::Generate,
370            &NewChapter {
371                name: chapter_name.clone(),
372                color: color.clone(),
373                course_id: course.id,
374                chapter_number,
375                front_page_id: None,
376                opens_at: None,
377                deadline: None,
378                course_module_id: Some(default_module.id),
379            },
380            user_id,
381            &spec_fetcher,
382            fetch_service_info,
383            Some(course_content::chapter_front_page_content(
384                &chapter_name,
385                &objectives,
386                color.as_deref(),
387            )),
388        )
389        .await?;
390
391        let chapter_draft = &drafts.chapters[index];
392        let topic_titles = chapter_topic_titles(planned_chapter);
393        let mut pages_to_insert = Vec::with_capacity(topic_titles.len() + 2);
394        pages_to_insert.push((
395            copy.introduction_title,
396            course_content::introduction_content(copy, chapter_draft),
397        ));
398        for (topic_index, title) in topic_titles.iter().enumerate() {
399            let outline = chapter_draft
400                .topic_outlines
401                .get(topic_index)
402                .cloned()
403                .unwrap_or_default();
404            pages_to_insert.push((*title, course_content::topic_content(copy, title, &outline)));
405        }
406        pages_to_insert.push((
407            copy.conclusion_title,
408            course_content::conclusion_content(copy, &objectives),
409        ));
410        let mut used_segments = HashSet::new();
411        for (title, content) in pages_to_insert {
412            let segment = unique_segment(&mut used_segments, title);
413            insert_generated_page(
414                &mut tx,
415                NewPage {
416                    url_path: chapters::page_path(chapter_number, &segment),
417                    title: title.to_string(),
418                    chapter_id: Some(chapter.id),
419                    ..generated_page(course.id, content)
420                },
421                user_id,
422                &spec_fetcher,
423                fetch_service_info,
424            )
425            .await?;
426        }
427    }
428
429    for (title, content) in [
430        (
431            copy.how_to_study_title,
432            course_content::how_to_study_content(copy, &drafts.course, &design.workload_activities),
433        ),
434        (
435            copy.contributors_title,
436            course_content::contributors_content(copy, analysis),
437        ),
438    ] {
439        insert_generated_page(
440            &mut tx,
441            NewPage {
442                url_path: format!("/{}", truncated_slug(title, "page")),
443                title: title.to_string(),
444                chapter_id: None,
445                ..generated_page(course.id, content)
446            },
447            user_id,
448            &spec_fetcher,
449            fetch_service_info,
450        )
451        .await?;
452    }
453
454    for (term, definition) in
455        course_content::glossary_entries(&design.style_guide.terminology_and_acronyms)
456    {
457        crate::glossary::insert(&mut tx, &term, &definition, course.id).await?;
458    }
459
460    if let Some(chatbot) = drafts
461        .course
462        .chatbot
463        .as_ref()
464        .filter(|_| source.has_planned_chatbot())
465    {
466        let model = default_chatbot_model(&mut tx).await?;
467        chatbot_configurations::insert(
468            &mut tx,
469            PKeyPolicy::Generate,
470            NewChatbotConf {
471                chatbot_name: chatbot.name.trim().to_string(),
472                course_id: Some(course.id),
473                model_id: model.id,
474                enabled_to_students: false,
475                // The course has no content to search yet; the teacher turns search on later.
476                use_azure_search: false,
477                maintain_azure_search_index: false,
478                prompt: chatbot.prompt.trim().to_string(),
479                initial_message: chatbot.initial_message.trim().to_string(),
480                initial_suggested_messages: Some(chatbot.suggested_messages.clone()),
481                ..Default::default()
482            },
483        )
484        .await?;
485    }
486
487    sqlx::query!(
488        r#"
489UPDATE course_designer_plans
490SET created_course_id = $2
491WHERE id = $1
492"#,
493        plan_id,
494        course.id
495    )
496    .execute(&mut *tx)
497    .await?;
498    insert_plan_event(
499        &mut tx,
500        plan_id,
501        Some(user_id),
502        "course_created",
503        plan.active_stage,
504        json!({ "course_id": course.id, "chapter_count": design.chapters.len() }),
505    )
506    .await?;
507    tx.commit().await?;
508    Ok(course)
509}

546/// The plan's saved Analysis and Design forms, empty when not saved yet.
547async fn get_analysis_and_design(
548    conn: &mut PgConnection,
549    plan_id: Uuid,
550) -> ModelResult<(AnalysisWorkspaceV1, DesignWorkspaceV1)> {
551    let rows = sqlx::query!(
552        r#"
553SELECT workspace_data
554FROM course_designer_plan_stages
555WHERE course_designer_plan_id = $1
556  AND stage IN ('analysis', 'design')
557  AND deleted_at IS NULL
558"#,
559        plan_id
560    )
561    .fetch_all(conn)
562    .await?;
563    let mut analysis = AnalysisWorkspaceV1::default();
564    let mut design = DesignWorkspaceV1::default();
565    for row in rows {
566        match parse_workspace_data(row.workspace_data)? {
567            Some(CourseDesignerStageWorkspace::AnalysisV1(a)) => analysis = a,
568            Some(CourseDesignerStageWorkspace::DesignV1(d)) => design = d,
569            None => {}
570        }
571    }
572    Ok((analysis, design))
573}

575/// The Analysis form's language is free text, so accept a code or a few common language names.
576fn language_code_from_analysis(language: Option<&str>) -> String {
577    let Some(language) = language.and_then(non_empty_trimmed) else {
578        return FALLBACK_LANGUAGE_CODE.to_string();
579    };
580    if is_ietf_language_code_like(language) {
581        return language.to_string();
582    }
583    match language.to_lowercase().as_str() {
584        "finnish" | "suomi" | "suomeksi" => "fi",
585        "swedish" | "svenska" | "ruotsi" => "sv",
586        _ => FALLBACK_LANGUAGE_CODE,
587    }
588    .to_string()
589}

591fn truncated_slug(title: &str, fallback: &str) -> String {
592    let slug: String = slugify(title).chars().take(MAX_SLUG_CHARS).collect();
593    let slug = slug.trim_end_matches('-');
594    if slug.is_empty() {
595        fallback.to_string()
596    } else {
597        slug.to_string()
598    }
599}

601fn unique_segment(used_segments: &mut HashSet<String>, title: &str) -> String {
602    let base = truncated_slug(title, "page");
603    let mut candidate = base.clone();
604    let mut suffix = 2;
605    while !used_segments.insert(candidate.clone()) {
606        candidate = format!("{base}-{suffix}");
607        suffix += 1;
608    }
609    candidate
610}

612/// A slug free in both `courses` and `course_language_groups`, which a new course takes from it.
613async fn unique_course_slug(conn: &mut PgConnection, name: &str) -> ModelResult<String> {
614    let base = truncated_slug(name, "course");
615    let mut candidate = base.clone();
616    let mut suffix = 2;
617    loop {
618        let is_taken = sqlx::query_scalar!(
619            r#"
620SELECT (
621    EXISTS (
622      SELECT 1
623      FROM courses
624      WHERE slug = $1
625        AND deleted_at IS NULL
626    )
627    OR EXISTS (
628      SELECT 1
629      FROM course_language_groups
630      WHERE slug = $1
631        AND deleted_at IS NULL
632    )
633  ) AS "is_taken!"
634"#,
635            candidate
636        )
637        .fetch_one(&mut *conn)
638        .await?;
639        if !is_taken {
640            return Ok(candidate);
641        }
642        candidate = format!("{base}-{suffix}");
643        suffix += 1;
644    }
645}

666/// In-progress plans due a weekly email: today (Helsinki time) is within the plan's schedule, and
667/// the last email is older than six days or none was sent. Six so a slightly early run still sends.
668pub async fn get_plan_ids_due_for_weekly_digest(
669    conn: &mut PgConnection,
670    now: DateTime<Utc>,
671) -> ModelResult<Vec<Uuid>> {
672    let ids = sqlx::query_scalar!(
673        r#"
674SELECT p.id
675FROM course_designer_plans p
676  JOIN course_designer_plan_stages s ON s.course_designer_plan_id = p.id
677  AND s.deleted_at IS NULL
678WHERE p.status = 'in_progress'
679  AND p.deleted_at IS NULL
680  AND (
681    p.last_weekly_stage_email_sent_at IS NULL
682    OR p.last_weekly_stage_email_sent_at < $1::TIMESTAMPTZ - INTERVAL '6 days'
683  )
684GROUP BY p.id
685HAVING ($1::TIMESTAMPTZ AT TIME ZONE 'Europe/Helsinki')::DATE BETWEEN MIN(s.planned_starts_on) AND MAX(s.planned_ends_on)
686ORDER BY p.id
687"#,
688        now
689    )
690    .fetch_all(conn)
691    .await?;
692    Ok(ids)
693}

695/// Gathers a plan's weekly email content: task completions since the previous email, or the past
696/// week for the first one. Locks the plan row, so call it in the transaction that queues the email
697/// and calls [`mark_weekly_digest_sent`].
698pub async fn get_weekly_digest(
699    conn: &mut PgConnection,
700    plan_id: Uuid,
701    now: DateTime<Utc>,
702) -> ModelResult<WeeklyDigest> {
703    let plan = sqlx::query!(
704        r#"
705SELECT
706  p.name,
707  p.active_stage AS "active_stage: CourseDesignerStage",
708  p.last_weekly_stage_email_sent_at,
709  s.planned_ends_on AS "active_stage_planned_ends_on?"
710FROM course_designer_plans p
711LEFT JOIN course_designer_plan_stages s
712  ON s.course_designer_plan_id = p.id
713  AND s.stage = p.active_stage
714  AND s.deleted_at IS NULL
715WHERE p.id = $1
716  AND p.deleted_at IS NULL
717FOR UPDATE OF p
718"#,
719        plan_id
720    )
721    .fetch_one(&mut *conn)
722    .await?;
723    let since = plan
724        .last_weekly_stage_email_sent_at
725        .unwrap_or(now - chrono::Duration::days(7));
726    let recipient_user_ids = sqlx::query_scalar!(
727        r#"
728SELECT m.user_id
729FROM course_designer_plan_members m
730JOIN users u ON u.id = m.user_id
731WHERE m.course_designer_plan_id = $1
732  AND m.receives_weekly_digest
733  AND m.deleted_at IS NULL
734  AND u.deleted_at IS NULL
735ORDER BY m.created_at, m.id
736"#,
737        plan_id
738    )
739    .fetch_all(&mut *conn)
740    .await?;
741    let task_counts = sqlx::query!(
742        r#"
743SELECT COUNT(*)::BIGINT AS "task_count!",
744  COUNT(*) FILTER (
745    WHERE t.is_completed
746      AND t.automatic_check IS NULL
747  )::BIGINT AS "completed_task_count!",
748  COALESCE(
749    ARRAY_AGG(t.automatic_check) FILTER (WHERE t.automatic_check IS NOT NULL),
750    '{}'
751  ) AS "automatic_checks!"
752FROM course_designer_plan_stage_tasks t
753JOIN course_designer_plan_stages s ON s.id = t.course_designer_plan_stage_id
754WHERE s.course_designer_plan_id = $1
755  AND s.deleted_at IS NULL
756  AND t.deleted_at IS NULL
757"#,
758        plan_id
759    )
760    .fetch_one(&mut *conn)
761    .await?;
762    let tasks_completed_since = sqlx::query_scalar!(
763        r#"
764SELECT t.title
765FROM course_designer_plan_stage_tasks t
766JOIN course_designer_plan_stages s ON s.id = t.course_designer_plan_stage_id
767WHERE s.course_designer_plan_id = $1
768  AND s.deleted_at IS NULL
769  AND t.deleted_at IS NULL
770  AND t.is_completed
771  AND t.completed_at > $2
772ORDER BY t.completed_at, t.order_number
773"#,
774        plan_id,
775        since
776    )
777    .fetch_all(&mut *conn)
778    .await?;
779    let course_status = course_designer_course_status::get_for_plan(&mut *conn, plan_id).await?;
780    let analysis_language = sqlx::query_scalar!(
781        r#"
782SELECT workspace_data->'payload'->>'language' AS language
783FROM course_designer_plan_stages
784WHERE course_designer_plan_id = $1
785  AND stage = 'analysis'
786  AND workspace_data->>'schema' = 'analysis_v1'
787  AND deleted_at IS NULL
788"#,
789        plan_id
790    )
791    .fetch_optional(&mut *conn)
792    .await?
793    .flatten();
794
795    Ok(WeeklyDigest {
796        plan_id,
797        plan_name: plan.name,
798        active_stage: plan.active_stage,
799        active_stage_planned_ends_on: plan.active_stage_planned_ends_on,
800        recipient_user_ids,
801        tasks_completed_since,
802        task_count: task_counts.task_count,
803        completed_task_count: task_counts.completed_task_count
804            + course_designer_course_status::count_passed_checks(
805                course_status.as_ref(),
806                &task_counts.automatic_checks,
807            ),
808        course_status,
809        language_code: language_code_from_analysis(analysis_language.as_deref()),
810    })
811}

813/// Records that the plan's weekly email went out, which keeps it out of the next run.
814pub async fn mark_weekly_digest_sent(
815    conn: &mut PgConnection,
816    plan_id: Uuid,
817    sent_at: DateTime<Utc>,
818) -> ModelResult<()> {
819    sqlx::query!(
820        r#"
821UPDATE course_designer_plans
822SET last_weekly_stage_email_sent_at = $2
823WHERE id = $1
824"#,
825        plan_id,
826        sent_at
827    )
828    .execute(conn)
829    .await?;
830    Ok(())
