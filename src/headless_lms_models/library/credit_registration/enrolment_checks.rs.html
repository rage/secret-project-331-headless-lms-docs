21/// What a check request did to the row.
22#[derive(Debug, Clone, Copy, PartialEq, Eq)]
23pub enum CheckRequestOutcome {
24    /// A check is due now.
25    CheckStarted,
26    /// The schedule restarted, but without the immediate check: the row was checked moments ago.
27    Rescheduled,
28    /// Asked too soon after the last request, or checked moments ago with the day's restarts used
29    /// up. Nothing changed.
30    TooSoon,
31    /// The row is not waiting for an enrolment.
32    NotWaiting,
33}
34
35impl CheckRequestOutcome {
36    /// Whether the request made a check due now.
37    pub fn started_check(self) -> bool {
38        self == Self::CheckStarted
39    }
40}

54async fn lock_schedule(conn: &mut PgConnection, id: Uuid) -> ModelResult<ScheduleFacts> {

62  enrolment_checks_stopped_at,

64  enrolment_check_restart_window_started_at,

71        "#,

93/// Writes a (re)started or advanced schedule. `next_attempt_at` is only moved for a row parked in
94/// `no_usable_enrolment`, and only ever earlier unless `replaces_next_attempt`: a row in a resolve
95/// state has a check on its way already, and one a roster just woke must stay woken.
96#[allow(clippy::too_many_arguments)]
97async fn write_schedule(
98    conn: &mut PgConnection,
99    id: Uuid,
100    group: EnrolmentCheckGroup,
101    anchor_at: DateTime<Utc>,
102    scheduled: Option<ScheduledEnrolmentCheck>,
103    source: EnrolmentCheckSource,
104    next_attempt_at: DateTime<Utc>,
105    replaces_next_attempt: bool,
106) -> ModelResult<()> {
107    sqlx::query!(
108        r#"
109UPDATE credit_registrations
110SET enrolment_check_group = GREATEST(enrolment_check_group, $2),
111  enrolment_check_anchor_at = $3,
112  enrolment_check_step = $4,
113  enrolment_check_due_at = $5,
114  is_enrolment_check_batched = $6,
115  enrolment_checks_stopped_at = CASE
116    WHEN $4::int IS NULL THEN COALESCE(enrolment_checks_stopped_at, now())
117  END,
118  enrolment_check_source = $7,
119  next_attempt_at = CASE
120    WHEN state <> 'no_usable_enrolment' THEN next_attempt_at
121    WHEN $9 THEN $8
122    ELSE LEAST(next_attempt_at, $8)
123  END
124WHERE id = $1
125        "#,
126        id,
127        group as EnrolmentCheckGroup,
128        anchor_at,
129        scheduled.map(|scheduled| scheduled.step),
130        scheduled.map(|scheduled| scheduled.due_at),
131        scheduled.is_some_and(|scheduled| scheduled.is_batched),
132        source as EnrolmentCheckSource,
133        next_attempt_at,
134        replaces_next_attempt,
135    )
136    .execute(conn)
137    .await?;
138    Ok(())
139}

141/// A student or teacher asking for a check, or Done being pressed: restarts the row on the
142/// check-requested ladder with its immediate check.
143///
144/// Shares one 30-minute limit between every kind of request. The immediate check is skipped when
145/// the row was checked within that limit, and past the daily restart cap a request gets its
146/// immediate check without restarting the ladder.
147pub async fn request_check(
148    conn: &mut PgConnection,
149    id: Uuid,
150    source: EnrolmentCheckSource,
151    now: DateTime<Utc>,
152) -> ModelResult<CheckRequestOutcome> {
153    let mut tx = conn.begin().await?;
154    let facts = lock_schedule(&mut tx, id).await?;
155    if !facts.is_waiting() {
156        return Ok(CheckRequestOutcome::NotWaiting);
157    }
158    if is_within(
159        facts.enrolment_check_requested_at,
160        now,
161        CHECK_REQUEST_MIN_INTERVAL,
162    ) {
163        return Ok(CheckRequestOutcome::TooSoon);
164    }
165    let checked_recently = is_within(facts.enrolment_checked_at, now, CHECK_REQUEST_MIN_INTERVAL);
166    let window_is_current = is_within(
167        facts.enrolment_check_restart_window_started_at,
168        now,
169        CHECK_REQUEST_RESTART_WINDOW,
170    );
171    let (window_started_at, restart_count) = if window_is_current {
172        (
173            facts
174                .enrolment_check_restart_window_started_at
175                .unwrap_or(now),
176            facts.enrolment_check_restart_count,
177        )
178    } else {
179        (now, 0)
180    };
181    let may_restart = restart_count < MAX_CHECK_REQUEST_RESTARTS_PER_DAY;
182    if !may_restart && checked_recently {
183        return Ok(CheckRequestOutcome::TooSoon);
184    }
185    sqlx::query!(
186        r#"
187UPDATE credit_registrations
188SET enrolment_check_requested_at = $2,
189  enrolment_check_restart_window_started_at = $3,
190  enrolment_check_restart_count = $4
191WHERE id = $1
192        "#,
193        id,
194        now,
195        window_started_at,
196        restart_count + i32::from(may_restart),
197    )
198    .execute(&mut *tx)
199    .await?;
200
201    let outcome = if may_restart {
202        let group = facts
203            .enrolment_check_group
204            .max(EnrolmentCheckGroup::CheckRequested);
205        let (scheduled, source, next_attempt_at, outcome) = if checked_recently {
206            let scheduled = next_check_after(group, now, now);
207            (
208                scheduled,
209                EnrolmentCheckSource::Schedule,
210                scheduled.map_or_else(never, |scheduled| scheduled.release_at()),
211                CheckRequestOutcome::Rescheduled,
212            )
213        } else {
214            (
215                first_check(group, now),
216                source,
217                now,
218                CheckRequestOutcome::CheckStarted,
219            )
220        };
221        write_schedule(
222            &mut tx,
223            id,
224            group,
225            now,
226            scheduled,
227            source,
228            next_attempt_at,
229            true,
230        )
231        .await?;
232        outcome
233    } else {
234        sqlx::query!(
235            r#"
236UPDATE credit_registrations
237SET enrolment_check_source = $2,
238  next_attempt_at = CASE
239    WHEN state = 'no_usable_enrolment' THEN now()
240    ELSE next_attempt_at
241  END
242WHERE id = $1
243            "#,
244            id,
245            source as EnrolmentCheckSource,
246        )
247        .execute(&mut *tx)
248        .await?;
249        CheckRequestOutcome::CheckStarted
250    };
251    tx.commit().await?;
252    Ok(outcome)
253}

255/// Whether a check request would be refused as too soon right now, which is when the buttons that
256/// make one are hidden.
257pub fn is_check_request_limited(
258    enrolment_check_requested_at: Option<DateTime<Utc>>,
259    enrolment_checked_at: Option<DateTime<Utc>>,
260    now: DateTime<Utc>,
261) -> bool {
262    is_within(
263        enrolment_check_requested_at,
264        now,
265        CHECK_REQUEST_MIN_INTERVAL,
266    ) || is_within(enrolment_checked_at, now, CHECK_REQUEST_MIN_INTERVAL)
267}

269/// A visit to the registration page while it showed the enrolment instructions. Moves a
270/// `completed` row onto the visited ladder; restarts a `visited` row at most once a day, and a
271/// stopped row of any group. Returns whether the schedule restarted.
272pub async fn record_visit(
273    conn: &mut PgConnection,
274    id: Uuid,
275    now: DateTime<Utc>,
276) -> ModelResult<bool> {
277    let mut tx = conn.begin().await?;
278    let facts = lock_schedule(&mut tx, id).await?;
279    if !facts.is_waiting() {
280        return Ok(false);
281    }
282    let restart_is_due = !is_within(
283        facts.enrolment_check_anchor_at,
284        now,
285        VISIT_RESTART_MIN_INTERVAL,
286    );
287    let restarts = facts.enrolment_check_group < EnrolmentCheckGroup::Visited
288        || (restart_is_due
289            && (facts.enrolment_checks_stopped_at.is_some()
290                || facts.enrolment_check_group == EnrolmentCheckGroup::Visited));
291    if !restarts {
292        return Ok(false);
293    }
294    let group = facts
295        .enrolment_check_group
296        .max(EnrolmentCheckGroup::Visited);
297    let scheduled = first_check(group, now);
298    write_schedule(
299        &mut tx,
300        id,
301        group,
302        now,
303        scheduled,
304        EnrolmentCheckSource::Schedule,
305        scheduled.map_or_else(never, |scheduled| scheduled.release_at()),
306        false,
307    )
308    .await?;
309    tx.commit().await?;
310    Ok(true)
311}

313/// Schedules the next check of a row an answer just parked in `no_usable_enrolment`: the first
314/// rung after now, or stopped once the ladder has run out. A row with no schedule yet, such as one
315/// import sent back, starts one now.
316///
317/// Reads the schedule under lock rather than from the snapshot the answer was decided on, so a
318/// request that restarted the row while the check was out is kept.
319pub async fn schedule_next_check(conn: &mut PgConnection, id: Uuid) -> ModelResult<()> {
320    let facts = lock_schedule(conn, id).await?;
321    if facts.state != CreditRegistrationState::NoUsableEnrolment {
322        return Ok(());
323    }
324    let now = Utc::now();
325    let anchor_at = facts.enrolment_check_anchor_at.unwrap_or(now);
326    let scheduled = next_check_after(facts.enrolment_check_group, anchor_at, now);
327    write_schedule(
328        conn,
329        id,
330        facts.enrolment_check_group,
331        anchor_at,
332        scheduled,
333        EnrolmentCheckSource::Schedule,
334        scheduled.map_or_else(never, |scheduled| scheduled.release_at()),
335        true,
336    )
337    .await
338}

351/// Writes the schedules rows start with, and ends the checks of the earlier attempts each new row
352/// replaces for its student and module.
353///
354/// Only rows still in `expected_state`: one a phase moved since the decision keeps whatever that
355/// phase wrote.
356pub async fn record_starts(
357    conn: &mut PgConnection,
358    starts: &[EnrolmentCheckStart],
359) -> ModelResult<()> {
360    if starts.is_empty() {
361        return Ok(());
362    }
363    let ids: Vec<Uuid> = starts
364        .iter()
365        .map(|start| start.credit_registration_id)
366        .collect();
367    let groups: Vec<EnrolmentCheckGroup> = starts.iter().map(|start| start.group).collect();
368    let anchors: Vec<DateTime<Utc>> = starts.iter().map(|start| start.anchor_at).collect();
369    let steps: Vec<i32> = starts.iter().map(|start| start.scheduled.step).collect();
370    let due_ats: Vec<DateTime<Utc>> = starts.iter().map(|start| start.scheduled.due_at).collect();
371    let batched: Vec<bool> = starts
372        .iter()
373        .map(|start| start.scheduled.is_batched)
374        .collect();
375    let sources: Vec<EnrolmentCheckSource> = starts.iter().map(|start| start.source).collect();
376    sqlx::query!(
377        r#"
378UPDATE credit_registrations cr
379SET enrolment_check_group = GREATEST(cr.enrolment_check_group, started.enrolment_check_group),
380  enrolment_check_anchor_at = started.anchor_at,
381  enrolment_check_step = started.step,
382  enrolment_check_due_at = started.due_at,
383  is_enrolment_check_batched = started.is_batched,
384  enrolment_checks_stopped_at = NULL,
385  enrolment_check_source = started.source
386FROM UNNEST(
387    $1::uuid [],
388    $2::enrolment_check_group [],
389    $3::timestamptz [],
390    $4::int [],
391    $5::timestamptz [],
392    $6::boolean [],
393    $7::enrolment_check_source []
394  ) AS started(
395    id,
396    enrolment_check_group,
397    anchor_at,
398    step,
399    due_at,
400    is_batched,
401    source
402  )
403WHERE cr.id = started.id
404  AND cr.state IN ('no_usable_enrolment', 'ready_to_submit')
405  AND cr.deleted_at IS NULL
406        "#,
407        &ids,
408        &groups as &[EnrolmentCheckGroup],
409        &anchors,
410        &steps,
411        &due_ats,
412        &batched,
413        &sources as &[EnrolmentCheckSource],
414    )
415    .execute(&mut *conn)
416    .await?;
417    sqlx::query!(
418        r#"
419UPDATE credit_registrations replaced
420SET enrolment_check_anchor_at = COALESCE(replaced.enrolment_check_anchor_at, now()),
421  enrolment_check_step = NULL,
422  enrolment_check_due_at = NULL,
423  is_enrolment_check_batched = FALSE,
424  enrolment_checks_stopped_at = COALESCE(replaced.enrolment_checks_stopped_at, now()),
425  next_attempt_at = $2
426FROM credit_registrations started
427WHERE started.id = ANY($1::uuid [])
428  AND replaced.user_id = started.user_id
429  AND replaced.course_module_id = started.course_module_id
430  AND replaced.id <> started.id
431  AND replaced.created_at < started.created_at
432  AND replaced.state = 'no_usable_enrolment'
433  AND replaced.superseded_by_id IS NULL
434  AND replaced.deleted_at IS NULL
435        "#,
436        &ids,
437        never(),
438    )
439    .execute(conn)
440    .await?;
441    Ok(())
442}

448    pub enrolment_ids: Vec<String>,
449}

451/// Wakes this module's rows waiting for an enrolment when the roster lists their student under an
452/// enrolment id the row has not seen, stopped rows included, and returns how many it touched.
453///
454/// A row never checked has seen nothing, so any listing wakes it. The listed ids count as seen from
455/// here on, so one listing wakes a row once. A row a later attempt has replaced is left alone. Only
456/// the clock moves: the enrolment is still resolved by a check rather than read off the roster.
457pub async fn wake_for_roster_listing(
458    conn: &mut PgConnection,
459    course_module_id: Uuid,
460    enrolees: &[RosterEnrolee],
461) -> ModelResult<u64> {
462    let mut user_ids = Vec::new();
463    let mut enrolment_ids: Vec<Option<String>> = Vec::new();
464    for enrolee in enrolees {
465        if enrolee.enrolment_ids.is_empty() {
466            user_ids.push(enrolee.user_id);
467            enrolment_ids.push(None);
468        }
469        for enrolment_id in &enrolee.enrolment_ids {
470            user_ids.push(enrolee.user_id);
471            enrolment_ids.push(Some(enrolment_id.clone()));
472        }
473    }
474    let res = sqlx::query!(
475        r#"
476WITH listed AS (
477  SELECT user_id,
478    ARRAY_REMOVE(ARRAY_AGG(enrolment_id), NULL) AS enrolment_ids
479  FROM UNNEST($2::uuid [], $3::text []) AS listing(user_id, enrolment_id)
480  GROUP BY user_id
481)
482UPDATE credit_registrations cr
483SET enrolment_check_source = CASE
484    WHEN cr.next_attempt_at > now() THEN 'roster_listing'
485    ELSE cr.enrolment_check_source
486  END,
487  next_attempt_at = LEAST(cr.next_attempt_at, now()),
488  seen_enrolment_ids = ARRAY(
489    SELECT DISTINCT UNNEST(COALESCE(cr.seen_enrolment_ids, '{}') || listed.enrolment_ids)
490  )
491FROM listed
492WHERE cr.course_module_id = $1
493  AND cr.user_id = listed.user_id
494  AND cr.state = 'no_usable_enrolment'
495  AND cr.superseded_by_id IS NULL
496  AND cr.deleted_at IS NULL
497  AND (
498    cr.seen_enrolment_ids IS NULL
499    OR NOT listed.enrolment_ids <@ cr.seen_enrolment_ids
500  )
501  AND NOT EXISTS (
502    SELECT 1
503    FROM credit_registrations later
504    WHERE later.user_id = cr.user_id
505      AND later.course_module_id = cr.course_module_id
506      AND later.created_at > cr.created_at
507      AND later.deleted_at IS NULL
508  )
509        "#,
510        course_module_id,
511        &user_ids,
512        &enrolment_ids as &[Option<String>],
513    )
514    .execute(conn)
515    .await?;
516    Ok(res.rows_affected())
517}

542/// An answered check of a row waiting for an enrolment, logged with the write its answer makes.
543#[derive(Clone, Copy)]
544pub struct EnrolmentCheckAnswer<'a> {
545    /// The row as it was claimed for the check.
546    pub checked: &'a CreditRegistration,
547    /// The enrolment the answer had to register against, however the row then settled.
548    pub usable_enrolment: Option<&'a RegistryEnrolment>,
549    pub listed_enrolments: &'a [RegistryEnrolment],
550}

552/// Logs what a check found and remembers the enrolments it saw for the roster wake-ups, given the
553/// row as the answer's write left it. A lookup that failed in transit was no check and leaves no
554/// trace.
555pub async fn record_enrolment_check(
556    conn: &mut PgConnection,
557    check: Option<&EnrolmentCheckAnswer<'_>>,
558    after: &CreditRegistration,
559) -> ModelResult<()> {
560    let Some(check) = check else {
561        return Ok(());
562    };
563    let row = check.checked;
564    let was_answered = after.state != CreditRegistrationState::NoUsableEnrolment
565        || after.enrolment_checked_at != row.enrolment_checked_at;
566    if !was_answered {
567        return Ok(());
568    }
569    credit_registration_enrolment_check_outcomes::insert(
570        conn,
571        &NewEnrolmentCheckOutcome {
572            credit_registration_id: row.id,
573            course_module_id: row.course_module_id,
574            enrolment_check_group: row.enrolment_check_group,
575            enrolment_check_step: row.enrolment_check_step,
576            source: row.enrolment_check_source,
577            due_at: row.enrolment_check_due_at,
578            checked_at: after.enrolment_checked_at.unwrap_or_else(Utc::now),
579            previous_checked_at: row.enrolment_checked_at,
580            is_enrolment_found: check.usable_enrolment.is_some(),
581            enrolled_at: check
582                .usable_enrolment
583                .and_then(|enrolment| enrolment.enrolment_date_time),
584        },
585    )
586    .await?;
587    let seen: Vec<String> = check
588        .listed_enrolments
589        .iter()
590        .map(|enrolment| enrolment.id.clone())
591        .collect();
592    add_seen_enrolment_ids(conn, row.id, &seen).await?;
593    Ok(())
594}

596/// Moves the schedules of a module's rows past a pause of `paused_for`, so the pause spends none
597/// of their ladder. A batched row stays on a batch boundary. Stopped rows are left as they are.
598pub async fn shift_past_pause(
599    conn: &mut PgConnection,
600    course_module_id: Uuid,
601    paused_for: TimeDelta,
602) -> ModelResult<u64> {
603    let res = sqlx::query!(
604        r#"
605UPDATE credit_registrations
606SET enrolment_check_anchor_at = enrolment_check_anchor_at + $2::interval,
607  enrolment_check_due_at = enrolment_check_due_at + $2::interval,
608  next_attempt_at = CASE
609    WHEN state <> 'no_usable_enrolment'
610    OR enrolment_check_source <> 'schedule' THEN next_attempt_at
611    WHEN is_enrolment_check_batched THEN to_timestamp(
612      CEIL(
613        EXTRACT(
614          EPOCH
615          FROM next_attempt_at + $2::interval
616        ) / $3::bigint
617      ) * $3::bigint
618    )
619    ELSE next_attempt_at + $2::interval
620  END
621WHERE course_module_id = $1
622  AND enrolment_check_anchor_at IS NOT NULL
623  AND enrolment_checks_stopped_at IS NULL
624  AND superseded_by_id IS NULL
625  AND deleted_at IS NULL
626        "#,
627        course_module_id,
628        paused_for.max(TimeDelta::zero()) as TimeDelta,
629        BATCH_INTERVAL.num_seconds(),
630    )
631    .execute(conn)
632    .await?;
633    Ok(res.rows_affected())
634}

636/// Brings the slow checks due within [`BATCH_PULL_FORWARD`] forward to now when another slow
637/// check in `scope` is released and not yet claimed: they cost nothing extra in the request that
638/// goes out for it. A row tried within [`TRANSIENT_FAILURE_RETRY`] keeps waiting out its failed
639/// lookup.
640///
641/// Its test for a released check shares its filters with the claim behind
642/// [`claim_due_for_resolve`](crate::credit_registrations::claim_due_for_resolve) and with
643/// [`count_due_enrolment_checks`](crate::credit_registrations::count_due_enrolment_checks); change
644/// all three together.
645pub async fn pull_forward_batched_checks(
646    conn: &mut PgConnection,
647    scope: &RegistrationScope,
648) -> ModelResult<()> {
649    sqlx::query!(
650        r#"
651WITH scoped AS (
652  SELECT cr.id,
653    cr.next_attempt_at,
654    cr.enrolment_check_due_at,
655    cr.last_attempt_at,
656    cr.enrolment_check_claimed_until
657  FROM credit_registrations cr
658    JOIN credit_registration_active_course_modules acm ON acm.course_module_id = cr.course_module_id
659  WHERE cr.state = 'no_usable_enrolment'
660    AND cr.is_enrolment_check_batched
661    AND cr.superseded_by_id IS NULL
662    AND cr.deleted_at IS NULL
663    AND ($1::uuid IS NULL OR cr.course_id = $1)
664    AND ($2::uuid IS NULL OR cr.user_id = $2)
665    AND (
666      cardinality($3::uuid []) = 0
667      OR cr.id = ANY($3::uuid [])
668    )
669)
670UPDATE credit_registrations cr
671SET next_attempt_at = now()
672FROM scoped
673WHERE cr.id = scoped.id
674  AND scoped.next_attempt_at > now()
675  AND scoped.enrolment_check_due_at > now()
676  AND scoped.enrolment_check_due_at <= now() + $4::interval
677  AND (
678    scoped.last_attempt_at IS NULL
679    OR scoped.last_attempt_at <= now() - $5::interval
680  )
681  AND EXISTS (
682    SELECT 1
683    FROM scoped released
684    WHERE released.next_attempt_at <= now()
685      AND (
686        released.enrolment_check_claimed_until IS NULL
687        OR released.enrolment_check_claimed_until <= now()
688      )
689  )
690        "#,
691        scope.course_id,
692        scope.user_id,
693        &scope.credit_registration_ids,
694        BATCH_PULL_FORWARD as TimeDelta,
695        TRANSIENT_FAILURE_RETRY as TimeDelta,
696    )
697    .execute(conn)
698    .await?;
699    Ok(())
700}
