1//! Everything a submission needs that does not live on the ledger row yet. One query rather than a
2//! lookup per row: the phases work in batches of up to a thousand.
3
4use std::collections::HashMap;
5
6use crate::prelude::*;
7
8use super::payload::CompletionFacts;
9
10/// The module's or mooc.fi target's fields, the linked student number and the completion, for one
11/// ledger row.
12#[derive(Debug, Clone)]
13pub struct SubmissionContext {
14    pub registration_id: Uuid,
15    /// `None` once the account's number has been unlinked.
16    pub student_number: Option<DbSecret>,
17    pub sisu_person_id: Option<DbSecret>,
18    pub uh_course_code: Option<String>,
19    pub ects_credits: Option<f32>,
20    pub completion: CompletionFacts,
21}
22
23/// The per-row facts each phase needs to act on a batch of ledger rows without a lookup per row.
24pub async fn get_submission_contexts(
25    conn: &mut PgConnection,
26    registration_ids: &[Uuid],
27) -> ModelResult<HashMap<Uuid, SubmissionContext>> {
28    let rows = sqlx::query!(
29        r#"
30-- Nullability is stated for the outer-joined columns rather than inferred: sqlx-cli 0.9.0 derives
31-- them the wrong way round, and an offline build of this query then fails to compile.
32SELECT cr.id,
33  vsn.student_number AS "student_number?",
34  vsn.sisu_person_id AS "sisu_person_id?",
35  s.uh_course_code,
36  s.ects_credits,
37  -- An unrecognised mooc.fi grade blocks the row before anything is built from this.
38  COALESCE(s.passed, FALSE) AS "passed!",
39  s.grade,
40  s.completion_date AS "completion_date!",
41  s.completion_language AS "completion_language!"
42FROM credit_registrations cr
43  JOIN credit_registration_subjects s ON s.subject_id = cr.subject_id
44  AND NOT s.is_withdrawn
45  LEFT JOIN course_modules cm ON cm.id = cr.course_module_id
46  LEFT JOIN verified_student_numbers vsn ON vsn.user_id = cr.user_id
47  AND vsn.deleted_at IS NULL
48WHERE cr.id = ANY($1::uuid [])
49  AND cr.deleted_at IS NULL
50  AND cm.deleted_at IS NULL
51        "#,
52        registration_ids,
53    )
54    .fetch_all(conn)
55    .await?;
56    Ok(rows
57        .into_iter()
58        .map(|row| {
59            (
60                row.id,
61                SubmissionContext {
62                    registration_id: row.id,
63                    student_number: row.student_number,
64                    sisu_person_id: row.sisu_person_id,
65                    uh_course_code: row.uh_course_code,
66                    ects_credits: row.ects_credits,
67                    completion: CompletionFacts {
68                        passed: row.passed,
69                        grade: row.grade,
70                        completion_date: row.completion_date,
71                        completion_language: row.completion_language,
72                    },
73                },
74            )
75        })
76        .collect())
77}
