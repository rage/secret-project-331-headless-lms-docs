1//! Which mooc.fi completions `moocfi-recheck` reads again, so import finds them fresh and never has
2//! to call mooc.fi itself.
3
4use chrono::TimeDelta;
5
6use crate::credit_registrations::{CreditRegistrationState, RegistrationScope};
7use crate::prelude::*;
8
9/// A row last read longer ago than this is read again. Half of
10/// [`MOOCFI_IMPORT_MAX_CHECK_AGE`](crate::credit_registrations::MOOCFI_IMPORT_MAX_CHECK_AGE), so a
11/// row on its way to import is re-read before import stops claiming it.
12pub const MOOCFI_RECHECK_AFTER: TimeDelta = TimeDelta::hours(1);
13
14/// How many lookups one iteration may make.
15pub const MOOCFI_RECHECK_LIMIT: i64 = 200;
16
17/// The states from which import is next: their rows are the ones a fresh read is for.
18const STATES_BEFORE_IMPORT: [CreditRegistrationState; 4] = [
19    CreditRegistrationState::ReadyToSubmit,
20    CreditRegistrationState::ResolvingEnrolment,
21    CreditRegistrationState::CheckingEnrolment,
22    CreditRegistrationState::NoUsableEnrolment,
23];
24
25/// One student's completions of one mooc.fi course, which one lookup reads.
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub struct MoocfiRecheckGroup {
28    pub user_id: Uuid,
29    pub moocfi_user_upstream_id: i32,
30    pub moocfi_course_id: Uuid,
31}
32
33/// The lookups due, stalest first: every student and mooc.fi course with a row before import, or
34/// blocked on a grade mooc.fi may since have fixed, last read or tried over
35/// [`MOOCFI_RECHECK_AFTER`] ago. Rows of a paused or switched-off target wait with their target.
36pub async fn get_due_recheck_groups(
37    conn: &mut PgConnection,
38    scope: &RegistrationScope,
39    limit: i64,
40) -> ModelResult<Vec<MoocfiRecheckGroup>> {
41    let res = sqlx::query_as!(
42        MoocfiRecheckGroup,
43        r#"
44SELECT ec.user_id,
45  ec.moocfi_user_upstream_id,
46  t.moocfi_course_id
47FROM credit_registrations cr
48  JOIN external_completions ec ON ec.id = cr.subject_id
49  JOIN external_credit_targets t ON t.id = ec.external_credit_target_id
50WHERE cr.deleted_at IS NULL
51  AND cr.superseded_by_id IS NULL
52  AND (
53    cr.state = ANY($1::credit_registration_state [])
54    OR (
55      cr.state = 'blocked'
56      AND ec.passed IS NULL
57    )
58  )
59  AND ec.deleted_at IS NULL
60  AND GREATEST(ec.upstream_checked_at, ec.upstream_check_failed_at) < now() - $2::interval
61  AND EXISTS (
62    SELECT 1
63    FROM credit_registration_active_units au
64    WHERE au.credit_unit_id = cr.credit_unit_id
65  )
66  AND ($3::uuid IS NULL OR cr.course_id = $3)
67  AND ($4::uuid IS NULL OR cr.external_credit_target_id = $4)
68  AND ($5::uuid IS NULL OR cr.user_id = $5)
69  AND (
70    cardinality($6::uuid []) = 0
71    OR cr.id = ANY($6::uuid [])
72  )
73GROUP BY ec.user_id,
74  ec.moocfi_user_upstream_id,
75  t.moocfi_course_id
76ORDER BY MIN(
77    GREATEST(ec.upstream_checked_at, ec.upstream_check_failed_at)
78  )
79LIMIT $7
80        "#,
81        &STATES_BEFORE_IMPORT as &[CreditRegistrationState],
82        MOOCFI_RECHECK_AFTER as TimeDelta,
83        scope.course_id,
84        scope.external_credit_target_id,
85        scope.user_id,
86        &scope.credit_registration_ids,
87        limit,
88    )
89    .fetch_all(conn)
90    .await?;
91    Ok(res)
92}
