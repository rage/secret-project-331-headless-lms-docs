11/// The stage a student sees.
12#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, Hash, ToSchema)]
13#[serde(rename_all = "snake_case")]
14pub enum StudentFacingCreditRegistrationStatus {
15    WaitingForCompletion,
16    NeedsStudentNumber,
17    /// We are still working out which enrolment to register against. Not [`Self::NeedsEnrolment`],
18    /// which is the answer that there is none.
19    LookingForEnrolment,
20    NeedsEnrolment,
21    /// An enrolment is settled and the attainment is on its way to the study registry.
22    Sending,
23    WaitingForSisu,
24    Registered,
25    Failed,
26    /// Nothing is happening and nothing will, until the student changes something.
27    NotRegistering,
28}
29
30impl StudentFacingCreditRegistrationStatus {
31    /// `preconditions` is only read for `pending`, whose whole point is that the ledger does not
32    /// record which of them the row is waiting on. Pass [`PendingPreconditions::ALL_MET`] only where
33    /// the row is known not to be pending.
34    ///
35    /// `enrolment_resolved` is whether the row has settled on an enrolment, which only
36    /// `failed_retryable` reads: that state does not record which half of the work failed, and a
37    /// retry while we are still looking for an enrolment is not a retry of the sending.
38    pub fn of(
39        state: CreditRegistrationState,
40        preconditions: PendingPreconditions,
41        enrolment_resolved: bool,
42    ) -> Self {
43        use CreditRegistrationPendingReason as Reason;
44        use CreditRegistrationState as State;
45        match state {
46            State::Pending => match preconditions.reason() {
47                Some(Reason::Completion) => Self::WaitingForCompletion,
48                Some(Reason::StudentNumber) => Self::NeedsStudentNumber,
49                // Nothing is outstanding, so the next precondition tick moves the row on.
50                None => Self::LookingForEnrolment,
51            },
52            State::ReadyToSubmit | State::ResolvingEnrolment | State::CheckingEnrolment => {
53                Self::LookingForEnrolment
54            }
55            State::Submitting => Self::Sending,
56            State::FailedRetryable => {
57                if enrolment_resolved {
58                    Self::Sending
59                } else {
60                    Self::LookingForEnrolment
61                }
62            }
63            State::NoUsableEnrolment => Self::NeedsEnrolment,
64            State::SubmissionUncertain | State::AwaitingVerification => Self::WaitingForSisu,
65            // not_improved means Sisu holds an equal or better attainment, so the credit exists.
66            State::Registered | State::Duplicate | State::NotImproved => Self::Registered,
67            State::Misregistered | State::FailedPermanent => Self::Failed,
68            State::Blocked | State::Cancelled => Self::NotRegistering,
69        }
70    }
71
72    /// Whether the pipeline still moves this row on its own; the status page polls while it does.
73    pub fn is_moving(self) -> bool {
74        matches!(
75            self,
76            Self::LookingForEnrolment | Self::Sending | Self::WaitingForSisu
77        )
78    }
79}

81/// The `(state, completion_eligible, has_verified_student_number, enrolment_resolved)` combinations
82/// a set of stages covers, as parallel arrays for a query to `UNNEST` and join against.
83///
84/// Enumerated from [`StudentFacingCreditRegistrationStatus::of`] rather than restated as a SQL
85/// predicate: a roster filtered to "failed" must return exactly the rows whose own badge says
86/// failed, and a new ledger state must not be able to fall out of one side of that.
87#[derive(Debug, Clone, Default, PartialEq)]
88pub struct StageMatch {
89    pub states: Vec<CreditRegistrationState>,
90    pub completion_eligible: Vec<bool>,
91    pub has_verified_student_number: Vec<bool>,
92    pub enrolment_resolved: Vec<bool>,
93}
94
95impl StageMatch {
96    /// Empty for an empty `stages`, which every query reads as "do not narrow".
97    pub fn of(stages: &[StudentFacingCreditRegistrationStatus]) -> Self {
98        let mut matched = Self::default();
99        if stages.is_empty() {
100            return matched;
101        }
102        for state in CreditRegistrationState::ALL {
103            for completion_eligible in [false, true] {
104                for has_verified_student_number in [false, true] {
105                    for enrolment_resolved in [false, true] {
106                        let preconditions = PendingPreconditions {
107                            completion_eligible,
108                            has_verified_student_number,
109                        };
110                        if stages.contains(&StudentFacingCreditRegistrationStatus::of(
111                            state,
112                            preconditions,
113                            enrolment_resolved,
114                        )) {
115                            matched.states.push(state);
116                            matched.completion_eligible.push(completion_eligible);
117                            matched
118                                .has_verified_student_number
119                                .push(has_verified_student_number);
120                            matched.enrolment_resolved.push(enrolment_resolved);
121                        }
122                    }
123                }
124            }
125        }
126        matched
127    }
128}

130#[cfg(test)]
131mod tests {
132    use super::*;
133    use CreditRegistrationState as State;
134    use StudentFacingCreditRegistrationStatus as Status;
135
136    /// A row waiting on the student must not keep the page polling; nothing changes until they act.
137    #[test]
138    fn only_the_states_the_pipeline_still_owns_keep_the_page_polling() {
139        for reason in [
140            PendingPreconditions {
141                completion_eligible: false,
142                ..PendingPreconditions::ALL_MET
143            },
144            PendingPreconditions {
145                has_verified_student_number: false,
146                ..PendingPreconditions::ALL_MET
147            },
148        ] {
149            assert!(
150                !Status::of(State::Pending, reason, false).is_moving(),
151                "{reason:?}"
152            );
153        }
154        let moving = [
155            // Nothing outstanding, so the recompute moves it on without the student doing anything.
156            State::Pending,
157            State::ReadyToSubmit,
158            State::ResolvingEnrolment,
159            State::CheckingEnrolment,
160            State::Submitting,
161            State::FailedRetryable,
162            State::SubmissionUncertain,
163            State::AwaitingVerification,
164        ];
165        for state in CreditRegistrationState::ALL {
166            for enrolment_resolved in [false, true] {
167                assert_eq!(
168                    Status::of(state, PendingPreconditions::ALL_MET, enrolment_resolved)
169                        .is_moving(),
170                    moving.contains(&state),
171                    "{state:?} {enrolment_resolved}"
172                );
173            }
174        }
175    }
176
177    /// The student's question is "do I have the credits", so every success terminal answers yes.
178    #[test]
179    fn the_success_set_is_one_stage() {
180        for state in CreditRegistrationState::ALL {
181            if state.is_success() {
182                assert_eq!(
183                    Status::of(state, PendingPreconditions::ALL_MET, true),
184                    Status::Registered,
185                    "{state:?}"
186                );
187            }
188        }
189    }
190
191    #[test]
192    fn the_wire_spelling_is_snake_case() {
193        assert_eq!(
194            serde_json::to_value(Status::NeedsStudentNumber).unwrap(),
195            serde_json::json!("needs_student_number")
196        );
197    }
