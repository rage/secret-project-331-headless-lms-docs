1//! The two terminal-state mails a student may get about a credit registration.
2//!
3//! There are exactly two, and each is sent at most once per ledger row. Idempotency lives in the
4//! `credit_registrations.{action_needed,registered}_email_delivery_id` columns rather than in the
5//! phase, so a re-tick, a restart or a row re-entering the state cannot mail twice. A
6//! grade-improvement attempt is a new row and does get its own mail.
7

17/// How many mails one iteration queues.
18pub const STUDENT_NOTIFICATION_LIMIT: i64 = 200;

20/// Which of the two student mails a row is owed, or already holds.
21#[derive(Debug, Serialize, Deserialize, PartialEq, Eq, Clone, Copy, ToSchema)]
22#[serde(rename_all = "snake_case")]
23pub enum CreditRegistrationNotificationKind {
24    /// The study registry had no usable enrolment, so the student has to act.
25    ActionNeeded,
26    /// The credit is in the study registry, whether we put it there or found it already recorded.
27    Registered,
28}
29
30impl CreditRegistrationNotificationKind {
31    pub fn email_template_type(self) -> EmailTemplateType {
32        match self {
33            Self::ActionNeeded => EmailTemplateType::CreditRegistrationActionNeeded,
34            Self::Registered => EmailTemplateType::CreditRegistrationRegistered,
35        }
36    }
37}

39/// One row owed a mail, with everything the message renders. `open_university_product_id` is the
40/// module's configured product, from which the action-needed mail's enrolment link is built.
41#[derive(Debug, Clone, PartialEq)]
42pub struct StudentNotificationToQueue {
43    pub credit_registration_id: Uuid,
44    pub kind: CreditRegistrationNotificationKind,
45    pub user_id: Uuid,
46    pub course_module_id: Uuid,
47    pub course_name: String,
48    pub course_language_code: String,
49    pub course_module_name: Option<String>,
50    pub first_name: Option<String>,
51    pub ects_credits: Option<f32>,
52    pub open_university_product_id: Option<String>,
53}

55/// Claims the rows owed a mail, locking them until the caller's transaction ends, so callers must
56/// pass a transaction. Never claims `abandoned_by_consent_withdrawal`, `cancelled`, `blocked` or
57/// any failure state: those get nothing.
58pub async fn claim_unnotified(
59    conn: &mut PgConnection,
60    scope: &RegistrationScope,
61    limit: i64,
62) -> ModelResult<Vec<StudentNotificationToQueue>> {
63    let res = sqlx::query!(
64        r#"
65SELECT cr.id AS "credit_registration_id!",
66  cr.state AS "state!: CreditRegistrationState",
67  cr.user_id AS "user_id!",
68  cr.course_module_id AS "course_module_id!",
69  c.name AS "course_name!",
70  c.language_code AS "course_language_code!",
71  cm.name AS "course_module_name?",
72  ud.first_name AS "first_name?",
73  cm.ects_credits AS "ects_credits?",
74  conf.open_university_product_id AS "open_university_product_id?"
75FROM credit_registrations cr
76  JOIN courses c ON c.id = cr.course_id
77  JOIN course_modules cm ON cm.id = cr.course_module_id
78  LEFT JOIN user_details ud ON ud.user_id = cr.user_id
79  LEFT JOIN course_module_suotar_configurations conf ON conf.course_module_id = cr.course_module_id
80  AND conf.deleted_at IS NULL
81WHERE cr.deleted_at IS NULL
82  AND (
83    (
84      cr.state = 'no_usable_enrolment'
85      AND cr.action_needed_email_delivery_id IS NULL
86    )
87    OR (
88      cr.state = ANY($5::credit_registration_state [])
89      AND cr.registered_email_delivery_id IS NULL
90    )
91  )
92  AND ($2::uuid IS NULL OR cr.course_id = $2)
93  AND ($3::uuid IS NULL OR cr.user_id = $3)
94  AND (
95    cardinality($4::uuid []) = 0
96    OR cr.id = ANY($4::uuid [])
97  )
98ORDER BY cr.state_entered_at
99FOR UPDATE OF cr SKIP LOCKED
100LIMIT $1
101        "#,
102        limit,
103        scope.course_id,
104        scope.user_id,
105        &scope.credit_registration_ids,
106        &CreditRegistrationState::SUCCESS_STATES as &[CreditRegistrationState],
107    )
108    .fetch_all(conn)
109    .await?;
110
111    Ok(res
112        .into_iter()
113        .map(|row| StudentNotificationToQueue {
114            kind: match row.state {
115                CreditRegistrationState::NoUsableEnrolment => {
116                    CreditRegistrationNotificationKind::ActionNeeded
117                }
118                _ => CreditRegistrationNotificationKind::Registered,
119            },
120            credit_registration_id: row.credit_registration_id,
121            user_id: row.user_id,
122            course_module_id: row.course_module_id,
123            course_name: row.course_name,
124            course_language_code: row.course_language_code,
125            course_module_name: row.course_module_name,
126            first_name: row.first_name,
127            ects_credits: row.ects_credits,
128            open_university_product_id: row.open_university_product_id,
129        })
130        .collect())
131}

133/// Records which delivery carries the mail, which is also what takes the row out of the queue.
134pub async fn set_email_delivery_id(
135    conn: &mut PgConnection,
136    credit_registration_id: Uuid,
137    kind: CreditRegistrationNotificationKind,
138    email_delivery_id: Uuid,
139) -> ModelResult<()> {
140    let action_needed = kind == CreditRegistrationNotificationKind::ActionNeeded;
141    sqlx::query!(
142        r#"
143UPDATE credit_registrations
144SET action_needed_email_delivery_id = CASE
145    WHEN $3 THEN $2
146    ELSE action_needed_email_delivery_id
147  END,
148  registered_email_delivery_id = CASE
149    WHEN $3 THEN registered_email_delivery_id
150    ELSE $2
151  END
152WHERE id = $1
153  AND deleted_at IS NULL
154        "#,
155        credit_registration_id,
156        email_delivery_id,
157        action_needed,
158    )
159    .execute(conn)
160    .await?;
161    Ok(())
162}

164/// One queued student mail and what we can honestly say about it.
165#[derive(Debug, Clone, PartialEq)]
166pub struct RegistrationNotificationEmail {
167    pub credit_registration_id: Uuid,
168    pub kind: CreditRegistrationNotificationKind,
169    /// The delivery the registration is pinned to. Stable for the life of the row: it is what stops a
170    /// second mail of this kind, so a changed id here means the guard was bypassed.
171    pub email_delivery_id: Uuid,
172    pub send_status: EmailSendStatusReport,
173}

175/// The mails queued for these rows, for the student, teacher and admin views that report on them.
176/// A row with neither mail queued yet contributes nothing.
177pub async fn get_for_registrations(
178    conn: &mut PgConnection,
179    credit_registration_ids: &[Uuid],
180) -> ModelResult<Vec<RegistrationNotificationEmail>> {
181    let rows = sqlx::query!(
182        r#"
183SELECT id,
184  action_needed_email_delivery_id,
185  registered_email_delivery_id
186FROM credit_registrations
187WHERE id = ANY($1::uuid [])
188  AND deleted_at IS NULL
189  AND (
190    action_needed_email_delivery_id IS NOT NULL
191    OR registered_email_delivery_id IS NOT NULL
192  )
193        "#,
194        credit_registration_ids
195    )
196    .fetch_all(&mut *conn)
197    .await?;
198
199    let delivery_ids: Vec<Uuid> = rows
200        .iter()
201        .flat_map(|row| {
202            [
203                row.action_needed_email_delivery_id,
204                row.registered_email_delivery_id,
205            ]
206        })
207        .flatten()
208        .collect();
209    let reports: HashMap<Uuid, EmailSendStatusReport> =
210        get_send_statuses(conn, &delivery_ids).await?;
211
212    let mut res = Vec::new();
213    for row in rows {
214        for (kind, delivery_id) in [
215            (
216                CreditRegistrationNotificationKind::ActionNeeded,
217                row.action_needed_email_delivery_id,
218            ),
219            (
220                CreditRegistrationNotificationKind::Registered,
221                row.registered_email_delivery_id,
222            ),
223        ] {
224            let Some((delivery_id, report)) =
225                delivery_id.and_then(|id| reports.get(&id).map(|report| (id, report)))
226            else {
227                continue;
228            };
229            res.push(RegistrationNotificationEmail {
230                credit_registration_id: row.id,
231                kind,
232                email_delivery_id: delivery_id,
233                send_status: report.clone(),
234            });
235        }
236    }
237    Ok(res)
238}
