1//! How long the pipeline waits before trying again. Generic scheduling math; which error codes are

10pub const SUBMIT_BASE_BACKOFF: TimeDelta = TimeDelta::minutes(1);
11pub const SUBMIT_MAX_BACKOFF: TimeDelta = TimeDelta::hours(6);
12/// After this long in failure a row stops being retried and becomes a support case.
13pub const SUBMIT_MAX_RETRY_AGE: TimeDelta = TimeDelta::days(7);
14/// When the first verify poll comes after the submission: about when the assessment item
15/// attainment usually shows.
16const VERIFY_WINDOW_START: TimeDelta = TimeDelta::hours(2);
17const VERIFY_DENSE_WINDOW_END: TimeDelta = TimeDelta::minutes(630);
18const VERIFY_WINDOW_END: TimeDelta = TimeDelta::hours(29);
19pub const VERIFY_WINDOW_INTERVAL: TimeDelta = TimeDelta::minutes(30);
20const VERIFY_LATE_WINDOW_INTERVAL: TimeDelta = TimeDelta::hours(1);
21pub const VERIFY_BASE_BACKOFF: TimeDelta = TimeDelta::hours(2);
22pub const VERIFY_MAX_BACKOFF: TimeDelta = TimeDelta::hours(6);
23/// After this, polling drops to daily and a human looks. Never a failure: the attainment may exist,
24/// and calling it failed would invite a second submission.
25pub const VERIFY_MAX_AGE: TimeDelta = TimeDelta::days(14);
26pub const VERIFY_GIVE_UP_POLL: TimeDelta = TimeDelta::days(1);
27pub const JITTER_MAX: TimeDelta = TimeDelta::seconds(30);
28
29/// The first wait before looking for an attainment we may or may not have created; it doubles
30/// after every fruitless look.
31pub const UNCERTAIN_RECHECK: TimeDelta = TimeDelta::minutes(15);
32pub const UNCERTAIN_MAX_RECHECK: TimeDelta = TimeDelta::hours(6);
33/// How long after the submission a human is asked to look in Sisu, well past the hour an
34/// attainment may take to show up. The row still never resubmits.
35pub const UNCERTAIN_ADMIN_AFTER: TimeDelta = TimeDelta::days(1);
36/// Suotar's `PENDING_WINDOW_MS`: how long it holds a submission it has not yet seen in Sisu as
37/// pending. Within it, a second import of the same completion can slip past Suotar's duplicate check.
38pub const SUOTAR_PENDING_WINDOW: TimeDelta = TimeDelta::hours(24);
39
40/// The wait after a registration's first failed report to mooc.fi, doubled per further failure.
41pub const MOOCFI_REPORT_BASE_BACKOFF: TimeDelta = TimeDelta::minutes(1);
42/// The longest wait between report attempts of one registration.
43pub const MOOCFI_REPORT_MAX_BACKOFF: TimeDelta = TimeDelta::hours(6);
44
45/// How long verify may see only the assessment item attainment before a human looks.
46pub const PARTIAL_REGISTRATION_ADMIN_AFTER: TimeDelta = TimeDelta::days(3);
47/// How many times Suotar may lose a submission (`notRegistered`) before a human looks. Each one
48/// still resubmits.
49pub const NOT_REGISTERED_REIMPORT_ADMIN_THRESHOLD: i32 = 3;
50
51/// A row still `submitting` this long belongs to a worker that died mid-call. Above the import
52/// timeout, so a live request is never condemned to `submission_uncertain`.
53pub const SUBMITTING_RECOVERY_GRACE: TimeDelta = TimeDelta::seconds(
54    SuotarEndpoint::ImportAttainments
55        .request_timeout()
56        .as_secs() as i64
57        + 15 * 60,
58);
59/// A row still `resolving_enrolment` this long belongs to a worker that died mid-call, and goes back
60/// to `ready_to_submit`; a parked row's claimed check expires after as long. Above the resolve
61/// timeout, and restarted before each resent half of a split batch, so a live answer still lands.
62pub const RESOLVING_RECOVERY_GRACE: TimeDelta = TimeDelta::seconds(
63    SuotarEndpoint::ResolveEnrolments
64        .request_timeout()
65        .as_secs() as i64
66        + 10 * 60,
67);

69fn doubling(base: TimeDelta, max: TimeDelta, count: i32) -> TimeDelta {
70    TimeDelta::seconds(exponential_backoff_secs(
71        base.num_seconds(),
72        max.num_seconds(),
73        count,
74    ))
75}
76
77/// The wait before the next report to mooc.fi after `attempt_count` failed ones.
78pub fn moocfi_report_backoff(attempt_count: i32) -> TimeDelta {
79    doubling(
80        MOOCFI_REPORT_BASE_BACKOFF,
81        MOOCFI_REPORT_MAX_BACKOFF,
82        attempt_count,
83    )
84}
85
86/// The wait before the next submit after `retry_count` failed ones.
87pub fn submit_backoff(retry_count: i32) -> TimeDelta {
88    doubling(SUBMIT_BASE_BACKOFF, SUBMIT_MAX_BACKOFF, retry_count)
89}

91/// The wait until the next verify poll while the assessment item attainment is not seen yet. Timed
92/// from the submission rather than by counting polls, so a delayed row still polls when it is due.
93pub fn verify_delay(submitted_at: Option<DateTime<Utc>>, now: DateTime<Utc>) -> TimeDelta {
94    let Some(submitted_at) = submitted_at else {
95        return VERIFY_WINDOW_INTERVAL;
96    };
97    let age = now - submitted_at;
98    if age < VERIFY_WINDOW_START {
99        VERIFY_WINDOW_START - age
100    } else if age < VERIFY_DENSE_WINDOW_END {
101        VERIFY_WINDOW_INTERVAL
102    } else if age < VERIFY_WINDOW_END {
103        VERIFY_LATE_WINDOW_INTERVAL
104    } else {
105        // Grows with the time past the window, so polls thin out until the cap.
106        (age - VERIFY_WINDOW_END).clamp(VERIFY_BASE_BACKOFF, VERIFY_MAX_BACKOFF)
107    }
108}
109
110/// `lookup_count` counts the look just made, so the wait after the first one is already doubled.
111pub fn uncertain_recheck_delay(lookup_count: i32) -> TimeDelta {
112    doubling(UNCERTAIN_RECHECK, UNCERTAIN_MAX_RECHECK, lookup_count)
113}

115/// Spreads a batch that failed together, so it does not come back as one thundering herd.
116pub fn next_attempt_at(now: DateTime<Utc>, delay: TimeDelta) -> DateTime<Utc> {
117    headless_lms_utils::backoff::next_attempt_at(now, delay.num_seconds(), JITTER_MAX.num_seconds())
118}

120pub fn submit_window_expired(first_failed_at: Option<DateTime<Utc>>, now: DateTime<Utc>) -> bool {
121    window_expired(first_failed_at, now, SUBMIT_MAX_RETRY_AGE.num_seconds())
122}
123
124pub fn verify_window_expired(submitted_at: Option<DateTime<Utc>>, now: DateTime<Utc>) -> bool {
125    window_expired(submitted_at, now, VERIFY_MAX_AGE.num_seconds())
126}
127
128/// Whether an uncertain submission has waited long enough to be handed to an admin; never without
129/// a submission time.
130pub fn uncertain_needs_admin(submitted_at: Option<DateTime<Utc>>, now: DateTime<Utc>) -> bool {
131    window_expired(submitted_at, now, UNCERTAIN_ADMIN_AFTER.num_seconds())
132}

134#[cfg(test)]
135mod tests {
136    use super::*;
137
138    #[test]
139    fn backoff_doubles_and_then_stops_growing() {
140        assert_eq!(submit_backoff(0), SUBMIT_BASE_BACKOFF);
141        assert_eq!(submit_backoff(1), SUBMIT_BASE_BACKOFF * 2);
142        assert_eq!(submit_backoff(3), SUBMIT_BASE_BACKOFF * 8);
143        assert_eq!(submit_backoff(30), SUBMIT_MAX_BACKOFF);
144        assert_eq!(submit_backoff(i32::MAX), SUBMIT_MAX_BACKOFF);
145    }
146
147    #[test]
148    fn verify_polls_follow_the_landing_window_from_the_submission() {
149        let now = Utc::now();
150        let delay = |age: TimeDelta| verify_delay(Some(now - age), now);
151        assert_eq!(delay(TimeDelta::zero()), VERIFY_WINDOW_START);
152        assert_eq!(delay(TimeDelta::minutes(90)), TimeDelta::minutes(30));
153        assert_eq!(delay(TimeDelta::hours(2)), VERIFY_WINDOW_INTERVAL);
154        assert_eq!(delay(TimeDelta::hours(10)), VERIFY_WINDOW_INTERVAL);
155        assert_eq!(delay(TimeDelta::hours(11)), VERIFY_LATE_WINDOW_INTERVAL);
156        assert_eq!(delay(TimeDelta::hours(28)), VERIFY_LATE_WINDOW_INTERVAL);
157        assert_eq!(delay(TimeDelta::hours(29)), VERIFY_BASE_BACKOFF);
158        assert_eq!(delay(TimeDelta::hours(33)), VERIFY_BASE_BACKOFF * 2);
159        assert_eq!(delay(TimeDelta::days(5)), VERIFY_MAX_BACKOFF);
160        assert_eq!(verify_delay(None, now), VERIFY_WINDOW_INTERVAL);
161    }
162
163    #[test]
164    fn the_retry_window_runs_from_the_first_failure() {
165        let now = Utc::now();
166        assert!(!submit_window_expired(None, now));
167        assert!(!submit_window_expired(Some(now - TimeDelta::days(6)), now));
168        assert!(submit_window_expired(Some(now - TimeDelta::days(8)), now));
169    }
170
171    #[test]
172    fn the_verify_window_runs_from_the_submission() {
173        let now = Utc::now();
174        assert!(!verify_window_expired(None, now));
175        assert!(!verify_window_expired(Some(now - TimeDelta::days(13)), now));
176        assert!(verify_window_expired(Some(now - TimeDelta::days(15)), now));
177    }
178
179    #[test]
180    fn jitter_never_shortens_a_backoff() {
181        let now = Utc::now();
182        for _ in 0..50 {
183            let scheduled = next_attempt_at(now, TimeDelta::minutes(1));
184            assert!(scheduled - now >= TimeDelta::minutes(1));
185            assert!(scheduled - now <= TimeDelta::minutes(1) + JITTER_MAX);
186        }
187    }
188}
