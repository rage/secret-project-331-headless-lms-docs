1//! Claiming the right to mail a Sisu person an account-linking link, and minting the token it

6//! address on their account here, so the click while logged in is what creates the binding.

40/// How long after a linking mail the person is left alone, across every course and address.
41pub const LINKING_MAIL_QUIET_PERIOD: TimeDelta = TimeDelta::days(1);
42
43/// How long after a linking mail for a course the same person may get another for it.
44pub const LINKING_MAIL_RESEND_INTERVAL: TimeDelta = TimeDelta::days(7);
45
46/// How many linking mails one person may ever get for one course, tokens that expired unused
47/// included.
48pub const MAX_LINKING_MAILS_PER_PERSON_AND_COURSE: i64 = 3;

50/// No override retires a mail claimed this recently, so a second click, tab or colleague cannot
51/// replace one that has only just gone out.
52pub const JUST_CLAIMED_PERIOD: TimeDelta = TimeDelta::minutes(1);

62/// One person Sisu's `list-by-course` returned, and the address we mail them at.
63#[derive(Debug, Clone)]
64pub struct DiscoveredPerson {
65    pub sisu_person_id: DbSecret,
66    pub student_number: DbSecret,
67    pub first_names: Option<DbSecret>,
68    pub last_name: Option<DbSecret>,
69    pub course_id: Uuid,
70    /// The one address in Sisu we mail, trimmed: each address mailed would spend one of the
71    /// person's mails for the course. See [`mail_address`].
72    pub address: Option<DbSecret>,
73}

90/// The address a linking mail to `person` goes to, if Sisu holds one: the primary, else the
91/// secondary.
92///
93/// Sisu verifies neither, so the confirmation page shows the Sisu name to make a planted address
94/// harder to exploit. Open University students without a University account often have only a
95/// secondary address, and Sisu clears old primaries.
96pub fn mail_address(person: &RosterPerson) -> Option<DbSecret> {
97    listed_addresses(person).next().map(DbSecret::new)
98}

100/// The person's non-empty Sisu addresses, trimmed, primary first.
101pub fn listed_addresses(person: &RosterPerson) -> impl Iterator<Item = &str> {
102    [&person.primary_email, &person.secondary_email]
103        .into_iter()
104        .flatten()
105        .map(|address| address.expose_secret().trim())
106        .filter(|address| !address.is_empty())
107}

109/// For each of the lowercased `addresses` that is the account email of someone with a completion
110/// the push path will register (`credit_registration_eligible_completions`, prerequisites met and
111/// not held for review) on one of `course_ids` created at or after `since`, the courses it is on.
112pub async fn registerable_completion_courses_by_account_email(
113    conn: &mut PgConnection,
114    course_ids: &[Uuid],
115    since: DateTime<Utc>,
116    addresses: &[String],
117) -> ModelResult<HashMap<String, BTreeSet<Uuid>>> {
118    let rows = sqlx::query!(
119        r#"
120SELECT DISTINCT lower(user_details.email) AS "email!",
121  ec.course_id AS "course_id!"
122FROM credit_registration_eligible_completions ec
123  JOIN user_details ON user_details.user_id = ec.user_id
124WHERE ec.course_id = ANY($1::uuid [])
125  AND ec.created_at >= $2
126  AND ec.fully_eligible
127  AND lower(user_details.email) = ANY($3::text [])
128        "#,
129        course_ids,
130        since,
131        addresses,
132    )
133    .fetch_all(conn)
134    .await?;
135    let mut courses_by_email: HashMap<String, BTreeSet<Uuid>> = HashMap::new();
136    for row in rows {
137        courses_by_email
138            .entry(row.email)
139            .or_default()
140            .insert(row.course_id);
141    }
142    Ok(courses_by_email)
143}

145/// What claiming a linking mail did for one person. Ordered from weakest to strongest, so the
146/// strongest of several claims for one person is their maximum.
147#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
148pub enum LinkingMailClaim {
149    NoAddress,
150    SuppressedByRateCap,
151    /// An earlier mail to the address still has a usable link, or another writer claimed it first.
152    SuppressedByDedup,
153    /// A mail the `link-emails` phase still owes.
154    Claimed,
155}

172/// Claims one slot and one unbound token per person whose address has no usable link from us yet,
173/// dedup before the rate caps. Returns one claim per input, in order.
174pub async fn claim_linking_mails_batch(
175    conn: &mut PgConnection,
176    people: &[DiscoveredPerson],
177) -> ModelResult<Vec<LinkingMailClaim>> {
178    let mut claims = vec![LinkingMailClaim::NoAddress; people.len()];
179    let sisu_person_ids: Vec<String> = people
180        .iter()
181        .filter(|person| person.address.is_some())
182        .map(|person| person.sisu_person_id.expose_secret().to_owned())
183        .collect();
184    if sisu_person_ids.is_empty() {
185        return Ok(claims);
186    }
187    let facts = get_existing_facts_for_persons(conn, &sisu_person_ids).await?;
188    let mut by_person: HashMap<&str, Vec<&ExistingLinkingMailFact>> = HashMap::new();
189    for fact in &facts {
190        by_person
191            .entry(fact.sisu_person_id.expose_secret())
192            .or_default()
193            .push(fact);
194    }
195    let now = Utc::now();
196
197    // Tokens are minted before any slot is claimed: the random value cannot come from SQL.
198    let mut new_tokens = Vec::new();
199    let mut new_slots = Vec::new();
200    let mut token_ids = Vec::new();
201    let mut token_owner: HashMap<Uuid, usize> = HashMap::new();
202    for (i, person) in people.iter().enumerate() {
203        let Some(address) = &person.address else {
204            continue;
205        };
206        let person_facts = by_person
207            .get(person.sisu_person_id.expose_secret())
208            .map(Vec::as_slice)
209            .unwrap_or(&[]);
210        if holds_address(person_facts, person.course_id, address.expose_secret()) {
211            claims[i] = LinkingMailClaim::SuppressedByDedup;
212            continue;
213        }
214        if !is_within_caps(person, person_facts, now) {
215            claims[i] = LinkingMailClaim::SuppressedByRateCap;
216            continue;
217        }
218        let token_id = Uuid::new_v4();
219        new_tokens.push(NewStudentNumberVerificationToken {
220            student_number: person.student_number.clone(),
221            sisu_person_id: person.sisu_person_id.clone(),
222            first_names: person.first_names.clone(),
223            last_name: person.last_name.clone(),
224            emailed_to: address.clone(),
225            course_id: Some(person.course_id),
226        });
227        token_owner.insert(token_id, i);
228        token_ids.push(token_id);
229        new_slots.push(NewAccountLinkingEmail {
230            student_number: person.student_number.clone(),
231            sisu_person_id: person.sisu_person_id.clone(),
232            course_id: person.course_id,
233            emailed_to: address.clone(),
234            student_number_verification_token_id: Some(token_id),
235            email_delivery_id: None,
236        });
237    }
238    if new_slots.is_empty() {
239        return Ok(claims);
240    }
241    insert_tokens_batch(conn, &token_ids, &new_tokens).await?;
242
243    replace_lapsed_slots(conn, &new_slots).await?;
244    let claimed_token_ids = claim_send_slots(conn, &new_slots, &token_ids).await?;
245    let lost_token_ids: Vec<Uuid> = token_owner
246        .keys()
247        .filter(|id| !claimed_token_ids.contains(id))
248        .copied()
249        .collect();
250    // A refused claim must leave no usable token behind: nobody would ever send that link.
251    void_tokens(conn, &lost_token_ids).await?;
252    for (token_id, person_index) in &token_owner {
253        claims[*person_index] = if claimed_token_ids.contains(token_id) {
254            LinkingMailClaim::Claimed
255        } else {
256            // Lost the race to another writer; same outcome for the recipient as dedup.
257            LinkingMailClaim::SuppressedByDedup
258        };
259    }
260    Ok(claims)
261}

263/// Retires the linking-mail rows the caps are counting for this person, so the ordinary claim path can
264/// take a slot again. No parameter relaxes a cap: the single writer of the ledger evaluates them from
265/// the rows that exist, so getting past one means soft-deleting rows, audited as its own action.
266pub async fn retire_capped_mails(
267    conn: &mut PgConnection,
268    actor_user_id: Uuid,
269    actor_role: &str,
270    course_id: Uuid,
271    student_number: &str,
272    reason: &str,
273) -> ModelResult<RetiredMails> {
274    let mut outcome = RetiredMails {
275        retired_count: 0,
276        kept_just_claimed: false,
277    };
278    let Some(person_id) = person_id_of_mails(conn, course_id, student_number).await? else {
279        return Ok(outcome);
280    };
281    let now = Utc::now();
282    let quiet_since = now - LINKING_MAIL_QUIET_PERIOD;
283    let mails = credit_registration_account_linking_emails::get_by_sisu_person_id(
284        conn,
285        person_id.expose_secret(),
286    )
287    .await?;
288    // This course's rows carry the dedup guard and the lifetime cap; a recent row on any course
289    // carries the quiet period, which is about the person's inbox rather than one course.
290    let (just_claimed, retirable): (Vec<_>, Vec<_>) = mails
291        .iter()
292        .filter(|mail| mail.course_id == course_id || mail.sent_at >= quiet_since)
293        .partition(|mail| mail.sent_at >= now - JUST_CLAIMED_PERIOD);
294    outcome.kept_just_claimed = !just_claimed.is_empty();
295    let retired: Vec<Uuid> = retirable.iter().map(|mail| mail.id).collect();
296    if retired.is_empty() {
297        return Ok(outcome);
298    }
299
300    let mut tx = conn.begin().await?;
301    credit_registration_account_linking_emails::soft_delete_batch(&mut tx, &retired).await?;
302    crate::credit_registration_admin_actions::record(
303        &mut tx,
304        &NewCreditRegistrationAdminAction {
305            target_id: Some(course_id),
306            reason: Some(reason.to_string()),
307            details: Some(serde_json::json!({
308                "student_number": student_number,
309                "retired_linking_email_ids": retired,
310            })),
311            affected_row_count: Some(i32::try_from(retired.len()).unwrap_or(i32::MAX)),
312            ..NewCreditRegistrationAdminAction::new(
313                CreditRegistrationAdminAction::OverrideRateCap,
314                CreditRegistrationAdminActionTarget::Course,
315                actor_user_id,
316                actor_role,
317            )
318        },
319    )
320    .await?;
321    tx.commit().await?;
322    outcome.retired_count = retired.len() as i64;
323    Ok(outcome)
324}

326async fn person_id_of_mails(
327    conn: &mut PgConnection,
328    course_id: Uuid,
329    student_number: &str,

340/// Whether the caps still allow this person another mail for this course.
341fn is_within_caps(
342    person: &DiscoveredPerson,
343    facts: &[&ExistingLinkingMailFact],
344    now: DateTime<Utc>,
345) -> bool {
346    // The quiet period is about the person's inbox, so it ignores the course.
347    if facts
348        .iter()
349        .any(|fact| fact.sent_at >= now - LINKING_MAIL_QUIET_PERIOD)
350    {
351        return false;
352    }
353    let course_facts: Vec<_> = facts
354        .iter()
355        .filter(|fact| fact.course_id == person.course_id)
356        .collect();
357    !course_facts
358        .iter()
359        .any(|fact| fact.sent_at >= now - LINKING_MAIL_RESEND_INTERVAL)
360        && (course_facts.len() as i64) < MAX_LINKING_MAILS_PER_PERSON_AND_COURSE
361}

363/// Whether a mail to this (person, course, address) still holds the address. The unique index
364/// behind [`claim_send_slots`] is what actually prevents a second one.
365fn holds_address(facts: &[&ExistingLinkingMailFact], course_id: Uuid, address: &str) -> bool {
366    facts.iter().any(|fact| {
367        fact.holds_address
368            && fact.course_id == course_id
369            && fact
370                .emailed_to
371                .expose_secret()
372                .eq_ignore_ascii_case(address)
373    })
374}

376/// Soft-deletes tokens whose slot lost the race, so no unusable link is left behind.
377async fn void_tokens(conn: &mut PgConnection, ids: &[Uuid]) -> ModelResult<()> {
378    if ids.is_empty() {
379        return Ok(());
380    }
381    sqlx::query!(
382        r#"
383UPDATE student_number_verification_tokens
384SET deleted_at = now()
385WHERE id = ANY($1::uuid [])
386  AND deleted_at IS NULL
387        "#,
388        ids,
389    )
390    .execute(conn)
391    .await?;
392    Ok(())
393}
