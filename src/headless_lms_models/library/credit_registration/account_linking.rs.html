1//! Claiming the right to mail a Sisu person an account-linking link, and minting the token it
2//! carries. Both caps and the dedup guard are evaluated here and nowhere else, and no argument
3//! switches them off; an override has to soft-delete a ledger row.
4//!
5//! The token is minted bound to no account: the recipient's Sisu address is routinely not the
6//! address on their account here, so the click while logged in is what creates the binding.

40/// How long after a linking mail the person is left alone, across every course and address.
41pub const LINKING_MAIL_QUIET_PERIOD: TimeDelta = TimeDelta::days(1);
42
43/// How long after a linking mail for a course the same person may get another for it.
44pub const LINKING_MAIL_RESEND_INTERVAL: TimeDelta = TimeDelta::days(7);
45
46/// How many linking mails one person may ever get for one course, tokens that expired unused
47/// included.
48pub const MAX_LINKING_MAILS_PER_PERSON_AND_COURSE: i64 = 3;

50/// One person Sisu's `list-by-course` returned, and the address we mail them at.
51#[derive(Debug, Clone)]
52pub struct DiscoveredPerson {
53    pub sisu_person_id: DbSecret,
54    pub student_number: DbSecret,
55    pub first_names: Option<DbSecret>,
56    pub last_name: Option<DbSecret>,
57    pub course_id: Uuid,
58    /// Their primary address in Sisu, trimmed: the one students are told to check, and each address
59    /// mailed would spend one of the person's mails for the course. See [`mail_address`].
60    pub address: Option<DbSecret>,
61}
62
63impl DiscoveredPerson {
64    /// `person` as listed on a roster of `course_id`'s; the caller decides what a missing address
65    /// means.
66    pub fn listed(person: &RosterPerson, course_id: Uuid) -> Self {
67        Self {
68            sisu_person_id: person.person_id.clone().into(),
69            student_number: person.student_number.clone().into(),
70            first_names: person.first_names.clone().map(Into::into),
71            last_name: person.last_name.clone().map(Into::into),
72            course_id,
73            address: mail_address(person),
74        }
75    }
76}

78/// The address a linking mail to `person` goes to, if Sisu holds one.
79pub fn mail_address(person: &RosterPerson) -> Option<DbSecret> {
80    person
81        .primary_email
82        .as_ref()
83        .map(|address| address.expose_secret().trim())
84        .filter(|address| !address.is_empty())
85        .map(DbSecret::new)
86}

88/// What claiming a linking mail did for one person. Ordered from weakest to strongest, so the
89/// strongest of several claims for one person is their maximum.
90#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
91pub enum LinkingMailClaim {
92    NoAddress,
93    SuppressedByRateCap,
94    /// An earlier mail to the address still has a usable link, or another writer claimed it first.
95    SuppressedByDedup,
96    /// A mail the `link-emails` phase still owes.
97    Claimed,
98}

100/// One person's [`claim_linking_mails_batch`].
101pub async fn claim_linking_mail(
102    conn: &mut PgConnection,
103    person: &DiscoveredPerson,
104) -> ModelResult<LinkingMailClaim> {
105    claim_linking_mails_batch(conn, std::slice::from_ref(person))
106        .await?
107        .into_iter()
108        .next()
109        .ok_or_else(missing_model_error(
110            ModelErrorType::Generic,
111            "Claiming linking mails answered nothing for the one person asked about.",
112        ))
113}

115/// Claims one slot and one unbound token per person whose address has no usable link from us yet,
116/// dedup before the rate caps. Returns one claim per input, in order.
117pub async fn claim_linking_mails_batch(
118    conn: &mut PgConnection,
119    people: &[DiscoveredPerson],
120) -> ModelResult<Vec<LinkingMailClaim>> {
121    let mut claims = vec![LinkingMailClaim::NoAddress; people.len()];
122    let sisu_person_ids: Vec<String> = people
123        .iter()
124        .filter(|person| person.address.is_some())
125        .map(|person| person.sisu_person_id.expose_secret().to_owned())
126        .collect();
127    if sisu_person_ids.is_empty() {
128        return Ok(claims);
129    }
130    let facts = get_existing_facts_for_persons(conn, &sisu_person_ids).await?;
131    let mut by_person: HashMap<&str, Vec<&ExistingLinkingMailFact>> = HashMap::new();
132    for fact in &facts {
133        by_person
134            .entry(fact.sisu_person_id.expose_secret())
135            .or_default()
136            .push(fact);
137    }
138    let now = Utc::now();
139
140    // Tokens are minted before any slot is claimed: the random value cannot come from SQL.
141    let mut new_tokens = Vec::new();
142    let mut new_slots = Vec::new();
143    let mut token_ids = Vec::new();
144    let mut token_owner: HashMap<Uuid, usize> = HashMap::new();
145    for (i, person) in people.iter().enumerate() {
146        let Some(address) = &person.address else {
147            continue;
148        };
149        let person_facts = by_person
150            .get(person.sisu_person_id.expose_secret())
151            .map(Vec::as_slice)
152            .unwrap_or(&[]);
153        if holds_address(person_facts, person.course_id, address.expose_secret()) {
154            claims[i] = LinkingMailClaim::SuppressedByDedup;
155            continue;
156        }
157        if !is_within_caps(person, person_facts, now) {
158            claims[i] = LinkingMailClaim::SuppressedByRateCap;
159            continue;
160        }
161        let token_id = Uuid::new_v4();
162        new_tokens.push(NewStudentNumberVerificationToken {
163            student_number: person.student_number.clone(),
164            sisu_person_id: person.sisu_person_id.clone(),
165            first_names: person.first_names.clone(),
166            last_name: person.last_name.clone(),
167            emailed_to: address.clone(),
168            course_id: Some(person.course_id),
169        });
170        token_owner.insert(token_id, i);
171        token_ids.push(token_id);
172        new_slots.push(NewAccountLinkingEmail {
173            student_number: person.student_number.clone(),
174            sisu_person_id: person.sisu_person_id.clone(),
175            course_id: person.course_id,
176            emailed_to: address.clone(),
177            student_number_verification_token_id: Some(token_id),
178            email_delivery_id: None,
179        });
180    }
181    if new_slots.is_empty() {
182        return Ok(claims);
183    }
184    insert_tokens_batch(conn, &token_ids, &new_tokens).await?;
185
186    replace_lapsed_slots(conn, &new_slots).await?;
187    let claimed_token_ids = claim_send_slots(conn, &new_slots, &token_ids).await?;
188    let lost_token_ids: Vec<Uuid> = token_owner
189        .keys()
190        .filter(|id| !claimed_token_ids.contains(id))
191        .copied()
192        .collect();
193    // A refused claim must leave no usable token behind: nobody would ever send that link.
194    void_tokens(conn, &lost_token_ids).await?;
195    for (token_id, person_index) in &token_owner {
196        claims[*person_index] = if claimed_token_ids.contains(token_id) {
197            LinkingMailClaim::Claimed
198        } else {
199            // Lost the race to another writer; same outcome for the recipient as dedup.
200            LinkingMailClaim::SuppressedByDedup
201        };
202    }
203    Ok(claims)
204}

206/// Retires the linking-mail rows the caps are counting for this person, so the ordinary claim path can
207/// take a slot again. No parameter relaxes a cap: the single writer of the ledger evaluates them from
208/// the rows that exist, so getting past one means soft-deleting rows, audited as its own action.
209pub async fn retire_capped_mails(
210    conn: &mut PgConnection,
211    actor_user_id: Uuid,
212    actor_role: &str,
213    course_id: Uuid,
214    student_number: &str,
215    reason: &str,
216) -> ModelResult<i64> {
217    let Some(person_id) = person_id_of_mails(conn, course_id, student_number).await? else {
218        return Ok(0);
219    };
220    let quiet_since = Utc::now() - LINKING_MAIL_QUIET_PERIOD;
221    let mails = credit_registration_account_linking_emails::get_by_sisu_person_id(
222        conn,
223        person_id.expose_secret(),
224    )
225    .await?;
226    // This course's rows carry the dedup guard and the lifetime cap; a recent row on any course
227    // carries the quiet period, which is about the person's inbox rather than one course.
228    let retired: Vec<Uuid> = mails
229        .iter()
230        .filter(|mail| mail.course_id == course_id || mail.sent_at >= quiet_since)
231        .map(|mail| mail.id)
232        .collect();
233    if retired.is_empty() {
234        return Ok(0);
235    }
236
237    let mut tx = conn.begin().await?;
238    credit_registration_account_linking_emails::soft_delete_batch(&mut tx, &retired).await?;
239    crate::credit_registration_admin_actions::record(
240        &mut tx,
241        &NewCreditRegistrationAdminAction {
242            target_id: Some(course_id),
243            reason: Some(reason.to_string()),
244            details: Some(serde_json::json!({
245                "student_number": student_number,
246                "retired_linking_email_ids": retired,
247            })),
248            affected_row_count: Some(i32::try_from(retired.len()).unwrap_or(i32::MAX)),
249            ..NewCreditRegistrationAdminAction::new(
250                CreditRegistrationAdminAction::OverrideRateCap,
251                CreditRegistrationAdminActionTarget::Course,
252                actor_user_id,
253                actor_role,
254            )
255        },
256    )
257    .await?;
258    tx.commit().await?;
259    Ok(retired.len() as i64)
260}

262async fn person_id_of_mails(
263    conn: &mut PgConnection,
264    course_id: Uuid,
265    student_number: &str,
266) -> ModelResult<Option<DbSecret>> {
267    let mails = credit_registration_account_linking_emails::get_by_course_id_and_student_number(
268        conn,
269        course_id,
270        student_number,
271    )
272    .await?;
273    Ok(mails.into_iter().next().map(|mail| mail.sisu_person_id))
274}

276/// Whether the caps still allow this person another mail for this course.
277fn is_within_caps(
278    person: &DiscoveredPerson,
279    facts: &[&ExistingLinkingMailFact],
280    now: DateTime<Utc>,
281) -> bool {
282    // The quiet period is about the person's inbox, so it ignores the course.
283    if facts
284        .iter()
285        .any(|fact| fact.sent_at >= now - LINKING_MAIL_QUIET_PERIOD)
286    {
287        return false;
288    }
289    let course_facts: Vec<_> = facts
290        .iter()
291        .filter(|fact| fact.course_id == person.course_id)
292        .collect();
293    !course_facts
294        .iter()
295        .any(|fact| fact.sent_at >= now - LINKING_MAIL_RESEND_INTERVAL)
296        && (course_facts.len() as i64) < MAX_LINKING_MAILS_PER_PERSON_AND_COURSE
297}

299/// Whether a mail to this (person, course, address) still holds the address. The unique index
300/// behind [`claim_send_slots`] is what actually prevents a second one.
301fn holds_address(facts: &[&ExistingLinkingMailFact], course_id: Uuid, address: &str) -> bool {
302    facts.iter().any(|fact| {
303        fact.holds_address
304            && fact.course_id == course_id
305            && fact
306                .emailed_to
307                .expose_secret()
308                .eq_ignore_ascii_case(address)
309    })
310}

312/// Soft-deletes tokens whose slot lost the race, so no unusable link is left behind.
313async fn void_tokens(conn: &mut PgConnection, ids: &[Uuid]) -> ModelResult<()> {
314    if ids.is_empty() {
315        return Ok(());
316    }
317    sqlx::query!(
318        r#"
319UPDATE student_number_verification_tokens
320SET deleted_at = now()
321WHERE id = ANY($1::uuid [])
322  AND deleted_at IS NULL
323        "#,
324        ids,
325    )
326    .execute(conn)
327    .await?;
328    Ok(())
329}
