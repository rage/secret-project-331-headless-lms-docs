21/// What a join or split returned, with the course language groups whose courses changed.
22pub struct MembershipChange<T> {
23    pub result: T,
24    pub changed_course_language_group_ids: [Uuid; 2],
25}

27/// A student with enrolment settings both in the moved course and in the target language
28/// versions. Only one settings row per student and group can exist, so the one with the later
29/// activity is kept and the other removed (and logged); points stay on both courses.
30#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
31pub struct EnrolmentConflict {
32    pub user_id: Uuid,
33    pub kept_course_id: Uuid,
34    pub removed_course_id: Uuid,
35    /// Later of the settings' last update and the student's latest exercise activity there.
36    pub kept_activity_at: DateTime<Utc>,
37    pub removed_activity_at: DateTime<Utc>,
38    pub keeps_moved_course: bool,
39}

41/// What joining `joined_course_id` into the group would change. Changes nothing.
42///
43/// `can_edit` of the returned courses is false; the caller fills it in. Fails with `NotFound` if
44/// the course is deleted or the group has no live courses.
45pub async fn join_preview(
46    conn: &mut PgConnection,
47    course_language_group_id: Uuid,
48    joined_course_id: Uuid,
49) -> ModelResult<JoinLanguageVersionPreview> {
50    let course = get_course(&mut *conn, joined_course_id).await?;
51    let language_versions = get_group_courses(&mut *conn, course_language_group_id).await?;
52    let blocked_reason = join_blocked_reason(&course, &language_versions, course_language_group_id);
53    let left_language_versions = if course.course_language_group_id == course_language_group_id {
54        vec![]
55    } else {
56        get_group_courses(&mut *conn, course.course_language_group_id)
57            .await?
58            .into_iter()
59            .filter(|c| c.id != joined_course_id)
60            .collect()
61    };
62    let plan = plan_move(
63        &mut *conn,
64        joined_course_id,
65        course.course_language_group_id,
66        Some(course_language_group_id),
67    )
68    .await?;
69    Ok(JoinLanguageVersionPreview {
70        course,
71        language_versions,
72        left_language_versions,
73        unlinked_page_count: plan.pages.unlinked_count(),
74        unlinked_exercise_count: plan.exercises.unlinked_count(),
75        blocked_reason,
76        expected_student_count: plan.live_enrolment_count,
77        moved_data: plan.moved_data(),
78        students_in_both_count: plan.enrolment_conflicts.len() as i64,
79    })
80}

82/// Moves `request.course_id` from its own group into this one with its students' per-group data,
83/// in one transaction, and logs the change. Its links to courses that stay in the old group are
84/// removed; the old group is soft-deleted when no live course is left in it.
85///
86/// Fails with `InvalidRequest` when the course already is a language version or the group
87/// already has a course in its language, and with `StaleVersion` when the course's student count
88/// is no longer `request.expected_student_count` or the course changed groups since the preview.
89pub async fn join_course(
90    conn: &mut PgConnection,
91    course_language_group_id: Uuid,
92    actor_user_id: Uuid,
93    request: &JoinLanguageVersionRequest,
94) -> ModelResult<MembershipChange<JoinLanguageVersionResult>> {
95    let mut tx = conn.begin().await?;
96    let from_group_id = get_course(&mut tx, request.course_id)
97        .await?
98        .course_language_group_id;
99    lock_groups(&mut tx, &[from_group_id, course_language_group_id]).await?;
100    let course = get_course(&mut tx, request.course_id).await?;
101    if course.course_language_group_id != from_group_id {
102        return Err(stale_error());
103    }
104    let language_versions = get_group_courses(&mut tx, course_language_group_id).await?;
105    if let Some(reason) = join_blocked_reason(&course, &language_versions, course_language_group_id)
106    {
107        return Err(invalid_request(match reason {
108            JoinBlockedReason::AlreadyLanguageVersion => {
109                "The course already is one of the language versions."
110            }
111            JoinBlockedReason::LanguageAlreadyPresent => {
112                "The language versions already have a course in this language."
113            }
114        }));
115    }
116    if course.student_count != Some(request.expected_student_count) {
117        return Err(stale_error());
118    }
119
120    let plan = plan_move(
121        &mut tx,
122        request.course_id,
123        from_group_id,
124        Some(course_language_group_id),
125    )
126    .await?;
127    let log_payload = apply_move(&mut tx, &plan, course_language_group_id).await?;
128    operation_logs::insert(
129        &mut tx,
130        actor_user_id,
131        LanguageVersionOperation::Join,
132        course_language_group_id,
133        log_payload,
134    )
135    .await?;
136    tx.commit().await?;
137    Ok(MembershipChange {
138        result: JoinLanguageVersionResult {
139            course_language_group_id,
140            moved_data: plan.moved_data(),
141            removed_enrolment_settings_count: plan.enrolment_conflicts.len() as i64,
142        },
143        changed_course_language_group_ids: [course_language_group_id, from_group_id],
144    })
145}

147/// What splitting `split_course_id` off the group would change. Changes nothing.
148///
149/// `can_edit` of the returned courses is false; the caller fills it in. Fails with
150/// `InvalidRequest` when the course is not a live course of the group or is its last one.
151pub async fn split_preview(
152    conn: &mut PgConnection,
153    course_language_group_id: Uuid,
154    split_course_id: Uuid,
155) -> ModelResult<SplitLanguageVersionPreview> {
156    let courses = get_group_courses(&mut *conn, course_language_group_id).await?;
157    let (course, remaining_courses) = split_off(courses, split_course_id)?;
158    let plan = plan_move(&mut *conn, split_course_id, course_language_group_id, None).await?;
159    Ok(SplitLanguageVersionPreview {
160        course,
161        remaining_courses,
162        expected_student_count: plan.live_enrolment_count,
163        moved_data: plan.moved_data(),
164        unlinked_page_count: plan.pages.unlinked_count(),
165        unlinked_exercise_count: plan.exercises.unlinked_count(),
166    })
167}

169/// Moves `request.course_id` out of the group into a new group of its own with its students'
170/// per-group data, unlinking its pages and exercises from the other versions, in one
171/// transaction, and logs the change. Students of the other versions stay; group-wide data (AI
172/// notice acknowledgements, mailing list) is copied.
173///
174/// Fails with `InvalidRequest` when the course is not a live course of the group or is its last
175/// one, and with `StaleVersion` when its student count is no longer
176/// `request.expected_student_count`.
177pub async fn split_course(
178    conn: &mut PgConnection,
179    course_language_group_id: Uuid,
180    actor_user_id: Uuid,
181    request: &SplitLanguageVersionRequest,
182) -> ModelResult<MembershipChange<SplitLanguageVersionResult>> {
183    let mut tx = conn.begin().await?;
184    lock_groups(&mut tx, &[course_language_group_id]).await?;
185    let courses = get_group_courses(&mut tx, course_language_group_id).await?;
186    let (course, _) = split_off(courses, request.course_id)?;
187    if course.student_count != Some(request.expected_student_count) {
188        return Err(stale_error());
189    }
190
191    let plan = plan_move(&mut tx, request.course_id, course_language_group_id, None).await?;
192    let slug = unique_group_slug(&mut tx, &course.slug).await?;
193    let new_group_id =
194        crate::course_language_groups::insert(&mut tx, PKeyPolicy::Generate, &slug).await?;
195    let log_payload = apply_move(&mut tx, &plan, new_group_id).await?;
196    operation_logs::insert(
197        &mut tx,
198        actor_user_id,
199        LanguageVersionOperation::Split,
200        course_language_group_id,
201        log_payload,
202    )
203    .await?;
204    tx.commit().await?;
205    Ok(MembershipChange {
206        result: SplitLanguageVersionResult {
207            course_language_group_id: new_group_id,
208            unlinked_page_count: plan.pages.unlinked_count(),
209            unlinked_exercise_count: plan.exercises.unlinked_count(),
210        },
211        changed_course_language_group_ids: [course_language_group_id, new_group_id],
212    })
213}

222fn join_blocked_reason(
223    course: &LanguageVersionCourse,
224    language_versions: &[LanguageVersionCourse],
225    course_language_group_id: Uuid,
226) -> Option<JoinBlockedReason> {
227    if course.course_language_group_id == course_language_group_id {
228        Some(JoinBlockedReason::AlreadyLanguageVersion)
229    } else if language_versions
230        .iter()
231        .any(|c| is_same_language(&c.language_code, &course.language_code))
232    {
233        Some(JoinBlockedReason::LanguageAlreadyPresent)
234    } else {
235        None
236    }
237}

239/// Splits `courses` into the course with the id and the rest.
240fn split_off(
241    courses: Vec<LanguageVersionCourse>,
242    course_id: Uuid,
243) -> ModelResult<(LanguageVersionCourse, Vec<LanguageVersionCourse>)> {
244    let (course, remaining): (Vec<_>, Vec<_>) =
245        courses.into_iter().partition(|c| c.id == course_id);
246    let course = course
247        .into_iter()
248        .next()
249        .ok_or_else(|| invalid_request("The course is not one of the language versions."))?;
250    if remaining.is_empty() {
251        return Err(invalid_request("The course is the only language version."));
252    }
253    Ok((course, remaining))
254}

300/// Locks the groups and their courses, in id order. Locking the courses also blocks new enrolment
301/// settings for them until the transaction ends (their foreign key check needs KEY SHARE).
302async fn lock_groups(conn: &mut PgConnection, group_ids: &[Uuid]) -> ModelResult<()> {
303    sqlx::query!(
304        "
305SELECT id
306FROM course_language_groups
307WHERE id = ANY($1)
308ORDER BY id
309FOR UPDATE
310        ",
311        group_ids
312    )
313    .fetch_all(&mut *conn)
314    .await?;
315    sqlx::query!(
316        "
317SELECT id
318FROM courses
319WHERE course_language_group_id = ANY($1)
320ORDER BY id
321FOR UPDATE
322        ",
323        group_ids
324    )
325    .fetch_all(conn)
326    .await?;
327    Ok(())
328}

330/// The base slug, or the first free `<base>-<n>`, among live course language groups.
331async fn unique_group_slug(conn: &mut PgConnection, base: &str) -> ModelResult<String> {
332    let taken: HashSet<String> = sqlx::query_scalar!(
333        "
334SELECT slug
335FROM course_language_groups
336WHERE starts_with(slug, $1)
337  AND deleted_at IS NULL
338        ",
339        base
340    )
341    .fetch_all(conn)

343    .into_iter()
344    .collect();
345    Ok(first_free_suffixed(base, |slug| taken.contains(slug)))

348/// Everything a move of one course between groups changes, worked out before changing anything
349/// so that previews and the operation report the same numbers.
350struct MovePlan {
351    course_id: Uuid,
352    from_group_id: Uuid,
353    /// No live course is left in the old group: group-wide rows move instead of being copied.
354    is_from_group_emptied: bool,
355    live_enrolment_count: i64,
356    hidden_enrolment_count: i64,
357    completion_count: i64,
358    enrolment_conflicts: Vec<EnrolmentConflict>,
359    moved_consents: Vec<PlannedRow>,
360    /// Losing consents of students with consents in both groups, from either group.
361    removed_consent_ids: Vec<Uuid>,
362    carried_ack_ids: Vec<Uuid>,
363    carried_token_ids: Vec<Uuid>,
364    /// Old-group tokens dropped because the target already has a mailing list.
365    dropped_token_ids: Vec<Uuid>,
366    carried_tag_ids: Vec<Uuid>,
367    pages: LinkPlan,
368    exercises: LinkPlan,
369}
370
371struct PlannedRow {
372    id: Uuid,
373    is_live: bool,
374}

376/// How the moved course's pages or exercises leave their old language groups.
377struct LinkPlan {
378    /// Groups holding no live items outside the target: they move to the target group.
379    moved_groups: Vec<MovedGroup>,
380    /// Items linked to live items that stay behind: each gets a new group of its own.
381    unlinked: Vec<UnlinkedItem>,
382}
383
384#[derive(Serialize)]
385struct MovedGroup {
386    id: Uuid,
387    previous_course_language_group_id: Uuid,
388}
389
390struct UnlinkedItem {
391    id: Uuid,
392    previous_group_id: Uuid,
393    is_live: bool,
394}

396impl LinkPlan {
397    fn unlinked_count(&self) -> i64 {
398        self.unlinked.iter().filter(|i| i.is_live).count() as i64
399    }
400}

402impl MovePlan {
403    /// Every kind, zero counts included.
404    fn moved_data(&self) -> Vec<StudentDataCount> {
405        let count = |kind, count: i64| StudentDataCount { kind, count };
406        vec![
407            count(
408                StudentDataKind::EnrolmentSettings,
409                self.live_enrolment_count,
410            ),
411            count(
412                StudentDataKind::HiddenFromMyCourses,
413                self.hidden_enrolment_count,
414            ),
415            count(StudentDataKind::Completions, self.completion_count),
416            count(
417                StudentDataKind::MarketingConsents,
418                self.moved_consents.iter().filter(|c| c.is_live).count() as i64,
419            ),
420            count(
421                StudentDataKind::AiNoticeAcknowledgements,
422                self.carried_ack_ids.len() as i64,
423            ),
424            count(
425                StudentDataKind::MailingListAccessTokens,
426                self.carried_token_ids.len() as i64,
427            ),
428            count(
429                StudentDataKind::MailchimpTags,
430                self.carried_tag_ids.len() as i64,
431            ),
432        ]
433    }
434}

436/// `to_group_id` `None` stands for a new, empty group (split).
437async fn plan_move(
438    conn: &mut PgConnection,
439    course_id: Uuid,
440    from_group_id: Uuid,
441    to_group_id: Option<Uuid>,
442) -> ModelResult<MovePlan> {
443    // The nil id matches no rows, which is what a group that doesn't exist yet holds.
444    let to_id = to_group_id.unwrap_or_else(Uuid::nil);
445    let is_from_group_emptied = !get_group_course_ids(&mut *conn, from_group_id)
446        .await?
447        .iter()
448        .any(|id| *id != course_id);
449    let mut target_course_ids = get_group_course_ids(&mut *conn, to_id).await?;
450    target_course_ids.push(course_id);
451
452    let enrolments = sqlx::query!(
453        r#"
454SELECT user_id,
455  (deleted_at IS NULL) AS "is_live!",
456  hidden
457FROM user_course_settings
458WHERE current_course_id = $1
459        "#,
460        course_id
461    )
462    .fetch_all(&mut *conn)
463    .await?;
464    let moved_user_ids: Vec<Uuid> = enrolments.iter().map(|e| e.user_id).collect();
465    let live_enrolment_count = enrolments.iter().filter(|e| e.is_live).count() as i64;
466    let hidden_enrolment_count = enrolments.iter().filter(|e| e.is_live && e.hidden).count() as i64;
467    let completion_count = sqlx::query_scalar!(
468        r#"
469SELECT COUNT(*) AS "count!"
470FROM course_module_completions
471WHERE course_id = $1
472  AND deleted_at IS NULL
473        "#,
474        course_id
475    )
476    .fetch_one(&mut *conn)
477    .await?;
478
479    let enrolment_conflicts = get_enrolment_conflicts(&mut *conn, course_id, to_group_id).await?;
480    let users_keeping_moved_course: HashSet<Uuid> = {
481        let losing: HashSet<Uuid> = enrolment_conflicts
482            .iter()
483            .filter(|c| !c.keeps_moved_course)
484            .map(|c| c.user_id)
485            .collect();
486        moved_user_ids
487            .iter()
488            .copied()
489            .filter(|id| !losing.contains(id))
490            .collect()
491    };
492    let (moved_consents, removed_consent_ids) = plan_consents(
493        &mut *conn,
494        course_id,
495        from_group_id,
496        to_id,
497        is_from_group_emptied,
498        &moved_user_ids,
499        &users_keeping_moved_course,
500    )
501    .await?;
502
503    let carried_ack_ids = sqlx::query_scalar!(
504        "
505SELECT a.id
506FROM user_ai_usage_notice_acknowledgements a
507WHERE a.course_language_group_id = $1
508  AND a.deleted_at IS NULL
509  AND (
510    $3
511    OR a.user_id = ANY($4)
512  )
513  AND NOT EXISTS (
514    SELECT 1
515    FROM user_ai_usage_notice_acknowledgements t
516    WHERE t.user_id = a.user_id
517      AND t.course_language_group_id = $2
518      AND t.deleted_at IS NULL
519  )
520        ",
521        from_group_id,
522        to_id,
523        is_from_group_emptied,
524        &moved_user_ids
525    )
526    .fetch_all(&mut *conn)
527    .await?;
528
529    let from_token_ids = sqlx::query_scalar!(
530        "
531SELECT id
532FROM marketing_mailing_list_access_tokens
533WHERE course_language_group_id = $1
534  AND deleted_at IS NULL
535        ",
536        from_group_id
537    )
538    .fetch_all(&mut *conn)
539    .await?;
540    let target_has_token = sqlx::query_scalar!(
541        r#"
542SELECT EXISTS (
543    SELECT 1
544    FROM marketing_mailing_list_access_tokens
545    WHERE course_language_group_id = $1
546      AND deleted_at IS NULL
547  ) AS "exists!"
548        "#,
549        to_id
550    )
551    .fetch_one(&mut *conn)
552    .await?;
553    // A surviving old group keeps using its own tokens, so only an emptied group drops any.
554    let (carried_token_ids, dropped_token_ids) = match (target_has_token, is_from_group_emptied) {
555        (false, _) => (from_token_ids, vec![]),
556        (true, true) => (vec![], from_token_ids),
557        (true, false) => (vec![], vec![]),
558    };
559    let carried_tag_ids = sqlx::query_scalar!(
560        "
561SELECT g.id
562FROM mailchimp_course_tags g
563WHERE g.marketing_mailing_list_access_token_id = ANY($1)
564  AND g.course_language_group_id = $2
565  AND g.deleted_at IS NULL
566  AND NOT EXISTS (
567    SELECT 1
568    FROM mailchimp_course_tags t
569    WHERE t.course_language_group_id = $3
570      AND t.tag_name = g.tag_name
571  )
572        ",
573        &carried_token_ids,
574        from_group_id,
575        to_id
576    )
577    .fetch_all(&mut *conn)
578    .await?;
579
580    let pages = plan_links(
581        &mut *conn,
582        LinkedItemKind::Page,
583        course_id,
584        to_id,
585        &target_course_ids,
586    )
587    .await?;
588    let exercises = plan_links(
589        &mut *conn,
590        LinkedItemKind::Exercise,
591        course_id,
592        to_id,
593        &target_course_ids,
594    )
595    .await?;
596
597    Ok(MovePlan {
598        course_id,
599        from_group_id,
600        is_from_group_emptied,
601        live_enrolment_count,
602        hidden_enrolment_count,
603        completion_count,
604        enrolment_conflicts,
605        moved_consents,
606        removed_consent_ids,
607        carried_ack_ids,
608        carried_token_ids,
609        dropped_token_ids,
610        carried_tag_ids,
611        pages,
612        exercises,
613    })
614}

616/// Conflicts between the course's enrolment settings, deleted ones included, and those in the
617/// target group (the primary key covers deleted rows too). Live settings beat deleted ones; then
618/// the later activity wins, the target's on a tie.
619async fn get_enrolment_conflicts(
620    conn: &mut PgConnection,
621    course_id: Uuid,
622    to_group_id: Option<Uuid>,
623) -> ModelResult<Vec<EnrolmentConflict>> {
624    let Some(to_group_id) = to_group_id else {
625        return Ok(vec![]);
626    };
627    let rows = sqlx::query!(
628        r#"
629SELECT m.user_id,
630  t.current_course_id AS target_course_id,
631  (m.deleted_at IS NULL) AS "moved_is_live!",
632  (t.deleted_at IS NULL) AS "target_is_live!",
633  GREATEST(
634    m.updated_at,
635    (
636      SELECT MAX(s.updated_at)
637      FROM user_exercise_states s
638      WHERE s.user_id = m.user_id
639        AND s.course_id = m.current_course_id
640        AND s.deleted_at IS NULL
641    )
642  ) AS "moved_activity_at!",
643  GREATEST(
644    t.updated_at,
645    (
646      SELECT MAX(s.updated_at)
647      FROM user_exercise_states s
648      WHERE s.user_id = t.user_id
649        AND s.course_id = t.current_course_id
650        AND s.deleted_at IS NULL
651    )
652  ) AS "target_activity_at!"
653FROM user_course_settings m
654  JOIN user_course_settings t ON t.user_id = m.user_id
655  AND t.course_language_group_id = $2
656WHERE m.current_course_id = $1
657  AND t.current_course_id <> $1
658ORDER BY m.user_id
659        "#,
660        course_id,
661        to_group_id
662    )
663    .fetch_all(conn)
664    .await?;
665    Ok(rows
666        .into_iter()
667        .map(|r| {
668            let keeps_moved_course = if r.moved_is_live != r.target_is_live {
669                r.moved_is_live
670            } else {
671                r.moved_activity_at > r.target_activity_at
672            };
673            let moved = (course_id, r.moved_activity_at);
674            let target = (r.target_course_id, r.target_activity_at);
675            let (kept, removed) = if keeps_moved_course {
676                (moved, target)
677            } else {
678                (target, moved)
679            };
680            EnrolmentConflict {
681                user_id: r.user_id,
682                kept_course_id: kept.0,
683                removed_course_id: removed.0,
684                kept_activity_at: kept.1,
685                removed_activity_at: removed.1,
686                keeps_moved_course,
687            }
688        })
689        .collect())
690}

692/// Consents follow their students' enrolment settings; a consent of a student without settings in
693/// the old group follows the course it was given in. One consent per student and group can exist
694/// (deleted ones included), so for students with one in both groups the consent of the group
695/// whose settings the student keeps wins, otherwise the live and then the newer one.
696async fn plan_consents(
697    conn: &mut PgConnection,
698    course_id: Uuid,
699    from_group_id: Uuid,
700    to_id: Uuid,
701    is_from_group_emptied: bool,
702    moved_user_ids: &[Uuid],
703    users_keeping_moved_course: &HashSet<Uuid>,
704) -> ModelResult<(Vec<PlannedRow>, Vec<Uuid>)> {
705    let candidates = sqlx::query!(
706        r#"
707SELECT umc.id,
708  umc.user_id,
709  (umc.deleted_at IS NULL) AS "is_live!",
710  umc.updated_at
711FROM user_marketing_consents umc
712WHERE umc.course_language_group_id = $1
713  AND (
714    $3
715    OR umc.user_id = ANY($4)
716    OR (
717      umc.course_id = $2
718      AND NOT EXISTS (
719        SELECT 1
720        FROM user_course_settings s
721        WHERE s.user_id = umc.user_id
722          AND s.course_language_group_id = $1
723      )
724    )
725  )
726        "#,
727        from_group_id,
728        course_id,
729        is_from_group_emptied,
730        moved_user_ids
731    )
732    .fetch_all(&mut *conn)
733    .await?;
734    let candidate_user_ids: Vec<Uuid> = candidates.iter().map(|c| c.user_id).collect();
735    let target_consents: HashMap<Uuid, _> = sqlx::query!(
736        r#"
737SELECT id,
738  user_id,
739  (deleted_at IS NULL) AS "is_live!",
740  updated_at
741FROM user_marketing_consents
742WHERE course_language_group_id = $1
743  AND user_id = ANY($2)
744        "#,
745        to_id,
746        &candidate_user_ids
747    )
748    .fetch_all(&mut *conn)
749    .await?
750    .into_iter()
751    .map(|r| (r.user_id, r))
752    .collect();
753    let users_enrolled_in_target: HashSet<Uuid> = sqlx::query_scalar!(
754        "
755SELECT user_id
756FROM user_course_settings
757WHERE course_language_group_id = $1
758  AND user_id = ANY($2)
759        ",
760        to_id,
761        &candidate_user_ids
762    )
763    .fetch_all(&mut *conn)
764    .await?
765    .into_iter()
766    .collect();
767
768    let mut moved = vec![];
769    let mut removed = vec![];
770    for candidate in candidates {
771        let Some(target) = target_consents.get(&candidate.user_id) else {
772            moved.push(PlannedRow {
773                id: candidate.id,
774                is_live: candidate.is_live,
775            });
776            continue;
777        };
778        let keeps_moved = if users_keeping_moved_course.contains(&candidate.user_id) {
779            true
780        } else if users_enrolled_in_target.contains(&candidate.user_id) {
781            false
782        } else if candidate.is_live != target.is_live {
783            candidate.is_live
784        } else {
785            candidate.updated_at > target.updated_at
786        };
787        if keeps_moved {
788            removed.push(target.id);
789            moved.push(PlannedRow {
790                id: candidate.id,
791                is_live: candidate.is_live,
792            });
793        } else {
794            removed.push(candidate.id);
795        }
796    }
797    Ok((moved, removed))
798}

800/// `target_course_ids`: the moved course and the target group's live courses.

816  JOIN pages p ON p.page_language_group_id = g.id

821    FROM pages o

826      AND o.course_id <> ALL($3)

828                ",
829                course_id,
830                to_id,
831                target_course_ids
832            )
833            .fetch_all(&mut *conn)
834            .await?;
835            let unlinked = sqlx::query_as!(
836                UnlinkedItem,

840  (p.deleted_at IS NULL) AS "is_live!"

874    WHERE o.exercise_language_group_id = g.id
875      AND o.deleted_at IS NULL
876      AND oc.deleted_at IS NULL
877      AND o.course_id <> ALL($3)

879                ",
880                course_id,
881                to_id,
882                target_course_ids
883            )
884            .fetch_all(&mut *conn)
885            .await?;
886            let unlinked = sqlx::query_as!(
887                UnlinkedItem,
888                r#"
889SELECT e.id,
890  e.exercise_language_group_id AS "previous_group_id!",
891  (e.deleted_at IS NULL) AS "is_live!"
892FROM exercises e
893WHERE e.course_id = $1
894  AND EXISTS (
895    SELECT 1
896    FROM exercises o
897      JOIN courses oc ON oc.id = o.course_id
898    WHERE o.exercise_language_group_id = e.exercise_language_group_id
899      AND o.deleted_at IS NULL
900      AND oc.deleted_at IS NULL
901      AND o.course_id <> ALL($2)
902  )
903                "#,
904                course_id,
905                target_course_ids
906            )
907            .fetch_all(conn)
908            .await?;
909            (moved_groups, unlinked)
910        }
911    };
912    Ok(LinkPlan {
913        moved_groups,

915    })

918/// Carries out the plan and returns the operation log payload. The caller holds the locks of
919/// [lock_groups] on both groups and has created `to_group_id` if it is new.
920async fn apply_move(
921    conn: &mut PgConnection,
922    plan: &MovePlan,
923    to_group_id: Uuid,
924) -> ModelResult<serde_json::Value> {
925    // The course and its students' settings change groups in separate statements.
926    sqlx::query!("SET CONSTRAINTS user_course_settings_current_course_group_fkey DEFERRED")
927        .execute(&mut *conn)
928        .await?;
929
930    let mut removed_enrolment_settings = vec![];
931    let removed_target_user_ids: Vec<Uuid> = plan
932        .enrolment_conflicts
933        .iter()
934        .filter(|c| c.keeps_moved_course)
935        .map(|c| c.user_id)
936        .collect();
937    let removed_moved_user_ids: Vec<Uuid> = plan
938        .enrolment_conflicts
939        .iter()
940        .filter(|c| !c.keeps_moved_course)
941        .map(|c| c.user_id)
942        .collect();
943    removed_enrolment_settings.extend(
944        sqlx::query_scalar!(
945            r#"
946DELETE FROM user_course_settings
947WHERE course_language_group_id = $1
948  AND user_id = ANY($2)
949RETURNING to_jsonb(user_course_settings.*) AS "row!"
950            "#,
951            to_group_id,
952            &removed_target_user_ids
953        )
954        .fetch_all(&mut *conn)
955        .await?,
956    );
957    removed_enrolment_settings.extend(
958        sqlx::query_scalar!(
959            r#"
960DELETE FROM user_course_settings
961WHERE current_course_id = $1
962  AND user_id = ANY($2)
963RETURNING to_jsonb(user_course_settings.*) AS "row!"
964            "#,
965            plan.course_id,
966            &removed_moved_user_ids
967        )
968        .fetch_all(&mut *conn)
969        .await?,
970    );
971    let removed_marketing_consents = sqlx::query_scalar!(
972        r#"
973DELETE FROM user_marketing_consents
974WHERE id = ANY($1)
975RETURNING to_jsonb(user_marketing_consents.*) AS "row!"
976        "#,
977        &plan.removed_consent_ids
978    )
979    .fetch_all(&mut *conn)
980    .await?;
981
982    sqlx::query!(
983        "UPDATE courses SET course_language_group_id = $2 WHERE id = $1",
984        plan.course_id,
985        to_group_id
986    )
987    .execute(&mut *conn)
988    .await?;
989    let moved_enrolment_user_ids = sqlx::query_scalar!(
990        "
991UPDATE user_course_settings
992SET course_language_group_id = $2
993WHERE current_course_id = $1
994RETURNING user_id
995        ",
996        plan.course_id,
997        to_group_id
998    )
999    .fetch_all(&mut *conn)
1000    .await?;
1001    let moved_consent_ids: Vec<Uuid> = plan.moved_consents.iter().map(|c| c.id).collect();
1002    sqlx::query!(
1003        "
1004UPDATE user_marketing_consents
1005SET course_language_group_id = $2
1006WHERE id = ANY($1)
1007        ",
1008        &moved_consent_ids,
1009        to_group_id
1010    )
1011    .execute(&mut *conn)
1012    .await?;
1013
1014    let carried = if plan.is_from_group_emptied {
1015        move_group_wide_rows(&mut *conn, plan, to_group_id).await?
1016    } else {
1017        copy_group_wide_rows(&mut *conn, plan, to_group_id).await?
1018    };
1019    if !plan.dropped_token_ids.is_empty() {
1020        sqlx::query!(
1021            "
1022UPDATE marketing_mailing_list_access_tokens
1023SET deleted_at = now()
1024WHERE id = ANY($1)
1025            ",
1026            &plan.dropped_token_ids
1027        )
1028        .execute(&mut *conn)
1029        .await?;
1030    }
1031
1032    let page_links =
1033        apply_link_plan(&mut *conn, LinkedItemKind::Page, &plan.pages, to_group_id).await?;
1034    let exercise_links = apply_link_plan(
1035        &mut *conn,
1036        LinkedItemKind::Exercise,
1037        &plan.exercises,
1038        to_group_id,
1039    )
1040    .await?;
1041
1042    if plan.is_from_group_emptied {
1043        sqlx::query!(
1044            "UPDATE course_language_groups SET deleted_at = now() WHERE id = $1",
1045            plan.from_group_id
1046        )
1047        .execute(&mut *conn)
1048        .await?;
1049    }
1050
1051    Ok(json!({
1052        "course_id": plan.course_id,
1053        "from_course_language_group_id": plan.from_group_id,
1054        "to_course_language_group_id": to_group_id,
1055        "from_course_language_group_deleted": plan.is_from_group_emptied,
1056        "moved_enrolment_settings_user_ids": moved_enrolment_user_ids,
1057        "enrolment_conflicts": plan.enrolment_conflicts,
1058        "removed_enrolment_settings": removed_enrolment_settings,
1059        "moved_marketing_consent_ids": moved_consent_ids,
1060        "removed_marketing_consents": removed_marketing_consents,
1061        "group_wide_rows": carried,
1062        "dropped_mailing_list_access_token_ids": plan.dropped_token_ids,
1063        "pages": page_links,
1064        "exercises": exercise_links,
1065    }))
1066}

1124        "
