24/// `base` if it is free, otherwise the first free one of `base-2`, `base-3`, ...
25pub(crate) fn first_free_suffixed(base: &str, is_taken: impl Fn(&str) -> bool) -> String {
26    let mut candidate = base.to_string();
27    let mut suffix = 2;
28    while is_taken(&candidate) {
29        candidate = format!("{base}-{suffix}");
30        suffix += 1;
31    }
32    candidate
33}
