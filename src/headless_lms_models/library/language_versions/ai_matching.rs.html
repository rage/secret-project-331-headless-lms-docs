16/// Excerpt lengths in characters, before [MAX_EXCERPT_CHARS_PER_TASK] shortens them.
17const PAGE_EXCERPT_CHARS: usize = 400;
18const EXERCISE_EXCERPT_CHARS: usize = 250;
19/// Bounds one prompt: a chapter with many exercises gets shorter excerpts.
20const MAX_EXCERPT_CHARS_PER_TASK: usize = 30_000;
21/// Read from the database per item; the excerpt is cut from this.
22const MAX_SOURCE_TEXT_CHARS: i32 = 2_000;

24/// A page or exercise the AI is asked to match.
25#[derive(Debug, Clone)]
26pub struct AiMatchingItem {

28    /// Page title or exercise name.

65/// One pick in the AI's answer. Ids are strings so that a mistyped id is caught here rather

82/// The AI's picks in one other course by this course's item id: `None` is an explicit "no

93/// Picks of an answer that [AiPicks::add_answer] dropped.
94#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
95pub struct DroppedPicks {
96    /// `item` is not an id of this course's items of the task and kind.
97    pub unknown_items: usize,
98    /// `counterpart` is not an id of the other course's items of the task and kind.
99    pub unknown_counterparts: usize,
100    /// A second pick of the same item or counterpart.
101    pub duplicates: usize,
102}
103
104impl DroppedPicks {
105    /// All dropped picks.
106    pub fn total(&self) -> usize {
107        self.unknown_items + self.unknown_counterparts + self.duplicates
108    }
109}
110
111impl AiPicks {
112    /// Adds the valid picks of `answer`. A pick is dropped when its ids are not among the task's
113    /// items of that kind, or when an earlier pick has its item or counterpart, keeping the picks
114    /// one-to-one.
115    pub fn add_answer(&mut self, task: &AiMatchingTask, answer: &AiMatchingAnswer) -> DroppedPicks {
116        let mut dropped = DroppedPicks::default();
117        add_picks(
118            &mut self.pages,
119            &mut dropped,
120            &task.this_pages,
121            &task.other_pages,
122            &answer.pages,
123        );
124        add_picks(
125            &mut self.exercises,
126            &mut dropped,
127            &task.this_exercises,
128            &task.other_exercises,
129            &answer.exercises,
130        );
131        dropped
132    }
133}

135fn add_picks(
136    res: &mut HashMap<Uuid, Option<Uuid>>,
137    dropped: &mut DroppedPicks,
138    this: &[AiMatchingItem],
139    other: &[AiMatchingItem],
140    picks: &[AiPick],
141) {
142    let find = |items: &[AiMatchingItem], id: &str| {
143        let id = Uuid::parse_str(id.trim()).ok()?;
144        items.iter().any(|item| item.id == id).then_some(id)
145    };
146    let mut used_other = HashSet::new();
147    let mut answered = HashSet::new();
148    for pick in picks {
149        let Some(item) = find(this, &pick.item) else {
150            dropped.unknown_items += 1;
151            continue;
152        };
153        let counterpart = match &pick.counterpart {
154            None => None,
155            Some(id) => match find(other, id) {
156                Some(id) => Some(id),
157                None => {
158                    dropped.unknown_counterparts += 1;
159                    continue;
160                }
161            },
162        };
163        if answered.contains(&item) || counterpart.is_some_and(|id| used_other.contains(&id)) {
164            dropped.duplicates += 1;
165            continue;
166        }
167        answered.insert(item);
168        used_other.extend(counterpart);
169        res.insert(item, counterpart);
170    }
171}

173/// Linked counterparts and copies stay as they are, so the AI is not asked about them.
174fn is_fixed<T>(m: &Match<'_, T>) -> bool {
175    m.is_linked || m.reason == Some(SuggestionReason::CopyHistory)
176}

178/// What to ask the AI about `course_id`'s pages and exercises: one task per other course and
179/// chapter number with something unlinked on both sides.
180pub fn ai_matching_tasks(
181    snapshot: &LanguageGroupSnapshot,
182    lineage: &CopyLineage,
183    course_id: Uuid,
184) -> Vec<AiMatchingTask> {
185    let index = CounterpartIndex::new(snapshot);
186    let language_of = |id: Uuid| {
187        snapshot
188            .courses
189            .iter()
190            .find(|c| c.id == id)
191            .map(|c| c.language_code.clone())
192            .unwrap_or_default()
193    };
194    let chapter_name = |course: Uuid, number: Option<i32>| {
195        let number = number?;
196        snapshot
197            .chapters
198            .iter()
199            .find(|c| c.course_id == course && c.chapter_number == number)
200            .map(|c| c.name.clone())
201    };
202    let mut res = Vec::new();
203    for other in index.course_ids().into_iter().filter(|id| *id != course_id) {
204        let page_matches = match_pages(&index, lineage, course_id, other);
205        let exercise_matches = match_exercises(&index, lineage, course_id, other, &page_matches);
206        let fixed_other_pages: HashSet<Uuid> = page_matches
207            .values()
208            .filter(|m| is_fixed(m))
209            .map(|m| m.item.page.id)
210            .collect();
211        let fixed_other_exercises: HashSet<Uuid> = exercise_matches
212            .values()
213            .filter(|m| is_fixed(m))
214            .map(|m| m.item.exercise.id)
215            .collect();
216        let free_this_pages: Vec<_> = index
217            .live_pages(course_id)
218            .filter(|p| !page_matches.get(&p.page.id).is_some_and(is_fixed))
219            .collect();
220        let free_other_pages: Vec<_> = index
221            .live_pages(other)
222            .filter(|p| {
223                !fixed_other_pages.contains(&p.page.id)
224                    && index.page_counterpart_in(&p.page, course_id).is_none()
225            })
226            .collect();
227        let free_this_exercises: Vec<_> = index
228            .live_exercises(course_id)
229            .filter(|e| !exercise_matches.get(&e.exercise.id).is_some_and(is_fixed))
230            .collect();
231        let free_other_exercises: Vec<_> = index
232            .live_exercises(other)
233            .filter(|e| {
234                !fixed_other_exercises.contains(&e.exercise.id)
235                    && index
236                        .exercise_counterpart_in(&e.exercise, course_id)
237                        .is_none()
238            })
239            .collect();
240
241        let mut chapter_numbers: Vec<Option<i32>> = free_this_pages
242            .iter()
243            .map(|p| p.page.chapter_number)
244            .chain(
245                free_this_exercises
246                    .iter()
247                    .map(|e| e.exercise.chapter_number),
248            )
249            .collect();
250        chapter_numbers.sort_unstable();
251        chapter_numbers.dedup();
252        for chapter_number in chapter_numbers {
253            let other_pages: Vec<AiMatchingItem> = free_other_pages
254                .iter()
255                .filter(|p| p.page.chapter_number == chapter_number)
256                .map(|p| AiMatchingItem {
257                    id: p.page.id,
258                    title: p.page.title.clone(),
259                    page_title: None,
260                    excerpt: String::new(),
261                    heuristic_pick: None,
262                })
263                .collect();
264            let other_exercises: Vec<AiMatchingItem> = free_other_exercises
265                .iter()
266                .filter(|e| e.exercise.chapter_number == chapter_number)
267                .map(|e| AiMatchingItem {
268                    id: e.exercise.id,
269                    title: e.exercise.name.clone(),
270                    page_title: Some(e.exercise.page_title.clone()),
271                    excerpt: String::new(),
272                    heuristic_pick: None,
273                })
274                .collect();
275            let listed_pick =
276                |items: &[AiMatchingItem], id: Uuid| items.iter().any(|o| o.id == id).then_some(id);
277            let this_pages: Vec<AiMatchingItem> = free_this_pages
278                .iter()
279                .filter(|p| p.page.chapter_number == chapter_number)
280                .map(|p| AiMatchingItem {
281                    id: p.page.id,
282                    title: p.page.title.clone(),
283                    page_title: None,
284                    excerpt: String::new(),
285                    heuristic_pick: page_matches.get(&p.page.id).and_then(|m| {
286                        listed_pick(&other_pages, m.item.page.id).map(|n| (n, m.sureness))
287                    }),
288                })
289                .collect();
290            let this_exercises: Vec<AiMatchingItem> = free_this_exercises
291                .iter()
292                .filter(|e| e.exercise.chapter_number == chapter_number)
293                .map(|e| AiMatchingItem {
294                    id: e.exercise.id,
295                    title: e.exercise.name.clone(),
296                    page_title: Some(e.exercise.page_title.clone()),
297                    excerpt: String::new(),
298                    heuristic_pick: exercise_matches.get(&e.exercise.id).and_then(|m| {
299                        listed_pick(&other_exercises, m.item.exercise.id).map(|n| (n, m.sureness))
300                    }),
301                })
302                .collect();
303            let has_page_question = !this_pages.is_empty() && !other_pages.is_empty();
304            let has_exercise_question = !this_exercises.is_empty() && !other_exercises.is_empty();
305            if !has_page_question && !has_exercise_question {
306                continue;
307            }
308            let known_page_pairs = page_matches
309                .iter()
310                .filter(|(_, m)| is_fixed(m) && m.item.page.chapter_number == chapter_number)
311                .filter_map(|(id, m)| {
312                    Some(AiKnownPair {
313                        this_title: index.page(*id)?.page.title.clone(),
314                        other_title: m.item.page.title.clone(),
315                    })
316                })
317                .collect();
318            let this_exercise_names: HashMap<Uuid, &str> = index
319                .live_exercises(course_id)
320                .map(|e| (e.exercise.id, e.exercise.name.as_str()))
321                .collect();
322            let known_exercise_pairs = exercise_matches
323                .iter()
324                .filter(|(_, m)| is_fixed(m) && m.item.exercise.chapter_number == chapter_number)
325                .filter_map(|(id, m)| {
326                    Some(AiKnownPair {
327                        this_title: this_exercise_names.get(id)?.to_string(),
328                        other_title: m.item.exercise.name.clone(),
329                    })
330                })
331                .collect();
332            res.push(AiMatchingTask {
333                other_course_id: other,
334                this_language_code: language_of(course_id),
335                other_language_code: language_of(other),
336                chapter_number,
337                this_chapter_name: chapter_name(course_id, chapter_number),
338                other_chapter_name: chapter_name(other, chapter_number),
339                this_pages,
340                other_pages,
341                this_exercises,
342                other_exercises,
343                known_page_pairs,
344                known_exercise_pairs,
345            });
346        }
347    }
348    res
349}

351/// Fills in the excerpts of the tasks' items from page content and exercise assignments.
352pub async fn fill_excerpts(
353    conn: &mut PgConnection,
354    tasks: &mut [AiMatchingTask],
355) -> ModelResult<()> {
356    let page_ids: Vec<Uuid> = tasks
357        .iter()
358        .flat_map(|t| t.this_pages.iter().chain(&t.other_pages))
359        .map(|p| p.id)
360        .collect();
361    let exercise_ids: Vec<Uuid> = tasks
362        .iter()
363        .flat_map(|t| t.this_exercises.iter().chain(&t.other_exercises))
364        .map(|e| e.id)
365        .collect();
366    let mut texts: HashMap<Uuid, String> = HashMap::new();
367    let page_rows = sqlx::query!(
368        r#"
369SELECT p.id,
370  left(p.content_search_original_text, $2) AS text
371FROM pages p
372WHERE p.id = ANY($1)
373  AND p.deleted_at IS NULL
374        "#,
375        &page_ids,
376        MAX_SOURCE_TEXT_CHARS
377    )
378    .fetch_all(&mut *conn)
379    .await?;
380    texts.extend(
381        page_rows
382            .into_iter()
383            .filter_map(|row| Some((row.id, row.text?))),
384    );
385    let exercise_rows = sqlx::query!(
386        r#"
387SELECT es.exercise_id AS "id!",
388  left(
389    string_agg(
390      block_text.value #>> '{}',
391      ' '
392      ORDER BY es.order_number,
393        et.order_number
394    ),
395    $2
396  ) AS text
397FROM exercise_slides es
398  JOIN exercise_tasks et ON et.exercise_slide_id = es.id
399  AND et.deleted_at IS NULL
400  CROSS JOIN LATERAL extract_searchable_text_from_document_schema(
401    CASE
402      WHEN jsonb_typeof(et.assignment) = 'array' THEN et.assignment
403      ELSE '[]'::jsonb
404    END
405  ) AS block_text(value)
406WHERE es.exercise_id = ANY($1)
407  AND es.deleted_at IS NULL
408GROUP BY es.exercise_id
409        "#,
410        &exercise_ids,
411        MAX_SOURCE_TEXT_CHARS
412    )
413    .fetch_all(&mut *conn)
414    .await?;
415    texts.extend(
416        exercise_rows
417            .into_iter()
418            .filter_map(|row| Some((row.id, row.text?))),
419    );
420
421    for task in tasks {
422        let page_count = task.this_pages.len() + task.other_pages.len();
423        let exercise_count = task.this_exercises.len() + task.other_exercises.len();
424        let wanted = page_count * PAGE_EXCERPT_CHARS + exercise_count * EXERCISE_EXCERPT_CHARS;
425        let scale = |chars: usize| {
426            if wanted <= MAX_EXCERPT_CHARS_PER_TASK {
427                chars
428            } else {
429                chars * MAX_EXCERPT_CHARS_PER_TASK / wanted
430            }
431        };
432        let (page_chars, exercise_chars) =
433            (scale(PAGE_EXCERPT_CHARS), scale(EXERCISE_EXCERPT_CHARS));
434        for item in task
435            .this_pages
436            .iter_mut()
437            .chain(task.other_pages.iter_mut())
438        {
439            item.excerpt = excerpt(texts.get(&item.id), page_chars);
440        }
441        for item in task
442            .this_exercises
443            .iter_mut()
444            .chain(task.other_exercises.iter_mut())
445        {
446            item.excerpt = excerpt(texts.get(&item.id), exercise_chars);
447        }
448    }
449    Ok(())
450}

452fn excerpt(text: Option<&String>, max_chars: usize) -> String {
453    let Some(text) = text else {
454        return String::new();
455    };
456    let plain = decode_html_entities(&strip_html_tags(text)).into_owned();
457    plain
458        .split_whitespace()
459        .collect::<Vec<_>>()
460        .join(" ")
461        .chars()
462        .take(max_chars)
463        .collect()
