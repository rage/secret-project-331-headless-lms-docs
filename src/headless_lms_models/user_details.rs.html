16/// Search box filter for user lists: every word of the search appears in the name or email, in any
17/// order, or the whole search is a near match for a misspelled name or email. Queries apply it as
18/// `(ud.name_search_helper || ' ' || ud.email_search_helper) LIKE ALL ($word_patterns)` OR
19/// `word_similarity($misspelling_term, ud.name_search_helper || ' ' || ud.email_search_helper) >= 0.5`.
20/// [`search_for_user_details_fuzzy`] is the ranked search over all users.
21#[derive(Debug, Clone, Default)]
22pub struct UserListSearch {
23    /// One `%word%` pattern per word, lowercased and LIKE-escaped.
24    pub word_patterns: Vec<String>,
25    /// `None` when the search contains a digit: a near match for a student number or for
26    /// `user_3@example.com` is a different person.
27    pub misspelling_term: Option<String>,
28}
29
30impl UserListSearch {
31    pub fn new(search: &str) -> Self {
32        let search = search.trim().to_lowercase();
33        let word_patterns = search
34            .split_whitespace()
35            .map(|word| format!("%{}%", escape_like_pattern(word)))
36            .collect();
37        let misspelling_term = (!search.chars().any(char::is_numeric)).then_some(search);
38        Self {
39            word_patterns,
40            misspelling_term,
41        }
42    }
43}

73/// How course design plans name a person: their full name, or their email when they have none.
74pub fn full_name_or_email(
75    first_name: Option<&str>,
76    last_name: Option<&str>,
77    email: &str,
78) -> String {
79    let name = [first_name, last_name]
80        .into_iter()
81        .flatten()
82        .map(str::trim)
83        .filter(|part| !part.is_empty())
84        .collect::<Vec<_>>()
85        .join(" ");
86    if name.is_empty() {
87        email.to_string()
88    } else {
89        name
90    }
91}

203/// Users whose id or email equals the search term, ignoring case and surrounding whitespace.
204///
205/// At most one row per kind, and cheap enough to answer before [`search_for_user_details_fuzzy`].
206pub async fn search_for_user_details_exact(
207    conn: &mut PgConnection,
208    search: &str,
209) -> ModelResult<Vec<UserDetail>> {
210    let search = search.trim();
211    let user_id = search.parse::<Uuid>().ok();
212    let email = (user_id.is_none() && search.contains('@')).then_some(search);
213    if user_id.is_none() && email.is_none() {
214        return Ok(Vec::new());
215    }
216    let res = sqlx::query_as!(
217        UserDetail,
218        "
219SELECT user_id,
220  created_at,
221  updated_at,
222  email,
223  first_name,
224  last_name,
225  country,
226  email_communication_consent,
227  email_verified_at,
228  email_verified_method
229FROM user_details
230WHERE user_id = $1
231  OR lower(email) = lower($2)
232",
233        user_id,
234        email,
235    )
236    .fetch_all(conn)
237    .await?;
238    Ok(res)
239}

241/// Typo-tolerant matches from [`search_for_user_details_fuzzy`], best first.
242#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize, ToSchema)]
243pub struct FuzzyUserSearchResults {
244    pub users: Vec<UserDetail>,
245    /// More users matched than `users` holds.
246    pub has_more: bool,
247}
248
249/// Users whose name or email local part resembles the search term, best match first, ties by id.
250///
251/// A term containing `@` is matched by its local part. Terms shorter than three characters and
252/// user ids match nothing; [`search_for_user_details_exact`] covers ids and whole emails. A common
253/// term takes seconds on production, so callers should show the exact matches first.
254pub async fn search_for_user_details_fuzzy(
255    conn: &mut PgConnection,
256    search: &str,
257) -> ModelResult<FuzzyUserSearchResults> {
258    let Some(term) = fuzzy_search_term(search) else {
259        return Ok(FuzzyUserSearchResults::default());
260    };
261    let mut tx = conn.begin().await?;
262    sqlx::query!(
263        "SELECT set_config('pg_trgm.word_similarity_threshold', $1, true)",
264        FUZZY_SEARCH_WORD_SIMILARITY_THRESHOLD.to_string(),
265    )
266    .fetch_one(&mut *tx)
267    .await?;
268    // `<%` uses the threshold set above. The expression must stay identical to the
269    // user_details_search_text_gin index. Not ordered by GiST trigram distance: a term close to no
270    // name walks most of such an index, 14s on production.
271    let mut users = sqlx::query_as!(
272        UserDetail,
273        r#"
274SELECT user_id,
275  created_at,
276  updated_at,
277  email,
278  first_name,
279  last_name,
280  country,
281  email_communication_consent,
282  email_verified_at,
283  email_verified_method
284FROM user_details
285WHERE $1 <% (name_search_helper || ' ' || split_part(email_search_helper, '@', 1))
286ORDER BY $1 <<-> (name_search_helper || ' ' || split_part(email_search_helper, '@', 1)),
287  user_id
288LIMIT $2
289"#,
290        term,
291        FUZZY_SEARCH_RESULT_LIMIT + 1,
292    )
293    .fetch_all(&mut *tx)
294    .await?;
295    // Rolled back rather than committed: it only read, and a commit inside a caller's transaction
296    // would keep the threshold for the rest of that transaction.
297    tx.rollback().await?;
298
299    let has_more = users.len() > FUZZY_SEARCH_RESULT_LIMIT as usize;
300    users.truncate(FUZZY_SEARCH_RESULT_LIMIT as usize);
301    Ok(FuzzyUserSearchResults { users, has_more })
302}

304/// The lowercased local part of the search term, or `None` when it should match nothing.
305fn fuzzy_search_term(search: &str) -> Option<String> {
306    let search = search.trim();
307    if search.parse::<Uuid>().is_ok() {
308        return None;
309    }
310    let term = search
311        .split('@')
312        .next()
313        .unwrap_or(search)
314        .trim()
315        .to_lowercase();
316    (term.chars().count() >= MIN_FUZZY_SEARCH_TERM_LENGTH).then_some(term)
317}

508/// Writes the whole profile form, including the derived `users.email_domain`.
509///
510/// On an address change the `clear_email_verification` trigger nulls `email_verified_at` and
511/// `email_verified_method`, so a caller wanting fresh proof must mail a new verification link.
512pub async fn update_user_info(
513    conn: &mut PgConnection,
514    user_id: Uuid,
515    email: &str,
516    first_name: &str,
517    last_name: &str,
518    country: &str,
519    email_communication_consent: bool,
520) -> Result<UserDetail, sqlx::Error> {
521    let mut tx = conn.begin().await?;
522    let updated_user = sqlx::query_as!(
523        UserDetail,
524        r#"
525UPDATE user_details
526SET email = $1,
527  first_name = $2,
528  last_name = $3,
529  country = $4,
530  email_communication_consent = $5
531WHERE user_id = $6
532RETURNING user_id,
533  created_at,
534  updated_at,
535  email,
536  first_name,
537  last_name,
538  country,
539  email_communication_consent,
540  email_verified_at,
541  email_verified_method
542"#,
543        email,
544        first_name,
545        last_name,
546        country,
547        email_communication_consent,
548        user_id,
549    )
550    .fetch_one(&mut *tx)
551    .await?;
552
553    sqlx::query!(
554        r#"
555UPDATE users
556SET email_domain = $1
557WHERE id = $2
558"#,
559        crate::users::email_domain_from_email(email),
560        user_id,
561    )
562    .execute(&mut *tx)
563    .await?;
564    tx.commit().await?;
565
566    Ok(updated_user)
567}

569/// Records a proof of control over the address currently in `email`, and grants the roles that were
570/// waiting for it (see [`crate::pending_roles`]).

599/// Records a proof of control from a code mailed at `code_created_at`, and grants the roles that
600/// were waiting for it, unless the account already has a proof or its details changed since then.
601/// An admin's assertion is replaced, since it is not proof (see [`has_mailbox_proof`]).
602///
603/// One conditional write, so an address change committing alongside cannot slip between the check
604/// and the write. Compare [`set_email_verified`], which records unconditionally.
605pub async fn set_email_verified_unless_changed_since(
606    conn: &mut PgConnection,
607    user_id: Uuid,
608    method: EmailVerificationMethod,
609    code_created_at: DateTime<Utc>,
610) -> ModelResult<()> {
611    let mut tx = conn.begin().await?;
612    let result = sqlx::query!(
613        r#"
614UPDATE user_details
615SET email_verified_at = now(),
616  email_verified_method = $2
617WHERE user_id = $1
618  AND (
619    email_verified_at IS NULL
620    OR email_verified_method = 'admin_asserted'
621  )
622  AND updated_at <= $3
623"#,
624        user_id,
625        method as EmailVerificationMethod,
626        code_created_at,
627    )
628    .execute(&mut *tx)
629    .await?;
630    if result.rows_affected() > 0 {
631        crate::pending_roles::activate_for_user(&mut tx, user_id).await?;
632    }
633    tx.commit().await?;
634    Ok(())
635}

654/// Whether the user has shown they control the address in `email`. An admin's say-so is not proof:
655/// it can attach any address to any account.
656pub async fn has_mailbox_proof(conn: &mut PgConnection, user_id: Uuid) -> ModelResult<bool> {

659        .is_some_and(|(_, method)| method != EmailVerificationMethod::AdminAsserted))
