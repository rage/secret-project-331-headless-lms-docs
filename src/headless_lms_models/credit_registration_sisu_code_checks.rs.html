3//! Asked by resolve-enrolments; read by the prerequisite gate and when a row is resolved.

12/// How long an answer that Sisu does not hold a code stands before the code is asked about again.
13pub const CHECK_MAX_AGE: TimeDelta = TimeDelta::days(1);
14
15/// One student and course code that is due to be asked about.
16#[derive(Debug, Clone)]
17pub struct DueCodeCheck {
18    pub user_id: Uuid,
19    pub student_number: DbSecret,
20    /// Upper-cased and trimmed.
21    pub course_code: String,
22    /// The ledger rows that wait on the answer, for the audit log.
23    pub credit_registration_ids: Vec<Uuid>,
24}
25
26/// Up to `limit` student and code pairs whose answer is missing or older than [`CHECK_MAX_AGE`]
27/// without an attainment: the equivalent codes of mooc.fi rows due a resolve, first or parked
28/// (as [`claim_due_for_resolve`](crate::credit_registrations::claim_due_for_resolve) takes them),
29/// and the prerequisite and its equivalents of rows held in `checking_enrolment` by
30/// `credit_registration_prerequisite_met`. Only students with a verified student number.
31pub async fn get_due(
32    conn: &mut PgConnection,
33    scope: &RegistrationScope,
34    limit: i64,
35) -> ModelResult<Vec<DueCodeCheck>> {
36    let rows = sqlx::query!(
37        r#"
38WITH wanted AS (
39  SELECT cr.id AS credit_registration_id,
40    cr.user_id,
41    UPPER(TRIM(equivalent)) AS course_code
42  FROM credit_registrations cr
43    JOIN credit_registration_active_units au ON au.credit_unit_id = cr.credit_unit_id
44    JOIN external_credit_target_codes code ON code.id = cr.external_credit_target_code_id
45    CROSS JOIN LATERAL UNNEST(code.equivalent_course_codes) AS equivalent
46  WHERE cr.state IN ('ready_to_submit', 'no_usable_enrolment')
47    AND cr.deleted_at IS NULL
48    AND cr.superseded_by_id IS NULL
49    AND cr.next_attempt_at <= now()
50    AND (
51      cr.enrolment_check_claimed_until IS NULL
52      OR cr.enrolment_check_claimed_until <= now()
53    )
54    AND ($2::uuid IS NULL OR cr.external_credit_target_id = $2)
55    AND ($3::uuid IS NULL OR cr.user_id = $3)
56    AND (
57      cardinality($4::uuid []) = 0
58      OR cr.id = ANY($4::uuid [])
59    )
60    AND $5::uuid IS NULL
61  UNION ALL
62  SELECT cr.id,
63    cr.user_id,
64    UPPER(TRIM(prerequisite_code))
65  FROM credit_registrations cr
66    JOIN credit_registration_active_units au ON au.credit_unit_id = cr.credit_unit_id
67    JOIN external_credit_target_codes code ON code.id = cr.external_credit_target_code_id
68    JOIN external_credit_target_codes prerequisite ON prerequisite.id = code.prerequisite_code_id
69    CROSS JOIN LATERAL UNNEST(
70      prerequisite.uh_course_code::text || prerequisite.equivalent_course_codes
71    ) AS prerequisite_code
72  WHERE cr.state = 'checking_enrolment'
73    AND code.prerequisite_code_id IS NOT NULL
74    AND cr.deleted_at IS NULL
75    AND cr.superseded_by_id IS NULL
76    AND NOT credit_registration_prerequisite_met(cr)
77    AND ($2::uuid IS NULL OR cr.external_credit_target_id = $2)
78    AND ($3::uuid IS NULL OR cr.user_id = $3)
79    AND (
80      cardinality($4::uuid []) = 0
81      OR cr.id = ANY($4::uuid [])
82    )
83    AND $5::uuid IS NULL
84)
85SELECT w.user_id AS "user_id!",
86  vsn.student_number AS "student_number!",
87  w.course_code AS "course_code!",
88  ARRAY_AGG(DISTINCT w.credit_registration_id) AS "credit_registration_ids!"
89FROM wanted w
90  JOIN verified_student_numbers vsn ON vsn.user_id = w.user_id
91  AND vsn.deleted_at IS NULL
92WHERE NOT EXISTS (
93    SELECT 1
94    FROM credit_registration_sisu_code_checks c
95    WHERE c.user_id = w.user_id
96      AND c.course_code = w.course_code
97      AND c.deleted_at IS NULL
98      AND (
99        c.attained_at IS NOT NULL
100        OR c.checked_at > now() - $6::interval
101      )
102  )
103GROUP BY w.user_id,
104  vsn.student_number,
105  w.course_code
106ORDER BY w.user_id,
107  w.course_code
108LIMIT $1
109        "#,
110        limit,
111        scope.external_credit_target_id,
112        scope.user_id,
113        &scope.credit_registration_ids,
114        // mooc.fi rows have no course, so a course scope leaves them all out.
115        scope.course_id,
116        CHECK_MAX_AGE as TimeDelta,
117    )
118    .fetch_all(conn)
119    .await?;
120    Ok(rows
121        .into_iter()
122        .map(|row| DueCodeCheck {
123            user_id: row.user_id,
124            student_number: row.student_number,
125            course_code: row.course_code,
126            credit_registration_ids: row.credit_registration_ids,
127        })
128        .collect())
129}

131/// Sisu's answer about one student and course code, for [`record_all`].
132#[derive(Debug, Clone)]
133pub struct SisuCodeAnswer {
134    pub user_id: Uuid,
135    pub course_code: String,
136    pub is_attained: bool,
137}
138
139/// Records Sisu's answers. `attained_at` is kept once set.
140pub async fn record_all(conn: &mut PgConnection, answers: &[SisuCodeAnswer]) -> ModelResult<()> {
141    let user_ids: Vec<Uuid> = answers.iter().map(|answer| answer.user_id).collect();
142    let course_codes: Vec<String> = answers
143        .iter()
144        .map(|answer| answer.course_code.clone())
145        .collect();
146    let is_attained: Vec<bool> = answers.iter().map(|answer| answer.is_attained).collect();
147    sqlx::query!(
148        r#"
149INSERT INTO credit_registration_sisu_code_checks (user_id, course_code, checked_at, attained_at)
150-- ON CONFLICT cannot update one row twice, so a pair answered twice keeps the attained answer.
151SELECT DISTINCT ON (answer.user_id, UPPER(TRIM(answer.course_code))) answer.user_id,
152  UPPER(TRIM(answer.course_code)),
153  now(),
154  CASE
155    WHEN answer.is_attained THEN now()
156  END
157FROM UNNEST($1::uuid [], $2::text [], $3::boolean []) AS answer(user_id, course_code, is_attained)
158ORDER BY answer.user_id,
159  UPPER(TRIM(answer.course_code)),
160  answer.is_attained DESC ON CONFLICT (user_id, course_code, deleted_at) DO
161UPDATE
162SET checked_at = EXCLUDED.checked_at,
163  attained_at = COALESCE(
164    credit_registration_sisu_code_checks.attained_at,
165    EXCLUDED.attained_at
166  )
167        "#,
168        &user_ids,
169        &course_codes,
170        &is_attained,
171    )
172    .execute(conn)
173    .await?;
174    Ok(())
175}

177/// What Sisu is known to hold of a ledger row's equivalent codes.
178#[derive(Debug, Clone, PartialEq, Eq)]
179pub enum EquivalentHolding {
180    /// Sisu holds this equivalent code, which counts as holding the row's own.
181    Held { course_code: String },
182    /// Every equivalent was asked about recently and none is held.
183    NotHeld,
184    /// Some equivalent has not been asked about recently, so the row cannot be resolved yet.
185    Unchecked,
186}
187
188/// [`EquivalentHolding`] for each of the rows whose code has equivalent codes; rows without any
189/// are left out.
190pub async fn get_equivalent_holdings(
191    conn: &mut PgConnection,
192    credit_registration_ids: &[Uuid],
193) -> ModelResult<HashMap<Uuid, EquivalentHolding>> {
194    let rows = sqlx::query!(
195        r#"
196SELECT cr.id,
197  (
198    SELECT MIN(c.course_code)
199    FROM credit_registration_sisu_code_checks c
200    WHERE c.user_id = cr.user_id
201      AND c.deleted_at IS NULL
202      AND c.attained_at IS NOT NULL
203      AND c.course_code IN (
204        SELECT UPPER(TRIM(equivalent))
205        FROM UNNEST(code.equivalent_course_codes) AS equivalent
206      )
207  ) AS held_course_code,
208  EXISTS (
209    SELECT 1
210    FROM UNNEST(code.equivalent_course_codes) AS equivalent
211    WHERE NOT EXISTS (
212        SELECT 1
213        FROM credit_registration_sisu_code_checks c
214        WHERE c.user_id = cr.user_id
215          AND c.deleted_at IS NULL
216          AND c.course_code = UPPER(TRIM(equivalent))
217          AND c.checked_at > now() - $2::interval
218      )
219  ) AS "has_unchecked!"
220FROM credit_registrations cr
221  JOIN external_credit_target_codes code ON code.id = cr.external_credit_target_code_id
222WHERE cr.id = ANY($1)
223  AND cardinality(code.equivalent_course_codes) > 0
224        "#,
225        credit_registration_ids,
226        CHECK_MAX_AGE as TimeDelta,
227    )
228    .fetch_all(conn)
229    .await?;
230    Ok(rows
231        .into_iter()
232        .map(|row| {
233            let holding = match row.held_course_code {
234                Some(course_code) => EquivalentHolding::Held { course_code },
235                None if row.has_unchecked => EquivalentHolding::Unchecked,
236                None => EquivalentHolding::NotHeld,
237            };
238            (row.id, holding)
239        })
240        .collect())
241}
