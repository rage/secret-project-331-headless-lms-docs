85/// Soft deletes a conversation, after which it accepts no further messages.
86///
87/// Deleting one that is already deleted is not an error, so a caller does not have to check first.
88/// See [crate::chatbot_configurations::delete] for removing the configuration a conversation
89/// belongs to.
90///
91/// Nothing in production deletes a conversation; this exists for tests.
92pub async fn delete(conn: &mut PgConnection, id: Uuid) -> ModelResult<()> {
93    sqlx::query!(
94        r#"
95UPDATE chatbot_conversations
96SET deleted_at = now()
97WHERE id = $1
98  AND deleted_at IS NULL
99        "#,
100        id
101    )
102    .execute(conn)
103    .await?;
104    Ok(())
105}

146pub async fn get_latest_conversation_for_user(
147    conn: &mut PgConnection,
148    user_id: Option<Uuid>,
149    anonymous_token: Option<String>,
150    chatbot_configuration_id: Uuid,
151) -> ModelResult<ChatbotConversation> {
152    if let (Some(_user_id), Some(_anonymous_token)) = (&user_id, &anonymous_token) {
153        return Err(model_err!(
154            InvalidRequest,
155            "User ID and anonymous token cannot both be present".to_string()
156        ));
157    }
158    let res = sqlx::query_as!(
159        ChatbotConversation,
160        r#"
161SELECT *
162FROM chatbot_conversations
163WHERE (
164    user_id = $1
165    OR anonymous_token = $2
166  )
167  AND chatbot_configuration_id = $3
168  AND deleted_at IS NULL
169ORDER BY created_at DESC
170LIMIT 1
171        "#,
172        user_id,
173        anonymous_token,
174        chatbot_configuration_id
175    )
176    .fetch_one(conn)
177    .await?;
178    Ok(res)
179}

251/// Gets specific conversation or latest conversation for the user, if any. If conversation_id is not provided then latest conversation is given. Also inlcudes information about the chatbot so that the chatbot ui can be rendered using the information.
252pub async fn get_conversation_info(
253    tx: &mut PgConnection,
254    user_id: Option<Uuid>,
255    anonymous_token: Option<String>,
256    chatbot_configuration_id: Uuid,
257    conversation_id: Option<Uuid>,
258) -> ModelResult<ChatbotConversationInfo> {
259    let chatbot_configuration =
260        crate::chatbot_configurations::get_by_id(tx, chatbot_configuration_id).await?;
261    let course = if let Some(course_id) = chatbot_configuration.course_id {
262        Some(crate::courses::get_course(tx, course_id).await?)
263    } else {
264        None
265    };
266
267    let current_conversation = if let Some(conversation_id) = conversation_id {
268        get_conversation_for_user(
269            tx,
270            user_id,
271            anonymous_token,
272            chatbot_configuration_id,
273            conversation_id,
274        )
275        .await
276        .optional()?
277    } else {
278        get_latest_conversation_for_user(tx, user_id, anonymous_token, chatbot_configuration_id)
279            .await
280            .optional()?
281    };
282
283    let conversation_id = current_conversation.as_ref().map(|c| c.id);
284
285    let current_conversation_messages = OptionFuture::from(conversation_id.map(|id| {
286        crate::chatbot_conversation_messages::get_by_conversation_id_for_display(tx, id)
287    }))
288    .await
289    .transpose()?;
290
291    let current_conversation_message_citations =
292        OptionFuture::from(conversation_id.map(|id| {
293            crate::chatbot_conversation_messages_citations::get_by_conversation_id(tx, id)
294        }))
295        .await
296        .transpose()?;
297
298    let suggested_messages = if chatbot_configuration.suggest_next_messages
299        && let Some(ccm) = &current_conversation_messages
300        // A suspended turn is waiting for the learner to answer the chatbot's own question, which
301        // is not a moment to suggest asking something else.
302        && !crate::chatbot_conversation_messages::turn_is_suspended(ccm)
303        && let Some(last_ccm) = ccm.last()
304    {
305        let sm = crate::chatbot_conversation_suggested_messages::get_by_conversation_message_id(
306            tx,
307            last_ccm.id.to_owned(),
308        )
309        .await?;
310        // return an empty vec if there are not yet any suggested messages
311        Some(sm)
312    } else {
313        None
314    };
315
316    Ok(ChatbotConversationInfo {
317        current_conversation,
318        current_conversation_messages,
319        current_conversation_message_citations,
320        suggested_messages,
321        // Don't want to expose everything from the chatbot configuration to the user because it contains private information like the prompt.
322        chatbot_name: chatbot_configuration.chatbot_name,
323        course_name: course.map(|course| course.name),
324        hide_citations: chatbot_configuration.hide_citations,
325    })
326}
