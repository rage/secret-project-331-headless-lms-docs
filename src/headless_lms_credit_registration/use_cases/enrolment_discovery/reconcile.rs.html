27/// Wakes the linked students' registrations on every module of one code's roster and claims a
28/// linking mail for everybody else, once per course on the code: mails and links are per course.
29/// Returns the linking mails claimed, for the caller's roster-fetch summary.
30pub(super) async fn reconcile_roster(
31    conn: &mut PgConnection,
32    listing: &CodeListing,
33    people: &[RosterPerson],
34    account_linking_since: Option<DateTime<Utc>>,
35) -> CreditRegistrationResult<i32> {
36    let distinct = distinct_people(people);
37    let linked_rows = load_linked_accounts(conn, &distinct).await?;
38    let linked = LinkedAccounts::new(&linked_rows);
39    let enrolees = roster_enrolees(people, &linked);
40    if !enrolees.is_empty() {
41        for module in &listing.modules {
42            wake_for_roster_listing(conn, module.course_module_id, &enrolees).await?;
43        }
44    }
45    let mut mailed_count = 0;
46    if let Some(since) = account_linking_since {
47        let considered = enrolled_since(&distinct, since);
48        let course_ids: BTreeSet<Uuid> = listing
49            .modules
50            .iter()
51            .map(|module| module.course_id)
52            .collect();
53        let counters = claim_linking_mails(conn, &considered, &linked, &course_ids).await?;
54        mailed_count = counters.mailed_count;
55        record_linking_counters(conn, listing.code.course_code.as_str(), &counters).await?;
56    }
57    for module in &listing.modules {
58        mark_listing_succeeded(
59            conn,
60            module.course_module_id,
61            account_linking_since.is_some(),
62        )
63        .await?;
64    }
65    Ok(mailed_count)
66}

68/// A person enrolled on several realisations of the code is listed once per realisation; keeps the
69/// most recent enrolment of each, one with no enrolment time counting as the oldest.
70fn distinct_people(people: &[RosterPerson]) -> Vec<&RosterPerson> {
71    let mut kept: Vec<&RosterPerson> = Vec::new();
72    let mut index_by_person: HashMap<&str, usize> = HashMap::new();
73    for person in people {
74        match index_by_person.get(person.person_id.expose_secret()) {
75            Some(&index) => {
76                if person.enrolled_at() > kept[index].enrolled_at() {
77                    kept[index] = person;
78                }
79            }
80            None => {
81                index_by_person.insert(person.person_id.expose_secret(), kept.len());
82                kept.push(person);
83            }
84        }
85    }
86    kept
87}

89/// The people the linking mails consider: those enrolled at or after `since`. Without a link we
90/// cannot tell who the rest are, so they are neither mailed nor counted.
91fn enrolled_since<'a>(people: &[&'a RosterPerson], since: DateTime<Utc>) -> Vec<&'a RosterPerson> {
92    people
93        .iter()
94        .copied()
95        .filter(|person| {
96            person
97                .enrolled_at()
98                .is_some_and(|enrolled_at| enrolled_at >= since)
99        })
100        .collect()
101}

103/// The accounts already linked to someone on the roster, by Sisu person id or student number.
104async fn load_linked_accounts(
105    conn: &mut PgConnection,
106    people: &[&RosterPerson],
107) -> CreditRegistrationResult<Vec<VerifiedStudentNumber>> {
108    let person_ids: Vec<String> = people
109        .iter()
110        .map(|person| person.person_id.expose_secret().to_owned())
111        .collect();
112    let student_numbers: Vec<String> = people
113        .iter()
114        .map(|person| person.student_number.expose_secret().to_owned())
115        .collect();
116    let mut linked = verified_student_numbers::get_by_sisu_person_ids(conn, &person_ids).await?;
117    // A study_registry link has no person id until resolve-person-ids reaches it.
118    let linked_ids: HashSet<_> = linked.iter().map(|row| row.id).collect();
119    let linked_by_number = verified_student_numbers::get_by_student_numbers(conn, &student_numbers)
120        .await?
121        .into_iter()
122        .filter(|row| !linked_ids.contains(&row.id))
123        .collect::<Vec<_>>();
124    linked.extend(linked_by_number);
125    Ok(linked)
126}

128/// The accounts linked to someone on one code's roster, indexed once for every module on it.
129struct LinkedAccounts<'a> {
130    rows: &'a [VerifiedStudentNumber],
131    person_ids: HashSet<&'a str>,
132    student_numbers: HashSet<&'a str>,
133}
134
135impl<'a> LinkedAccounts<'a> {
136    fn new(rows: &'a [VerifiedStudentNumber]) -> Self {
137        Self {
138            rows,
139            person_ids: rows
140                .iter()
141                .filter_map(|row| expose_option(&row.sisu_person_id))
142                .collect(),
143            student_numbers: rows
144                .iter()
145                .map(|row| row.student_number.expose_secret())
146                .collect(),
147        }
148    }
149
150    /// Whether some account is linked to `person`, by Sisu person id or student number.
151    fn is_linked(&self, person: &RosterPerson) -> bool {
152        self.person_ids.contains(person.person_id.expose_secret())
153            || self
154                .student_numbers
155                .contains(person.student_number.expose_secret())
156    }
157}

159/// The linked accounts on the roster, each with every enrolment id the registry lists them under.
160fn roster_enrolees(people: &[RosterPerson], linked: &LinkedAccounts<'_>) -> Vec<RosterEnrolee> {
161    let mut ids_by_person_id: HashMap<&str, Vec<String>> = HashMap::new();
162    let mut person_id_by_student_number: HashMap<&str, &str> = HashMap::new();
163    for person in people {
164        let person_id = person.person_id.expose_secret();
165        person_id_by_student_number.insert(person.student_number.expose_secret(), person_id);
166        let ids = ids_by_person_id.entry(person_id).or_default();
167        if let Some(id) = person
168            .enrolment
169            .as_ref()
170            .and_then(|enrolment| enrolment.id.clone())
171        {
172            ids.push(id);
173        }
174    }
175    linked
176        .rows
177        .iter()
178        .filter_map(|row| {
179            let person_id = expose_option(&row.sisu_person_id)
180                .filter(|person_id| ids_by_person_id.contains_key(person_id))
181                .or_else(|| {
182                    person_id_by_student_number
183                        .get(row.student_number.expose_secret())
184                        .copied()
185                })?;
186            Some(RosterEnrolee {
187                user_id: row.user_id,
188                enrolment_ids: ids_by_person_id.get(person_id).cloned().unwrap_or_default(),
189            })
190        })
191        .collect()
192}

194/// Claims a linking mail on each of `course_ids` for everyone in `people` we hold no link for, and
195/// counts each person once by the strongest claim any course got for them.
196async fn claim_linking_mails(
197    conn: &mut PgConnection,
198    people: &[&RosterPerson],
199    linked: &LinkedAccounts<'_>,
200    course_ids: &BTreeSet<Uuid>,
201) -> CreditRegistrationResult<AccountLinkingCodeCounters> {
202    let (mut counters, candidates) = linking_candidates(people, linked);
203    let mut strongest = vec![LinkingMailClaim::NoAddress; candidates.len()];
204    if !candidates.is_empty() {
205        for &course_id in course_ids {
206            let discovered: Vec<DiscoveredPerson> = candidates
207                .iter()
208                .map(|person| DiscoveredPerson::listed(person, course_id))
209                .collect();
210            let claims = claim_linking_mails_batch(conn, &discovered).await?;
211            for (person_strongest, claim) in strongest.iter_mut().zip(claims) {
212                *person_strongest = (*person_strongest).max(claim);
213            }
214        }
215    }
216    for claim in strongest {
217        match claim {
218            LinkingMailClaim::Claimed => counters.mailed_count += 1,
219            LinkingMailClaim::SuppressedByDedup => counters.suppressed_by_dedup_count += 1,
220            LinkingMailClaim::SuppressedByRateCap => counters.suppressed_by_rate_cap_count += 1,
221            LinkingMailClaim::NoAddress => {}
222        }
223    }
224    Ok(counters)
225}

227/// The people on a roster a linking mail may go to, and the counters of those it may not: already
228/// linked, or with no address to mail.
229fn linking_candidates<'a>(
230    people: &[&'a RosterPerson],
231    linked: &LinkedAccounts<'_>,
232) -> (AccountLinkingCodeCounters, Vec<&'a RosterPerson>) {
233    let mut counters = AccountLinkingCodeCounters {
234        listed_person_count: i32::try_from(people.len()).unwrap_or(i32::MAX),
235        ..AccountLinkingCodeCounters::default()
236    };
237    let mut candidates = Vec::new();
238    for &person in people {
239        if linked.is_linked(person) {
240            counters.already_linked_count += 1;
241        } else if mail_address(person).is_none() {
242            // The only genuinely unreachable population, and the reason it has a counter of its own.
243            counters.no_address_count += 1;
244        } else {
245            candidates.push(person);
246        }
247    }
248    (counters, candidates)
249}
