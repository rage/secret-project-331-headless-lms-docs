25/// The credits outside Sisu this attempt's grade competes with for the module: what we registered
26/// from the student's other attempts, which Suotar's copy of Sisu may predate, and the credits our
27/// records hold.
28pub(super) struct CompetingCredits {
29    registered_by_us: Vec<LiveSuccessForModule>,
30    recorded_locally: Vec<RecordedCredit>,
31}
32
33impl CompetingCredits {
34    /// Locks our registered successes only for a listed answer, the one that can freeze a payload
35    /// superseding them, and reads the recorded credits only for an enrolment error, the one answer
36    /// they can overrule. Call inside the transaction that writes the resolution.
37    pub(super) async fn load(
38        tx: &mut PgConnection,
39        lookup: &AnsweredLookup<'_>,
40    ) -> CreditRegistrationResult<Self> {
41        let row_id = lookup.claim().id();
42        let registered_by_us = match lookup.result() {
43            EnrolmentLookupResult::Listed { .. } => {
44                lock_live_successes_for_same_module(tx, row_id).await?
45            }
46            EnrolmentLookupResult::Refused { .. } => Vec::new(),
47        };
48        let recorded_locally = if lookup.result().is_enrolment_error() {
49            get_recorded_credits_for_same_module(tx, row_id).await?
50        } else {
51            Vec::new()
52        };
53        Ok(Self {
54            registered_by_us,
55            recorded_locally,
56        })
57    }
58
59    /// The registrations a frozen payload replaces once it is registered.
60    fn supersedes(&self) -> Vec<Uuid> {
61        self.registered_by_us
62            .iter()
63            .map(|replaced| replaced.id)
64            .collect()
65    }
66}

68/// What the answer comes to once Sisu is known not to hold the credit: nothing sent when our own
69/// records hold a credit the grade would not beat, a failure when there is nothing to register
70/// against, and otherwise the payload frozen for import.
71pub(super) fn resolve<'a>(
72    lookup: &AnsweredLookup<'a>,
73    competing: &CompetingCredits,
74) -> Decision<'a> {
75    let ours = lookup.our_grade();
76    let registered_grades: Vec<_> = competing
77        .registered_by_us
78        .iter()
79        .map(|success| success.credit.held_grade())
80        .collect();
81    if !registered_grades.is_empty() && !improves_on_all(&registered_grades, ours) {
82        return unsent_duplicate(
83            lookup,
84            "A grade at least as good is already registered for this module from another \
85             attempt, so nothing was submitted.",
86        );
87    }
88    // Suotar's copy of Sisu may predate a pull-path registration, so a student who holds the
89    // credit would be told to enrol again. A better grade still gets the error, since an
90    // improvement needs an enrolment too.
91    let recorded_grades: Vec<_> = competing
92        .recorded_locally
93        .iter()
94        .map(RecordedCredit::held_grade)
95        .collect();
96    if lookup.result().is_enrolment_error()
97        && !recorded_grades.is_empty()
98        && !improves_on_all(&recorded_grades, ours)
99    {
100        return unsent_duplicate(
101            lookup,
102            "The study registry has no usable enrolment, but our records already hold a credit at \
103             least as good for this module, so nothing was submitted.",
104        );
105    }
106    let facts = lookup.claim().facts(Utc::now());
107    let chosen = match lookup.result() {
108        EnrolmentLookupResult::Refused {
109            code,
110            error_message,
111        } => {
112            return Decision::new(submit_error_outcome(
113                EnrolmentLookup::OPERATION,
114                code,
115                &facts,
116            ))
117            .with_row_error(error_message);
118        }
119        EnrolmentLookupResult::Listed {
120            chosen: Err(reason),
121            ..
122        } => {
123            return Decision::new(submit_error_outcome(
124                EnrolmentLookup::OPERATION,
125                reason.error_code(),
126                &facts,
127            ))
128            .with_message(reason.message());
129        }
130        EnrolmentLookupResult::Listed {
131            chosen: Ok(chosen), ..
132        } => chosen,
133    };
134    let submission = lookup.submission();
135    let absent = DbSecret::new("");
136    let built = build_payload_snapshot(
137        &submission.completion,
138        PayloadSources {
139            student_number: submission.student_number.as_ref().unwrap_or(&absent),
140            sisu_person_id: submission.sisu_person_id.as_ref(),
141            uh_course_code: submission.uh_course_code.as_deref(),
142            ects_credits: submission.ects_credits,
143            enrolment: Some(chosen),
144        },
145    );
146    match built {
147        Ok(built) => freeze(built, competing.supersedes()),
148        Err(code) => Decision::new(submit_error_outcome(
149            EnrolmentLookup::OPERATION,
150            code,
151            &facts,
152        )),
153    }
154}

156/// The row is not sent, because a credit it would not beat is already held, and moves to
157/// `duplicate`. The grade we would have sent stays on the row as the one a later regrade has to
158/// beat.
159pub(super) fn unsent_duplicate<'a>(
160    lookup: &AnsweredLookup<'a>,
161    message: &'static str,
162) -> Decision<'a> {
163    Decision::new(Outcome::to(CreditRegistrationState::Duplicate))
164        .with_message(message)
165        .with_payload(PayloadChange::Unsent {
166            weighed_grade: map_grade(lookup.our_grade()).ok(),
167        })
168}

170/// Only now does the row become claimable by `import`: the payload is frozen and the event records
171/// when the enrolment was resolved.
172fn freeze(built: BuiltPayload, supersedes: Vec<Uuid>) -> Decision<'static> {
173    let message = freeze_message(&built, &supersedes);
174    let decision = Decision::new(Outcome::to(CreditRegistrationState::CheckingEnrolment))
175        .with_payload(PayloadChange::Frozen {
176            snapshot: Box::new(built.snapshot),
177            supersedes,
178        });
179    match message {
180        Some(message) => decision.with_message(message),
181        None => decision,
182    }
183}
184
185/// The timeline line for a freeze: that it replaces another attempt's registered grade, and that
186/// its credits were adjusted to fit the enrolment.
187fn freeze_message(built: &BuiltPayload, supersedes: &[Uuid]) -> Option<String> {
188    let superseded_message = (!supersedes.is_empty()).then(|| {
189        format!(
190            "This attempt's grade {} beats the one registered from another attempt, which this \
191             one supersedes once it is registered.",
192            built.snapshot.grade_id
193        )
194    });
195    let clamped_message = built.clamped_credits_from.map(|from| {
196        format!(
197            "Credits adjusted from {from} to {} to fit the enrolment's range.",
198            built.snapshot.credits
199        )
200    });
201    [superseded_message, clamped_message]
202        .into_iter()
203        .flatten()
204        .reduce(|first, second| format!("{first} {second}"))
205}

268    /// The frozen payload and the registrations it supersedes.
269    fn frozen(decision: Decision<'_>) -> (PayloadSnapshot, Vec<Uuid>) {
270        assert_eq!(decision.outcome.to_state, State::CheckingEnrolment);
271        match decision.atomic.payload {
272            Some(PayloadChange::Frozen {
273                snapshot,
274                supersedes,
275            }) => (*snapshot, supersedes),
276            other => panic!("not frozen: {other:?}"),
277        }
278    }
