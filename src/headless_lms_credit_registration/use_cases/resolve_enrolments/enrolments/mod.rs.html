44/// What a row's lookup was built from: the submission context, and the criteria the lookup's
45/// enrolment must fit. The answer is applied against what was asked, not against a second read of
46/// the database.
47pub(super) struct LookupBasis {
48    submission: SubmissionContext,
49    criteria: EnrolmentCriteria,
50    /// Sisu holds one of the equivalent older codes of the row's code, which counts as holding the
51    /// code itself.
52    holds_equivalent: bool,
53}
54
55/// A row claimed for its enrolment lookup.
56pub(super) type Resolvable = Claimed<LookupBasis>;
57
58impl RegistryBatchFlow for ResolveEnrolments {
59    type Extra = LookupBasis;
60    type Request = EnrolmentLookup;
61
62    const ALL_UNAVAILABLE_ERROR: &'static str = "Every item of the batch came back unavailable.";
63    const REFUSAL: RefusalPolicy<LookupBasis> = RefusalPolicy::RequestLevel;
64
65    async fn claim(
66        ctx: &BatchFlowContext<'_>,
67        conn: &mut PgConnection,
68        limit: usize,
69    ) -> CreditRegistrationResult<Prepared<LookupBasis, EnrolmentLookup>> {
70        let claimed =
71            claim_due_for_resolve(conn, ctx.scope, i64::try_from(limit).unwrap_or(i64::MAX))
72                .await?;
73        let ids: Vec<_> = claimed.iter().map(|row| row.id).collect();
74        let mut contexts = get_submission_contexts(conn, &ids).await?;
75        let holdings = get_equivalent_holdings(conn, &ids).await?;
76
77        let mut prepared = Prepared::new();
78        for row in claimed {
79            let Some(context) = contexts.remove(&row.id) else {
80                warn!(
81                    credit_registration_id = %row.id,
82                    "Credit registration has no completion or module to submit for"
83                );
84                let claim = ClaimedRegistration::left_in_place(row);
85                let applied =
86                    write_unasked_move(conn, &claim, missing_context(&claim.facts(Utc::now())))
87                        .await?;
88                prepared.record_applied(claim.id(), applied);
89                continue;
90            };
91            // Left for the next iteration's person lookup rather than frozen without the person.
92            if context.student_number.is_some() && context.sisu_person_id.is_none() {
93                debug!(
94                    credit_registration_id = %row.id,
95                    "Leaving row claimable: waiting for a Sisu person id"
96                );
97                continue;
98            }
99            let holding = holdings.get(&row.id);
100            if context.student_number.is_some() && holding == Some(&EquivalentHolding::Unchecked) {
101                debug!(
102                    credit_registration_id = %row.id,
103                    "Leaving row claimable: Sisu has not been asked about its equivalent codes"
104                );
105                continue;
106            }
107            match enrolment_lookup(&context) {
108                Ok(Askable { request, criteria }) => {
109                    let row = Claimed {
110                        claim: claim_for_lookup(row),
111                        extra: LookupBasis {
112                            submission: context,
113                            criteria,
114                            holds_equivalent: matches!(
115                                holding,
116                                Some(EquivalentHolding::Held { .. })
117                            ),
118                        },
119                    };
120                    prepared.send(row, request);
121                }
122                Err(problem) => {
123                    match &problem {
124                        Unaskable::Config(code) => {
125                            warn!(
126                                credit_registration_id = %row.id,
127                                error_code = ?code,
128                                "Course module is not configured for credit registration"
129                            );
130                        }
131                        Unaskable::NoStudentNumber => {
132                            debug!(
133                                credit_registration_id = %row.id,
134                                "No verified student number; sending back to pending"
135                            );
136                        }
137                    }
138                    // The claim holds the row's lock, so the guard only confirms the state it read.
139                    let claim = ClaimedRegistration::left_in_place(row);
140                    let applied = write_unasked_move(conn, &claim, problem.unasked_move()).await?;
141                    prepared.record_applied(claim.id(), applied);
142                }
143            }
144        }
145        hold_in_flight(
146            conn,
147            prepared.sendable().iter().map(|entry| &entry.row.claim),
148        )
149        .await?;
150        Ok(prepared)
151    }

153    async fn apply_answer(
154        conn: &mut PgConnection,
155        resolvable: &Resolvable,
156        answer: Option<&EnrolmentAnswer>,
157        audit: &ExchangeAudit,
158    ) -> CreditRegistrationResult<Applied> {
159        let claim = &resolvable.claim;
160        let row = claim.registration();
161        let enrolments = answer
162            .map(|answer| answer.enrolments.as_slice())
163            .unwrap_or_default();
164        let chosen = select_enrolment(enrolments, resolvable.extra.criteria);
165        let check = row
166            .enrolment_check_anchor_at
167            .is_some()
168            .then(|| EnrolmentCheckAnswer {
169                checked: row,
170                usable_enrolment: chosen.ok(),
171                listed_enrolments: enrolments,
172            });
173        let Some(answer) = answer else {
174            let outcome = unanswered_item_outcome(
175                EnrolmentLookup::OPERATION,
176                row.state,
177                &claim.facts(Utc::now()),
178            );
179            let decision = Decision::new(outcome)
180                .with_message("Sisu did not answer for this item.")
181                .with_enrolment_check(check);
182            return write_decision(conn, claim, decision, audit).await;
183        };
184        let lookup = AnsweredLookup::new(
185            resolvable,
186            EnrolmentLookupResult::read(answer, chosen),
187            &answer.existing_attainments,
188        );
189        apply_enrolment_answer(
190            conn,
191            &lookup,
192            resolvable.extra.holds_equivalent,
193            check,
194            audit,
195        )
196        .await
197    }

199    async fn keep_in_flight(
200        conn: &mut PgConnection,
201        rows: &[&Resolvable],
202    ) -> CreditRegistrationResult<()> {
203        keep_lookups_in_flight(conn, rows.iter().map(|row| &row.claim)).await
204    }

207/// Writes an answered lookup: settles the row as `duplicate` when a credit it would not beat is
208/// already held, fails it when there is nothing to register against, and otherwise freezes the
209/// payload and queues the row for import.
210///
211/// What Sisu itself holds is weighed first: if the attainment exists the credit does, so sending
212/// the student off to re-enrol would be wrong as well as unnecessary. An attainment of an
213/// equivalent older code counts as one of the row's own; its grade is not compared.
214async fn apply_enrolment_answer(
215    conn: &mut PgConnection,
216    lookup: &AnsweredLookup<'_>,
217    holds_equivalent: bool,
218    check: Option<EnrolmentCheckAnswer<'_>>,
219    audit: &ExchangeAudit,
220) -> CreditRegistrationResult<Applied> {
221    if holds_equivalent {
222        let decision = unsent_duplicate(
223            lookup,
224            "Sisu already has an older course that counts as this one, so nothing was submitted.",
225        )
226        .with_enrolment_check(check);
227        return write_decision(conn, lookup.claim(), decision, audit).await;
228    }
229    // Before the transaction below: a Sisu attainment must not be written inside one.
230    if let Some(attained) = lookup.held_in_sisu() {
231        let decision = unsent_duplicate(
232            lookup,
233            "Sisu already has an equal or better grade for this course, so nothing was submitted.",
234        )
235        .with_sisu_attainment(Some(attained))
236        .with_enrolment_check(check);
237        return write_decision(conn, lookup.claim(), decision, audit).await;
238    }
239
240    // Our other attempts' successes are locked in the transaction that freezes the payload, so two
241    // attempts cannot both decide theirs is the grade to send.
242    let mut tx = conn.begin().await?;
243    let competing = CompetingCredits::load(&mut tx, lookup).await?;
244    let decision = resolve(lookup, &competing).with_enrolment_check(check);
245    write_decision_committing_if_written(tx, lookup.claim(), decision, audit).await
246}
