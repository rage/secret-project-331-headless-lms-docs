1//! Asks Sisu, through the enrolment lookup's existing attainments, whether students hold course
2//! codes no ledger row of theirs is under: the equivalent older codes of a mooc.fi row's code,
3//! which the enrolment lookup then settles the row against, and the prerequisite of a row waiting
4//! for it, with that code's equivalents. Runs after the enrolment lookup, which takes up a row it
5//! skipped for unasked equivalents on a later tick.
6
7use headless_lms_models::credit_registration_sisu_code_checks::{self, SisuCodeAnswer, get_due};
8use headless_lms_models::library::credit_registration::enrolment_selection::attained_candidates;
9use secrecy::ExposeSecret;
10use uuid::Uuid;
11
12use crate::error::CreditRegistrationResult;
13use crate::registry::{
14    BatchEntry, BatchOptions, BatchReply, CourseCode, EnrolmentLookup, RegistryOperation,
15    StudentNumber, StudyRegistry,
16};
17use crate::use_cases::batch_flow::BatchFlowContext;
18use crate::workflow::Counts;
19
20/// The student and code one lookup is about.
21struct CodeCheck {
22    user_id: Uuid,
23    course_code: String,
24}
25
26/// Asks about up to the registry's allowance of due student and code pairs and records each answer.
27pub(super) async fn run<R: StudyRegistry>(
28    ctx: &BatchFlowContext<'_>,
29    registry: &mut R,
30) -> CreditRegistrationResult<Counts> {
31    let limit = registry.allowance(RegistryOperation::ResolveEnrolments);
32    if limit == 0 {
33        return Ok(Counts::default());
34    }
35    let due = {
36        let mut conn = ctx.pool.acquire().await?;
37        get_due(
38            &mut conn,
39            ctx.scope,
40            i64::try_from(limit).unwrap_or(i64::MAX),
41        )
42        .await?
43    };
44    let mut registration_ids = Vec::new();
45    let mut entries = Vec::new();
46    for check in due {
47        // A code Suotar could not be asked about is left unasked, and its rows wait.
48        let Some(course_code) = CourseCode::parse(&check.course_code) else {
49            continue;
50        };
51        registration_ids.extend(check.credit_registration_ids);
52        entries.push(BatchEntry {
53            row: CodeCheck {
54                user_id: check.user_id,
55                course_code: check.course_code,
56            },
57            request: EnrolmentLookup {
58                student_number: StudentNumber::new(
59                    check.student_number.expose_secret().to_string(),
60                ),
61                course_code,
62            },
63        });
64    }
65    if entries.is_empty() {
66        return Ok(Counts::default());
67    }
68    let options = BatchOptions {
69        may_split: false,
70        is_resent_half: false,
71        all_unavailable_error: "Every Sisu course code check came back unavailable.",
72        registration_ids,
73    };
74    let BatchReply::Answered(rows) = registry.resolve_enrolments(entries, options).await else {
75        // Unrecorded, so the same pairs are due again next tick.
76        return Ok(Counts::default());
77    };
78    let answers: Vec<SisuCodeAnswer> = rows
79        .into_iter()
80        // Any answer settles the question for now, a refusal such as an unknown old code included.
81        .filter_map(|row| {
82            let answer = row.answer?;
83            Some(SisuCodeAnswer {
84                user_id: row.row.user_id,
85                course_code: row.row.course_code,
86                is_attained: !attained_candidates(&answer.existing_attainments).is_empty(),
87            })
88        })
89        .collect();
90    let recorded = i64::try_from(answers.len()).unwrap_or(i64::MAX);
91    let mut conn = ctx.pool.acquire().await?;
92    credit_registration_sisu_code_checks::record_all(&mut conn, &answers).await?;
93    Ok(Counts::processed(recorded))
94}
