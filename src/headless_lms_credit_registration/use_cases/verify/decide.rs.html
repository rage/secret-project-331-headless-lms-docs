19/// What a poll's answer comes to, where that needs no more than the answer.
20pub(super) enum PollAnswer<'a> {
21    Decided(Box<Decision<'a>>),
22    /// Only the assessment item attainment is there yet; the outcome depends on when a poll first
23    /// saw that.
24    PartiallyRegistered,
25    /// Suotar has no trace of the submission; the outcome depends on how often that happened.
26    NotRegistered,
27}
28
29/// Only a course unit attainment registers the row, and only `notRegistered` sends it back towards
30/// import; anything else keeps it polling.
31pub(super) fn decide_poll<'a>(
32    state: CreditRegistrationState,
33    answer: Option<&'a VerificationAnswer>,
34    facts: &RowFacts,
35) -> PollAnswer<'a> {
36    let Some(answer) = answer else {
37        return PollAnswer::Decided(Box::new(still_polling_decision(state, facts, None)));
38    };
39    let decision = match &answer.reading {
40        VerificationReading::Registered { attainment } => Decision::new(Outcome {
41            // Confirmed, so whatever an operator was asked to look at is settled.
42            needs_admin_attention: Some(AdminAttention::Clear),
43            ..Outcome::to(CreditRegistrationState::Registered)
44        })
45        .with_sisu_attainment(Some(attainment)),
46        VerificationReading::PartiallyRegistered => return PollAnswer::PartiallyRegistered,
47        // `submissionPending`: polled on as usual, since the attainment usually shows up long
48        // before `retryAfter`, which only bounds when a resubmission becomes safe.
49        VerificationReading::Pending {
50            resubmit_not_before,
51        } => still_polling_decision(
52            state,
53            facts,
54            Some("Sisu is still processing the submission."),
55        )
56        .with_resubmit_not_before(*resubmit_not_before),
57        VerificationReading::NotRegistered => return PollAnswer::NotRegistered,
58        VerificationReading::Failed { code } => {
59            Decision::new(verify_error_outcome(state, *code, facts))
60        }
61        VerificationReading::Inconclusive => still_polling_decision(state, facts, None),
62    };
63    PollAnswer::Decided(Box::new(decision))
64}

66/// Keeps polling in place; past the verify window the expiry is the news, so it replaces
67/// `message`.
68fn still_polling_decision<'a>(
69    state: CreditRegistrationState,
70    facts: &RowFacts,
71    message: Option<&'static str>,
72) -> Decision<'a> {
73    let decision = Decision::new(verify_inconclusive_outcome(state, facts));
74    if verify_window_expired(facts.submitted_at, facts.now) {
75        decision.with_message(format!(
76            "Not confirmed within {} days of sending; now checked once a day.",
77            VERIFY_MAX_AGE.num_days()
78        ))
79    } else if let Some(message) = message {
80        decision.with_message(message)
81    } else {
82        decision
83    }
84}

86/// A poll that found only the assessment item attainment, which a poll first saw at
87/// `partially_registered_at`.
88pub(super) fn partially_registered_decision(
89    facts: &RowFacts,
90    partially_registered_at: DateTime<Utc>,
91) -> Decision<'static> {
92    Decision::new(verify_partial_outcome(facts, partially_registered_at)).with_message(
93        "Sisu has the assessment item attainment; waiting for the course unit attainment before \
94         counting it registered.",
95    )
96}

98/// A poll that found no trace of the submission.
99pub(super) fn not_registered_decision(facts: &RowFacts, reimport_count: i32) -> Decision<'static> {
100    Decision::new(verify_not_registered_outcome(facts, reimport_count))
101        .with_message("Sisu has no trace of the submission, so it will be sent again.")
102}

104/// Settles an uncertain row as `duplicate` if the lookup found the attainment it would have created,
105/// and otherwise leaves it uncertain.
106pub(super) fn decide_recovery<'a>(
107    row: &CreditRegistration,
108    facts: &RowFacts,
109    answer: Option<&'a EnrolmentAnswer>,
110) -> Decision<'a> {
111    // An enrolment error still lists the attainments, and the enrolment may be gone by now.
112    let found = answer
113        .zip(row.attainment_date)
114        .zip(row.frozen_grade())
115        .and_then(|((answer, attainment_date), grade)| {
116            attainment_matching_submission(
117                &answer.existing_attainments,
118                attainment_date,
119                row.submitted_at,
120                &grade,
121            )
122        });
123    let Some(attainment) = found else {
124        return Decision::new(uncertain_recheck_outcome(facts)).with_message(
125            "No matching attainment yet, so whether the submission landed is still unknown.",
126        );
127    };
128    Decision::new(Outcome {
129        needs_admin_attention: Some(AdminAttention::Clear),
130        ..Outcome::to(CreditRegistrationState::Duplicate)
131    })
132    .with_sisu_attainment(Some(attainment))
133    .with_message(
134        "The credits this submission would have created are in Sisu, so it was registered after \
135         all.",
136    )
137}

165    fn decided<'a>(state: State, answer: Option<&'a VerificationAnswer>) -> Decision<'a> {
166        match decide_poll(state, answer, &facts()) {
167            PollAnswer::Decided(decision) => *decision,
168            PollAnswer::PartiallyRegistered => panic!("partially registered"),
169            PollAnswer::NotRegistered => panic!("not registered"),
170        }
171    }
