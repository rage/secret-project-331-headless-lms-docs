1//! The account-linking action an admin or a teacher sets off by hand: re-running the linking mail
2//! send path for one person on one course or mooc.fi target. Not a phase: no schedule and no
3//! allowance, and no breaker learns from its calls, so one click cannot trip the workers'.
4//!
5//! The addresses come from the study registry rather than the ledger, and the claim goes through
6//! [`claim_linking_mail`], so the caps and dedup guard apply exactly as they do to the worker.

30/// What one resend attempt came to.
31#[derive(Debug, Clone, Copy, PartialEq, Eq)]
32pub enum ResendOutcome {
33    /// A slot was claimed; the `link-emails` phase queues the message on its next run.
34    Queued,
35    /// Every address the study registry holds for them has already had its mail for this course.
36    AlreadyMailedToEveryKnownAddress,
37    /// A cap refused it: either the quiet period or the per-course lifetime limit.
38    RefusedByRateCap,
39    /// An override left a mail claimed moments ago in place, and the caps counting it refused.
40    JustSent,
41    NoAddressInStudyRegistry,
42    /// The study registry does not list them under any course code of this course.
43    NotOnTheCourseRoster,
44    /// The number is already linked to an account, so no linking mail is owed.
45    AlreadyLinked,
46    /// We could not ask the study registry, so nothing was decided.
47    StudyRegistryUnavailable,
48}

50/// The outcome of one resend attempt, plus how many capped mails an override retired to get there.
51pub struct ResendAttempt {
52    pub outcome: ResendOutcome,
53    /// Always zero without an override; only the admin-facing endpoint can pass one.
54    pub retired_mail_count: i64,
55    /// Where the queued mail goes; `None` unless `outcome` is `Queued`.
56    pub emailed_to: Option<SecretString>,
57}

59/// An admin's go-ahead to get past the linking-mail caps by retiring the mails they count; see
60/// [`retire_capped_mails`].
61pub struct RateCapOverride<'a> {
62    pub actor_user_id: Uuid,
63    pub actor_role: &'a str,
64    pub reason: &'a str,
65}

67/// [`crate::account_linking::resend_linking_mail_for_target`] through `registry`.
68pub(crate) async fn resend_for_target<R: InteractiveStudyRegistry>(
69    pool: &PgPool,
70    registry: &R,
71    source: LinkingMailSource,
72    student_number: &SecretString,
73    rate_cap_override: Option<RateCapOverride<'_>>,
74) -> CreditRegistrationResult<ResendAttempt> {
75    let already_linked = {
76        let mut conn = pool.acquire().await?;
77        verified_student_numbers::get_by_student_number(&mut conn, student_number.expose_secret())
78            .await?
79            .is_some()
80    };
81    if already_linked {
82        return Ok(ResendAttempt {
83            outcome: ResendOutcome::AlreadyLinked,
84            retired_mail_count: 0,
85            emailed_to: None,
86        });
87    }
88    resend_linking_mail(
89        pool,
90        registry,
91        source,
92        student_number,
93        rate_cap_override.as_ref(),
94    )
95    .await
96}

98/// Claims a linking mail for the one person on the course's or target's roster with this student
99/// number.
100///
101/// The override retires the capped mails only once the registry has named an address to mail, in
102/// the claim's transaction, so a lookup that decides nothing leaves the caps in place.
103async fn resend_linking_mail<R: InteractiveStudyRegistry>(
104    pool: &PgPool,
105    registry: &R,
106    source: LinkingMailSource,
107    student_number: &SecretString,
108    rate_cap_override: Option<&RateCapOverride<'_>>,
109) -> CreditRegistrationResult<ResendAttempt> {
110    let not_retired = |outcome| ResendAttempt {
111        outcome,
112        retired_mail_count: 0,
113        emailed_to: None,
114    };
115    let course_codes = source_course_codes(pool, source).await?;
116    if course_codes.is_empty() {
117        return Ok(not_retired(ResendOutcome::NotOnTheCourseRoster));
118    }
119
120    let search = registry
121        .search_course_rosters(&course_codes, &StudentNumber::new(student_number.clone()))
122        .await;
123    let Some(person) = search.person else {
124        return Ok(not_retired(if search.has_unanswered_code {
125            ResendOutcome::StudyRegistryUnavailable
126        } else {
127            ResendOutcome::NotOnTheCourseRoster
128        }));
129    };
130
131    let discovered = DiscoveredPerson::listed(&person, source);
132    if discovered.address.is_none() {
133        return Ok(not_retired(ResendOutcome::NoAddressInStudyRegistry));
134    }
135
136    let mut conn = pool.acquire().await?;
137    let mut tx = conn.begin().await?;
138    let retired = match rate_cap_override {
139        Some(rate_cap_override) => {
140            retire_capped_mails(
141                &mut tx,
142                rate_cap_override.actor_user_id,
143                rate_cap_override.actor_role,
144                source,
145                student_number.expose_secret(),
146                rate_cap_override.reason,
147            )
148            .await?
149        }
150        None => RetiredMails {
151            retired_count: 0,
152            kept_just_claimed: false,
153        },
154    };
155    let claim = claim_linking_mail(&mut tx, &discovered).await?;
156    tx.commit().await?;
157    debug!(source_id = %source.id(), ?claim, "Linking mail resend claim result");
158    let outcome = match claim {
159        LinkingMailClaim::Claimed => ResendOutcome::Queued,
160        _ if retired.kept_just_claimed => ResendOutcome::JustSent,
161        LinkingMailClaim::SuppressedByRateCap => ResendOutcome::RefusedByRateCap,
162        LinkingMailClaim::SuppressedByDedup => ResendOutcome::AlreadyMailedToEveryKnownAddress,
163        LinkingMailClaim::NoAddress => ResendOutcome::NoAddressInStudyRegistry,
164    };
165    let emailed_to = (outcome == ResendOutcome::Queued)
166        .then(|| discovered.address.clone().map(SecretString::from))
167        .flatten();
168    Ok(ResendAttempt {
169        outcome,
170        retired_mail_count: retired.retired_count,
171        emailed_to,
172    })
173}

175/// The distinct course codes of the course's modules that take part in credit registration.
176async fn active_course_codes(
177    pool: &PgPool,
178    course_id: Uuid,
179) -> CreditRegistrationResult<Vec<CourseCode>> {
180    let mut conn = pool.acquire().await?;
181    Ok(get_active_modules_for_course(&mut conn, course_id)
182        .await?
183        .iter()
184        .filter_map(|module| CourseCode::parse(&module.uh_course_code))
185        .collect::<BTreeSet<_>>()
186        .into_iter()
187        .collect())
188}
189
190/// The distinct course codes `source` mails about; none for a paused or disabled target.
191async fn source_course_codes(
192    pool: &PgPool,
193    source: LinkingMailSource,
194) -> CreditRegistrationResult<Vec<CourseCode>> {
195    match source {
196        LinkingMailSource::Course(course_id) => active_course_codes(pool, course_id).await,
197        LinkingMailSource::ExternalCreditTarget(id) => {
198            let mut conn = pool.acquire().await?;
199            let target = external_credit_targets::get_by_id(&mut conn, id).await?;
200            if !target.is_active() {
201                return Ok(Vec::new());
202            }
203            Ok(
204                external_credit_target_codes::get_by_target_id(&mut conn, id)
205                    .await?
206                    .iter()
207                    .filter_map(|code| CourseCode::parse(&code.uh_course_code))
208                    .sorted()
209                    .dedup()
210                    .collect(),
211            )
212        }
213    }
214}

216/// What listing a course's or mooc.fi target's rosters came to.
217pub enum EnroleeListing {
218    /// No active module or target with a course code, so there is no roster to ask for.
219    NoCourseCodes,
220    /// The study registry gave no usable answer for some code.
221    StudyRegistryUnavailable,
222    Listed(Vec<ListedEnrolee>),
223}

225/// [`crate::account_linking::list_unlinked_on_course_rosters`] through `registry`.
226pub(crate) async fn unlinked_on_course_rosters<R: InteractiveStudyRegistry>(
227    pool: &PgPool,
228    registry: &R,
229    course_id: Uuid,
230) -> CreditRegistrationResult<Option<Vec<RosterPerson>>> {
231    Ok(
232        match enrolees_of_source(pool, registry, LinkingMailSource::Course(course_id)).await? {
233            EnroleeListing::NoCourseCodes => Some(Vec::new()),
234            EnroleeListing::StudyRegistryUnavailable => None,
235            EnroleeListing::Listed(enrolees) => Some(
236                enrolees
237                    .into_iter()
238                    .filter(|enrolee| enrolee.linked_user_id.is_none())
239                    .map(|enrolee| enrolee.person)
240                    .collect(),
241            ),
242        },
243    )
244}

246/// [`crate::account_linking::list_enrolees`] through `registry`.
247pub(crate) async fn enrolees_of_source<R: InteractiveStudyRegistry>(
248    pool: &PgPool,
249    registry: &R,
250    source: LinkingMailSource,
251) -> CreditRegistrationResult<EnroleeListing> {
252    let codes = source_course_codes(pool, source).await?;
253    if codes.is_empty() {
254        return Ok(EnroleeListing::NoCourseCodes);
255    }
256    let mut rosters = Vec::new();
257    for code in codes {
258        let Some(listed) = registry.fetch_course_roster(&code).await else {
259            return Ok(EnroleeListing::StudyRegistryUnavailable);
260        };
261        rosters.push((code.as_str().to_owned(), listed));
262    }
263    let mut conn = pool.acquire().await?;
264    Ok(EnroleeListing::Listed(
265        list_with_linked_accounts(&mut conn, &rosters).await?,
266    ))
267}

269/// [`crate::account_linking::list_unlinked_enrolled_before`] through `registry`.
270pub(crate) async fn unlinked_enrolled_before<R: InteractiveStudyRegistry>(
271    pool: &PgPool,
272    registry: &R,
273    course_code: &str,
274    since: DateTime<Utc>,
275) -> CreditRegistrationResult<Option<Vec<RosterPerson>>> {
276    let Some(code) = CourseCode::parse(course_code) else {
277        return Ok(Some(Vec::new()));
278    };
279    let Some(people) = registry.fetch_course_roster(&code).await else {
280        return Ok(None);
281    };
282    let mut conn = pool.acquire().await?;
283    Ok(Some(
284        list_unlinked_enrolled_before(&mut conn, &people, since).await?,
285    ))
