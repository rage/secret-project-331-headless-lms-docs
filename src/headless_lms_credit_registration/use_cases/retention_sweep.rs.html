1//! The `retention-sweep` phase: clearing out records past their retention window.
2
3use chrono::{Duration, Utc};
4use headless_lms_models::credit_registration_enrolment_check_outcomes;
5use headless_lms_models::credit_registration_events::redact_details_older_than;
6use headless_lms_models::student_number_verification_tokens::soft_delete_expired;
7use headless_lms_models::suotar_api_calls::{RETENTION_DAYS, delete_older_than};
8use sqlx::PgPool;
9
10use crate::error::CreditRegistrationResult;
11use crate::workflow::Counts;
12
13/// How much one retention sweep removes from each table.
14const SWEEP_LIMIT: i64 = 500;
15/// Above the tens of thousands of checks a day the hourly sweep has to keep up with.
16const OUTCOME_SWEEP_LIMIT: i64 = 5000;
17/// Each enrolment check answer writes an event, so one batch an hour would fall behind the checks;
18/// separate transactions keep each batch's row locks short.
19const MAX_REDACTION_BATCHES: usize = 20;
20
21/// The call log's 90-day window, which also ends the unredacted life of the ledger's copies of those
22/// exchanges, the enrolment check log's own longer window, and the expired linking tokens.
23///
24/// Bounded per iteration and run hourly, so the first sweep after the window opens clears a backlog
25/// over several hours instead of in one statement that locks every `credit_registration_events` row
26/// referencing it.
27pub(crate) async fn run(pool: &PgPool) -> CreditRegistrationResult<Counts> {
28    let mut conn = pool.acquire().await?;
29    let cutoff = Utc::now() - Duration::days(RETENTION_DAYS);
30    let purged_calls = delete_older_than(&mut conn, cutoff, SWEEP_LIMIT).await?;
31    let mut redacted_events = 0;
32    for _ in 0..MAX_REDACTION_BATCHES {
33        let redacted = redact_details_older_than(&mut conn, cutoff, SWEEP_LIMIT).await?;
34        redacted_events += redacted;
35        if redacted < SWEEP_LIMIT.unsigned_abs() {
36            break;
37        }
38    }
39    let purged_outcomes = credit_registration_enrolment_check_outcomes::delete_older_than(
40        &mut conn,
41        Utc::now() - Duration::days(credit_registration_enrolment_check_outcomes::RETENTION_DAYS),
42        OUTCOME_SWEEP_LIMIT,
43    )
44    .await?;
45    let retired_tokens = soft_delete_expired(&mut conn, SWEEP_LIMIT).await?;
46    if purged_calls > 0 || redacted_events > 0 || purged_outcomes > 0 || retired_tokens > 0 {
47        info!(
48            purged_calls,
49            redacted_events,
50            purged_outcomes,
51            retired_tokens,
52            "Retention sweep cleared expired credit registration records"
53        );
54    }
55    Ok(Counts::processed(
56        i64::try_from(purged_calls + redacted_events + purged_outcomes + retired_tokens)
57            .unwrap_or(i64::MAX),
58    ))
59}
