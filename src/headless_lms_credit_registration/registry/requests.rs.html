72/// ECTS credits as they go on the wire: finite, and rounded to the tenth credits never go finer
73/// than.
74#[derive(Debug, Clone, Copy, PartialEq)]
75pub(crate) struct Credits(f64);
76
77impl Credits {
78    /// `None` for a value that is not finite, which Suotar refuses. Rounding also removes the
79    /// f32-to-f64 widening error, which would send 2.7 as 2.700000047683716.
80    pub(crate) fn from_stored(credits: f32) -> Option<Self> {
81        credits
82            .is_finite()
83            .then(|| Self((f64::from(credits) * 10.0).round() / 10.0))
84    }
85
86    pub(crate) fn as_f64(self) -> f64 {
87        self.0
88    }
89}

120
121    #[test]
122    fn stored_credits_go_out_rounded_to_a_tenth_and_only_when_finite() {
123        assert_eq!(Credits::from_stored(2.7).map(Credits::as_f64), Some(2.7));
124        assert_eq!(Credits::from_stored(f32::NAN), None);
125        assert_eq!(Credits::from_stored(f32::INFINITY), None);
126    }
