1//! One batch request to the study registry and what came of it, row by row.
2
3use uuid::Uuid;
4
5use super::ids::StudentNumber;
6use super::{RegistryError, RegistryOperation, StudyRegistry};
7
8/// What one item of a batch operation asks: the item decides the operation it goes out under and the
9/// answer it gets back.
10pub(crate) trait BatchRequest: Sized {
11    const OPERATION: RegistryOperation;
12    type Answer;
13
14    /// Sends `entries` in one request of [`Self::OPERATION`].
15    async fn send<Registry: StudyRegistry, K>(
16        registry: &mut Registry,
17        entries: Vec<BatchEntry<K, Self>>,
18        options: BatchOptions,
19    ) -> BatchReply<K, Self, Self::Answer>;
20}
21
22/// One claimed row of a batch and the item it asks.
23pub(crate) struct BatchEntry<K, R> {
24    pub row: K,
25    pub request: R,
26}
27
28/// How one batch request is sent and accounted for.
29#[derive(Debug, Clone)]
30pub(crate) struct BatchOptions {
31    /// Every row takes the shared request-level refusal, so a malformed-request refusal of a batch
32    /// of several rows comes back as [`BatchReply::RefusedAsMalformed`] for the caller to split.
33    pub may_split: bool,
34    /// A half of a batch refused as malformed, which the limiter lets through past its allowance.
35    pub is_resent_half: bool,
36    /// The iteration's error when every item comes back unavailable.
37    pub all_unavailable_error: &'static str,
38    /// The rows' registrations, which the call is tagged with in the audit log.
39    pub registration_ids: Vec<Uuid>,
40}
41
42/// What came of one batch request.
43pub(crate) enum BatchReply<K, R, A> {
44    /// Every row, in request order; a row the registry did not answer has no answer.
45    Answered(Vec<AnsweredRow<K, A>>),
46    /// The request failed as a whole.
47    Refused {
48        rows: Vec<RefusedRow<K>>,
49        error: RegistryError,
50        refused_for: RefusedFor,
51    },
52    /// A splittable batch of several rows refused as malformed. Suotar validates every item before
53    /// acting on any, so nothing was acted on, and no breaker counts it; the rows come back as
54    /// sent, for the caller to resend in halves.
55    RefusedAsMalformed {
56        entries: Vec<BatchEntry<K, R>>,
57        error: RegistryError,
58    },
59}
60
61/// Whose a whole-request refusal is.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub(crate) enum RefusedFor {
64    WholeBatch,
65    /// A malformed request refused with the row alone in it, which resending cannot fix.
66    RowAlone,
67}
68
69pub(crate) struct AnsweredRow<K, A> {
70    pub row: K,
71    pub answer: Option<A>,
72    pub audit: ExchangeAudit,
73}
74
75pub(crate) struct RefusedRow<K> {
76    pub row: K,
77    pub audit: ExchangeAudit,
78}
79
80/// The exchange behind one row's answer, as its ledger event records it. No `Debug`: the bodies are
81/// unscrubbed, and are scrubbed only on their way into the event row.
82pub(crate) struct ExchangeAudit {
83    /// `suotar_api_calls.id`; `None` on a refusal, or when the call row could not be written.
84    pub call_id: Option<Uuid>,
85    /// What the row's item went out under in that request.
86    pub request_item_id: String,
87    pub request: serde_json::Value,
88    /// The row's item exactly as it arrived.
89    pub response: Option<serde_json::Value>,
90    /// The number the request carried, which may no longer be the linked one.
91    pub sent_student_number: Option<StudentNumber>,
92}
