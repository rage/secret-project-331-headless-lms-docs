7/// A student number sent to the study registry. Its `Debug` is redacted.
8#[derive(Debug)]
9pub(crate) struct StudentNumber(SecretString);
10
11impl StudentNumber {
12    pub(crate) fn new(number: impl Into<SecretString>) -> Self {
13        Self(number.into())
14    }
15
16    pub(crate) fn as_secret(&self) -> &SecretString {
17        &self.0
18    }
19
20    pub(crate) fn expose(&self) -> &str {
21        self.0.expose_secret()
22    }
23}

25/// A University of Helsinki course code as the study registry is asked about it: trimmed, and never
26/// empty.
27#[derive(Debug, Clone, PartialEq, Eq, Hash, PartialOrd, Ord)]
28pub(crate) struct CourseCode(String);
29
30impl CourseCode {
31    /// `None` for a code that is blank once trimmed.
32    pub(crate) fn parse(code: &str) -> Option<Self> {
33        let code = code.trim();
34        (!code.is_empty()).then(|| Self(code.to_string()))
35    }
36
37    pub(crate) fn as_str(&self) -> &str {
38        &self.0
39    }
40}
41
42impl Borrow<str> for CourseCode {
43    fn borrow(&self) -> &str {
44        &self.0
45    }
46}
