16/// What a manual action needs from the controller that runs it.
17pub struct ManualActionContext<'a> {
18    pool: &'a PgPool,
19    suotar_client: &'a SuotarClient,
20    caller: &'a str,
21}
22
23impl<'a> ManualActionContext<'a> {
24    /// `caller` is the `worker_name` the action's study registry calls are logged under, one per
25    /// endpoint that sets the action off, e.g. `admin-resend`.
26    pub fn new(pool: &'a PgPool, suotar_client: &'a SuotarClient, caller: &'a str) -> Self {
27        Self {
28            pool,
29            suotar_client,
30            caller,
31        }
32    }
33}
34
35/// Shared by the teacher- and admin-facing resend endpoints: refuses a target that is already
36/// linked, otherwise reruns the send path exactly as the worker would, applying
37/// `rate_cap_override`, if any, only once the registry has named an address to mail.
38pub async fn resend_linking_mail_for_target(
39    ctx: &ManualActionContext<'_>,
40    course_id: Uuid,
41    student_number: &SecretString,
42    rate_cap_override: Option<RateCapOverride<'_>>,
43) -> CreditRegistrationResult<ResendAttempt> {
44    let registry = InteractiveSuotar::new(ctx.suotar_client, ctx.caller.to_string());
45    account_linking::resend_for_target(
46        ctx.pool,
47        &registry,
48        course_id,
49        student_number,
50        rate_cap_override,
51    )
52    .await
53}

55/// Looks one student number up in the study registry without changing anything: no ledger row, no
56/// claimed mail slot, just the call log row every study registry call writes.
57///
58/// `Ok(None)` means the registry answered `personNotFound`; `Err` means it gave no usable answer.
59pub async fn look_up_person(
60    ctx: &ManualActionContext<'_>,
61    student_number: &SecretString,
62) -> Result<Option<RegistryPerson>, PersonLookupError> {
63    InteractiveSuotar::new(ctx.suotar_client, ctx.caller.to_string())
64        .look_up_person(&StudentNumber::new(student_number.clone()))
65        .await
66}

72pub async fn list_unlinked_enrolled_before(
