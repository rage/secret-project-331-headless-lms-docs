8/// A pipeline phase. [`CreditRegistrationPhase::as_str`] is canonical: it is
9/// `credit_registration_phase_state.phase`, the tick endpoint's `?phase=` and the audit log's
10/// `target_phase`.
11#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
12pub enum CreditRegistrationPhase {
13    Materialize,
14    Preconditions,
15    ResolveEnrolments,
16    Import,
17    Verify,
18    LegacyMirror,
19    StudentNotifications,
20    EnrolmentDiscovery,
21    LinkEmails,
22    ConfigValidation,
23    RetentionSweep,
24    LedgerSnapshot,
25}

27/// The worker process that runs a phase's loop. The two are separate OS processes, each with its
28/// own circuit breakers and limiter.
29#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
30pub enum WorkerProcess {
31    /// Owns the ledger.
32    CreditRegistrar,
33    /// Owns everything about credit registration except the ledger.
34    SuotarSyncer,
35}
36
37impl WorkerProcess {
38    pub const ALL: [Self; 2] = [Self::CreditRegistrar, Self::SuotarSyncer];
39
40    /// `credit_registration_phase_state.process_name`, and the caller the audit log names.
41    pub fn as_str(self) -> &'static str {
42        match self {
43            Self::CreditRegistrar => "credit-registrar",
44            Self::SuotarSyncer => "suotar-syncer",
45        }
46    }
47}

49/// Everything fixed about one phase, declared in one place.
50#[derive(Debug)]
51pub struct PhaseSpec {
52    /// See [`CreditRegistrationPhase::as_str`].
53    pub name: &'static str,
54    pub process: WorkerProcess,
55    pub scope: ScopeSupport,
56    /// What one iteration asks of the study registry, one after the other. Empty for a
57    /// database-only phase, which no breaker ever holds back.
58    /// `study_registry_endpoints` names them as Suotar endpoints.
59    pub operations: &'static [RegistryOperation],
60    /// The circuit breakers that pause the phase.
61    pub breakers: &'static [BreakerTarget],
62    /// The ledger states this phase is the one to move a row out of; see
63    /// [`CreditRegistrationPhase::owned_states`].
64    pub owned_states: &'static [CreditRegistrationState],
65    /// The phase does nothing but account linking, and so is skipped while the deployment has
66    /// linking switched off. `enrolment-discovery` is not one: with linking off it still wakes
67    /// linked students' registrations and only leaves out the mails.
68    pub is_account_linking_only: bool,
69}

201impl CreditRegistrationPhase {
202    /// Every phase, in pipeline order.
203    pub const ALL: [Self; 12] = [
204        Self::Materialize,
205        Self::Preconditions,
206        Self::ResolveEnrolments,
207        Self::Import,
208        Self::Verify,
209        Self::LegacyMirror,
210        Self::StudentNotifications,
211        Self::EnrolmentDiscovery,
212        Self::LinkEmails,
213        Self::ConfigValidation,
214        Self::RetentionSweep,
215        Self::LedgerSnapshot,
216    ];
217
218    /// Everything fixed about the phase.
219    pub fn spec(self) -> &'static PhaseSpec {
220        match self {
221            Self::Materialize => &MATERIALIZE,
222            Self::Preconditions => &PRECONDITIONS,
223            Self::ResolveEnrolments => &RESOLVE_ENROLMENTS,
224            Self::Import => &IMPORT,
225            Self::Verify => &VERIFY,
226            Self::LegacyMirror => &LEGACY_MIRROR,
227            Self::StudentNotifications => &STUDENT_NOTIFICATIONS,
228            Self::EnrolmentDiscovery => &ENROLMENT_DISCOVERY,
229            Self::LinkEmails => &LINK_EMAILS,
230            Self::ConfigValidation => &CONFIG_VALIDATION,
231            Self::RetentionSweep => &RETENTION_SWEEP,
232            Self::LedgerSnapshot => &LEDGER_SNAPSHOT,
233        }
234    }
235
236    /// The phase's canonical name; see [`CreditRegistrationPhase`].
237    pub fn as_str(self) -> &'static str {
238        self.spec().name
239    }
240
241    /// The phase [`Self::as_str`] names `name`, if any.
242    pub fn from_phase_name(name: &str) -> Option<Self> {
243        Self::ALL.into_iter().find(|phase| phase.as_str() == name)
244    }
245
246    /// Where the phase sits in [`Self::ALL`]'s pipeline order, for sorting phases by it.
247    pub fn pipeline_index(self) -> usize {
248        Self::ALL
249            .iter()
250            .position(|phase| *phase == self)
251            .unwrap_or(Self::ALL.len())
252    }
253
254    /// The ledger states this phase is the one to move a row out of.
255    ///
256    /// Empty for the phases whose work is not a ledger state at all: `materialize`'s queue is
257    /// completions with no row yet, and the syncer's phases work on course modules. Narrower than
258    /// what `preconditions` may claim, which is every non-terminal row: these are the states nothing
259    /// else advances. How many of their rows are waiting on the phase is [`Self::queue_depth`].
260    pub fn owned_states(self) -> &'static [CreditRegistrationState] {
261        self.spec().owned_states
262    }
263
264    /// The live rows waiting on this phase: the Workers tab's "queue depth it is responsible for",
265    /// and the depth the failing-phase alert asks about before calling a quiet phase wedged.
266    ///
267    /// `depth_of` is the live count of a state. `due_enrolment_checks` stands in for
268    /// `no_usable_enrolment`, whose other rows wait for their schedule rather than for the phase.
269    pub fn queue_depth(
270        self,
271        depth_of: impl Fn(CreditRegistrationState) -> i64,
272        due_enrolment_checks: i64,
273    ) -> i64 {
274        self.owned_states()
275            .iter()
276            .map(|&state| {
277                if state == CreditRegistrationState::NoUsableEnrolment {
278                    due_enrolment_checks
279                } else {
280                    depth_of(state)
281                }
282            })
283            .sum()
284    }
285}

287/// Which of the scope's dimensions a phase's claim query can apply. Declared rather than assumed,
288/// so a phase added later cannot quietly ignore a scope and sweep the whole database.
289#[derive(Debug, Clone, Copy, PartialEq, Eq)]
290pub struct ScopeSupport {
291    pub course: bool,
292    pub user: bool,
293    pub registration_ids: bool,
294}
295
296impl ScopeSupport {
297    pub const NONE: Self = Self {
298        course: false,
299        user: false,
300        registration_ids: false,
301    };
302    /// The phases that claim ledger rows, which carry all three keys themselves.
303    pub const LEDGER: Self = Self {
304        course: true,
305        user: true,
306        registration_ids: true,
307    };
308
309    pub(crate) fn covers(self, scope: &RegistrationScope) -> bool {
310        let requested_unsupported = (scope.course_id.is_some() && !self.course)
311            || (scope.user_id.is_some() && !self.user)
312            || (!scope.credit_registration_ids.is_empty() && !self.registration_ids);
313        !requested_unsupported
314    }
315}

317#[cfg(test)]
318mod tests {
319    use headless_lms_models::credit_registration_phase_state::PHASES;
320    use uuid::Uuid;
321
322    use super::*;
323
324    /// A mismatch with the seeded `credit_registration_phase_state` rows makes a tick or a
325    /// heartbeat silently target a row that does not exist.
326    #[test]
327    fn phase_names_match_the_seeded_rows() {
328        let from_enum: Vec<&str> = CreditRegistrationPhase::ALL
329            .iter()
330            .map(|phase| phase.as_str())
331            .collect();
332        assert_eq!(from_enum, PHASES);
333        assert_eq!(from_enum.len(), 12);
334    }
335
336    /// A phase that cannot honour the narrowing it was handed has to say so, or a caller that
337    /// believes it narrowed the run gets a silently wrong answer.
338    #[test]
339    fn a_phase_refuses_a_scope_it_cannot_apply() {
340        let ids = RegistrationScope {
341            credit_registration_ids: vec![Uuid::new_v4()],
342            ..RegistrationScope::default()
343        };
344        assert!(
345            !CreditRegistrationPhase::Materialize
346                .spec()
347                .scope
348                .covers(&ids)
349        );
350        assert!(CreditRegistrationPhase::Import.spec().scope.covers(&ids));
351        assert!(
352            !CreditRegistrationPhase::RetentionSweep
353                .spec()
354                .scope
355                .covers(&RegistrationScope::for_course(Uuid::new_v4()))
356        );
357    }
