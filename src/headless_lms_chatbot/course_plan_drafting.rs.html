41/// Name the course-level structured output goes by; the test-mode mock Azure API answers by it.
42pub const COURSE_RESPONSE_FORMAT_NAME: &str = "CoursePlanCourseDraft";

45/// Introduces the learning goals in the course-level prompt, one [`LIST_ITEM_PREFIX`] line each.

61
62Rules:
63- Write in the language whose code is given as "Course language", whatever language the plan is written in.
64- Use only what the plan says. Do not invent facts, numbers, names, dates, sources or claims about the course.
65- Follow the plan's writing style and terminology when it gives them.
66- Plain text only: no Markdown, no HTML."#;

68const COURSE_INSTRUCTIONS: &str = r#"Draft the course-level texts:
69- description: 2 to 3 sentences for the course catalog on what the course teaches and for whom.
70- hero_subtitle: one short sentence under the course name on the front page.
71- about_text: one paragraph for the "About this course" section of the front page.
72- learning_goal_headings: one short heading (at most 6 words) per course learning goal, in the order listed, summarising that goal.
73- how_to_study_paragraphs: 1 to 3 short paragraphs for the "How to study this course" page: how the course is organised into chapters and pages, the planned content formats and exercises, and that students can study at their own pace when the plan says the course is asynchronous. Do not mention workload numbers; they are listed separately."#;

75const CHATBOT_INSTRUCTIONS: &str = r#"- chatbot: the course plans an AI chatbot. Give it a short name, a system prompt that keeps it a helpful study companion for this course's topics and learning goals without doing students' exercises for them, a short first message to students, and 3 to 4 short example messages a student could send first."#;

77const CHAPTER_INSTRUCTIONS: &str = r#"Draft texts for one chapter of the course:
78- prior_learning_heading: a short heading for a box on the chapter's introduction page that activates what students already know about the chapter's topics.
79- prior_learning_questions: 2 to 3 short reflection questions for that box. They are not graded.
80- topic_outlines: one outline per topic of this chapter, in the order listed. Each outline has 2 to 5 sections, each with a short heading for a part of the topic's page and 1 to 4 short points on what that part should cover. Points name what to cover; they are notes for the teacher, not finished sentences for students. Keep within the chapter's learning objectives and the topic; do not repeat other topics."#;

82/// Drafts every text of a course from `source`. Fails as a whole when any request fails twice or
83/// the deadline passes, so a course is never created from partial drafts.
84pub async fn draft_course_from_plan(
85    app_config: &ApplicationConfiguration,
86    task_lm: &TaskLMSpec,
87    source: &CourseFromPlanSource,
88) -> ChatbotResult<CourseFromPlanDrafts> {
89    let plan_text = plan_text(source);
90    let drafting = async {
91        let course = draft_course(app_config, task_lm, source, &plan_text);
92        let chapters = futures::stream::iter(&source.design.chapters)
93            .map(|chapter| draft_chapter(app_config, task_lm, &plan_text, chapter))
94            .buffered(MAX_CONCURRENT_REQUESTS - 1)
95            .try_collect::<Vec<_>>();
96        futures::future::try_join(course, chapters).await
97    };
98    let (course, chapters) = tokio::time::timeout(DEADLINE, drafting)
99        .await
100        .map_err(|_| {
101            chatbot_err!(
102                FailedAzureResponse,
103                "Drafting the course with AI took too long.".to_string()
104            )
105        })??;
106    Ok(CourseFromPlanDrafts { course, chapters })
107}

109async fn draft_course(
110    app_config: &ApplicationConfiguration,
111    task_lm: &TaskLMSpec,
112    source: &CourseFromPlanSource,
113    plan_text: &str,
114) -> ChatbotResult<CourseDraft> {
115    let has_chatbot = source.has_planned_chatbot();
116    let mut instructions = COURSE_INSTRUCTIONS.to_string();
117    if has_chatbot {
118        let _ = write!(instructions, "\n{CHATBOT_INSTRUCTIONS}");
119    }
120    let goal_count = source.learning_goals().len();
121    request_with_retry(
122        app_config,
123        task_lm,
124        &instructions,
125        plan_text,
126        course_format(has_chatbot),
127        |draft: &CourseDraft| {
128            if draft.learning_goal_headings.len() != goal_count {
129                return Err(format!(
130                    "Expected {goal_count} learning goal headings, got {}.",
131                    draft.learning_goal_headings.len()
132                ));
133            }
134            if has_chatbot && draft.chatbot.is_none() {
135                return Err("The chatbot draft is missing.".to_string());
136            }
137            Ok(())
138        },
139    )
140    .await
141}

143async fn draft_chapter(
144    app_config: &ApplicationConfiguration,
145    task_lm: &TaskLMSpec,
146    plan_text: &str,
147    chapter: &DesignChapter,
148) -> ChatbotResult<ChapterDraft> {
149    let topics = chapter_topic_titles(chapter);
150    let mut prompt = format!(
151        "{plan_text}\n## This chapter\nTitle: {}\nLearning objectives:\n",
152        chapter.title.trim()
153    );
154    for objective in learning_objectives(chapter) {
155        let _ = writeln!(prompt, "{LIST_ITEM_PREFIX}{objective}");
156    }
157    let _ = writeln!(prompt, "\n{TOPICS_HEADING}");
158    for topic in &topics {
159        let _ = writeln!(prompt, "{LIST_ITEM_PREFIX}{topic}");
160    }
161    request_with_retry(
162        app_config,
163        task_lm,
164        CHAPTER_INSTRUCTIONS,
165        &prompt,
166        chapter_format(),
167        |draft: &ChapterDraft| {
168            if draft.topic_outlines.len() == topics.len() {
169                Ok(())
170            } else {
171                Err(format!(
172                    "Expected {} topic outlines, got {}.",
173                    topics.len(),
174                    draft.topic_outlines.len()
175                ))
176            }
177        },
178    )
179    .await
180}

182/// Sends the request up to [`ATTEMPTS_PER_REQUEST`] times, until an answer parses as `T` and
183/// passes `validate`.
184async fn request_with_retry<T: DeserializeOwned>(
185    app_config: &ApplicationConfiguration,
186    task_lm: &TaskLMSpec,
187    instructions: &str,
188    prompt: &str,
189    format: LLMRequestResponseFormatParam,
190    validate: impl Fn(&T) -> Result<(), String>,
191) -> ChatbotResult<T> {
192    let mut last_error = None;
193    for _ in 0..ATTEMPTS_PER_REQUEST {
194        let answer: ChatbotResult<T> =
195            request(app_config, task_lm, instructions, prompt, format.clone()).await;
196        match answer {
197            Ok(answer) => match validate(&answer) {
198                Ok(()) => return Ok(answer),
199                Err(message) => {
200                    warn!("Course plan drafting answer was invalid: {message}");
201                    last_error = Some(chatbot_err!(FailedAzureResponse, message));
202                }
203            },
204            Err(e) => {
205                warn!("Course plan drafting request failed: {e}");
206                last_error = Some(e);
207            }
208        }
209    }
210    Err(last_error.unwrap_or_else(|| {
211        chatbot_err!(
212            FailedAzureResponse,
213            "Course plan drafting made no request.".to_string()
214        )
215    }))
216}

218async fn request<T: DeserializeOwned>(
219    app_config: &ApplicationConfiguration,
220    task_lm: &TaskLMSpec,
221    instructions: &str,
222    prompt: &str,
223    format: LLMRequestResponseFormatParam,
224) -> ChatbotResult<T> {
225    let input = vec![
226        APIInputMessage {
227            message_type: InputItem::Message {
228                role: MessageRole::System,
229                content: MessageContent::Text(format!("{COMMON_RULES}\n\n{instructions}")),
230            },
231        },
232        APIInputMessage {
233            message_type: InputItem::Message {
234                role: MessageRole::User,
235                content: MessageContent::Text(prompt.to_string()),
236            },
237        },
238    ];
239    let (params, max_output_tokens) = if model_is_thinking(task_lm.model_type) {
240        (
241            LLMRequestParams::GPTThinking(ThinkingParams {
242                reasoning: Some(Reasoning {
243                    effort: ReasoningEffortLevel::Low,
244                    summary: None,
245                    context: None,
246                }),
247            }),
248            Some(16000),
249        )
250    } else {
251        (
252            LLMRequestParams::GPTNonThinking(NonThinkingParams {
253                temperature: None,
254                top_p: None,
255                frequency_penalty: None,
256                presence_penalty: None,
257            }),
258            Some(8000),
259        )
260    };
261    request_structured_json(
262        input,
263        task_lm.model.to_owned(),
264        params,
265        max_output_tokens,
266        format,
267        app_config,
268        || {
269            chatbot_err!(
270                FailedAzureResponse,
271                "Invalidly structured response from Azure"
272            )
273        },
274    )
275    .await
276}

278/// The plan as the prompts present it. Blank fields are left out.
279fn plan_text(source: &CourseFromPlanSource) -> String {
280    let analysis = &source.analysis;
281    let design = &source.design;
282    let style = &design.style_guide;
283    let mut text = String::new();
284    let mut field = |label: &str, value: Option<&str>| {
285        if let Some(value) = value.and_then(non_empty_trimmed) {
286            let _ = writeln!(text, "{label}: {value}");
287        }
288    };
289    field("Course name", Some(&source.course_name));
290    field("Course language", Some(&source.language_code));
291    field("Target group", analysis.target_group.as_deref());
292    field(
293        "Students' background",
294        analysis.students_demographic_data.as_deref(),
295    );
296    field(
297        "Topics students wished for",
298        analysis.wishes_topics.as_deref(),
299    );
300    field(
301        "The team's vision for the course",
302        source.great_mooc_vision.as_deref(),
303    );
304    field("Must-know content", Some(&design.core_content_must_know));
305    field(
306        "Should-know content",
307        Some(&design.core_content_should_know),
308    );
309    field(
310        "Nice-to-know content",
311        Some(&design.core_content_nice_to_know),
312    );
313    field("Exercise plan", Some(&design.exercise_notes));
314    field("Writing style", Some(&style.writing_style));
315    field("Active or passive voice", Some(&style.active_passive_voice));
316    field(
317        "Terminology and acronyms",
318        Some(&style.terminology_and_acronyms),
319    );
320    let formats = &design.planned_content_formats;
321    let planned_formats: Vec<&str> = [
322        (formats.text, "text"),
323        (formats.infographics, "infographics"),
324        (formats.photos, "photos"),
325        (formats.videos, "videos"),
326        (formats.podcast, "podcasts"),
327        (formats.thinglink, "ThingLink"),
328        (formats.simulations, "simulations"),
329        (formats.chatbots, "AI chatbot"),
330    ]
331    .into_iter()
332    .filter_map(|(is_planned, name)| is_planned.then_some(name))
333    .collect();
334    if !planned_formats.is_empty() {
335        let _ = writeln!(
336            text,
337            "Planned content formats: {}",
338            planned_formats.join(", ")
339        );
340    }
341    let study_modes: Vec<&str> = [
342        (analysis.mode_synchronous, "synchronous"),
343        (analysis.mode_asynchronous, "asynchronous"),
344    ]
345    .into_iter()
346    .filter_map(|(is_planned, name)| is_planned.then_some(name))
347    .collect();
348    if !study_modes.is_empty() {
349        let _ = writeln!(text, "Study mode: {}", study_modes.join(", "));
350    }
351
352    let _ = writeln!(text, "\n{LEARNING_GOALS_HEADING}");
353    for goal in source.learning_goals() {
354        let _ = writeln!(text, "{LIST_ITEM_PREFIX}{goal}");
355    }
356    let _ = writeln!(text, "\n## Chapters");
357    for (index, chapter) in design.chapters.iter().enumerate() {
358        let _ = writeln!(text, "{}. {}", index + 1, chapter.title.trim());
359        let topics = chapter_topic_titles(chapter);
360        if !topics.is_empty() {
361            let _ = writeln!(text, "   Topics: {}", topics.join("; "));
362        }
363    }
364    text
365}

367/// Mirrors [`CourseDraft`]; `chatbot` is asked for only when the course plans one.
368fn course_format(has_chatbot: bool) -> LLMRequestResponseFormatParam {
369    let mut properties = IndexMap::from([
370        (
371            "description".to_string(),
372            string_property("Catalog description."),
373        ),
374        (
375            "hero_subtitle".to_string(),
376            string_property("Front page subtitle."),
377        ),
378        (
379            "about_text".to_string(),
380            string_property("About this course paragraph."),
381        ),
382        (
383            "learning_goal_headings".to_string(),
384            string_array_property(Some("One heading per learning goal, in order.")),
385        ),
386        (
387            "how_to_study_paragraphs".to_string(),
388            string_array_property(Some("Paragraphs of the How to study page.")),
389        ),
390    ]);
391    if has_chatbot {
392        properties.insert(
393            "chatbot".to_string(),
394            SchemaPropertyType::Object(Schema::strict_object(
395                IndexMap::from([
396                    ("name".to_string(), string_property("Chatbot name.")),
397                    (
398                        "prompt".to_string(),
399                        string_property("Chatbot system prompt."),
400                    ),
401                    (
402                        "initial_message".to_string(),
403                        string_property("First message to students."),
404                    ),
405                    (
406                        "suggested_messages".to_string(),
407                        string_array_property(Some("Example first messages from a student.")),
408                    ),
409                ]),
410                None,
411            )),
412        );
413    }
414    LLMRequestResponseFormatParam::JsonSchema {
415        name: COURSE_RESPONSE_FORMAT_NAME.to_string(),
416        schema: Schema::strict_object(properties, None),
417        strict: true,
418    }
419}

421/// Mirrors [`ChapterDraft`].
422fn chapter_format() -> LLMRequestResponseFormatParam {
423    let section = Schema::strict_object(
424        IndexMap::from([
425            ("heading".to_string(), string_property("Section heading.")),
426            (
427                "points".to_string(),
428                string_array_property(Some("What the section should cover.")),
429            ),
430        ]),
431        None,
432    );
433    let outline = Schema::strict_object(
434        IndexMap::from([(
435            "sections".to_string(),
436            SchemaPropertyType::ArrayProperty(ArrayProperty {
437                type_field: JSONType::Array,
438                items: ArrayItem::Schema(section),
439                description: None,
440            }),
441        )]),
442        None,
443    );
444    LLMRequestResponseFormatParam::JsonSchema {
445        name: CHAPTER_RESPONSE_FORMAT_NAME.to_string(),
446        schema: Schema::strict_object(
447            IndexMap::from([
448                (
449                    "prior_learning_heading".to_string(),
450                    string_property("Heading of the prior learning box."),
451                ),
452                (
453                    "prior_learning_questions".to_string(),
454                    string_array_property(Some("Reflection questions.")),
455                ),
456                (
457                    "topic_outlines".to_string(),
458                    SchemaPropertyType::ArrayProperty(ArrayProperty {
459                        type_field: JSONType::Array,
460                        items: ArrayItem::Schema(outline),
461                        description: Some("One outline per topic, in order.".to_string()),
462                    }),
463                ),
464            ]),
465            None,
466        ),
467        strict: true,
468    }
