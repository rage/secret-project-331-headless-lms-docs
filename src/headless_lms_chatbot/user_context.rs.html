11/// Context for a chatbot turn, in two halves:
12/// - who is asking: `user_id`, `course_id`, `course_name`, and the roles fetched from them.
13/// - what the configuration offers: `enabled_tool_categories`, read once per request from the
14///   same configuration row the turn is assembled from.
15pub struct ChatbotTurnContext {
16    pub user_id: Option<Uuid>,
17    pub course_id: Option<Uuid>,
18    pub course_name: Option<String>,
19    /// The conversation this turn belongs to. `None` only in tests that build a context without
20    /// one; every production call site has a conversation id in scope.
21    pub conversation_id: Option<Uuid>,
22    /// The course design plan the conversation is about, set only once `user_id` has been verified
23    /// as its member. Plan tools act on this plan only, never on one the model names.
24    pub course_plan: Option<CourseDesignerPlan>,
25    pub enabled_tool_categories: EnabledToolCategories,
26    roles: OnceCell<Vec<Role>>,
27}
28
29impl ChatbotTurnContext {
30    /// Collects what a chatbot request needs to know about its caller and about what the
31    /// chatbot's configuration offers.
32    pub fn new(
33        user_id: Option<Uuid>,
34        course_id: Option<Uuid>,
35        course_name: Option<String>,
36        conversation_id: Uuid,
37        configuration: &ChatbotConfiguration,
38    ) -> Self {
39        Self {
40            user_id,
41            course_id,
42            course_name,
43            conversation_id: Some(conversation_id),
44            course_plan: None,
45            enabled_tool_categories: EnabledToolCategories::from_configuration(configuration),
46            roles: OnceCell::new(),
47        }
48    }
49
50    /// Like [Self::new], for a turn of the course design assistant about a plan. The caller must
51    /// have loaded `course_plan` through a membership check of `user_id` and checked that the
52    /// conversation is about it: plan tools trust it without asking the database again.
53    pub fn new_for_course_plan(
54        user_id: Uuid,
55        course_name: Option<String>,
56        conversation_id: Uuid,
57        configuration: &ChatbotConfiguration,
58        course_plan: CourseDesignerPlan,
59    ) -> Self {
60        Self {
61            course_plan: Some(course_plan),
62            ..Self::new(
63                Some(user_id),
64                configuration.course_id,
65                course_name,
66                conversation_id,
67                configuration,
68            )
69        }
70    }
71
72    /// The resource an offer-time check runs against, before any call has named a target: the
73    /// course this chatbot belongs to, or global permissions for a chatbot that belongs to none.
74    pub fn turn_resource(&self) -> Resource {
75        match self.course_id {
76            Some(course_id) => Resource::Course(course_id),
77            None => Resource::GlobalPermissions,
78        }
79    }
80
81    /// The caller's roles, fetched the first time an authorization check needs them.
82    ///
83    /// One roles query per request however many authorization checks ask for them, and none at
84    /// all for a request whose tools need no role.
85    pub(crate) async fn roles(&self, conn: &mut PgConnection) -> ChatbotResult<&[Role]> {
86        let roles = self
87            .roles
88            .get_or_try_init(|| async {
89                fetch_user_roles(conn, self.user_id)
90                    .await
91                    .map_err(ChatbotError::from)
92            })
93            .await?;
94        Ok(roles)
95    }
96
97    #[cfg(test)]
98    pub(crate) fn with_roles(
99        user_id: Option<Uuid>,
100        course_id: Option<Uuid>,
101        course_name: Option<String>,
102        roles: Vec<Role>,
103    ) -> Self {
104        Self::with_roles_and_categories(
105            user_id,
106            course_id,
107            course_name,
108            roles,
109            EnabledToolCategories::all(),
110        )
111    }
112
113    /// Like [Self::with_roles], but with a specific enabled-category set instead of everything
114    /// enabled — for tests of the category gate itself rather than of authorization.
115    #[cfg(test)]
116    pub(crate) fn with_roles_and_categories(
117        user_id: Option<Uuid>,
118        course_id: Option<Uuid>,
119        course_name: Option<String>,
120        roles: Vec<Role>,
121        enabled_tool_categories: EnabledToolCategories,
122    ) -> Self {
123        Self {
124            user_id,
125            course_id,
126            course_name,
127            conversation_id: None,
128            course_plan: None,
129            enabled_tool_categories,
130            roles: OnceCell::new_with(Some(roles)),
131        }
132    }
