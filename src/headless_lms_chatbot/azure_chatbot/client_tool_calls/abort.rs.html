1//! Why a suspended tool call is closed without ever being carried out, and the text the model
2//! reads for it.
3
4use crate::chatbot_error::ChatbotResult;
5use crate::chatbot_tools::tool_permission::ToolPermission;
6use crate::prelude::*;
7use crate::user_context::ChatbotUserContext;
8
9/// Why a tool call's output was written without running the call.
10pub(crate) enum ToolCallAbortReason {
11    /// The turn that made this call died; a later request is repairing it.
12    Hanging,
13    /// The learner sent a new message instead of answering it.
14    Replaced,
15    /// The permission this call needed was revoked while it waited for the client.
16    PermissionRevoked,
17}
18
19impl ToolCallAbortReason {
20    /// The output the model reads for a call closed this way.
21    ///
22    /// Prompt text: keep every variant byte-identical, or the model's behavior on seeing it changes.
23    pub(crate) fn model_output(&self) -> &'static str {
24        match self {
25            Self::Hanging => "Unexpected error encountered, tool call aborted.",
26            Self::Replaced => {
27                "The user sent a new message instead of answering this tool call, so it was never carried out and returned no data. Answer their new message; ask again only if you still need this."
28            }
29            Self::PermissionRevoked => {
30                "The tool call was aborted and returned no data, because the user is no longer allowed to use this tool: their permissions changed while the call was waiting for them. Do not call this tool again in this conversation. Answer without it, or tell the user that you cannot do that step for them."
31            }
32        }
33    }
34}
35
36/// The abort output for `tool_name`, if `permission` no longer holds for `user_context` — checked
37/// both when the server plans to run a call and when a client's answer to one arrives, since
38/// nothing bounds how long a call waits and a role can be revoked while it does.
39pub(crate) async fn permission_revoked_output(
40    conn: &mut PgConnection,
41    user_context: &ChatbotUserContext,
42    permission: ToolPermission,
43    tool_name: &str,
44) -> ChatbotResult<Option<&'static str>> {
45    if permission.is_satisfied_by(conn, user_context).await? {
46        return Ok(None);
47    }
48    warn!(
49        tool_name,
50        "Aborting a chatbot tool call: its caller no longer holds the permission it requires"
51    );
52    Ok(Some(ToolCallAbortReason::PermissionRevoked.model_output()))
53}
