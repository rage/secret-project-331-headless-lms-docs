35/// Names this feature's structured output to Azure. The test-mode mock Azure API picks its canned
36/// answer for this feature by this name.
37pub const RESPONSE_FORMAT_NAME: &str = "LanguageVersionMatchingResponse";
38
39/// Headings and line prefixes of the prompt, which lists items as `- id: <uuid> "title"`.
40/// The test-mode mock Azure API reads the heuristic's picks back out of the prompt by these.
41pub const THIS_PAGES_HEADING: &str = "## Pages of course A";
42pub const OTHER_PAGES_HEADING: &str = "## Pages of course B";
43pub const THIS_EXERCISES_HEADING: &str = "## Exercises of course A";
44pub const OTHER_EXERCISES_HEADING: &str = "## Exercises of course B";
45pub const HEURISTIC_PICK_PREFIX: &str = "Position heuristic suggests: ";
46/// How the prompt names [SuggestionSureness::VerySure] after a heuristic pick.
47pub const VERY_SURE_LABEL: &str = "very sure";
48pub const ITEM_ID_PREFIX: &str = "- id: ";

50/// Requests in flight at once; one per chapter and other course.
51const MAX_CONCURRENT_REQUESTS: usize = 8;
52/// Every request must finish by then; the ingress for this endpoint allows 300 s.
53const DEADLINE: Duration = Duration::from_secs(150);

55const SYSTEM_PROMPT: &str = r#"You match the pages and exercises of two language versions of the same online course: course A and its translation, course B. You get one chapter at a time, or the pages outside chapters.
56
57For each page of course A, pick the page of course B that is its translation. Do the same for the exercises. Base the decision on what the titles, names and text excerpts mean, not on their order or position: translations are often reordered, and some pages or exercises exist in only one language (for example review quizzes or feedback surveys added to one version only). When nothing in course B corresponds, answer null as the counterpart.
58
59Answer in this JSON format, with one entry per page and one per exercise of course A, and an empty list when course A has none of that kind. Copy every id exactly as listed; a counterpart is the id of an item of the same kind in course B, or null:
60{
61  "pages": [
62    {"item": "<id of a page of course A>", "counterpart": "<id of a page of course B>"},
63    {"item": "<id of another page of course A>", "counterpart": null}
64  ],
65  "exercises": [
66    {"item": "<id of an exercise of course A>", "counterpart": "<id of an exercise of course B>"}
67  ]
68}
69
70Rules:
71- Answer every page and exercise of course A exactly once.
72- Each page or exercise of course B can be the counterpart of at most one item of course A.
73- Pages match only pages and exercises only exercises.
74- An exercise's counterpart is usually on the counterpart of its page, but not always.
75- "Known pairs" are already linked; they show how the courses correspond, and their items are not listed for matching.
76- Some items come with the suggestion of a position heuristic. It only compares positions and exercise counts, so it is imprecise and often wrong when the versions differ. Treat it as a weak hint and overrule it whenever the titles or texts say otherwise.
77
78Example (made-up course, by title; your answer uses the ids):
79Pages of course A: "Silmukat", "Ehtolauseet", "Kertaus", "Palautekysely"
80Pages of course B: "Conditional statements", "Loops", "Feedback survey"
81Matches: "Silmukat" -> "Loops" and "Ehtolauseet" -> "Conditional statements" (the order differs, the meaning matches); "Kertaus" -> null (a review quiz that course B does not have); "Palautekysely" -> "Feedback survey".
82Names can differ in wording too: "Kertaus" means "Review", and "Tehtävä: summa" can be the counterpart of "Exercise: Sum of numbers"."#;

84/// The structured output the matching request answers in; mirrors [AiMatchingAnswer].
85fn response_format() -> LLMRequestResponseFormatParam {
86    let picks = |description: &str| {
87        SchemaPropertyType::ArrayProperty(ArrayProperty {
88            type_field: JSONType::Array,
89            items: ArrayItem::Schema(Schema::strict_object(
90                IndexMap::from([
91                    (
92                        "item".to_string(),
93                        SchemaPropertyType::Item(JsonItem {
94                            type_field: JSONType::String,
95                            description: Some("Id of the item in course A.".to_string()),
96                        }),
97                    ),
98                    (
99                        "counterpart".to_string(),
100                        nullable_string_property(Some(
101                            "Id of its counterpart in course B, or null for none.",
102                        )),
103                    ),
104                ]),
105                None,
106            )),
107            description: Some(description.to_string()),
108        })
109    };
110    LLMRequestResponseFormatParam::JsonSchema {
111        name: RESPONSE_FORMAT_NAME.to_string(),
112        schema: Schema::strict_object(
113            IndexMap::from([
114                (
115                    "pages".to_string(),
116                    picks("One entry per page of course A."),
117                ),
118                (
119                    "exercises".to_string(),
120                    picks("One entry per exercise of course A."),
121                ),
122            ]),
123            None,
124        ),
125        strict: true,
126    }
127}

129fn sureness_label(sureness: SuggestionSureness) -> &'static str {
130    match sureness {
131        SuggestionSureness::VerySure => VERY_SURE_LABEL,
132        SuggestionSureness::Likely => "likely",
133        SuggestionSureness::Unsure | SuggestionSureness::AiSuggestion => "unsure",
134    }
135}

137fn write_items(prompt: &mut String, heading: &str, items: &[AiMatchingItem]) {
138    let _ = writeln!(prompt, "\n{heading}");
139    if items.is_empty() {
140        let _ = writeln!(prompt, "(none)");
141    }
142    for item in items {
143        let _ = write!(prompt, "{ITEM_ID_PREFIX}{} \"{}\"", item.id, item.title);
144        if let Some(page_title) = &item.page_title {
145            let _ = write!(prompt, " on page \"{page_title}\"");
146        }
147        let _ = writeln!(prompt);
148        if !item.excerpt.is_empty() {
149            let _ = writeln!(prompt, "  Text: {}", item.excerpt);
150        }
151        if let Some((id, sureness)) = item.heuristic_pick {
152            let _ = writeln!(
153                prompt,
154                "  {HEURISTIC_PICK_PREFIX}{id} ({})",
155                sureness_label(sureness)
156            );
157        }
158    }
159}

161fn write_known_pairs(prompt: &mut String, kind: &str, pairs: &[AiKnownPair]) {
162    if pairs.is_empty() {
163        return;
164    }
165    let _ = writeln!(prompt, "\nKnown pairs of {kind} (A = B):");
166    for pair in pairs {
167        let _ = writeln!(
168            prompt,
169            "- \"{}\" = \"{}\"",
170            pair.this_title, pair.other_title
171        );
172    }
173}

175fn user_prompt(task: &AiMatchingTask) -> String {
176    let chapter = |name: &Option<String>| match (task.chapter_number, name) {
177        (Some(number), Some(name)) => format!("chapter {number} \"{name}\""),
178        (Some(number), None) => format!("chapter {number}"),
179        (None, _) => "pages outside chapters".to_string(),
180    };
181    let mut prompt = format!(
182        "Course A is in language \"{}\", {}. Course B is in language \"{}\", {}.\n",
183        task.this_language_code,
184        chapter(&task.this_chapter_name),
185        task.other_language_code,
186        chapter(&task.other_chapter_name),
187    );
188    write_known_pairs(&mut prompt, "pages", &task.known_page_pairs);
189    write_known_pairs(&mut prompt, "exercises", &task.known_exercise_pairs);
190    write_items(&mut prompt, THIS_PAGES_HEADING, &task.this_pages);
191    write_items(&mut prompt, OTHER_PAGES_HEADING, &task.other_pages);
192    write_items(&mut prompt, THIS_EXERCISES_HEADING, &task.this_exercises);
193    write_items(&mut prompt, OTHER_EXERCISES_HEADING, &task.other_exercises);
194    prompt
195}

197async fn request_matching(
198    app_config: &ApplicationConfiguration,
199    task_lm: &TaskLMSpec,
200    task: &AiMatchingTask,
201) -> ChatbotResult<AiMatchingAnswer> {
202    let input = vec![
203        APIInputMessage {
204            message_type: InputItem::Message {
205                role: MessageRole::System,
206                content: MessageContent::Text(SYSTEM_PROMPT.to_string()),
207            },
208        },
209        APIInputMessage {
210            message_type: InputItem::Message {
211                role: MessageRole::User,
212                content: MessageContent::Text(user_prompt(task)),
213            },
214        },
215    ];
216    let (params, max_output_tokens) = if model_is_thinking(task_lm.model_type) {
217        (
218            LLMRequestParams::GPTThinking(ThinkingParams {
219                reasoning: Some(Reasoning {
220                    effort: ReasoningEffortLevel::Low,
221                    summary: None,
222                    context: None,
223                }),
224            }),
225            Some(16000),
226        )
227    } else {
228        (
229            LLMRequestParams::GPTNonThinking(NonThinkingParams {
230                temperature: None,
231                top_p: None,
232                frequency_penalty: None,
233                presence_penalty: None,
234            }),
235            Some(8000),
236        )
237    };
238    request_structured_json(
239        input,
240        task_lm.model.to_owned(),
241        params,
242        max_output_tokens,
243        response_format(),
244        app_config,
245        || {
246            chatbot_err!(
247                FailedAzureResponse,
248                "Invalidly structured response from Azure"
249            )
250        },
251    )
252    .await
253}

255/// The AI's counterpart picks for `tasks`, and how many tasks got no usable answer (failed or
256/// missed the deadline). Never fails as a whole: a failed task leaves its items to the heuristic.
257pub async fn suggest_counterparts(
258    app_config: &ApplicationConfiguration,
259    task_lm: &TaskLMSpec,
260    tasks: &[AiMatchingTask],
261) -> (AiPicksByCourse, usize) {
262    let deadline = Instant::now() + DEADLINE;
263    let answers: Vec<(&AiMatchingTask, Option<AiMatchingAnswer>)> = futures::stream::iter(tasks)
264        .map(|task| async move {
265            let answer = match tokio::time::timeout_at(
266                deadline,
267                request_matching(app_config, task_lm, task),
268            )
269            .await
270            {
271                Ok(Ok(answer)) => Some(answer),
272                Ok(Err(e)) => {
273                    warn!("Language version matching request failed: {e}");
274                    None
275                }
276                Err(_) => {
277                    warn!("Language version matching request missed the deadline");
278                    None
279                }
280            };
281            (task, answer)
282        })
283        .buffer_unordered(MAX_CONCURRENT_REQUESTS)
284        .collect()
285        .await;
286    let mut picks = AiPicksByCourse::new();
287    let mut failed_count = 0;
288    for (task, answer) in answers {
289        let Some(answer) = answer else {
290            failed_count += 1;
291            continue;
292        };
293        let dropped = picks
294            .entry(task.other_course_id)
295            .or_default()
296            .add_answer(task, &answer);
297        if dropped.total() > 0 {
298            warn!(
299                other_course_id = %task.other_course_id,
300                chapter_number = ?task.chapter_number,
301                unknown_items = dropped.unknown_items,
302                unknown_counterparts = dropped.unknown_counterparts,
303                duplicates = dropped.duplicates,
304                "Dropped invalid picks from a language version matching answer"
305            );
306        }
307    }
308    (picks, failed_count)
309}
