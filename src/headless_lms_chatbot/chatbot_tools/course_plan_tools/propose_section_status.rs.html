24/// Proposes marking a section of a stage ready or unfinished, with an optional comment.
25pub struct ProposeSectionStatusTool;
26
27pub struct ProposeSectionStatusArguments {
28    stage: CourseDesignerStage,
29    section_key: String,
30    current_status: Option<CourseDesignerSectionStatus>,
31    current_comment: String,
32    proposed_status: CourseDesignerSectionStatus,
33    proposed_comment: Option<String>,
34}

36#[derive(Deserialize)]
37struct RawArguments {
38    section_key: String,
39    current_status: String,
40    current_comment: String,
41    proposed_status: String,
42    proposed_comment: String,
43}

45fn parse_status(value: &str) -> ChatbotResult<Option<CourseDesignerSectionStatus>> {
46    match value.trim() {
47        "not_marked" => Ok(None),
48        other => other.parse().map(Some).map_err(|_| {
49            chatbot_err!(
50                InvalidToolArguments,
51                format!(
52                    "'{other}' is not a section status. Valid values: ready, unfinished, not_marked."
53                )
54            )
55        }),
56    }
57}

59impl ChatbotToolDeclaration for ProposeSectionStatusTool {

64    fn get_tool_definition() -> AzureLLMFunctionToolDefinition {

66            tool_type: LLMToolType::Function,
67            name: Self::NAME.to_string(),
68            description: format!(
69                "Propose marking a section of a stage as ready or unfinished, with an optional comment such as what is still missing. Sections: {}. The user applies or discards it; nothing is saved before that.",
70                KNOWN_SECTION_KEYS.join(", ")

72            parameters: Schema::strict_object(

74                    (
75                        "section_key".to_string(),
76                        text_property(

79                    ),
80                    (
81                        "current_status".to_string(),
82                        text_property(
83                            "ready, unfinished, or not_marked when get_course_plan shows no mark for it. Checked before anything is saved.",
84                        ),
85                    ),
86                    (
87                        "current_comment".to_string(),
88                        text_property(
89                            "The mark's current comment from get_course_plan; empty string for none.",

93                        "proposed_status".to_string(),

96                    (
97                        "proposed_comment".to_string(),

108impl ConfirmableActionTool for ProposeSectionStatusTool {
109    type Arguments = ProposeSectionStatusArguments;
110    plan_proposal_items!();
111
112    fn parse_arguments(arguments: &str) -> ChatbotResult<Self::Arguments> {
113        let raw: RawArguments = parse_raw_arguments(arguments)?;
114        let section_key = raw.section_key.trim().to_string();
115        if !KNOWN_SECTION_KEYS.contains(&section_key.as_str()) {
116            return Err(chatbot_err!(
117                InvalidToolArguments,
118                format!(
119                    "'{section_key}' is not a section. Sections: {}.",
120                    KNOWN_SECTION_KEYS.join(", ")
121                )
122            ));
123        }
124        // Every known key starts with its stage's name.
125        let stage = parse_stage(section_key.split('.').next().unwrap_or_default())?;
126        let proposed_status = parse_status(&raw.proposed_status)?.ok_or_else(|| {
127            chatbot_err!(
128                InvalidToolArguments,
129                "proposed_status must be ready or unfinished.".to_string()
130            )
131        })?;
132        Ok(ProposeSectionStatusArguments {
133            stage,
134            section_key,
135            current_status: parse_status(&raw.current_status)?,
136            current_comment: raw.current_comment,
137            proposed_status,
138            proposed_comment: non_empty_trimmed(&raw.proposed_comment).map(str::to_string),
139        })
140    }
141
142    async fn execute(
143        conn: &mut PgConnection,
144        _app_config: &ApplicationConfiguration,
145        arguments: &Self::Arguments,
146        authorization: &ToolAuthorization<Self>,
147    ) -> ChatbotResult<(ExecutedAction, Self::Facts)> {
148        let plan_id = authorized_plan_id(authorization)?;
149        let user_id = authorization.acting_user_id();
150        let mut tx = conn.begin().await?;
151        course_designer_plan_members::lock_editable_plan_for_user(&mut tx, plan_id, user_id)
152            .await?;
153        let statuses = course_designer_section_statuses::get_for_plan(&mut tx, plan_id).await?;
154        let stored = statuses
155            .iter()
156            .find(|s| s.stage == arguments.stage && s.section_key == arguments.section_key);
157        if stored.map(|s| s.status) != arguments.current_status {
158            return Err(stale_proposal_error("section's mark"));
159        }
160        verify_current_value(
161            "section's comment",
162            stored.and_then(|s| s.comment.as_deref()).unwrap_or(""),
163            &arguments.current_comment,
164        )?;
165
166        course_designer_section_statuses::upsert_for_user(
167            &mut tx,
168            plan_id,
169            user_id,
170            arguments.stage,
171            &arguments.section_key,
172            arguments.proposed_status,
173            arguments.proposed_comment.clone(),
174        )
175        .await?;
176        tx.commit().await?;
177
178        let status = arguments.proposed_status.as_str();
179        Ok((
180            ExecutedAction {
181                output: format!("Marked {} as {status}.", arguments.section_key),
182                client_payload: Some(updated_plan_payload(conn, plan_id, user_id).await?),
183                audit: ActionAuditFields {
184                    target_user_id: None,
185                    course_id: None,
186                    summary: format!(
187                        "Marked {} {status} in course design plan {plan_id}",
188                        arguments.section_key
189                    ),
190                },
191            },
192            (),
193        ))
194    }
