27/// Proposes changes to one task: its title, description, completion or assignee.
28pub struct ProposeUpdateTaskTool;
29
30pub struct ProposeUpdateTaskArguments {
31    task_id: Uuid,
32    task_title: String,
33    changes: Vec<TaskFieldChange>,
34}
35
36#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
37enum TaskField {
38    Title,
39    Description,
40    IsCompleted,
41    Assignee,
42}
43
44struct TaskFieldChange {
45    field: TaskField,
46    current_value: String,
47    proposed_value: String,
48}
49
50#[derive(Deserialize)]
51struct RawArguments {
52    task_id: String,
53    task_title: String,
54    changes: Vec<RawChange>,
55}
56
57#[derive(Deserialize)]
58struct RawChange {
59    field: String,
60    current_value: String,
61    proposed_value: String,
62}

69    fn get_tool_definition() -> AzureLLMFunctionToolDefinition {
70        AzureLLMFunctionToolDefinition {
71            tool_type: LLMToolType::Function,
72            name: Self::NAME.to_string(),
73            description: "Propose changing an existing task: rename it, change or clear its description, mark it done or not done, or assign it to a member. The user applies or discards the changes; nothing is saved before that.".to_string(),
74            parameters: Schema::strict_object(
75                IndexMap::from([
76                    ("task_id".to_string(), text_property("The task's id from get_course_plan.")),
77                    ("task_title".to_string(), text_property("The task's current title, exactly as get_course_plan returned it. Shown to the user and checked before anything is saved.")),
78                    (
79                        "changes".to_string(),
80                        SchemaPropertyType::ArrayProperty(ArrayProperty {
81                            type_field: JSONType::Array,
82                            items: ArrayItem::Schema(Schema::strict_object(
83                                IndexMap::from([
84                                    ("field".to_string(), text_property("title, description, is_completed or assignee.")),
85                                    ("current_value".to_string(), text_property("The field's current value from get_course_plan: text for title and description, 'true' or 'false' for is_completed, the member's name for assignee. Empty string for none.")),
86                                    ("proposed_value".to_string(), text_property("The new value in the same form. Empty string clears the description or unassigns the task; a title cannot be empty.")),
87                                ]),
88                                None,
89                            )),
90                            description: Some("1 to 4 changes, each to a different field.".to_string()),
91                        }),
92                    ),
93                ]),
94                None,
95            ),
96            strict: true,
97        }
98    }

101impl ConfirmableActionTool for ProposeUpdateTaskTool {
102    type Arguments = ProposeUpdateTaskArguments;
103    plan_proposal_items!();
104
105    fn parse_arguments(arguments: &str) -> ChatbotResult<Self::Arguments> {
106        let raw: RawArguments = parse_raw_arguments(arguments)?;
107        let task_id = parse_required_uuid("task_id", &raw.task_id)?;
108        let task_title = non_empty_trimmed(&raw.task_title)
109            .map(str::to_string)
110            .ok_or_else(|| {
111                chatbot_err!(
112                    InvalidToolArguments,
113                    "task_title must not be empty.".to_string()
114                )
115            })?;
116        if raw.changes.is_empty() || raw.changes.len() > 4 {
117            return Err(chatbot_err!(
118                InvalidToolArguments,
119                "Propose 1 to 4 changes.".to_string()
120            ));
121        }
122        let mut seen = HashSet::new();
123        let changes = raw
124            .changes
125            .into_iter()
126            .map(|change| {
127                let field = match change.field.trim() {
128                    "title" => TaskField::Title,
129                    "description" => TaskField::Description,
130                    "is_completed" => TaskField::IsCompleted,
131                    "assignee" => TaskField::Assignee,
132                    other => {
133                        return Err(chatbot_err!(
134                            InvalidToolArguments,
135                            format!("'{other}' is not a task field. Valid values: title, description, is_completed, assignee.")
136                        ));
137                    }
138                };
139                if !seen.insert(field) {
140                    return Err(chatbot_err!(
141                        InvalidToolArguments,
142                        "Each field can be changed only once.".to_string()
143                    ));
144                }
145                if field == TaskField::Title && change.proposed_value.trim().is_empty() {
146                    return Err(chatbot_err!(
147                        InvalidToolArguments,
148                        "A task title cannot be empty.".to_string()
149                    ));
150                }
151                if field == TaskField::IsCompleted {
152                    parse_bool(&change.proposed_value)?;
153                }
154                Ok(TaskFieldChange {
155                    field,
156                    current_value: change.current_value,
157                    proposed_value: change.proposed_value,
158                })
159            })
160            .collect::<ChatbotResult<Vec<_>>>()?;
161        Ok(ProposeUpdateTaskArguments {
162            task_id,
163            task_title,
164            changes,
165        })
166    }
167
168    async fn execute(
169        conn: &mut PgConnection,
170        _app_config: &ApplicationConfiguration,
171        arguments: &Self::Arguments,
172        authorization: &ToolAuthorization<Self>,
173    ) -> ChatbotResult<(ExecutedAction, Self::Facts)> {
174        let plan_id = authorized_plan_id(authorization)?;
175        let user_id = authorization.acting_user_id();
176        let mut tx = conn.begin().await?;
177        course_designer_plan_members::lock_editable_plan_for_user(&mut tx, plan_id, user_id)
178            .await?;
179        let task = course_designer_plan_members::get_plan_tasks(&mut tx, plan_id)
180            .await?
181            .into_iter()
182            .find(|task| task.id == arguments.task_id)
183            .ok_or_else(|| {
184                chatbot_err!(
185                    ToolUseError,
186                    "The task no longer exists. Call get_course_plan again.".to_string()
187                )
188            })?;
189        verify_current_value("task title", &task.title, &arguments.task_title)?;
190
191        let members =
192            course_designer_plan_members::get_plan_members_with_details(&mut tx, plan_id, user_id)
193                .await?;
194        let names: HashMap<Uuid, String> = members
195            .iter()
196            .map(|member| (member.user_id, member.display_name()))
197            .collect();
198
199        let mut title = None;
200        let mut description = None;
201        let mut is_completed = None;
202        let mut assignee = None;
203        for change in &arguments.changes {
204            match change.field {
205                TaskField::Title => {
206                    verify_current_value("task title", &task.title, &change.current_value)?;
207                    title = Some(change.proposed_value.trim().to_string());
208                }
209                TaskField::Description => {
210                    verify_current_value(
211                        "task description",
212                        task.description.as_deref().unwrap_or(""),
213                        &change.current_value,
214                    )?;
215                    description = Some(change.proposed_value.trim().to_string());
216                }
217                TaskField::IsCompleted => {
218                    if parse_bool(&change.current_value).ok() != Some(task.is_completed) {
219                        return Err(stale_proposal_error("task's completion"));
220                    }
221                    is_completed = Some(parse_bool(&change.proposed_value)?);
222                }
223                TaskField::Assignee => {
224                    let current_name = task
225                        .assignee_user_id
226                        .and_then(|id| names.get(&id).cloned())
227                        .unwrap_or_default();
228                    if current_name.to_lowercase() != change.current_value.trim().to_lowercase() {
229                        return Err(stale_proposal_error("task's assignee"));
230                    }
231                    let proposed = non_empty_trimmed(&change.proposed_value)
232                        .map(|name| resolve_member_by_name(&members, name))
233                        .transpose()?;
234                    assignee = Some(proposed);
235                }
236            }
237        }
238
239        if title.is_some() || description.is_some() || is_completed.is_some() {
240            course_designer_plan_members::update_stage_task_for_user(
241                &mut tx,
242                plan_id,
243                arguments.task_id,
244                user_id,
245                title,
246                description,
247                is_completed,
248            )
249            .await?;
250        }
251        if let Some(assignee) = assignee {
252            course_designer_plan_members::set_task_assignee_for_user(
253                &mut tx,
254                plan_id,
255                arguments.task_id,
256                user_id,
257                assignee,
258            )
259            .await?;
260        }
261        tx.commit().await?;
262
263        let fields = arguments
264            .changes
265            .iter()
266            .map(|change| match change.field {
267                TaskField::Title => "title",
268                TaskField::Description => "description",
269                TaskField::IsCompleted => "is_completed",
270                TaskField::Assignee => "assignee",
271            })
272            .collect::<Vec<_>>()
273            .join(", ");
274        Ok((
275            ExecutedAction {
276                output: format!(
277                    "Updated the task \"{}\". Changed fields: {fields}.",
278                    arguments.task_title
279                ),
280                client_payload: Some(updated_plan_payload(conn, plan_id, user_id).await?),
281                audit: ActionAuditFields {
282                    target_user_id: assignee.flatten(),
283                    course_id: None,
284                    summary: format!(
285                        "Updated task {} of course design plan {plan_id}: {fields}",
286                        arguments.task_id
287                    ),
288                },
289            },
290            (),
291        ))
292    }
293}

295fn parse_bool(value: &str) -> ChatbotResult<bool> {
296    match value.trim().to_lowercase().as_str() {
297        "true" => Ok(true),
298        "false" => Ok(false),
299        other => Err(chatbot_err!(
300            InvalidToolArguments,
301            format!("'{other}' is not 'true' or 'false'.")
302        )),
303    }
