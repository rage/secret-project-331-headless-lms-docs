28/// The [ChatbotToolDeclaration](crate::chatbot_tools::ChatbotToolDeclaration) items every plan

43/// The [ConfirmableActionTool](crate::chatbot_tools::action_tools::ConfirmableActionTool) items
44/// every proposal tool shares, so each one writes only its arguments and `execute`.
45macro_rules! plan_proposal_items {
46    () => {
47        type Facts = ();
48
49        fn call_requirements(
50            _arguments: &Self::Arguments,
51            _user_context: &$crate::user_context::ChatbotTurnContext,
52        ) -> Vec<$crate::chatbot_tools::tool_authorization::ToolRequirement> {
53            $crate::chatbot_tools::course_plan_tools::plan_member_requirements()
54        }
55
56        fn declined_output(_arguments: &Self::Arguments) -> String {
57            $crate::chatbot_tools::course_plan_tools::DECLINED_OUTPUT.to_string()
58        }
59
60        fn output_description_instructions(
61            _arguments: &Self::Arguments,
62            facts: Option<&()>,
63            _app_config: &headless_lms_base::config::ApplicationConfiguration,
64        ) -> Option<String> {
65            Some($crate::chatbot_tools::course_plan_tools::outcome_instructions(facts.is_some()))
66        }
67    };
68}

84pub(crate) fn parse_stage(value: &str) -> ChatbotResult<CourseDesignerStage> {
85    value.trim().to_lowercase().parse().map_err(|_| {
86        chatbot_err!(
87            InvalidToolArguments,
88            format!(
89                "'{value}' is not a stage. Valid values: analysis, design, development, implementation, evaluation."
90            )
91        )
92    })
93}

95/// Parses tool arguments into their raw shape, reporting failure to the model.
96pub(crate) fn parse_raw_arguments<T: DeserializeOwned>(arguments: &str) -> ChatbotResult<T> {
97    serde_json::from_str(arguments).map_err(|e| {
98        chatbot_err!(
99            InvalidToolArguments,
100            format!("Couldn't parse tool arguments. Arguments: {arguments}"),
101            e
102        )
103    })
104}

106/// The plan member going by `name` as the plan pages show it, case-insensitively.
107///
108/// Fails when no member, or more than one, has that name.
109pub(crate) fn resolve_member_by_name(
110    members: &[PlanMemberWithDetails],
111    name: &str,
112) -> ChatbotResult<Uuid> {
113    let wanted = name.trim().to_lowercase();
114    let matching: Vec<Uuid> = members
115        .iter()
116        .filter(|member| member.display_name().to_lowercase() == wanted)
117        .map(|member| member.user_id)
118        .collect();
119    match matching.as_slice() {
120        [user_id] => Ok(*user_id),
121        [] => Err(chatbot_err!(
122            ToolUseError,
123            format!(
124                "No plan member is named {name}. Nothing was saved. Call get_course_plan for the members' names."
125            )
126        )),
127        _ => Err(chatbot_err!(
128            ToolUseError,
129            format!(
130                "More than one plan member is named {name}. Nothing was saved; ask the user to assign the task themselves."
131            )
132        )),
133    }
134}

136/// Refuses to apply a proposal whose "current" value no longer matches the plan, so a change
137/// confirmed from a stale card cannot overwrite an edit made after the model read the plan.
138pub(crate) fn verify_current_value(what: &str, actual: &str, supplied: &str) -> ChatbotResult<()> {
139    if actual.trim() == supplied.trim() {
140        Ok(())
141    } else {
142        Err(stale_proposal_error(what))
143    }
144}
145
146pub(crate) fn stale_proposal_error(what: &str) -> ChatbotError {
147    chatbot_err!(
148        ToolUseError,
149        format!(
150            "The {what} has changed since the plan was read, so nothing was applied. Call get_course_plan and propose the change again."
151        )
152    )
153}

155/// What every executed plan tool sends the browser: the plan as the plan pages load it, so they
156/// can show the change without refetching.
157pub(crate) async fn updated_plan_payload(
158    conn: &mut PgConnection,
159    plan_id: Uuid,
160    user_id: Uuid,
161) -> ChatbotResult<serde_json::Value> {
162    let details = course_designer_plans::get_plan_details_for_user(conn, plan_id, user_id).await?;
163    Ok(serde_json::json!({ "plan_details": details }))
164}

168/// Instructions shared by every proposal tool's outcome.
169pub(crate) fn outcome_instructions(applied: bool) -> String {
170    if applied {
171        "The user applied the change and it is now saved in the plan. Confirm it briefly in the user's language.".to_string()
172    } else {
173        "Acknowledge briefly in the user's language. Never claim the change was made.".to_string()
174    }
