1//! Path lookup, id filling and comparison over the JSON form of a stage workspace, for applying
2//! the model's field-level edits.
3
4use std::collections::HashSet;
5
6use serde_json::{Map, Value};
7use uuid::Uuid;
8
9/// Lists whose items are objects with a client-generated `id`, keyed by the field holding them.
10const LISTS_WITH_IDS: [&str; 3] = ["chapters", "topics", "workload_activities"];
11
12/// The value at `segments`, where a segment names an object key or, inside a list with ids, the
13/// `id` of an item.
14pub(super) fn resolve_field<'a>(
15    root: &'a mut Value,
16    segments: &[&str],
17) -> Result<&'a mut Value, String> {
18    let mut current = root;
19    for segment in segments {
20        current = match current {
21            Value::Object(map) => map
22                .get_mut(*segment)
23                .ok_or_else(|| format!("there is no field {segment}"))?,
24            Value::Array(items) => items
25                .iter_mut()
26                .find(|item| item.get("id").and_then(Value::as_str) == Some(*segment))
27                .ok_or_else(|| format!("there is no item with id {segment}"))?,
28            _ => return Err(format!("{segment} goes inside a value that has no fields")),
29        };
30    }
31    Ok(current)
32}

34/// Gives proposed list items the ids of the current items they replace, matched by title, and new
35/// ids to the rest, so the model never has to invent or copy ids correctly.
36pub(super) fn fill_ids(key: Option<&str>, proposed: &mut Value, current: &Value) {
37    match proposed {
38        Value::Array(items) if key.is_some_and(|key| LISTS_WITH_IDS.contains(&key)) => {
39            let current_items = current.as_array().cloned().unwrap_or_default();
40            let mut used_ids = HashSet::new();
41            for item in items.iter_mut() {
42                if let Value::String(title) = item {
43                    let mut object = Map::new();
44                    object.insert("title".to_string(), Value::String(title.clone()));
45                    *item = Value::Object(object);
46                }
47                let Value::Object(object) = item else {
48                    continue;
49                };
50                let supplied_id = object
51                    .get("id")
52                    .and_then(Value::as_str)
53                    .and_then(|id| Uuid::parse_str(id).ok())
54                    .filter(|id| !id.is_nil());
55                let matched = match supplied_id {
56                    Some(id) => current_items
57                        .iter()
58                        .find(|candidate| id_of(candidate) == Some(id)),
59                    None => {
60                        let title = object
61                            .get("title")
62                            .and_then(Value::as_str)
63                            .map(normalized_title);
64                        current_items.iter().find(|candidate| {
65                            id_of(candidate).is_some_and(|id| !used_ids.contains(&id))
66                                && title.is_some()
67                                && candidate
68                                    .get("title")
69                                    .and_then(Value::as_str)
70                                    .map(normalized_title)
71                                    == title
72                        })
73                    }
74                };
75                let id = supplied_id
76                    .or_else(|| matched.and_then(id_of))
77                    .unwrap_or_else(Uuid::new_v4);
78                used_ids.insert(id);
79                object.insert("id".to_string(), Value::String(id.to_string()));
80                let matched = matched.cloned().unwrap_or(Value::Null);
81                fill_nested(object, &matched);
82            }
83        }
84        Value::Object(object) => {
85            if let Some(id) = id_of(current) {
86                object.insert("id".to_string(), Value::String(id.to_string()));
87            }
88            fill_nested(object, current);
89        }
90        _ => {}
91    }
92}

94fn fill_nested(object: &mut Map<String, Value>, current: &Value) {
95    for (key, value) in object.iter_mut() {
96        if key != "id" {
97            fill_ids(Some(key), value, current.get(key).unwrap_or(&Value::Null));
98        }
99    }
100}

102fn id_of(value: &Value) -> Option<Uuid> {
103    value
104        .get("id")
105        .and_then(Value::as_str)
106        .and_then(|id| Uuid::parse_str(id).ok())
107}

109fn normalized_title(title: &str) -> String {
110    title.trim().to_lowercase()
111}

113/// Whether the model's copy of a current value describes `actual`: ids and surrounding whitespace
114/// are ignored, and every empty value (null, "", [], false, 0) counts as the same.
115pub(super) fn same_content(actual: &Value, supplied: &Value) -> bool {
116    if is_empty(actual) && is_empty(supplied) {
117        return true;
118    }
119    match (actual, supplied) {
120        (Value::String(a), Value::String(b)) => a.trim() == b.trim(),
121        (Value::Number(a), Value::Number(b)) => a.as_f64() == b.as_f64(),
122        (Value::Array(a), Value::Array(b)) => {
123            a.len() == b.len() && a.iter().zip(b).all(|(a, b)| same_content(a, b))
124        }
125        // A list item the model wrote as its title alone.
126        (Value::Object(a), Value::String(_)) => a
127            .get("title")
128            .is_some_and(|title| same_content(title, supplied)),
129        (Value::Object(a), Value::Object(b)) => a
130            .iter()
131            .filter(|(key, _)| key.as_str() != "id")
132            .all(|(key, value)| same_content(value, b.get(key).unwrap_or(&Value::Null))),
133        (a, b) => a == b,
134    }
135}

137/// Whether `saved` holds everything `proposed` set. Fields `proposed` left out may have been
138/// filled with defaults; numbers compare by value (`5` and `5.0`).
139pub(super) fn keeps_proposed(saved: &Value, proposed: &Value) -> bool {
140    match (saved, proposed) {
141        (Value::Number(a), Value::Number(b)) => a.as_f64() == b.as_f64(),
142        (Value::Array(a), Value::Array(b)) => {
143            a.len() == b.len() && a.iter().zip(b).all(|(a, b)| keeps_proposed(a, b))
144        }
145        (Value::Object(a), Value::Object(b)) => b
146            .iter()
147            .all(|(key, value)| a.get(key).is_some_and(|saved| keeps_proposed(saved, value))),
148        (a, b) => a == b,
149    }
150}

152fn is_empty(value: &Value) -> bool {
153    match value {
154        Value::Null => true,
155        Value::Bool(b) => !b,
156        Value::Number(n) => n.as_f64() == Some(0.0),
157        Value::String(s) => s.trim().is_empty(),
158        Value::Array(items) => items.is_empty(),
159        Value::Object(map) => map
160            .iter()
161            .all(|(key, value)| key == "id" || is_empty(value)),
162    }
