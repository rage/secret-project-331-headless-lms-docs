27#[derive(Debug)]
28pub struct CourseFinderState {

32#[derive(Deserialize, Clone, Debug)]
33pub struct CourseFinderArguments {
34    #[serde(deserialize_with = "empty_vec_as_none")]
35    description: Option<Vec<String>>,
36    #[serde(deserialize_with = "empty_vec_as_none")]
37    prerequisites: Option<Vec<String>>,
38    #[serde(deserialize_with = "empty_vec_as_none")]
39    audiences: Option<Vec<String>>,
40}
41#[derive(Serialize, Deserialize, Clone, Debug)]
42pub struct CourseOccurrences {
43    course: Course,
44    occurrences: usize,

47pub type CourseFinderTool = ToolProperties<CourseFinderState>;

49impl ChatbotTool for CourseFinderTool {
50    type Arguments = CourseFinderArguments;
51
52    async fn from_db_and_arguments(
53        conn: &mut PgConnection,
54        app_config: &ApplicationConfiguration,
55        arguments: Self::Arguments,
56        _user_context: &ChatbotUserContext,
57    ) -> ChatbotResult<Self> {
58        let audience_courses = if let Some(audiences) = &arguments.audiences {
59            let audience_embeddings = create_embeddings(app_config, audiences.clone())
60                .await?
61                .to_owned();
62
63            get_course_ids_by_audience_vectors(conn, audience_embeddings, audiences.clone()).await?
64        } else {
65            vec![]
66        };
67
68        let prerequisite_courses = if let Some(prerequisites) = &arguments.prerequisites {
69            let prerequisite_embeddings = create_embeddings(app_config, prerequisites.clone())
70                .await?
71                .to_owned();
72
73            get_course_ids_by_prerequisite_vectors(
74                conn,
75                prerequisite_embeddings,
76                prerequisites.clone(),
77            )
78            .await?
79        } else {
80            vec![]
81        };
82
83        let description_courses = if let Some(description) = &arguments.description {
84            let description_embeddings = create_embeddings(app_config, description.clone())
85                .await?
86                .to_owned();
87
88            get_by_description_vectors(conn, description_embeddings, description.clone()).await?
89        } else {
90            vec![]
91        };
92
93        let course_ids = [description_courses, audience_courses, prerequisite_courses].concat();
94
95        let mut counts: HashMap<Uuid, usize> = HashMap::new();
96
97        for id in &course_ids {
98            *counts.entry(*id).or_insert(0) += 1;
99        }
100
101        let courses = courses::get_by_ids(conn, &course_ids).await?;
102
103        let mut course_occurrences: Vec<CourseOccurrences> = courses
104            .into_iter()
105            .map(|course| CourseOccurrences {
106                occurrences: counts[&course.id],
107                course,
108            })
109            .collect();
110
111        course_occurrences.sort_by_key(|b| std::cmp::Reverse(b.occurrences));
112
113        Ok(CourseFinderTool {
114            state: CourseFinderState {
115                courses: course_occurrences,
116            },
117        })
118    }
119
120    fn output(&self) -> String {
121        serde_json::to_string(&self.state.courses)
122            .unwrap_or_else(|_| "No courses found".to_string())
123    }
124
125    fn output_description_instructions(&self) -> Option<String> {
126        Some("Do not return the whole JSON of the courses to the user. Present the most suitable courses based on the user query. Use the course names and course descriptions to give a list and a very brief and summarized description of each course to the user. If there are duplicate courses ignore them. You can also mention why the course could be suitable to the user based on their request.".to_string())
127    }
128}

162fn empty_vec_as_none<'de, D>(deserializer: D) -> Result<Option<Vec<String>>, D::Error>
163where
164    D: Deserializer<'de>,
165{
166    let opt = Option::<Vec<String>>::deserialize(deserializer)?;
167
168    Ok(opt.and_then(|vec| {
169        let vec: Vec<String> = vec
170            .into_iter()
171            .map(|s| s.trim().to_owned())
172            .filter(|s| !s.is_empty())
173            .collect();
174
175        if vec.is_empty() { None } else { Some(vec) }
176    }))
177}
