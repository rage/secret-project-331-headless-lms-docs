24#[derive(Debug)]
25pub struct CourseFinderState {
26    courses: Vec<CourseOccurrences>,
27}

29#[derive(Serialize, Deserialize, Clone, Debug)]
30pub struct CourseFinderArguments {
31    #[serde(deserialize_with = "empty_vec_as_none")]
32    description: Option<Vec<String>>,
33    #[serde(deserialize_with = "empty_vec_as_none")]
34    prerequisites: Option<Vec<String>>,
35    #[serde(deserialize_with = "empty_vec_as_none")]
36    audiences: Option<Vec<String>>,
37}
38#[derive(Serialize, Deserialize, Clone, Debug)]
39pub struct CourseOccurrences {
40    course: Course,
41    occurrences: usize,
42}

44pub type CourseFinderTool = ToolProperties<CourseFinderState, CourseFinderArguments>;

46impl ChatbotTool for CourseFinderTool {
47    type State = CourseFinderState;
48    type Arguments = CourseFinderArguments;
49
50    fn parse_arguments(args_string: String) -> ChatbotResult<Self::Arguments> {
51        serde_json::from_str::<Self::Arguments>(&args_string).map_err(|e| {
52            chatbot_err!(
53                InvalidToolArguments,
54                format!("Couldn't parse tool arguments. Arguments: {args_string}"),
55                e
56            )
57        })
58    }
59
60    fn get_arguments(&self) -> &Self::Arguments {
61        &self.arguments
62    }
63
64    async fn from_db_and_arguments(
65        conn: &mut PgConnection,
66        app_config: &ApplicationConfiguration,
67        arguments: Self::Arguments,
68        _user_context: &ChatbotUserContext,
69    ) -> ChatbotResult<Self> {
70        let audience_courses = if let Some(audiences) = &arguments.audiences {
71            let audience_embeddings = create_embeddings(app_config, audiences.clone())
72                .await?
73                .to_owned();
74
75            get_course_ids_by_audience_vectors(conn, audience_embeddings, audiences.clone()).await?
76        } else {
77            vec![]
78        };
79
80        let prerequisite_courses = if let Some(prerequisites) = &arguments.prerequisites {
81            let prerequisite_embeddings = create_embeddings(app_config, prerequisites.clone())
82                .await?
83                .to_owned();
84
85            get_course_ids_by_prerequisite_vectors(
86                conn,
87                prerequisite_embeddings,
88                prerequisites.clone(),
89            )
90            .await?
91        } else {
92            vec![]
93        };
94
95        let description_courses = if let Some(description) = &arguments.description {
96            let description_embeddings = create_embeddings(app_config, description.clone())
97                .await?
98                .to_owned();
99
100            get_by_description_vectors(conn, description_embeddings, description.clone()).await?
101        } else {
102            vec![]
103        };
104
105        let course_ids = [description_courses, audience_courses, prerequisite_courses].concat();
106
107        let mut counts: HashMap<Uuid, usize> = HashMap::new();
108
109        for id in &course_ids {
110            *counts.entry(*id).or_insert(0) += 1;
111        }
112
113        let courses = courses::get_by_ids(conn, &course_ids).await?;
114
115        let mut course_occurrences: Vec<CourseOccurrences> = courses
116            .into_iter()
117            .map(|course| CourseOccurrences {
118                occurrences: counts[&course.id],
119                course,
120            })
121            .collect();
122
123        course_occurrences.sort_by_key(|b| std::cmp::Reverse(b.occurrences));
124
125        Ok(CourseFinderTool {
126            state: CourseFinderState {
127                courses: course_occurrences,
128            },
129            arguments,
130        })
131    }
132
133    fn output(&self) -> String {
134        serde_json::to_string(&self.state.courses)
135            .unwrap_or_else(|_| "No courses found".to_string())
136    }

190fn empty_vec_as_none<'de, D>(deserializer: D) -> Result<Option<Vec<String>>, D::Error>
191where
192    D: Deserializer<'de>,
193{
194    let opt = Option::<Vec<String>>::deserialize(deserializer)?;
195
196    Ok(opt.and_then(|vec| {
197        let vec: Vec<String> = vec
198            .into_iter()
199            .map(|s| s.trim().to_owned())
200            .filter(|s| !s.is_empty())
201            .collect();
202
203        if vec.is_empty() { None } else { Some(vec) }
204    }))
