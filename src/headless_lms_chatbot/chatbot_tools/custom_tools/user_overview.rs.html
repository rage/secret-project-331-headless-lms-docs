135impl UserOverviewFacet {
136    fn wire_name(self) -> &'static str {
137        match self {
138            Self::Profile => "profile",
139            Self::Roles => "roles",
140            Self::Enrollments => "enrollments",
141            Self::CheatingFlags => "cheating_flags",
142            Self::EmailDeliveries => "email_deliveries",
143        }
144    }
145
146    fn from_wire_name(s: &str) -> Option<Self> {
147        match s {
148            "profile" => Some(Self::Profile),
149            "roles" => Some(Self::Roles),
150            "enrollments" => Some(Self::Enrollments),
151            "cheating_flags" => Some(Self::CheatingFlags),
152            "email_deliveries" => Some(Self::EmailDeliveries),
153            _ => None,
154        }
155    }
156}

163impl<'de> serde::Deserialize<'de> for UserOverviewArguments {
164    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
165    where
166        D: serde::Deserializer<'de>,
167    {
168        #[derive(Deserialize)]
169        struct Raw {
170            user_id: String,
171            facets: Vec<String>,
172        }
173        let raw = Raw::deserialize(deserializer)?;
174        let user_id = Uuid::from_str(&raw.user_id).map_err(serde::de::Error::custom)?;
175
176        let mut facets = Vec::new();
177        for wire_name in &raw.facets {
178            let facet = UserOverviewFacet::from_wire_name(wire_name).ok_or_else(|| {
179                serde::de::Error::custom(format!(
180                    "Unknown facet '{wire_name}'. Valid facets: profile, roles, enrollments, cheating_flags, email_deliveries."
181                ))
182            })?;
183            if !facets.contains(&facet) {
184                facets.push(facet);
185            }
186        }
187        if facets.is_empty() {
188            return Err(serde::de::Error::custom(
189                "At least one facet must be requested.",
190            ));
191        }
192
193        Ok(UserOverviewArguments { user_id, facets })
194    }
195}

234impl ChatbotTool for UserOverviewTool {
235    type Arguments = UserOverviewArguments;
236
237    fn call_requirements(
238        arguments: &Self::Arguments,
239        _user_context: &ChatbotTurnContext,
240    ) -> Vec<ToolRequirement> {
241        vec![ToolRequirement::on_user(
242            Action::ViewUserProgressOrDetails,
243            arguments.user_id,
244        )]
245    }
246
247    async fn from_db_and_arguments(
248        conn: &mut PgConnection,
249        app_config: &ApplicationConfiguration,
250        arguments: Self::Arguments,
251        _user_context: &ChatbotTurnContext,
252    ) -> ChatbotResult<Self> {
253        let base_url = app_config.base_url.trim_end_matches('/').to_string();
254        let user_id = arguments.user_id;
255        let user_detail = user_details::get_user_details_by_user_id(conn, user_id)
256            .await
257            .map_err(|e| {
258                chatbot_err!(ToolUseError, format!("No user found with id {user_id}."), e)
259            })?;
260
261        let mut facets = IndexMap::new();
262        for facet in &arguments.facets {
263            let value = match facet {
264                UserOverviewFacet::Profile => {
265                    let user = users::get_by_id(conn, user_id).await?;
266                    UserOverviewFacetValue::Profile(ProfileFacet {
267                        user_id: user_detail.user_id,
268                        email: user_detail.email.clone(),
269                        first_name: user_detail.first_name.clone(),
270                        last_name: user_detail.last_name.clone(),
271                        country: user_detail.country.clone(),
272                        email_communication_consent: user_detail.email_communication_consent,
273                        email_verified_at: user_detail.email_verified_at,
274                        email_verified_method: user_detail.email_verified_method,
275                        upstream_id: user.upstream_id,
276                        email_domain: user.email_domain,
277                        created_at: user.created_at,
278                        deleted_at: user.deleted_at,
279                    })
280                }
281                UserOverviewFacet::Roles => {
282                    let roles = get_roles(conn, user_id).await?;
283                    UserOverviewFacetValue::Roles(
284                        roles
285                            .into_iter()
286                            .filter(|role| !role.is_global && role.organization_id.is_none())
287                            .map(|role| RoleFacet {
288                                role: role.role,
289                                course_id: role.course_id,
290                                course_instance_id: role.course_instance_id,
291                                exam_id: role.exam_id,
292                            })
293                            .collect(),
294                    )
295                }
296                UserOverviewFacet::Enrollments => {
297                    let enrollments = get_course_enrollments_info_for_user(conn, user_id).await?;
298                    UserOverviewFacetValue::Enrollments(
299                        enrollments
300                            .course_enrollments
301                            .into_iter()
302                            .map(|enrollment| {
303                                let instance_name = enrollment
304                                    .course_instances
305                                    .iter()
306                                    .map(|instance| {
307                                        instance
308                                            .name
309                                            .clone()
310                                            .unwrap_or_else(|| "Default".to_string())
311                                    })
312                                    .collect::<Vec<_>>()
313                                    .join(", ");
314                                EnrollmentFacet {
315                                    course_id: enrollment.course_id,
316                                    course_name: enrollment.course.name,
317                                    instance_name,
318                                    first_enrolled_at: enrollment.first_enrolled_at,
319                                    is_current: enrollment.is_current,
320                                    completed_modules_count: enrollment
321                                        .course_module_completions
322                                        .len(),
323                                    completions_needing_review_count: enrollment
324                                        .course_module_completions_needing_review,
325                                }
326                            })
327                            .collect(),
328                    )
329                }
330                UserOverviewFacet::CheatingFlags => {
331                    let flags = get_suspected_cheater_info_for_user(conn, user_id).await?;
332                    let course_ids: Vec<Uuid> = flags.iter().map(|f| f.course_id).collect();
333                    let course_names: std::collections::HashMap<Uuid, String> =
334                        courses::get_by_ids(conn, &course_ids)
335                            .await?
336                            .into_iter()
337                            .map(|c| (c.id, c.name))
338                            .collect();
339                    let mut rows = Vec::with_capacity(flags.len());
340                    for flag in flags {
341                        let course_name = course_names.get(&flag.course_id).cloned();
342                        rows.push(CheatingFlagFacet {
343                            course_id: flag.course_id,
344                            course_name,
345                            status: flag.status,
346                            total_points: flag.total_points,
347                            total_duration_seconds: flag.total_duration_seconds,
348                            threshold_seconds: flag.threshold_seconds,
349                            created_at: flag.first_flagged_at,
350                        });
351                    }
352                    UserOverviewFacetValue::CheatingFlags(rows)
353                }
354                UserOverviewFacet::EmailDeliveries => {
355                    let deliveries =
356                        get_recent_deliveries_for_user(conn, user_id, EMAIL_DELIVERY_LIMIT).await?;
357                    UserOverviewFacetValue::EmailDeliveries(
358                        deliveries
359                            .into_iter()
360                            .map(|delivery| EmailDeliveryFacet {
361                                email_template_type: delivery.email_template_type,
362                                created_at: delivery.created_at,
363                                status: delivery.status,
364                                retry_count: delivery.retry_count,
365                                last_attempt_at: delivery.last_attempt_at,
366                                failure_code: delivery.failure_code,
367                                failure_is_transient: delivery.failure_is_transient,
368                            })
369                            .collect(),
370                    )
371                }
372            };
373            facets.insert(facet.wire_name().to_string(), value);
374        }
375
376        Ok(UserOverviewTool {
377            state: UserOverviewState {
378                facets,
379                base_url,
380                user_id,
381            },
382        })
383    }

390    fn output_description_instructions(&self) -> Option<String> {
391        let mut notes: Vec<String> = Vec::new();
392        let base_url = &self.state.base_url;
393        let user_id = self.state.user_id;
394
395        let has_profile = self
396            .state
397            .facets
398            .contains_key(UserOverviewFacet::Profile.wire_name());
399        let has_enrollments = self
400            .state
401            .facets
402            .contains_key(UserOverviewFacet::Enrollments.wire_name());
403        if has_profile || has_enrollments {
404            notes.push(format!(
405                "{base_url}/manage/users/{user_id} shows this user's enrollment list and profile \
406                 fields for cross-checking."
407            ));
408        }
409
410        if let Some(UserOverviewFacetValue::CheatingFlags(flags)) = self
411            .state
412            .facets
413            .get(UserOverviewFacet::CheatingFlags.wire_name())
414            && !flags.is_empty()
415        {
416            let tab = |status: &SuspectedCheaterStatus| match status {
417                SuspectedCheaterStatus::Flagged => "suspected",
418                SuspectedCheaterStatus::ConfirmedCheating => "confirmed",
419                SuspectedCheaterStatus::Dismissed => "dismissed",
420            };
421            let course_links = flags
422                .iter()
423                .map(|f| {
424                    format!(
425                        "{base_url}/manage/courses/{}/other/cheaters/{}",
426                        f.course_id,
427                        tab(&f.status)
428                    )
429                })
430                .collect::<Vec<_>>()
431                .join(", ");
432            notes.push(format!(
433                "cheating_flags rows are not current suspicions: the list includes resolved \
434                 cases. Only status 'Flagged' is awaiting review and actionable, and only via \
435                 update_cheating_status (read that tool's own description before recommending \
436                 any action); 'ConfirmedCheating' and 'Dismissed' are terminal, closed cases. A \
437                 flag reflects a system heuristic, not evidence of misconduct \u{2014} never \
438                 quote, imply, or hint at a cheating suspicion in any text meant for the \
439                 student. created_at is when the flag was first raised, not the latest status \
440                 change. Verify at {base_url}/manage/users/{user_id}, and per flag at: \
441                 {course_links}. That per-course table is keyed by user_id only, with no name \
442                 or email column, so match on the UUID."
443            ));
444        }
445
446        if let Some(UserOverviewFacetValue::Enrollments(enrollments)) = self
447            .state
448            .facets
449            .get(UserOverviewFacet::Enrollments.wire_name())
450            && !enrollments.is_empty()
451        {
452            let course_links = enrollments
453                .iter()
454                .map(|e| {
455                    format!(
456                        "{base_url}/manage/courses/{}/user-status-summary/{user_id}",
457                        e.course_id
458                    )
459                })
460                .collect::<Vec<_>>()
461                .join(", ");
462            notes.push(format!(
463                "enrollments.completed_modules_count counts completion rows, including failed \
464                 and under-review ones, and can double-count a module completed twice for a \
465                 grade improvement \u{2014} it is not the number of modules passed. \
466                 completions_needing_review_count > 0 is internal and explains a missing or \
467                 delayed certificate; never surface it to the student. is_current reflects which \
468                 language version of the course the user last selected, not whether they are \
469                 currently enrolled \u{2014} is_current: false can still be an active enrollment. \
470                 instance_name 'Default' is the course's unnamed default instance, not missing \
471                 data; multiple names mean multiple enrollments in the same course. Enrollments \
472                 in deleted courses are silently omitted, so their absence here does not mean \
473                 the student was never enrolled. Per enrollment, the deeper per-course view is \
474                 at: {course_links} \u{2014} its \"X of Y modules\" figure is derived the same \
475                 way as completed_modules_count, so it is not an independent check of the \
476                 caveat above."
477            ));
478        }
479
480        if let Some(UserOverviewFacetValue::Roles(roles)) =
481            self.state.facets.get(UserOverviewFacet::Roles.wire_name())
482            && !roles.is_empty()
483        {
484            let role_links = roles
485                .iter()
486                .map(|role| {
487                    if let Some(course_instance_id) = role.course_instance_id {
488                        format!(
489                            "{base_url}/manage/course-instances/{course_instance_id}/permissions"
490                        )
491                    } else if let Some(course_id) = role.course_id {
492                        format!("{base_url}/manage/courses/{course_id}/permissions")
493                    } else if let Some(exam_id) = role.exam_id {
494                        format!("{base_url}/manage/exams/{exam_id}/permissions")
495                    } else {
496                        format!("{base_url}/manage/permissions")
497                    }
498                })
499                .collect::<Vec<_>>()
500                .join(", ");
501            notes.push(format!(
502                "roles only lists course-, course-instance-, or exam-scoped roles for this \
503                 user \u{2014} organization-wide and platform-wide roles are left out. UserRole \
504                 values are distinct capabilities, not a seniority hierarchy. A role on the \
505                 course in question means this person is staff there, not a student. Verify \
506                 each row's scope at: {role_links} (built from that row's own scope id) \u{2014} \
507                 the user page's role badges carry no scope attribution, so scope can only be \
508                 confirmed on these pages."
509            ));
510        }
511
512        if let Some(UserOverviewFacetValue::EmailDeliveries(deliveries)) = self
513            .state
514            .facets
515            .get(UserOverviewFacet::EmailDeliveries.wire_name())
516            && !deliveries.is_empty()
517        {
518            notes.push(
519                "email_deliveries.status: 'Sent' means handed to the mail relay, not confirmed \
520                 delivered; 'Queued' has not been handed over yet; 'Retrying' has failed at \
521                 least once but is still within the 3-day retry window; 'SendFailed' means \
522                 retries have stopped. Distinguish a recurring failure_code (delivery keeps \
523                 failing) from a send with no failure recorded (probably just landed in spam). \
524                 failure_is_transient absent means nothing has failed, not that severity is \
525                 unknown; last_attempt_at absent means no attempt has been made. This is only \
526                 the 20 most recent deliveries addressed to this account, not its full history."
527                    .to_string(),
528            );
529        }
530
531        if notes.is_empty() {
532            None
533        } else {
534            Some(notes.join(" "))
535        }
536    }
