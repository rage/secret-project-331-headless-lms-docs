30#[derive(serde::Deserialize)]
31pub struct DocumentLookupArguments {
32    /// Required by the tool's schema, but the lookup resolves the document by id or filepath and
33    /// never reads this back; kept only so a call missing it fails to deserialize.
34    #[allow(dead_code)]
35    title: String,
36    filepath: Option<String>,
37    #[serde(deserialize_with = "deserialize_to_optional_uuid_and_errors_to_none")]
38    page_id: Option<Uuid>,
39    format: String,
40}
41
42/// Deserializes an optional string field and parses it into an Uuid, doing this
43/// optionally without failure. If an error occurs, a None is returned. This is
44/// desired for the page_id field in DocumentLookupArguments, which is generated
45/// by an LLM and can be None or a non-Uuid string in some cases. If any errors
46/// should occur, they are emitted in ChatbotTools's from_db_and_arguments.
47fn deserialize_to_optional_uuid_and_errors_to_none<'de, D>(
48    deserializer: D,
49) -> Result<Option<Uuid>, D::Error>
50where
51    D: Deserializer<'de>,
52{
53    let res = String::deserialize(deserializer)
54        .ok()
55        .and_then(|s| Uuid::from_str(&s).ok());
56    Ok(res)
57}

59/// Truncates page content until its estimated token count fits the budget we are willing to hand
60/// to the LLM. Scaling the byte length by the token ratio always shrinks the content, so this
61/// terminates in a pass or two.
62fn shorten_page_content(mut content: String) -> String {
63    const MAX_TOKENS: i32 = 25_000;
64    loop {
65        let tokens = estimate_tokens(&content);
66        if tokens <= MAX_TOKENS {
67            return content;
68        }
69        let max_bytes = content.len() * (MAX_TOKENS as usize - 1_000) / tokens as usize;
70        content = truncate_utf8_at_boundary(&content, max_bytes).to_string();
71    }
72}

122/// Look up a document (page) from the course the chatbot is on.
123impl ChatbotTool for DocumentLookupTool {
124    type Arguments = DocumentLookupArguments;
125
126    async fn from_db_and_arguments(
127        conn: &mut PgConnection,
128        _app_config: &ApplicationConfiguration,
129        arguments: Self::Arguments,
130        user_context: &ChatbotUserContext,
131    ) -> ChatbotResult<Self> {
132        let Some(course_id) = user_context.course_id else {
133            return Err(chatbot_err!(
134                ToolUseError,
135                "Course id is missing.".to_string()
136            ));
137        };
138
139        let page_id = if let Some(id) = &arguments.page_id {
140            id.to_owned()
141        } else if let Some(f) = &arguments.filepath {
142            let res = parse_document_filepath(f);
143            match res {
144                Ok(d) => d.page_id,
145                Err(e) => Err(chatbot_err!(
146                    InvalidToolArguments,
147                    "Couldn't parse document file path and no valid page id was provided, unable to look up document.",
148                    e
149                ))?,
150            }
151        } else {
152            return Err(chatbot_err!(
153                InvalidToolArguments,
154                format!(
155                    "Unable to call document_lookup tool. No filepath or page id provided. One of them is needed to find the document."
156                )
157            ));
158        };
159        let page_content =
160            headless_lms_models::course_page_markdown_content::get_course_page_content_by_page_id(
161                conn, page_id,
162            )
163            .await?;
164
165        let document =
166            // A page of another course is not the learner's to read, so it reads as not found.
167            if page_content.course_id == course_id {
168                if arguments.format == "json" {
169                    let s = shorten_page_content(serde_json::to_string(&page_content.json_content)?);
170                    Some(s)
171                } else {
172                // use markdown content if there is any. else use json as string
173                if let Some(content) = page_content.markdown_content {
174                    let s = shorten_page_content(content);
175                    Some(s)
176                } else {
177                    let base = "Markdown content not found. Page JSON content:\n\n".to_string();
178                    let s = shorten_page_content(serde_json::to_string(&page_content.json_content)?);
179                    Some(base + &s)
180                }
181                }
182
183            } else {
184                None
185            };
186
187        Ok(DocumentLookupTool {
188            state: DocumentLookupState { document },
189        })
190    }
191
192    fn output(&self) -> String {
193        if let Some(d) = &self.state.document {
194            d.to_string()
195        } else {
196            "Document not found.".to_string()
197        }
198    }
199
200    fn output_description_instructions(&self) -> Option<String> {
201        Some("Do not return the whole document to the user. Use the document as a source of more information for answering the user etc. If you need to cite the content of this document, cite the Azure search result of the document.".to_string())
202    }
203}

205#[cfg(test)]
206mod tests {
207    use super::*;

209    #[test]
210    fn shorten_page_content_shortens_prose() {
211        let input = "The quick brown fox jumps over the lazy dog. ".repeat(4600);
212        assert!(input.len() > 200_000);
213
214        let shortened = shorten_page_content(input);
215
216        assert!(estimate_tokens(&shortened) <= 25_000);
217    }
218
219    #[test]
220    fn shorten_page_content_shortens_punctuation_heavy_json() {
221        let input = format!(
222            "[{}]",
223            r#"{"id":"1","name":"block","attributes":{"content":"Hei, mitä kuuluu?"}},"#
224                .repeat(2500)
225        );
226        assert!(input.len() > 150_000);
227
228        let shortened = shorten_page_content(input);
229
230        assert!(estimate_tokens(&shortened) <= 25_000);
231    }
232
233    #[test]
234    fn shorten_page_content_leaves_short_content_alone() {
235        let input = "Short enough.".to_string();
236
237        assert_eq!(shorten_page_content(input.clone()), input);
238    }
