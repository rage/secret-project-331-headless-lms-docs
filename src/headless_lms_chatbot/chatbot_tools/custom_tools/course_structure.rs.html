34#[derive(Serialize, Deserialize, Debug)]
35#[serde(rename_all = "snake_case")]
36pub enum PageType {
37    CourseFrontPage,
38    TopLevelPage,
39    ChapterFrontPage,
40    GenericPage,
41}
42
43impl PageType {
44    /// Determine page type based on page's position in course structure
45    fn determine(
46        order_number: i32,
47        chapter_number: Option<i32>,
48        module_number: Option<i32>,
49    ) -> Self {
50        if chapter_number.is_none() && module_number.is_none() && order_number == 0 {
51            PageType::CourseFrontPage
52        } else if chapter_number.is_none() && module_number.is_none() && order_number != 0 {
53            PageType::TopLevelPage
54        } else if chapter_number.is_some() && order_number == 0 {
55            PageType::ChapterFrontPage
56        } else {
57            PageType::GenericPage
58        }
59    }
60}

113impl ChatbotTool for CourseStructureTool {
114    type Arguments = CourseStructureArguments;
115
116    fn call_requirements(
117        arguments: &Self::Arguments,
118        user_context: &ChatbotTurnContext,
119    ) -> Vec<ToolRequirement> {
120        material_requirements(resolve_course_scope(user_context, arguments.course_id).ok())
121    }
122
123    /// A model that treats this tool as parameterless sends an empty argument string, which is
124    /// not valid JSON, so that keeps working alongside `course_id`.
125    fn parse_arguments(args_string: String) -> ChatbotResult<Self::Arguments> {
126        if args_string.trim().is_empty() {
127            return Ok(CourseStructureArguments { course_id: None });
128        }
129        serde_json::from_str(&args_string).map_err(|e| {
130            chatbot_err!(
131                InvalidToolArguments,
132                format!("Couldn't parse tool arguments. Arguments: {args_string}"),
133                e
134            )
135        })
136    }
137
138    async fn from_db_and_arguments(
139        conn: &mut PgConnection,
140        _app_config: &ApplicationConfiguration,
141        arguments: Self::Arguments,
142        user_context: &ChatbotTurnContext,
143    ) -> ChatbotResult<Self>
144    where
145        Self: Sized,
146    {
147        let course_id_from_argument = arguments.course_id.is_some();
148        let course_id = resolve_course_scope(user_context, arguments.course_id)?;
149
150        let mut pages_info = pages::get_page_info_special_for_course(conn, course_id).await?;
151        pages_info.sort_by_key(|x| {
152            // map module number 0 to 1 so that pages without a module
153            // are ordered first. same for chapters.
154            // order by module first, then chapter, then page number.
155            x.module_number.map(|x| x + 1).unwrap_or(0) * 100
156                + x.chapter_number.map(|x| x + 1).unwrap_or(0) * 10
157                + x.order_number
158        });
159
160        let info: Vec<PageDocumentInfo> = pages_info
161            .into_iter()
162            .map(|p| {
163                let blocks = p.blocks_cloned();
164                let Ok(b) = blocks else {
165                    // no block content
166                    return PageDocumentInfo {
167                        page_id: p.page_id,
168                        url_path: p.url_path,
169                        page_title: p.page_title,
170                        page_type: PageType::determine(
171                            p.order_number,
172                            p.chapter_number,
173                            p.module_number,
174                        ),
175                        chapter_title: p.chapter_title,
176                        learning_objectives: None,
177                        chapter_number: p.chapter_number,
178                        module_name: p.module_name,
179                    };
180                };
181                let learning_objectives = get_learning_objectives(b).ok();
182                PageDocumentInfo {
183                    page_id: p.page_id,
184                    url_path: p.url_path,
185                    page_title: p.page_title,
186                    page_type: PageType::determine(
187                        p.order_number,
188                        p.chapter_number,
189                        p.module_number,
190                    ),
191                    chapter_title: p.chapter_title,
192                    learning_objectives,
193                    chapter_number: p.chapter_number,
194                    module_name: p.module_name,
195                }
196            })
197            .collect();
198
199        Ok(CourseStructureTool {
200            state: CourseStructureState {
201                course_pages_info: info,
202                course_id_from_argument,
203            },
204        })
205    }
206
207    fn output(&self) -> String {
208        serde_json::to_string(&self.state.course_pages_info).unwrap_or("Not found.".to_string())
209    }
210
211    fn output_description_instructions(&self) -> Option<String> {
212        let closing = if self.state.course_id_from_argument {
213            "This is a course the admin is looking up on behalf of a user, not the one this chat is running on. Look up a listed page's content with document_lookup using its page_id, or search the course's pages with course_material_search."
214        } else {
215            "The user has access to the course structure, so you shouldn't give it to them: they know it already. You can give an overview if asked. Look up a listed page's content with document_lookup using its page_id, or search the course's pages with course_material_search."
216        };
217        Some(format!(
218            "Use the course structure to find out more about the course and answer the user's questions. The learning objectives listed on the course front page or top level pages are objectives for the whole course. Learning objectives listed on a chapter front page encompass the whole chapter, and objectives listed on a generic page are for the page only. {closing}"
219        ))
220    }
