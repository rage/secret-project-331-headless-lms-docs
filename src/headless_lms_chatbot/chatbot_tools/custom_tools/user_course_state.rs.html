107            Self::Resets => "resets",

114fn parse_facets(raw: &[String]) -> ChatbotResult<Vec<UserCourseStateFacet>> {
115    if raw.is_empty() {
116        return Err(chatbot_err!(
117            InvalidToolArguments,
118            "facets must not be empty. Valid facets: progress, completions, submissions, reviews, resets, certificates, credit_registration.".to_string()
119        ));
120    }
121    let mut seen = std::collections::HashSet::new();
122    let mut facets = Vec::new();
123    for wire in raw {
124        let facet = UserCourseStateFacet::from_wire(wire).ok_or_else(|| {
125            chatbot_err!(
126                InvalidToolArguments,
127                format!(
128                    "Unknown facet '{wire}'. Valid facets: {}.",
129                    UserCourseStateFacet::ALL_WIRE_NAMES.join(", ")
130                )
131            )
132        })?;
133        if seen.insert(facet) {
134            facets.push(facet);
135        }
136    }
137    Ok(facets)
138}

156/// Manual, not derived: ids and facets need validation `#[derive(Deserialize)]` can't express,
157/// and this is what [ChatbotTool::Arguments]'s `DeserializeOwned` bound is satisfied by
158/// (`parse_arguments` below is overridden and never calls it, but the bound still has to hold).
159impl<'de> Deserialize<'de> for UserCourseStateArguments {
160    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
161    where
162        D: serde::Deserializer<'de>,
163    {
164        let raw = RawArguments::deserialize(deserializer)?;
165        build_arguments(raw).map_err(serde::de::Error::custom)
166    }
167}
168
169fn build_arguments(raw: RawArguments) -> ChatbotResult<UserCourseStateArguments> {
170    let user_id = Uuid::from_str(&raw.user_id).map_err(|e| {
171        chatbot_err!(
172            InvalidToolArguments,
173            format!("'{}' is not a valid user_id.", raw.user_id),
174            e
175        )
176    })?;
177    let course_id = Uuid::from_str(&raw.course_id).map_err(|e| {
178        chatbot_err!(
179            InvalidToolArguments,
180            format!("'{}' is not a valid course_id.", raw.course_id),
181            e
182        )
183    })?;
184    let facets = parse_facets(&raw.facets)?;
185    Ok(UserCourseStateArguments {
186        user_id,
187        course_id,
188        exercise_id: raw.exercise_id,
189        facets,
190    })
191}

271    async fn from_db_and_arguments(
272        conn: &mut PgConnection,
273        app_config: &ApplicationConfiguration,
274        arguments: Self::Arguments,
275        _user_context: &ChatbotTurnContext,
276    ) -> ChatbotResult<Self> {
277        let base_url = app_config.base_url.trim_end_matches('/').to_string();
278        let course = headless_lms_models::courses::get_course(conn, arguments.course_id)
279            .await
280            .map_err(|e| {
281                chatbot_err!(
282                    InvalidToolArguments,
283                    format!("No course found with id {}.", arguments.course_id),
284                    e
285                )
286            })?;
287        let user_detail = user_details::get_user_details_by_user_id(conn, arguments.user_id)
288            .await
289            .map_err(|e| {
290                chatbot_err!(
291                    InvalidToolArguments,
292                    format!("No user found with id {}.", arguments.user_id),
293                    e
294                )
295            })?;
296
297        if let Some(exercise_id) = arguments.exercise_id {
298            let exercise = exercises::get_by_id(conn, exercise_id).await.map_err(|e| {
299                chatbot_err!(
300                    InvalidToolArguments,
301                    format!("No exercise found with id {exercise_id}."),
302                    e
303                )
304            })?;
305            if exercise.course_id != Some(arguments.course_id) {
306                return Err(chatbot_err!(
307                    InvalidToolArguments,
308                    format!(
309                        "Exercise {exercise_id} does not belong to course {}.",
310                        arguments.course_id
311                    )
312                ));
313            }
314        }
315
316        let mut facets = IndexMap::new();
317
318        // Fetched once and shared across facets instead of once per facet, since a single call
319        // commonly requests several facets that would otherwise repeat the same query.
320        let course_exercises = if arguments.facets.iter().any(|f| {
321            matches!(
322                f,
323                UserCourseStateFacet::Submissions | UserCourseStateFacet::Reviews
324            )
325        }) {
326            Some(exercises::get_exercises_by_course_id(conn, arguments.course_id).await?)
327        } else {
328            None
329        };
330        let completions = if arguments.facets.iter().any(|f| {
331            matches!(
332                f,
333                UserCourseStateFacet::Completions
334                    | UserCourseStateFacet::Certificates
335                    | UserCourseStateFacet::CreditRegistration
336            )
337        }) {
338            Some(
339                course_module_completions::get_all_by_course_id_and_user_id(
340                    conn,
341                    arguments.course_id,
342                    arguments.user_id,
343                )
344                .await?,
345            )
346        } else {
347            None
348        };
349
350        for facet in &arguments.facets {
351            let value = match facet {
352                UserCourseStateFacet::Progress => UserCourseStateFacetValue::Progress(
353                    progress_facet(conn, arguments.user_id, arguments.course_id).await?,
354                ),
355                UserCourseStateFacet::Completions => {
356                    let completions = completions.as_ref().ok_or_else(|| {
357                        chatbot_err!(
358                            ToolUseError,
359                            "expected completions to have been prefetched".to_string()
360                        )
361                    })?;
362                    UserCourseStateFacetValue::Completions(
363                        completions_facet(
364                            conn,
365                            arguments.user_id,
366                            arguments.course_id,
367                            completions,
368                        )
369                        .await?,
370                    )
371                }
372                UserCourseStateFacet::Submissions => {
373                    let course_exercises = course_exercises.as_ref().ok_or_else(|| {
374                        chatbot_err!(
375                            ToolUseError,
376                            "expected course_exercises to have been prefetched".to_string()
377                        )
378                    })?;
379                    UserCourseStateFacetValue::Submissions(
380                        submissions_facet(
381                            conn,
382                            arguments.user_id,
383                            arguments.course_id,
384                            arguments.exercise_id,
385                            course_exercises,
386                        )
387                        .await?,
388                    )
389                }
390                UserCourseStateFacet::Reviews => {
391                    let course_exercises = course_exercises.as_ref().ok_or_else(|| {
392                        chatbot_err!(
393                            ToolUseError,
394                            "expected course_exercises to have been prefetched".to_string()
395                        )
396                    })?;
397                    UserCourseStateFacetValue::Reviews(
398                        reviews_facet(
399                            conn,
400                            arguments.user_id,
401                            arguments.course_id,
402                            arguments.exercise_id,
403                            course_exercises,
404                        )
405                        .await?,
406                    )
407                }
408                UserCourseStateFacet::Resets => UserCourseStateFacetValue::Resets(
409                    resets_facet(conn, arguments.user_id, arguments.course_id).await?,
410                ),
411                UserCourseStateFacet::Certificates => {
412                    let completions = completions.as_ref().ok_or_else(|| {
413                        chatbot_err!(
414                            ToolUseError,
415                            "expected completions to have been prefetched".to_string()
416                        )
417                    })?;
418                    UserCourseStateFacetValue::Certificates(
419                        certificates_facet(
420                            conn,
421                            arguments.user_id,
422                            arguments.course_id,
423                            completions,
424                        )
425                        .await?,
426                    )
427                }
428                UserCourseStateFacet::CreditRegistration => {
429                    let completions = completions.as_ref().ok_or_else(|| {
430                        chatbot_err!(
431                            ToolUseError,
432                            "expected completions to have been prefetched".to_string()
433                        )
434                    })?;
435                    UserCourseStateFacetValue::CreditRegistration(
436                        credit_registration_facet(conn, completions).await?,
437                    )
438                }
439            };
440            facets.insert(facet.wire_name().to_string(), value);
441        }
442
443        Ok(UserCourseStateTool {
444            state: UserCourseStateState {
445                output: UserCourseStateOutput {
446                    user_email: user_detail.email.clone(),
447                    course_name: course.name.clone(),
448                    facets,
449                },
450                base_url,
451                user_id: arguments.user_id,
452                course_id: arguments.course_id,
453            },
454        })
455    }

461    fn output_description_instructions(&self) -> Option<String> {
462        let facets = &self.state.output.facets;
463        let mut notes = Vec::new();
464        let base_url = &self.state.base_url;
465
466        if facets.contains_key(UserCourseStateFacet::Progress.wire_name())
467            || facets.contains_key(UserCourseStateFacet::Completions.wire_name())
468            || facets.contains_key(UserCourseStateFacet::Submissions.wire_name())
469            || facets.contains_key(UserCourseStateFacet::Reviews.wire_name())
470        {
471            notes.push(format!(
472                "{base_url}/manage/courses/{}/user-status-summary/{} shows progress, module completions, \
473                submissions and reviews for this student on one page - check the numbers there before telling \
474                the admin anything a student will act on.",
475                self.state.course_id, self.state.user_id,
476            ));
477        }
478
479        if facets.contains_key(UserCourseStateFacet::Progress.wire_name()) {
480            notes.push(
481                "progress: unlike the other facets this one serializes explicit nulls rather than omitting fields. \
482                score_required / attempted_exercises_required are null when the module's completion policy is Manual \
483                (staff grants completion), not \"nothing required\". Totals are computed across all chapters, including \
484                ones not yet opened, so a score can be measured against material the student cannot see yet. \
485                score_given: 0 does not distinguish \"scored zero\" from \"never attempted\"; attempted_exercises only \
486                counts submitted/completed activity, so a drafted-but-unsubmitted answer counts as zero. The unnamed \
487                default module is reported under the course's own name. Exam exercises have no module and are excluded \
488                from every number here.".to_string(),
489            );
490        }
491
492        if let Some(UserCourseStateFacetValue::Completions(completions)) =
493            facets.get(UserCourseStateFacet::Completions.wire_name())
494        {
495            notes.push(
496                "needs_to_be_reviewed and anything derived from it is internal: it silently blocks certificates and \
497                grade visibility, and the student must never learn it exists - explain effects (\"the completion is \
498                still being processed\") without the cause. The user-status-summary page above does show this \
499                needs-review state (with a banner) to staff, so that page is where an admin can confirm the \
500                blocked-certificate cause directly; the no-cause-to-student rule above only governs what you tell \
501                the student.".to_string(),
502            );
503            if completions
504                .raw_completions
505                .iter()
506                .any(|c| c.needs_to_be_reviewed)
507            {
508                notes.push(
509                    "module_completion_statuses.completed: false can mean a completion exists in raw_completions but \
510                    needs_to_be_reviewed - this is deliberate so a flagged student cannot infer suspicion, not a data \
511                    contradiction. When that happens, passed is also null on the status entry as a consequence of \
512                    completed: false, not a separate signal.".to_string(),
513                );
514            }
515            notes.push(
516                "raw_completions lists every completion row, including failed and superseded ones; \
517                module_completion_statuses reports only the best one per module (grade improvement wins), so the two \
518                can legitimately disagree in count - a regrade always writes a new row rather than editing one in \
519                place. passed: false is a recorded failure, not \"still in progress\". grade: null means the module \
520                is pass/fail, not that a grade is missing. completion_granter distinguishes staff-granted completions \
521                (exempt from cheating flags) from automatic ones. eligible_for_ects: false blocks credit registration \
522                regardless of the module's ECTS setting. completion_registration_attempt_date absent means the \
523                student never opened the open-university registration form - pair with the credit_registration facet.".to_string(),
524            );
525        }
526
527        if let Some(UserCourseStateFacetValue::Submissions(submissions)) =
528            facets.get(UserCourseStateFacet::Submissions.wire_name())
529        {
530            notes.push(
531                "Quote submission timestamps exactly when the question is whether an answer saved."
532                    .to_string(),
533            );
534            notes.push(format!(
535                "submissions are timestamps only, with no score or answer content. An absent row can mean nothing \
536                was submitted, or that a reset soft-deleted the submissions it would have come from - cross-check \
537                the resets facet before telling anyone an answer never saved. exercise_name absent means the exercise \
538                has since been deleted from the course. exercise_attempts is present only when exercise_id was \
539                passed; its absence means \"not requested\", never \"no attempts\". Each row's exercise_id links to \
540                {base_url}/manage/exercises/<exercise_id>/submissions, the exercise-wide submission view.",
541            ));
542            if let Some(attempts) = &submissions.exercise_attempts {
543                notes.push(
544                    "attempt_count sums submissions across all slides, but out_of_tries is decided by the worst \
545                    single slide reaching the cap, so attempt_count can exceed max_tries_per_slide with \
546                    out_of_tries: false, or be below it with out_of_tries: true.".to_string(),
547                );
548                if attempts.limit_number_of_tries && attempts.max_tries_per_slide.is_none() {
549                    notes.push(
550                        "This exercise has limit_number_of_tries: true but no max_tries_per_slide, which means the \
551                        limit does not actually exist - do not tell the student they are out of tries based on \
552                        limit_number_of_tries alone.".to_string(),
553                    );
554                }
555            }
556        }
557
558        if let Some(UserCourseStateFacetValue::Reviews(_)) =
559            facets.get(UserCourseStateFacet::Reviews.wire_name())
560        {
561            notes.push(
562                "in_review_stages only queries PeerReview, SelfReview, WaitingForPeerReviews and \
563                WaitingForManualGrading; it omits ReviewedAndLocked (reviewed, scored, and permanently unanswerable \
564                because the model solution was revealed) and the chapter-locking stages, so an empty list does not \
565                mean nothing is blocking the student - only a reset clears ReviewedAndLocked, and the \
566                user-status-summary page above shows ReviewedAndLocked as the exercise's own state, so check there \
567                when this list is empty. NotStarted never \
568                appears here and is the answerable state; any listed stage means the student cannot submit. Who is \
569                blocking differs by stage: PeerReview/SelfReview waits on the student's own reviews, \
570                WaitingForPeerReviews waits on other students, WaitingForManualGrading waits on the teacher. \
571                teacher_grading_decisions are the latest decision per still-live exercise state, so a reset that \
572                soft-deletes the state makes an earlier decision disappear from this list - an empty list is not \
573                proof the student was never graded down, and a listed decision may already be superseded by a later \
574                submission. hidden: true means the decision is hidden from the student - never quote its \
575                justification back to them; hidden absent means unrecorded, not safe to quote. A FullPoints decision \
576                can be system-generated by the peer-review timeout auto-pass. In peer_review_queue, no entry for an \
577                exercise usually means the student has not yet given their own required peer reviews, not that \
578                nothing is pending; peer_review_priority is higher-served-sooner, not a queue position.".to_string(),
579            );
580        }
581
582        if let Some(UserCourseStateFacetValue::Resets(resets)) =
583            facets.get(UserCourseStateFacet::Resets.wire_name())
584        {
585            notes.push(
586                "reset_by absent means either the system did it (e.g. the automatic peer-review reset) or the \
587                acting staff account has since been deleted - it does not mean the actor is unknown. reason is \
588                shown to the student verbatim in the course material as the reset notice, so treat it as \
589                user-facing text, not an internal field.".to_string(),
590            );
591            notes.push(format!(
592                "{base_url}/manage/users/{} lists this student's exercise resets across every course, not just this \
593                one - a superset of this facet worth checking when a reset elsewhere might explain something here.",
594                self.state.user_id,
595            ));
596            if !resets.resets.is_empty() {
597                notes.push(
598                    "A reset soft-deletes the affected submissions, gradings, peer-review queue entries and \
599                    exercise states, and unlocks the affected chapters, but it does not touch \
600                    course_module_completions - a module can still show completed/graded while progress and \
601                    submissions show nothing. A reset restores tries; it does not revoke completions.".to_string(),
602                );
603            }
604        }
605
606        if let Some(UserCourseStateFacetValue::Certificates(certificates)) =
607            facets.get(UserCourseStateFacet::Certificates.wire_name())
608        {
609            notes.push(
610                "certificates.configurations lists only single-module (\"default\") certificate configurations - an \
611                empty array does not mean the course has no certificate. modules_blocked_by_pending_review is the \
612                internal cause (a pending suspected-cheater review) and must never be told to the student; \
613                missing_module_completions is the safe, student-explainable list. generated_certificates empty does \
614                not mean generation failed - certificates are only created when the student explicitly clicks \
615                generate, nothing generates them automatically; empty plus eligible: true means they can download it \
616                now. verification_id both verifies and grants access to the certificate image - only share it with \
617                the certificate's owner.".to_string(),
618            );
619            let certificates_search_url = search_url(
620                base_url,
621                &format!(
622                    "/manage/courses/{}/students/certificates",
623                    self.state.course_id
624                ),
625                &self.state.output.user_email,
626            );
627            notes.push(format!(
628                "{certificates_search_url} lists this student's generated certificates (issued date, \
629                verification URL, certificate image) for cross-checking generated_certificates.",
630            ));
631            if certificates
632                .configurations
633                .iter()
634                .any(|c| c.eligible && !c.missing_module_completions.is_empty())
635            {
636                notes.push(
637                    "eligible checks completions across the whole platform and ignores passed, while \
638                    missing_module_completions counts an unpassed completion as missing - so eligible: true \
639                    alongside a non-empty missing_module_completions is an expected combination here, not a bug; \
640                    trust missing_module_completions when explaining what is left to do.".to_string(),
641                );
642            }
643        }
644
645        if facets.contains_key(UserCourseStateFacet::CreditRegistration.wire_name()) {
646            notes.push(
647                "credit_registration has one row per completion, keyed by course_module_id - a module with no \
648                completion produces no row at all, so an empty array can mean \"nothing completed yet\" rather than \
649                \"nothing registered\". study_registry: \"This platform\" means this platform recorded the \
650                attainment itself, not that an external registry accepted it. registered_at is the row's own \
651                created_at, not the registration date shown to the student in the registry.".to_string(),
652            );
653            let completions_search_url = search_url(
654                base_url,
655                &format!(
656                    "/manage/courses/{}/students/completions",
657                    self.state.course_id
658                ),
659                &self.state.output.user_email,
660            );
661            notes.push(format!(
662                "{completions_search_url} shows per-module grade and registration status for \
663                cross-checking, and \
664                {base_url}/manage/credit-registration/registrations?user_id={} is the richer per-user view \
665                (attempt chain, event timeline, API calls, attainment ids).",
666                self.state.user_id,
667            ));
668        }
669
670        if notes.is_empty() {
671            return None;
672        }
673        Some(notes.join(" "))
674    }
675}

690/// An absolute `{base_url}{path}` URL with a single percent-encoded `search` query parameter.
691/// `search` can contain characters (e.g. a `+` in an email's local part) that are not safe to
692/// interpolate into a query string directly.
693fn search_url(base_url: &str, path: &str, search: &str) -> String {
694    url::Url::parse(&format!("{base_url}{path}"))
695        .map(|mut url| {
696            url.query_pairs_mut().append_pair("search", search);
697            url.to_string()
698        })
699        .unwrap_or_else(|_| format!("{base_url}{path}"))
700}

780async fn submissions_facet(
781    conn: &mut PgConnection,
782    user_id: Uuid,
783    course_id: Uuid,
784    exercise_id: Option<Uuid>,
785    course_exercises: &[Exercise],
786) -> ChatbotResult<SubmissionsFacet> {
787    let times =
788        exercise_slide_submissions::get_user_course_submission_times(conn, user_id, course_id)
789            .await?;
790    let exercise_names = exercise_name_index(course_exercises);
791
792    let submissions = times
793        .iter()
794        .filter(|t| exercise_id.is_none_or(|id| t.exercise_id == id))
795        .map(|t| SubmissionRow {
796            created_at: t.created_at,
797            exercise_id: t.exercise_id,
798            exercise_name: exercise_names.get(&t.exercise_id).map(|s| s.to_string()),
799            course_module_id: t.course_module_id,
800        })
801        .collect();
802
803    let mut exercise_attempts = None;
804
805    if let Some(exercise_id) = exercise_id
806        && let Some(exercise) = course_exercises.iter().find(|e| e.id == exercise_id)
807    {
808        let counts_per_slide =
809            exercise_slide_submissions::get_exercise_slide_submission_counts_for_exercise_user(
810                conn,
811                exercise_id,
812                CourseOrExamId::Course(course_id),
813                user_id,
814            )
815            .await?;
816        let attempt_count: i64 = counts_per_slide.values().sum();
817        // A slide is out of tries once its own count reaches the per-slide cap, whichever slide
818        // gets there first - matching the per-slide limit the course-material endpoint enforces.
819        let max_slide_attempt_count = counts_per_slide.values().copied().max().unwrap_or(0);
820        let out_of_tries = exercise.limit_number_of_tries
821            && exercise
822                .max_tries_per_slide
823                .is_some_and(|max| max_slide_attempt_count >= max as i64);
824        exercise_attempts = Some(ExerciseAttempts {
825            attempt_count,
826            max_tries_per_slide: exercise.max_tries_per_slide,
827            limit_number_of_tries: exercise.limit_number_of_tries,
828            out_of_tries,
829        });
830    }
831
832    Ok(SubmissionsFacet {
833        submissions,
834        exercise_attempts,
835    })
836}

1035async fn certificates_facet(
1036    conn: &mut PgConnection,
1037    user_id: Uuid,
1038    course_id: Uuid,
1039    raw_completions: &[CourseModuleCompletion],
1040) -> ChatbotResult<CertificatesFacet> {
1041    let configurations =
1042        certificate_configurations::get_default_certificate_configurations_and_requirements_by_course(
1043            conn, course_id,
1044        )
1045        .await?;
1046    let modules = course_modules::get_by_course_id(conn, course_id).await?;
1047    let module_names: HashMap<Uuid, String> = modules
1048        .iter()
1049        .map(|m| {
1050            (
1051                m.id,
1052                m.name
1053                    .clone()
1054                    .unwrap_or_else(|| "Default module".to_string()),
1055            )
1056        })
1057        .collect();
1058
1059    let mut configurations_json = Vec::new();
1060    for configuration in &configurations {
1061        let eligible = configuration
1062            .requirements
1063            .has_user_completed_all_requirements(conn, user_id)
1064            .await?;
1065        let mut missing_module_completions = Vec::new();
1066        // A completion can exist but still be withheld from certificate eligibility because it
1067        // needs_to_be_reviewed (e.g. a suspected-cheater flag) - surfaced here separately so
1068        // support can explain "still processing" without naming the real, internal reason.
1069        let mut modules_blocked_by_pending_review = Vec::new();
1070        for module_id in &configuration.requirements.course_module_ids {
1071            let module_name = module_names
1072                .get(module_id)
1073                .cloned()
1074                .unwrap_or_else(|| module_id.to_string());
1075            match raw_completions
1076                .iter()
1077                .find(|c| c.course_module_id == *module_id)
1078            {
1079                None => missing_module_completions.push(module_name),
1080                Some(c) if c.needs_to_be_reviewed => {
1081                    modules_blocked_by_pending_review.push(module_name)
1082                }
1083                Some(c) if !c.passed => missing_module_completions.push(module_name),
1084                Some(_) => {}
1085            }
1086        }
1087        configurations_json.push(CertificateConfigurationRow {
1088            certificate_configuration_id: configuration.certificate_configuration.id,
1089            eligible,
1090            missing_module_completions,
1091            modules_blocked_by_pending_review,
1092        });
1093    }
1094
1095    let generated_certificates = generated_certificates::get_all_by_user_id(conn, user_id)
1096        .await?
1097        .into_iter()
1098        .filter(|c| c.course_id == course_id)
1099        .map(|c| GeneratedCertificateRow {
1100            verification_id: c.verification_id,
1101            name_on_certificate: c.name_on_certificate,
1102            created_at: c.created_at,
1103        })
1104        .collect();
1105
1106    Ok(CertificatesFacet {
1107        configurations: configurations_json,
1108        generated_certificates,
1109    })
1110}
