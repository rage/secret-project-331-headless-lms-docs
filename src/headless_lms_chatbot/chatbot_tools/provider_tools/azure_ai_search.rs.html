15/// Separates the content fields Azure concatenates into one chunk. Baked into the format the
16/// search index was written with, so changing it silently breaks every indexed document.
17pub const CONTENT_FIELD_SEPARATOR: &str = ",|||,";
18
19pub fn get_azure_ai_search_tool_definition(
20    app_config: &ApplicationConfiguration,
21    course_id: Uuid,
22    use_semantic_reranking: bool,
23) -> ChatbotResult<AzureAISearchToolDefinition> {
24    let index_name = Url::parse(&app_config.base_url)?
25        .host_str()
26        .ok_or_else(|| {
27            chatbot_err!(
28                AzureRequestBuildError,
29                "Invalid application base url, no host"
30            )
31        })?
32        .replace(".", "-");
33    let search_config = azure_search_configuration(app_config)?;
34
35    let query_type = if use_semantic_reranking {
36        "vector_semantic_hybrid"
37    } else {
38        "vector_simple_hybrid"
39    };
40
41    let semantic_configuration = format!("{}-semantic-configuration", &index_name);
42
43    Ok(AzureAISearchToolDefinition {
44        data_type: "azure_ai_search".to_string(),
45        azure_ai_search: AzureAISearch {
46            indexes: vec![SearchIndex {
47                index_name,
48                project_connection_id: search_config.search_connection_id.to_owned(),
49                query_type: query_type.to_string(),
50                semantic_configuration,
51                embedding_dependency: EmbeddingDependency {
52                    dep_type: "deployment_name".to_string(),
53                    deployment_name: search_config.vectorizer_deployment_id.clone(),
54                },
55                in_scope: false,
56                top_k: 15,
57                strictness: 3,
58                filter: Some(SearchFilter::eq("course_id", course_id.to_string()).to_odata()?),
59                fields_mapping: FieldsMapping {
60                    content_fields_separator: CONTENT_FIELD_SEPARATOR.to_string(),
61                    content_fields: vec!["chunk_context".to_string(), "chunk".to_string()],
62                    filepath_field: "filepath".to_string(),
63                    title_field: "title".to_string(),
64                    url_field: "url".to_string(),
65                    vector_fields: vec!["text_vector".to_string()],
66                },
67            }],
68        },
69    })
70}
