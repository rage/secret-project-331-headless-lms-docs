1//! The confirm-and-execute primitive: a client tool whose confirmed answer runs a privileged
2//! mutation on the server instead of only rendering data the client already had.
3//!
4//! [ClientChatbotTool](crate::chatbot_tools::ClientChatbotTool) cannot carry this: its
5//! `parse_response`/`output` are synchronous and get no [PgConnection], no
6//! [ApplicationConfiguration] and no proof of who is acting. [ConfirmableActionTool] is the one place that
7//! threads those in, so every action tool gets the same confirm-parsing, exactly-once guard,
8//! transaction handling and audit write instead of a hand-rolled copy each.
9

27/// The one answer shape every action tool accepts.
28#[derive(Deserialize)]
29pub struct ConfirmAnswer {
30    pub confirmed: bool,
31}

33/// What executing a confirmed action produced.
34pub struct ExecutedAction {
35    /// Model-facing description of what happened. Never contains secrets.
36    pub output: String,
37    /// Data for the confirming admin's browser only (e.g. the reset link). Sent as a stream
38    /// event, never persisted, never shown to the model.
39    pub client_payload: Option<serde_json::Value>,
40    /// What the audit row records about the action's target and effect.
41    pub audit: ActionAuditFields,
42}

44/// What the audit row records about a confirmed action, beyond who ran it and what tool it was.
45pub struct ActionAuditFields {
46    pub target_user_id: Option<Uuid>,
47    pub course_id: Option<Uuid>,
48    /// One human-readable sentence, e.g. "Reset 3 exercises for jane@example.com in Course X".
49    /// Must never contain secrets: this is stored in `chatbot_action_logs`.
50    pub summary: String,
51}

53/// A client tool whose confirmed answer performs a privileged mutation on the server.
54///
55/// Suspends the turn like [ClientChatbotTool](crate::chatbot_tools::ClientChatbotTool); the
56/// client answers with [ConfirmAnswer]. A declined answer produces [Self::declined_output]
57/// without touching the database. A confirmed answer runs [Self::execute] inside the same
58/// transaction as the audit insert and the recorded tool output, guarded by
59/// `lock_unanswered_for_execution` so it can run at most once.
60pub trait ConfirmableActionTool: ChatbotToolDeclaration {
61    type Arguments;
62
63    /// Extra facts [Self::execute] gathers while performing the mutation, used only to decide
64    /// what [Self::output_description_instructions] should mention for this particular call. `()`
65    /// for a tool with nothing worth gating on.
66    type Facts;
67
68    fn parse_arguments(arguments: &str) -> ChatbotResult<Self::Arguments>;
69
70    /// What the caller must be allowed to do against what this call actually targets. Checked
71    /// before the turn suspends on the call and again immediately before the mutation runs.
72    fn call_requirements(
73        arguments: &Self::Arguments,
74        user_context: &ChatbotTurnContext,
75    ) -> Vec<ToolRequirement>;
76
77    /// Performs the mutation. Must re-verify every model-supplied display field against the
78    /// database (case-insensitively for emails) and refuse with a descriptive error rather than
79    /// mutate on a mismatch, so a wrong or stale display can never mutate a row it doesn't
80    /// describe. `authorization` both proves the confirming admin was checked against
81    /// this call's own requirements and names them, so the mutation cannot run ahead of the
82    /// check and the audit row cannot credit anyone else.
83    fn execute(
84        conn: &mut PgConnection,
85        app_config: &ApplicationConfiguration,
86        arguments: &Self::Arguments,
87        authorization: &ToolAuthorization<Self>,
88    ) -> impl std::future::Future<Output = ChatbotResult<(ExecutedAction, Self::Facts)>> + Send
89    where
90        Self: Sized;
91
92    fn declined_output(_arguments: &Self::Arguments) -> String {
93        "The admin declined the action. Nothing was changed. Do not retry unless asked to."
94            .to_string()
95    }
96
97    /// Just-in-time instructions for the LLM on how to describe the outcome. `facts` is `None` on
98    /// a declined answer (nothing executed) and `Some` after a successful [Self::execute], so
99    /// instructions can be conditioned on what this call actually did instead of always
100    /// including everything that could ever be true. `app_config` is available on both branches
101    /// (unlike `facts`) so a verification link can be offered even on a decline.
102    fn output_description_instructions(
103        arguments: &Self::Arguments,
104        facts: Option<&Self::Facts>,
105        app_config: &ApplicationConfiguration,
106    ) -> Option<String>;
107}
108
109/// Unicode-aware case-insensitive equality for a model-supplied display field against its
110/// current database value. `eq_ignore_ascii_case` would reject a match for cased letters outside
111/// ASCII (e.g. Finnish å/Ö) that differ only in case, spuriously refusing a genuine match.
112pub fn display_field_matches(actual: &str, supplied: &str) -> bool {
113    actual.to_lowercase() == supplied.to_lowercase()
114}
115
116/// Re-verifies a model-supplied display field against `actual`, per [ConfirmableActionTool::execute]'s
117/// contract: refuses with a descriptive error naming `field_name` and `rerun_tool` instead of
118/// mutating on a mismatch.
119pub fn verify_display_field(
120    field_name: &str,
121    actual: &str,
122    supplied: &str,
123    rerun_tool: &str,
124) -> ChatbotResult<()> {
125    if display_field_matches(actual, supplied) {
126        Ok(())
127    } else {
128        Err(chatbot_err!(
129            ToolUseError,
130            format!(
131                "The {field_name} does not match the current record. Re-run {rerun_tool} and try again."
132            )
133        ))
134    }
135}
